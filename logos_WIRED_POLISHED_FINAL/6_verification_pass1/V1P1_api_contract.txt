    def _parse_python_file(self, file_path: str) -> List[APIEndpoint]:
        """
        Internal helper to parse a single Python file using AST.
        Searches for FastAPI route decorators and extracts metadata.
        """
        results = []
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                tree = ast.parse(f.read())

            for node in ast.walk(tree):
                if isinstance(node, ast.FunctionDef):
                    for decorator in node.decorator_list:
                        endpoint = self._extract_endpoint_from_decorator(decorator, node, file_path)
                        if endpoint:
                            results.append(endpoint)
        except Exception as e:
            logger.error(f"Failed to parse Python file {file_path}: {e}")
        
        return results

    def _extract_endpoint_from_decorator(self, decorator: Any, node: ast.FunctionDef, file_path: str) -> Optional[APIEndpoint]:
        """
        Extracts API metadata from a FastAPI decorator node.
        Handles .get(), .post(), .put(), .delete(), and .patch().
        """
        # Handle cases like @router.get("/")
        if isinstance(decorator, ast.Call) and isinstance(decorator.func, ast.Attribute):
            method = decorator.func.attr.upper()
            if method in ["GET", "POST", "PUT", "DELETE", "PATCH"]:
                path = ""
                if decorator.args and isinstance(decorator.args[0], ast.Constant):
                    path = decorator.args[0].value
                
                # Extract response model if present
                response_model = None
                for keyword in decorator.keywords:
                    if keyword.arg == "response_model" and isinstance(keyword.value, ast.Name):
                        response_model = keyword.value.id
                
                return APIEndpoint(
                    path=path,
                    method=method,
                    function_name=node.name,
                    request_model=self._infer_request_model(node),
                    response_model=response_model,
                    file_path=file_path,
                    line_number=node.lineno,
                    tags=[]
                )
        return None

    def _infer_request_model(self, node: ast.FunctionDef) -> Optional[str]:
        """
        Examines function arguments to find Pydantic models used for request bodies.
        """
        for arg in node.args.args:
            if arg.annotation and isinstance(arg.annotation, ast.Name):
                # Simple heuristic: if it's not a primitive, it might be a model
                if arg.annotation.id not in ["int", "str", "float", "bool", "dict", "list"]:
                    return arg.annotation.id
        return None

    def scan_frontend_calls(self) -> List[FrontendCall]:
        """
        Recursively scans the frontend directory for API service calls.
        Uses regex patterns to find common API patterns in TypeScript files.
        """
        logger.info(f"Starting frontend scan in: {self.frontend_root}")
        calls = []

    def _parse_typescript_file(self, file_path: str, pattern: re.Pattern) -> List[FrontendCall]:
        """
        Parses a TypeScript file using regex to extract API call information.
        """
        results = []
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                content = f.read()
                lines = content.splitlines()
                
                for i, line in enumerate(lines):
                    matches = pattern.finditer(line)
                    for match in matches:
                        results.append(FrontendCall(
                            method=match.group(1).upper(),
                            interface_name=match.group(2),
                            url=match.group(3),
                            file_path=file_path,
                            line_number=i + 1,
                            is_async="async" in line or "await" in line
                        ))
        except Exception as e:
            logger.error(f"Failed to parse TypeScript file {file_path}: {e}")
        
        return results

    def _find_matching_endpoint(self, call: FrontendCall) -> Optional[APIEndpoint]:
        """
        Attempts to match a frontend URL call to a backend path template.
        Handles path parameters like /texts/{id} vs /texts/123.
        """
        for endpoint in self.found_endpoints:
            if endpoint.method != call.method:
                continue
            
            # Convert FastAPI path template to regex
            # e.g., /texts/{id}/chapter/{num} -> /texts/[^/]+/chapter/[^/]+
            pattern = re.sub(r'\{[^}]+\}', r'[^/]+', endpoint.path)
            # Ensure exact match
            pattern = f"^{pattern}$"
            
            if re.match(pattern, call.url):
                return endpoint
        return None

    def _is_endpoint_used(self, endpoint: APIEndpoint) -> bool:
        """Checks if a specific backend endpoint is referenced anywhere in the frontend."""
        for call in self.found_calls:
            if call.method != endpoint.method:
                continue
            pattern = re.sub(r'\{[^}]+\}', r'[^/]+', endpoint.path)
            if re.match(f"^{pattern}$", call.url):
                return True
        return False

    async def _validate_database_alignment(self, results: Dict[str, Any]) -> None:

        actually match the column names in the Railway PostgreSQL database.
        """

            if missing:
                results["db_schema_violations"].append({
                    "table": "texts",
                    "missing_columns": list(missing),
                    "severity": "CRITICAL"
                })

            # Check 'translator_profiles' table
            columns = await conn.fetch("SELECT column_name FROM information_schema.columns WHERE table_name = 'translator_profiles'")
            trans_cols = {row['column_name'] for row in columns}
            required_trans_cols = {'id', 'name', 'style_vector', 'works_translated'}
            missing_trans = required_trans_cols - trans_cols
            if missing_trans:
                results["db_schema_violations"].append({
                    "table": "translator_profiles",
                    "missing_columns": list(missing_trans),
                    "severity": "CRITICAL"
                })

"""
LOGOS - The Bible for Classical Studies
Endpoint Matcher - URI Pattern Recognition and Normalization

This module specializes in the complex task of matching dynamic URL patterns
between Python (FastAPI) and TypeScript (React/Axios). It handles path parameters,
query strings, and base URL stripping to ensure accurate contract mapping.

Author: Senior Software Architect
"""

import re
import logging
from typing import List, Dict, Optional, Tuple

logger = logging.getLogger(__name__)

class EndpointMatcher:
    """
    Advanced URI matching engine for LOGOS.
    Resolves differences between backend route templates and frontend literal calls.
    """

    def __init__(self):
        # Cache for compiled regex patterns to improve performance
        self._pattern_cache: Dict[str, re.Pattern] = {}

    def normalize_path(self, path: str) -> str:
        """
        Cleans and normalizes a URI path.
        - Removes trailing slashes
        - Ensures leading slash
        - Removes query parameters
        """
        if not path:
            return "/"
        
        # Remove query string
        path = path.split("?")[0]
        
        # Strip trailing slash unless it's just the root
        if len(path) > 1 and path.endswith("/"):
            path = path[:-1]
            
        # Ensure leading slash
        if not path.startswith("/"):
            path = "/" + path
            
        return path

    def template_to_regex(self, template: str) -> re.Pattern:
        """
        Converts a FastAPI style path template to a regex pattern.
        Example: /texts/{text_id}/chapters/{num} 
        becomes: ^/texts/(?P<text_id>[^/]+)/chapters/(?P<num>[^/]+)$
        """
        if template in self._pattern_cache:
            return self._pattern_cache[template]

        # Escape special regex characters except for braces
        escaped = re.escape(template)
        # Revert escaped braces for parameter identification
        escaped = escaped.replace(r'\{', '{').replace(r'\}', '}')
        
        # Replace {param} with named capture group
        pattern_str = re.sub(r'\{([a-zA-Z0-9_]+)\}', r'(?P<\1>[^/]+)', escaped)
        
        # Add anchors for exact matching
        pattern = re.compile(f"^{pattern_str}$")
        self._pattern_cache[template] = pattern
        return pattern

    def match(self, template: str, actual_url: str) -> Tuple[bool, Dict[str, str]]:
        """
        Checks if an actual URL matches a template and extracts parameters.
        
        :param template: The backend route template (e.g., /author/{id})
        :param actual_url: The frontend call URL (e.g., /author/42)
        :return: (is_match, extracted_params)
        """
        normalized_template = self.normalize_path(template)
        normalized_url = self.normalize_path(actual_url)
        
        regex = self.template_to_regex(normalized_template)
        match = regex.match(normalized_url)
        
        if match:
            return True, match.groupdict()
        return False, {}

    def find_best_match(self, actual_url: str, method: str, 
                        available_endpoints: List[Any]) -> Optional[Any]:
        """
        Iterates through all known backend endpoints to find the best match for a frontend call.
        
        :param actual_url: The URL found in frontend code.
        :param method: HTTP Verb (GET, POST, etc.)
        :param available_endpoints: List of APIEndpoint objects.
        """
        normalized_url = self.normalize_path(actual_url)
        
        for endpoint in available_endpoints:
            if endpoint.method.upper() != method.upper():
                continue
                
            is_match, _ = self.match(endpoint.path, normalized_url)
            if is_match:
                return endpoint
                
        return None

    def validate_translator_param(self, params: Dict[str, str]) -> bool:
        """
        Specific business logic for LOGOS:
        If a 'translator' parameter is present in the URL, it must match 
        the 38 approved classical translators to prevent copyright issues.
        """
        valid_translators = {
            "Jowett", "Dryden", "Dakyns", "Pope", "Murray", "Butler", "Church_Brodribb", 
            "Cowper", "Butcher_Lang", "Lang_Leaf_Myers", "Conington", "Goodwin", "Storr", 
            "Roberts", "Aubrey_Stewart", "Williams", "Dryden_et_al", "Brookes_More", 
            "Ross", "Rawlinson", "Moore", "Evelyn-White", "Morshead", "Heseltine", 
            "Crawley", "Long", "Lindsay", "Jebb", "Macaulay", "Leonard", "Adlington", 
            "Smith", "Morris", "Butcher", "Derby", "Kenyon", "Hickie", "Anonymous"
        }
        
        if 'translator' in params:
            val = params['translator']
            if val not in valid_translators:
                logger.warning(f"Forbidden or unknown translator detected in URL: {val}")
                return False
        return True

    def check_for_collisions(self, endpoints: List[Any]) -> List[Dict[str, str]]:
        """
        Identifies backend routes that might collide due to overlapping patterns.
        Example: /texts/{id} and /texts/search might collide if not ordered correctly.
        """
        collisions = []
        for i, ep1 in enumerate(endpoints):
            for j, ep2 in enumerate(endpoints):
                if i >= j:
                    continue
                
                if ep1.method == ep2.method:
                    # Very basic check: do the paths look similar?
                    p1 = re.sub(r'\{[^}]+\}', '{X}', ep1.path)
                    p2 = re.sub(r'\{[^}]+\}', '{X}', ep2.path)
                    if p1 == p2:
                        collisions.append({
                            "endpoint_a": f"{ep1.method} {ep1.path}",
                            "endpoint_b": f"{ep2.method} {ep2.path}",
                            "file_a": ep1.file_path,
                            "file_b": ep2.file_path,
                            "severity": "HIGH"
                        })
        return collisions

    def analyze_api_coverage(self, endpoints: List[Any], calls: List[Any]) -> Dict[str, Any]:
        """
        Calculates coverage statistics for the API contract.
        """
        covered_endpoints = set()
        uncovered_endpoints = []
        
        for ep in endpoints:
            found = False
            for call in calls:
                is_match, _ = self.match(ep.path, call.url)
                if is_match and ep.method == call.method:
                    covered_endpoints.add(f"{ep.method} {ep.path}")
                    found = True
                    break
            if not found:
                uncovered_endpoints.append(f"{ep.method} {ep.path}")
                
        coverage_pct = (len(covered_endpoints) / len(endpoints)) * 100 if endpoints else 0
        
        return {
            "coverage_percentage": coverage_pct,
            "total_endpoints": len(endpoints),
            "covered_count": len(covered_endpoints),
            "uncovered_endpoints": uncovered_endpoints
        }

# filepath: verification/pass1/type_matcher.py
"""
LOGOS - The Bible for Classical Studies
Type Matcher - Cross-Language Type Consistency

This module compares Pydantic models (Python) with TypeScript Interfaces/Types.
It ensures that the data structures returned by the backend match the 
expectations of the frontend components.

Author: Senior Software Architect
"""

import re
import ast
import logging
from typing import Dict, List, Any, Optional, Set

logger = logging.getLogger(__name__)

class TypeMatcher:
    """
    Validates structural type consistency between Python and TypeScript.
    """

    def __init__(self):
        # Maps Python types to TS types
        self.type_map = {
            "str": "string",
            "int": "number",
            "float": "number",
            "bool": "boolean",
            "Any": "any",
            "Dict": "Record",
            "List": "Array",
            "Optional": "null | "
        }

    def parse_pydantic_model(self, file_content: str, model_name: str) -> Dict[str, str]:
        """
        Extracts fields and types from a Pydantic model using AST.
        """
        fields = {}
        try:
            tree = ast.parse(file_content)
            for node in ast.walk(tree):
                if isinstance(node, ast.ClassDef) and node.name == model_name:
                    for item in node.body:
                        if isinstance(item, ast.AnnAssign) and isinstance(item.target, ast.Name):
                            field_name = item.target.id
                            field_type = self._get_type_name(item.annotation)
                            fields[field_name] = field_type
        except Exception as e:
            logger.error(f"Error parsing Pydantic model {model_name}: {e}")
            
        return fields

    def _get_type_name(self, node: Any) -> str:
        """Recursively resolves AST type annotations to string representations."""
        if isinstance(node, ast.Name):
            return node.id
        elif isinstance(node, ast.Subscript):
            value = self._get_type_name(node.value)
            slice_val = self._get_type_name(node.slice)
            return f"{value}[{slice_val}]"
        elif isinstance(node, ast.Attribute):
            return node.attr
        elif isinstance(node, ast.Constant):
            return str(node.value)
        return "Unknown"

    def parse_ts_interface(self, file_content: str, interface_name: str) -> Dict[str, str]:
        """
        Extracts fields and types from a TypeScript interface using regex.
        Note: In a full production system, this would use a proper TS parser (like swc),
        but regex is used here for direct implementation within the Python validator.
        """
        fields = {}
        # Match interface Body: interface Name { ... }
        pattern = rf"interface\s+{interface_name}\s*\{{([^}}]+)\}}"
        match = re.search(pattern, file_content, re.MULTILINE | re.DOTALL)
        
        if match:
            body = match.group(1)
            # Match individual lines: fieldName: type;
            field_pattern = r"(\w+)\s*(\?|:)\s*([^;]+);"
            field_matches = re.finditer(field_pattern, body)
            for fm in field_matches:
                name = fm.group(1)
                t_type = fm.group(3).strip()
                fields[name] = t_type
                
        return fields

    def compare_models(self, py_fields: Dict[str, str], ts_fields: Dict[str, str]) -> List[str]:
        """
        Compares a Python Pydantic model with a TS interface.
        Returns a list of discrepancies.
        """
        discrepancies = []
        
        # Check for missing fields in TS
        for field, py_type in py_fields.items():
            if field not in ts_fields:
                discrepancies.append(f"Missing field in TS: '{field}' (expected by Python model)")
            else:
                # Basic type compatibility check
                ts_type = ts_fields[field]
                if not self._are_types_compatible(py_type, ts_type):
                    discrepancies.append(
                        f"Type mismatch for '{field}': Python '{py_type}' vs TS '{ts_type}'"
                    )
                    
        # Check for extra fields in TS
        for field in ts_fields:
            if field not in py_fields:
                discrepancies.append(f"Extra field in TS: '{field}' (not in Python model)")
                
        return discrepancies

    def _are_types_compatible(self, py_type: str, ts_type: str) -> bool:
        """
        Determines if a Python type and TS type are functionally equivalent.
        Example: List[str] is compatible with string[] or Array<string>.
        """
        # Normalize types
        py_type = py_type.lower()
        ts_type = ts_type.lower()
        
        # Primitive mappings
        if "str" in py_type and "string" in ts_type: return True
        if ("int" in py_type or "float" in py_type) and "number" in ts_type: return True
        if "bool" in py_type and "boolean" in ts_type: return True
        
        # Collection mappings
        if "list" in py_type and ("[]" in ts_type or "array" in ts_type): return True
        
        # Complex nested types (simplified for this pass)
        if py_type == ts_type: return True
        
        return False

    def validate_database_types(self, model_fields: Dict[str, str], db_schema: Dict[str, str]) -> List[str]:
        """
        Ensures that the Pydantic model types align with the PostgreSQL column types.
        """
        errors = []
        type_conversions = {
            "integer": ["int", "int64"],
            "character varying": ["str", "string"],
            "text": ["str", "string"],
            "boolean": ["bool"],
            "timestamp with time zone": ["datetime"],
            "numeric": ["float", "decimal"]
        }
        
        for field, py_type in model_fields.items():
            if field in db_schema:
                db_type = db_schema[field].lower()
                allowed_py = type_conversions.get(db_type, [])
                
                # Check if py_type (or part of it) is in allowed_py
                match_found = False
                for allowed in allowed_py:
                    if allowed in py_type.lower():
                        match_found = True
                        break
                
                if not match_found and allowed_py:
                    errors.append(
                        f"DB Type Conflict: Column '{field}' is '{db_type}' but model uses '{py_type}'"
                    )
        return errors

class ContractReporter:
    """
    Responsible for formatting and persisting validation results.

    def __init__(self, output_dir: str = "reports/contract"):
        self.output_dir = output_dir
        self.timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

    def generate_markdown_summary(self, results: Dict[str, Any]) -> str:
        """Creates a formatted Markdown report for GitHub Actions or developer review."""
        
        lines = [
            "# LOGOS API Contract Validation Report",
            f"**Timestamp:** {datetime.now().isoformat()}",
            f"**Status:** {'❌ FAILED' if self._has_critical_errors(results) else '✅ PASSED'}",
            "",
            "## 📊 Executive Summary",
            f"- **Total Backend Endpoints:** {results.get('total_backend_endpoints', 0)}",
            f"- **Total Frontend Calls:** {results.get('total_frontend_calls', 0)}",
            f"- **Unmapped Frontend Calls:** {len(results.get('unmapped_frontend_calls', []))}",
            f"- **Orphaned Backend Endpoints:** {len(results.get('orphaned_backend_endpoints', []))}",
            f"- **Database Schema Violations:** {len(results.get('db_schema_violations', []))}",
            ""
        ]

        # Critical Violations Section
        if results.get('db_schema_violations'):
            lines.append("## 🚨 Database Schema Violations")
            for violation in results['db_schema_violations']:
                lines.append(f"- **Table:** `{violation['table']}`")
                lines.append(f"  - Missing Columns: {', '.join(violation['missing_columns'])}")
            lines.append("")

        # Unmapped Frontend Calls
        if results.get('unmapped_frontend_calls'):
            lines.append("## 🔍 Unmapped Frontend Calls (No Backend Match)")
            lines.append("| Method | URL | File | Line |")
            lines.append("|--------|-----|------|------|")
            for call in results['unmapped_frontend_calls']:
                lines.append(f"| {call['method']} | `{call['url']}` | `{call['file_path']}` | {call['line_number']} |")
            lines.append("")

        # Orphaned Endpoints
        if results.get('orphaned_backend_endpoints'):
            lines.append("## 🧹 Orphaned Backend Endpoints (Unused)")
            for ep in results['orphaned_backend_endpoints']:
                lines.append(f"- `{ep['method']} {ep['path']}` (Function: `{ep['function_name']}` in `{ep['file_path']}`)")
            lines.append("")

        # Footer
        lines.append("---")
        lines.append("*Generated by LOGOS Contract Validator v1.0*")
        
        return "\n".join(lines)

    def _has_critical_errors(self, results: Dict[str, Any]) -> bool:
        """Determines if the validation results contain blocking issues."""
        if results.get('db_schema_violations'):
            return True
        if results.get('unmapped_frontend_calls'):
            return True
        return False

    def save_reports(self, results: Dict[str, Any]) -> None:
        """Saves both JSON and Markdown reports to the output directory."""
        import os
        if not os.path.exists(self.output_dir):
            os.makedirs(self.output_dir)

        # Save JSON
        json_path = os.path.join(self.output_dir, f"report_{self.timestamp}.json")
        try:
            with open(json_path, "w", encoding="utf-8") as f:
                json.dump(results, f, indent=4)
            logger.info(f"JSON report saved: {json_path}")
        except Exception as e:
            logger.error(f"Failed to save JSON report: {e}")

        # Save Markdown
        md_path = os.path.join(self.output_dir, f"summary_{self.timestamp}.md")
        try:
            md_content = self.generate_markdown_summary(results)
            with open(md_path, "w", encoding="utf-8") as f:
                f.write(md_content)
            logger.info(f"Markdown summary saved: {md_path}")
        except Exception as e:
            logger.error(f"Failed to save Markdown report: {e}")
