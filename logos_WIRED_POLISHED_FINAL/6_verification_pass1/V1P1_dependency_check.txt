        logger.info(f"Found {len(py_files)} Python files to analyze.")
        return py_files

    def extract_imports_from_file(self, file_path: Path) -> Set[str]:
        """
        Uses AST to parse a Python file and extract all top-level imports.
        
        Args:
            file_path (Path): Path to the python file.
            
        Returns:
            Set[str]: Set of module names imported.
        """
        imports = set()
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                tree = ast.parse(f.read())
            
            for node in ast.walk(tree):
                if isinstance(node, ast.Import):
                    for alias in node.names:
                        imports.add(alias.name.split('.')[0])
                elif isinstance(node, ast.ImportFrom):
                    if node.module:
                        imports.add(node.module.split('.')[0])
            
            return imports
        except Exception as e:
            logger.error(f"Failed to parse imports in {file_path}: {str(e)}")
            self.validation_errors.append(f"PARSE_ERROR in {file_path}: {str(e)}")
            return set()

    def validate_dependency_resolution(self, module_name: str) -> bool:
        """
        Checks if a specific module can be resolved in the current environment.
        
        Args:
            module_name (str): Name of the module to check.
            
        Returns:
            bool: True if resolvable, False otherwise.
        """
        if module_name in sys.builtin_module_names:
            return True
            
        try:
            spec = importlib.util.find_spec(module_name)
            return spec is not None
        except (ImportError, ValueError):
            return False

    async def run_full_validation(self) -> Dict[str, Any]:
        """
        Executes the full dependency validation suite.
        
        Returns:
            Dict[str, Any]: Comprehensive validation report.

        logger.info("Starting LOGOS Python Dependency Validation Suite...")
        
        # 1. Database Integrity Check

        db_violations = []

            db_violations = await self.verify_translator_table_integrity()
        
        # 2. File Scanning and Import Extraction
        py_files = self.scan_for_python_files()
        all_imports = set()
        for py_file in py_files:
            file_imports = self.extract_imports_from_file(py_file)
            all_imports.update(file_imports)
        
        # 3. Resolution Check
        unresolved = []
        for imp in all_imports:
            if not self.validate_dependency_resolution(imp):
                # Filter out local project modules if they aren't installed
                if not (self.project_root / imp).exists():
                    unresolved.append(imp)
        
        # 4. Forbidden String Check (Code level)
        code_violations = []
        for py_file in py_files:
            violations = self.check_file_for_forbidden_strings(py_file)
            code_violations.extend(violations)

        await self.close_database()
        
        report = {
            "status": "PASS" if not (unresolved or db_violations or code_violations) else "FAIL",
            "files_scanned": len(py_files),
            "total_unique_imports": len(all_imports),
            "unresolved_dependencies": unresolved,
            "database_violations": db_violations,
            "code_violations": code_violations,
            "errors": self.validation_errors
        }
        
        self.log_report(report)
        return report

    def check_file_for_forbidden_strings(self, file_path: Path) -> List[str]:

            List[str]: List of violation descriptions.

                    # Case-insensitive check for maximum safety
                    if forbidden.lower() in content.lower():
                        # Exclude the validator itself from reporting its own constant list
                        if "FORBIDDEN_TRANSLATORS" not in content:
                            msg = f"FORBIDDEN_ENTITY_FOUND: '{forbidden}' in {file_path}"
                            logger.error(msg)
                            violations.append(msg)
        except Exception as e:
            logger.error(f"Could not read file {file_path} for string check: {e}")

    def log_report(self, report: Dict[str, Any]) -> None:

/**
 * LOGOS - The Bible for Classical Studies
 * NPM Dependency Validator - TypeScript Implementation
 * 
 * This module ensures the integrity of the frontend environment,
 * validating package.json, lockfiles, and node_modules consistency.
 */

import * as fs from 'fs';
import * as path from 'path';
import * as crypto from 'crypto';

interface PackageJson {
    name: string;
    version: string;
    dependencies?: Record<string, string>;
    devDependencies?: Record<string, string>;
    peerDependencies?: Record<string, string>;
}

interface ValidationResult {
    isValid: boolean;
    errors: string[];
    warnings: string[];
    metadata: {
        dependencyCount: number;
        checkedAt: string;
    };
}

export class NpmDependencyValidator {
    private projectRoot: string;
    private errors: string[] = [];
    private warnings: string[] = [];

    constructor(projectRoot: string) {
        this.projectRoot = projectRoot;
        console.log(`[LOGOS] Initializing NPM Validator at: ${projectRoot}`);
    }

    /**
     * Main entry point for the validation process.
     */
    public async validate(): Promise<ValidationResult> {
        try {
            const packageJsonPath = path.join(this.projectRoot, 'package.json');
            const lockFilePath = path.join(this.projectRoot, 'package-lock.json');

            if (!fs.existsSync(packageJsonPath)) {
                throw new Error("package.json not found in project root.");
            }

            const packageJson: PackageJson = JSON.parse(fs.readFileSync(packageJsonPath, 'utf8'));
            
            this.validatePackageStructure(packageJson);
            this.checkNodeModules(packageJson);
            this.verifyLockfileConsistency(packageJsonPath, lockFilePath);
            this.scanForForbiddenStrings();

            const allDeps = {
                ...(packageJson.dependencies || {}),
                ...(packageJson.devDependencies || {})
            };

            return {
                isValid: this.errors.length === 0,
                errors: this.errors,
                warnings: this.warnings,
                metadata: {
                    dependencyCount: Object.keys(allDeps).length,
                    checkedAt: new Date().toISOString()
                }
            };

        } catch (error: any) {
            this.errors.push(`FATAL_VALIDATION_ERROR: ${error.message}`);
            return {
                isValid: false,
                errors: this.errors,
                warnings: this.warnings,
                metadata: { dependencyCount: 0, checkedAt: new Date().toISOString() }
            };
        }
    }

    /**
     * Validates the basic structure and required fields of package.json.
     */
    private validatePackageStructure(pkg: PackageJson): void {
        if (!pkg.name || pkg.name !== 'logos-classical-studies') {
            this.warnings.push(`Unexpected project name: ${pkg.name}`);
        }
        if (!pkg.version) {
            this.errors.push("Missing version field in package.json");
        }
    }

    /**
     * Checks if all declared dependencies are present in node_modules.
     */
    private checkNodeModules(pkg: PackageJson): void {
        const nodeModulesPath = path.join(this.projectRoot, 'node_modules');
        
        if (!fs.existsSync(nodeModulesPath)) {
            this.errors.push("node_modules directory is missing. Run 'npm install'.");
            return;
        }

        const checkDeps = (deps?: Record<string, string>) => {
            if (!deps) return;
            for (const dep of Object.keys(deps)) {
                const depPath = path.join(nodeModulesPath, dep);
                if (!fs.existsSync(depPath)) {
                    this.errors.push(`Missing dependency in node_modules: ${dep}`);
                }
            }
        };

        checkDeps(pkg.dependencies);
        checkDeps(pkg.devDependencies);
    }

    /**
     * Ensures that the package-lock.json is in sync with package.json.
     */
    private verifyLockfileConsistency(pkgPath: string, lockPath: string): void {
        if (!fs.existsSync(lockPath)) {
            this.warnings.push("package-lock.json is missing. Version pinning cannot be verified.");
            return;
        }

        try {
            const lockData = JSON.parse(fs.readFileSync(lockPath, 'utf8'));
            // Logic to verify checksums or version ranges would go here in a full implementation
            if (lockData.lockfileVersion < 2) {
                this.warnings.push("Old lockfile version detected. Consider upgrading to npm v7+.");
            }
        } catch (e) {
            this.errors.push(`Failed to parse lockfile: ${e}`);
        }
    }

    /**
     * Scans source files for forbidden translator names (Chapman, etc).
     */
    private scanForForbiddenStrings(): void {
        const forbidden = ["Chapman", "Lattimore", "Fagles", "Wilson"];
        const srcPath = path.join(this.projectRoot, 'src');

        if (!fs.existsSync(srcPath)) return;

        const walkSync = (dir: string) => {
            const files = fs.readdirSync(dir);
            files.forEach(file => {
                const filePath = path.join(dir, file);
                const stat = fs.statSync(filePath);
                if (stat.isDirectory()) {
                    walkSync(filePath);
                } else if (file.endsWith('.ts') || file.endsWith('.tsx')) {
                    const content = fs.readFileSync(filePath, 'utf8');
                    forbidden.forEach(name => {
                        if (content.includes(name) && !filePath.includes('npm_deps_validator.ts')) {
                            this.errors.push(`FORBIDDEN_TRANSLATOR_FOUND: '${name}' in ${filePath}`);
                        }
                    });
                }
            });
        };

        walkSync(srcPath);
    }

    /**
     * Checks for circular dependencies in the package structure.
     */
    public detectCircularDependencies(): string[] {
        // Simplified circular detection logic for production monitoring
        const circles: string[] = [];
        console.log("[LOGOS] Running circular dependency check...");
        // In a real production system, we would integrate 'madge' or similar here.
        return circles;
    }

    /**
     * Validates that no insecure protocols are used in dependency URLs.
     */
    private auditDependencyUrls(pkg: PackageJson): void {
        const check = (deps?: Record<string, string>) => {
            if (!deps) return;
            for (const [name, version] of Object.entries(deps)) {
                if (version.startsWith('http://')) {
                    this.errors.push(`SECURITY_RISK: Insecure HTTP URL for dependency '${name}'`);
                }
            }
        };
        check(pkg.dependencies);
        check(pkg.devDependencies);
    }
}

import logging

import platform
from typing import Dict, Any, List, Optional

logger = logging.getLogger(__name__)

class VersionChecker:
    """
    Ensures that the environment meets the minimum version requirements
    for Python, PostgreSQL, and critical libraries.
    """

    MIN_PYTHON_VERSION = "3.9.0"
    MIN_POSTGRES_VERSION = "13.0"
    DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"

    REQUIRED_LIBS = {
        "asyncpg": "0.25.0",
        "pydantic": "2.0.0",
        "numpy": "1.20.0",
        "pandas": "1.3.0"
    }

    def __init__(self):
        self.results: Dict[str, Any] = {}
        self.is_compatible = True

    async def check_all(self) -> bool:
        """Runs all version compatibility checks."""
        logger.info("Starting comprehensive version compatibility check...")
        
        self.check_python_version()
        self.check_library_versions()
        await self.check_postgres_version()
        
        self.log_results()
        return self.is_compatible

    def check_python_version(self) -> None:
        """Validates the current Python interpreter version."""
        current = platform.python_version()
        logger.info(f"Checking Python version: {current}")
        
        if version.parse(current) < version.parse(self.MIN_PYTHON_VERSION):
            msg = f"Incompatible Python version: {current}. Required: >= {self.MIN_PYTHON_VERSION}"
            logger.error(msg)
            self.results["python"] = {"status": "FAIL", "current": current, "required": self.MIN_PYTHON_VERSION}
            self.is_compatible = False
        else:
            self.results["python"] = {"status": "PASS", "current": current}

    def check_library_versions(self) -> None:
        """Checks installed library versions against requirements."""
        lib_results = {}
        for lib, min_ver in self.REQUIRED_LIBS.items():
            try:
                module = __import__(lib)
                current = getattr(module, "__version__", "unknown")
                if current == "unknown":
                    import importlib.metadata
                    current = importlib.metadata.version(lib)

                if version.parse(current) < version.parse(min_ver):
                    logger.error(f"Library {lib} version mismatch: {current} < {min_ver}")
                    lib_results[lib] = {"status": "FAIL", "current": current, "required": min_ver}
                    self.is_compatible = False
                else:
                    lib_results[lib] = {"status": "PASS", "current": current}
            except ImportError:
                logger.error(f"Required library {lib} is not installed.")
                lib_results[lib] = {"status": "MISSING"}
                self.is_compatible = False
            except Exception as e:
                logger.error(f"Error checking {lib}: {e}")
                lib_results[lib] = {"status": "ERROR", "message": str(e)}
                self.is_compatible = False
        
        self.results["libraries"] = lib_results

    async def check_postgres_version(self) -> None:
        """Connects to the DB and verifies the PostgreSQL version."""
        try:
            conn = await asyncpg.connect(self.DATABASE_URL)
            server_version = conn.get_server_version()
            # server_version is a named tuple: ServerVersion(major=15, minor=3, patch=0)
            ver_str = f"{server_version.major}.{server_version.minor}"
            
            logger.info(f"Database version detected: PostgreSQL {ver_str}")
            
            if server_version.major < 13:
                logger.error(f"PostgreSQL version {ver_str} is below minimum {self.MIN_POSTGRES_VERSION}")
                self.results["database"] = {"status": "FAIL", "current": ver_str, "required": self.MIN_POSTGRES_VERSION}
                self.is_compatible = False
            else:
                self.results["database"] = {"status": "PASS", "current": ver_str}
            
            await conn.close()
        except Exception as e:
            logger.error(f"Failed to check database version: {e}")
            self.results["database"] = {"status": "ERROR", "message": str(e)}
            self.is_compatible = False

    def log_results(self) -> None:
        """Prints a summary of the version check."""
        logger.info("-" * 40)
        logger.info("VERSION COMPATIBILITY SUMMARY")
        logger.info(f"Overall Status: {'SUCCESS' if self.is_compatible else 'FAILURE'}")
        for key, val in self.results.items():
            logger.info(f"{key.upper()}: {val}")
        logger.info("-" * 40)

                    "severity": "MEDIUM"
                })
                logger.warning("Database connection is using superuser privileges.")
            await conn.close()
        except Exception as e:
            logger.error(f"DB permission audit failed: {e}")

    async def verify_data_integrity(self) -> None:
        """Performs checksum validation on critical reference tables."""
        logger.info("Verifying translator table integrity...")
        try:
            conn = await asyncpg.connect(self.DATABASE_URL)
            # Ensure the approved list matches the DB
            rows = await conn.fetch("SELECT name FROM translator_profiles")
            db_names = {row['name'] for row in rows}
            
            for forbidden in self.FORBIDDEN_TRANSLATORS:
                if forbidden in db_names:
                    self.findings.append({
                        "type": "DATABASE_INTEGRITY",
                        "message": f"Forbidden translator '{forbidden}' exists in production database!",
                        "severity": "BLOCKER"
                    })
                    logger.critical(f"DATABASE CORRUPTION: Forbidden entity '{forbidden}' detected in production!")
            
            await conn.close()
        except Exception as e:
            logger.error(f"Data integrity check failed: {e}")

    def generate_security_report(self) -> str:
        """Generates a formatted string report of findings."""
        if not self.findings:
            return "SECURITY STATUS: CLEAN. No vulnerabilities detected."
        
        report = ["SECURITY AUDIT REPORT", "=" * 30]
        for issue in self.findings:
            report.append(f"SEVERITY: {issue['severity']}")
            report.append(f"TYPE: {issue['type']}")
            report.append(f"INFO: {issue.get('message', issue.get('entity', 'N/A'))}")
            if 'file' in issue:
                report.append(f"FILE: {issue['file']}")
            report.append("-" * 20)
        
        return "\n".join(report)
