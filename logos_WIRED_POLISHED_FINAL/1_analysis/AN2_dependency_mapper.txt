@dataclass
class DependencyNode:
    """Represents a single node (file or module) in the dependency graph."""
    id: str
    name: str
    path: str
    node_type: str  # 'python', 'typescript', 'database_table', 'api_endpoint'
    metadata: Dict[str, Any] = field(default_factory=dict)
    imports: List[str] = field(default_factory=list)
    exported_symbols: List[str] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        """Converts the node to a serializable dictionary for JSON reporting."""
        return asdict(self)

class DependencyGraph:
    """
    Manages the directed graph of dependencies.
    Provides methods for cycle detection, pathfinding, and report generation.
    """

    def __init__(self):
        self.nodes: Dict[str, DependencyNode] = {}
        self.edges: List[Tuple[str, str]] = []  # List of (source_id, target_id)
        self.adjacency_list: Dict[str, Set[str]] = {}
        logger.info("Initialized new DependencyGraph instance.")

    def add_node(self, node: DependencyNode) -> None:
        """Adds a node to the graph with validation."""
        if not node or not node.id:
            logger.error("Attempted to add an invalid node to the graph.")
            return
        
        if node.id in self.nodes:
            logger.warning(f"Node with ID {node.id} already exists. Updating metadata.")
            self.nodes[node.id].metadata.update(node.metadata)
        else:
            self.nodes[node.id] = node
            self.adjacency_list[node.id] = set()
            logger.debug(f"Added node: {node.id} ({node.node_type})")

    def add_edge(self, source_id: str, target_id: str) -> None:
        """Creates a directed edge between two nodes."""
        if source_id not in self.nodes:
            logger.error(f"Source node {source_id} does not exist. Cannot add edge.")
            return
        
        # We allow edges to targets that aren't nodes yet (external dependencies)
        if target_id not in self.nodes:
            logger.debug(f"Target node {target_id} not in graph. Adding placeholder.")
            placeholder = DependencyNode(
                id=target_id, 
                name=target_id.split('.')[-1], 
                path="external", 
                node_type="external"
            )
            self.add_node(placeholder)

        if target_id not in self.adjacency_list[source_id]:
            self.edges.append((source_id, target_id))
            self.adjacency_list[source_id].add(target_id)
            logger.debug(f"Added edge: {source_id} -> {target_id}")

    def detect_cycles(self) -> List[List[str]]:
        """
        Detects circular dependencies using Depth First Search.
        Returns a list of cycles found.
        """
        cycles = []
        visited = set()
        stack = []
        path = []

        def visit(node_id: str):
            if node_id in path:
                cycle_start_idx = path.index(node_id)
                cycles.append(path[cycle_start_idx:] + [node_id])
                return
            
            if node_id in visited:
                return

            visited.add(node_id)
            path.append(node_id)
            
            for neighbor in self.adjacency_list.get(node_id, []):
                visit(neighbor)
            
            path.pop()

        for node_id in self.nodes:
            if node_id not in visited:
                visit(node_id)
        
        if cycles:
            logger.warning(f"Detected {len(cycles)} circular dependencies in the graph.")
        return cycles

    def find_orphans(self) -> List[str]:
        """Identifies nodes that have no incoming or outgoing edges."""
        has_incoming = set()
        for _, target in self.edges:
            has_incoming.add(target)
            
        orphans = []
        for node_id in self.nodes:
            if node_id not in has_incoming and not self.adjacency_list.get(node_id):
                orphans.append(node_id)
        
        logger.info(f"Found {len(orphans)} orphan nodes.")
        return orphans

    def get_node_depth(self, node_id: str) -> int:
        """Calculates the maximum depth of a node from the roots."""
        if node_id not in self.nodes:
            return -1
        
        # Simple BFS to find depth
        depths = {node_id: 0}
        queue = [node_id]
        max_depth = 0
        
        while queue:
            current = queue.pop(0)
            current_depth = depths[current]
            max_depth = max(max_depth, current_depth)
            
            for neighbor in self.adjacency_list.get(current, []):
                if neighbor not in depths:
                    depths[neighbor] = current_depth + 1
                    queue.append(neighbor)
        
        return max_depth

    def get_impact_set(self, node_id: str) -> Set[str]:
        """Finds all nodes that depend on the given node (upstream impact)."""
        impacted = set()
        
        # Reverse edges logic
        reverse_adj = {n: set() for n in self.nodes}
        for src, tgt in self.edges:
            if tgt in reverse_adj:
                reverse_adj[tgt].add(src)
        
        stack = [node_id]
        while stack:
            curr = stack.pop()
            for parent in reverse_adj.get(curr, []):
                if parent not in impacted:
                    impacted.add(parent)
                    stack.append(parent)
                    
        return impacted

    def validate_translator_usage(self) -> List[Dict[str, Any]]:
        """Checks nodes for usage of forbidden translators."""
        violations = []
        for node_id, node in self.nodes.items():
            content = node.metadata.get('raw_content', '')
            for forbidden in FORBIDDEN_TRANSLATORS:
                if forbidden in content:
                    violations.append({
                        "node": node_id,
                        "translator": forbidden,
                        "path": node.path
                    })
                    logger.critical(f"COPYRIGHT VIOLATION: {forbidden} found in {node.path}")
        return violations

    async def sync_with_database(self):
        """Cross-references identified tables with the actual PostgreSQL schema."""
        try:
            conn = await asyncpg.connect(DATABASE_URL)
            logger.info("Connected to Railway PostgreSQL for schema validation.")
            
            rows = await conn.fetch("""
                SELECT table_name 
                FROM information_schema.tables 
                WHERE table_schema = 'public'
            """)
            
            existing_tables = {row['table_name'] for row in rows}
            
            for node_id, node in self.nodes.items():
                if node.node_type == 'database_table':
                    if node.name not in existing_tables:
                        logger.error(f"Table {node.name} identified in code but missing in DB!")
                        node.metadata['db_status'] = 'missing'
                    else:
                        node.metadata['db_status'] = 'verified'
            
            await conn.close()
        except Exception as e:
            logger.error(f"Failed to sync with database: {str(e)}")

    def generate_json_report(self, output_path: str):
        """Exports the entire graph state to a JSON file for CI/CD analysis."""
        report = {
            "timestamp": datetime.datetime.now().isoformat(),
            "summary": {
                "total_nodes": len(self.nodes),
                "total_edges": len(self.edges),
                "cycles_found": len(self.detect_cycles())
            },
            "nodes": {nid: n.to_dict() for nid, n in self.nodes.items()},
            "edges": self.edges
        }
        
        try:
            with open(output_path, 'w') as f:
                json.dump(report, f, indent=2)
            logger.info(f"Dependency report generated at {output_path}")
        except IOError as e:
            logger.error(f"Failed to write report: {str(e)}")

    def get_subgraph(self, node_ids: List[str]) -> 'DependencyGraph':
        """Creates a new graph containing only the specified nodes and their connections."""
        subgraph = DependencyGraph()
        for nid in node_ids:
            if nid in self.nodes:
                subgraph.add_node(self.nodes[nid])
        
        for src, tgt in self.edges:
            if src in subgraph.nodes and tgt in subgraph.nodes:
                subgraph.add_edge(src, tgt)
        
        return subgraph

    def filter_by_type(self, node_type: str) -> List[str]:
        """Returns IDs of nodes matching a specific type."""
        return [nid for nid, node in self.nodes.items() if node.node_type == node_type]

    def get_statistics(self) -> Dict[str, Any]:
        """Calculates graph metrics for architectural health assessment."""
        if not self.nodes:
            return {"error": "Graph is empty"}
            
        in_degrees = {nid: 0 for nid in self.nodes}
        out_degrees = {nid: 0 for nid in self.nodes}
        
        for src, tgt in self.edges:
            out_degrees[src] += 1
            if tgt in in_degrees:
                in_degrees[tgt] += 1
                
        return {
            "max_out_degree": max(out_degrees.values()) if out_degrees else 0,
            "max_in_degree": max(in_degrees.values()) if in_degrees else 0,
            "avg_out_degree": sum(out_degrees.values()) / len(self.nodes),
            "isolated_nodes": len([n for n in in_degrees if in_degrees[n] == 0 and out_degrees[n] == 0])
        }

    def clear(self):
        """Resets the graph state."""
        self.nodes.clear()
        self.edges.clear()
        self.adjacency_list.clear()
        logger.info("DependencyGraph state cleared.")

class PythonImportAnalyzer:
    """
    Analyzes Python source files to extract import dependencies using AST.
    """

    def __init__(self, root_dir: str):
        self.root_dir = root_dir
        self.graph = DependencyGraph()
        logger.info(f"PythonImportAnalyzer initialized for root: {root_dir}")

    def analyze_file(self, file_path: str) -> Optional[DependencyNode]:
        """Parses a single Python file and extracts its imports."""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            tree = ast.parse(content)
            module_path = os.path.relpath(file_path, self.root_dir).replace(os.sep, '.').replace('.py', '')
            
            node = DependencyNode(
                id=module_path,
                name=os.path.basename(file_path),
                path=file_path,
                node_type='python',
                metadata={'raw_content': content}
            )

            for item in tree.body:
                if isinstance(item, ast.Import):
                    for alias in item.names:
                        node.imports.append(alias.name)
                elif isinstance(item, ast.ImportFrom):
                    module = item.module or ''
                    if item.level > 0:
                        # Handle relative imports
                        node.imports.append(f"{'.' * item.level}{module}")
                    else:
                        node.imports.append(module)
                elif isinstance(item, (ast.FunctionDef, ast.ClassDef)):
                    node.exported_symbols.append(item.name)

            return node
        except Exception as e:
            logger.error(f"Failed to analyze Python file {file_path}: {str(e)}")
            return None

    def scan_directory(self):
        """Walks the root directory and processes all Python files."""
        logger.info(f"Scanning directory: {self.root_dir}")
        for root, _, files in os.walk(self.root_dir):
            for file in files:
                if file.endswith('.py'):
                    full_path = os.path.join(root, file)
                    node = self.analyze_file(full_path)
                    if node:
                        self.graph.add_node(node)
                        for imp in node.imports:
                            self.graph.add_edge(node.id, imp)

class TypeScriptImportAnalyzer:
    """
    Analyzes TypeScript/React source files to extract import dependencies using regex.
    """

    def __init__(self, root_dir: str):
        self.root_dir = root_dir
        self.graph = DependencyGraph()
        # Regex to match: import { x } from './y' or import x from 'y'
        self.import_pattern = re.compile(r"import\s+(?:[\w\s{},*]+from\s+)?['\"]([^'\"]+)['\"]")
        logger.info(f"TypeScriptImportAnalyzer initialized for root: {root_dir}")

    def analyze_file(self, file_path: str) -> Optional[DependencyNode]:
        """Parses a single TS/TSX file and extracts its imports."""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            relative_path = os.path.relpath(file_path, self.root_dir)
            node_id = relative_path.replace(os.sep, '/')
            
            node = DependencyNode(
                id=node_id,
                name=os.path.basename(file_path),
                path=file_path,
                node_type='typescript',
                metadata={'raw_content': content}
            )

            matches = self.import_pattern.findall(content)
            for match in matches:
                node.imports.append(match)
            
            # Look for component definitions
            comp_matches = re.findall(r"const\s+(\w+)\s*:\s*React\.FC", content)
            node.exported_symbols.extend(comp_matches)

            return node
        except Exception as e:
            logger.error(f"Failed to analyze TypeScript file {file_path}: {str(e)}")
            return None

    def scan_directory(self):
        """Walks the root directory and processes all TS/TSX files."""
        for root, _, files in os.walk(self.root_dir):
            for file in files:
                if file.endswith(('.ts', '.tsx')):
                    full_path = os.path.join(root, file)
                    node = self.analyze_file(full_path)
                    if node:
                        self.graph.add_node(node)
                        for imp in node.imports:
                            self.graph.add_edge(node.id, imp)

async def generate_dependency_report():
    """
    Orchestrates the full analysis of the LOGOS V23 build.
    """
    logger.info("Starting LOGOS V23 Dependency Analysis...")
    
    # Initialize analyzers
    py_analyzer = PythonImportAnalyzer(os.getcwd())
    ts_analyzer = TypeScriptImportAnalyzer(os.path.join(os.getcwd(), "frontend"))
    
    # Perform scans
    py_analyzer.scan_directory()
    ts_analyzer.scan_directory()
    
    # Merge graphs into a master graph
    master_graph = DependencyGraph()
    for node in py_analyzer.graph.nodes.values():
        master_graph.add_node(node)
    for src, tgt in py_analyzer.graph.edges:
        master_graph.add_edge(src, tgt)
        
    for node in ts_analyzer.graph.nodes.values():
        master_graph.add_node(node)
    for src, tgt in ts_analyzer.graph.edges:
        master_graph.add_edge(src, tgt)
        
    # Check for forbidden translators
    violations = master_graph.validate_translator_usage()
    if violations:
        logger.critical(f"Build failed due to copyright violations: {violations}")
    
    # Database sync
    await master_graph.sync_with_database()
    
    # Final report
    master_graph.generate_json_report("dependency_report_v23.json")
    
    logger.info("Analysis complete. Report generated.")

if __name__ == "__main__":
    asyncio.run(generate_dependency_report())

class ImportResolver:
    """
    Responsible for validating and fixing imports across Python and TypeScript.
    """

    def __init__(self, base_path: str):
        self.base_path = pathlib.Path(base_path).resolve()
        self.python_paths = [str(self.base_path)] + sys.path
        self.missing_modules: Set[str] = set()
        logger.info(f"ImportResolver initialized at {self.base_path}")

    def resolve_python_import(self, module_name: str, current_file: str) -> Tuple[bool, Optional[str]]:
        """
        Attempts to locate a Python module on the filesystem.
        Handles relative imports and standard library modules.
        """
        try:
            # Handle relative imports
            if module_name.startswith('.'):
                current_dir = pathlib.Path(current_file).parent
                # Count leading dots
                dot_count = 0
                for char in module_name:
                    if char == '.':
                        dot_count += 1
                    else:
                        break
                
                # Navigate up the directory tree
                target_dir = current_dir
                for _ in range(dot_count - 1):
                    target_dir = target_dir.parent
                
                # Resolve the remaining path
                sub_path = module_name[dot_count:].replace('.', os.sep)
                potential_file = target_dir / f"{sub_path}.py"
                potential_dir = target_dir / sub_path / "__init__.py"
                
                if potential_file.exists():
                    return True, str(potential_file)
                if potential_dir.exists():
                    return True, str(potential_dir)
                
                return False, None

            # Handle absolute imports
            spec = None
            try:
                spec = importlib.util.find_spec(module_name)
            except (ModuleNotFoundError, ValueError):
                pass

            if spec and spec.origin:
                return True, spec.origin
            
            # Manual check in base_path if spec fails
            parts = module_name.split('.')
            manual_path = self.base_path.joinpath(*parts)
            if manual_path.with_suffix('.py').exists():
                return True, str(manual_path.with_suffix('.py'))
            if manual_path.joinpath('__init__.py').exists():
                return True, str(manual_path.joinpath('__init__.py'))

            self.missing_modules.add(module_name)
            return False, None

        except Exception as e:
            logger.error(f"Error resolving Python import {module_name}: {str(e)}")
            return False, None

    def resolve_typescript_import(self, import_path: str, current_file: str) -> Tuple[bool, Optional[str]]:
        """
        Resolves TypeScript imports, accounting for tsconfig paths and extensions.
        """
        try:
            current_dir = pathlib.Path(current_file).parent
            
            # Clean up path (remove aliases if any, though we assume standard relative paths here)
            clean_path = import_path
            
            # Check for relative imports
            if clean_path.startswith('.'):
                # Possible extensions in priority order
                extensions = ['.tsx', '.ts', '.d.ts', '/index.tsx', '/index.ts']
                
                for ext in extensions:
                    potential_path = (current_dir / clean_path).with_suffix('')
                    # Manually append because with_suffix might not handle /index correctly
                    full_potential = pathlib.Path(str(potential_path) + ext)
                    
                    if full_potential.exists():
                        return True, str(full_potential)
            
            # External or absolute paths
            return False, None

        except Exception as e:
            logger.error(f"Error resolving TypeScript import {import_path}: {str(e)}")
            return False, None

    def find_missing_imports(self, project_data: Dict[str, Any]) -> List[Dict[str, str]]:
        """
        Scans project metadata to identify all broken links.
        """
        broken_links = []
        nodes = project_data.get('nodes', {})
        
        for node_id, node in nodes.items():
            current_file = node.get('path')
            node_type = node.get('node_type')
            
            for imp in node.get('imports', []):
                resolved = False
                if node_type == 'python':
                    resolved, _ = self.resolve_python_import(imp, current_file)
                elif node_type == 'typescript':
                    resolved, _ = self.resolve_typescript_import(imp, current_file)
                
                if not resolved and imp and not imp.startswith(('http', 'https')):
                    broken_links.append({
                        "file": current_file,
                        "import": imp,
                        "type": node_type
                    })
        
        logger.warning(f"Found {len(broken_links)} broken import links.")
        return broken_links

    def suggest_import_fixes(self, broken_import: str, node_type: str) -> List[str]:
        """
        Uses fuzzy matching to suggest corrections for broken imports.
        """
        suggestions = []
        all_known_files = []
        
        # Collect all files in the project to match against
        for root, _, files in os.walk(self.base_path):
            for f in files:
                if node_type == 'python' and f.endswith('.py'):
                    rel = os.path.relpath(os.path.join(root, f), self.base_path)
                    all_known_files.append(rel.replace(os.sep, '.').replace('.py', ''))
                elif node_type == 'typescript' and f.endswith(('.ts', '.tsx')):
                    rel = os.path.relpath(os.path.join(root, f), self.base_path)
                    all_known_files.append(rel.replace(os.sep, '/'))

        # Get top 3 matches
        suggestions = difflib.get_close_matches(broken_import, all_known_files, n=3, cutoff=0.5)
        return suggestions

    async def validate_database_references(self, sql_queries: List[str]) -> List[str]:
        """
        Connects to the Railway DB and ensures all table names in strings exist.
        """
        invalid_tables = []
        try:
            conn = await asyncpg.connect(DATABASE_URL)
            # Query for actual tables
            rows = await conn.fetch("SELECT table_name FROM information_schema.tables WHERE table_schema = 'public'")
            valid_tables = {row['table_name'] for row in rows}
            
            # Simple regex to find table names in SELECT/FROM/JOIN
            table_regex = re.compile(r"(?:FROM|JOIN|UPDATE|INTO)\s+([a-zA-Z0-9_]+)", re.IGNORECASE)
            
            for query in sql_queries:
                matches = table_regex.findall(query)
                for table in matches:
                    if table.lower() not in valid_tables and table.lower() not in ['dual', 'values']:
                        invalid_tables.append(table)
            
            await conn.close()
        except Exception as e:
            logger.error(f"Database validation failed: {str(e)}")
            
        return list(set(invalid_tables))

    def check_circular_dependencies(self, import_map: Dict[str, List[str]]) -> List[List[str]]:
        """
        Deep check for cycles in the import map.
        """
        cycles = []
        visited = set()
        path = []

        def dfs(node):
            if node in path:
                idx = path.index(node)
                cycles.append(path[idx:] + [node])
                return
            
            if node in visited:
                return
            
            visited.add(node)
            path.append(node)
            
            for neighbor in import_map.get(node, []):
                dfs(neighbor)
            
            path.pop()

        for node in import_map:
            dfs(node)
            
        return cycles

    def generate_fix_script(self, broken_links: List[Dict[str, str]]) -> str:
        """
        Generates a shell script or log of actions to fix the identified issues.
        """
        script = ["# LOGOS Import Fix Suggestions", f"# Generated: {pathlib.Path().cwd()}"]
        
        for link in broken_links:
            suggestions = self.suggest_import_fixes(link['import'], link['type'])
            if suggestions:
                script.append(f"# File: {link['file']}")
                script.append(f"# Broken: {link['import']}")
                script.append(f"# Suggestions: {', '.join(suggestions)}")
                script.append("")
            else:
                script.append(f"# CRITICAL: No suggestions for {link['import']} in {link['file']}")
                
        return "\n".join(script)

    def verify_no_forbidden_content(self, file_path: str) -> bool:

                    if forbidden in content:

                        return False
            return True
        except Exception as e:
            logger.error(f"Could not verify {file_path}: {str(e)}")
            return False

    def get_module_summary(self) -> Dict[str, Any]:
        """Provides a statistical overview of the codebase imports."""
        py_count = 0
        ts_count = 0
        for root, _, files in os.walk(self.base_path):
            for f in files:
                if f.endswith('.py'): py_count += 1
                if f.endswith(('.ts', '.tsx')): ts_count += 1
                
        return {
            "python_files": py_count,

/**
 * LOGOS - The Bible for Classical Studies
 * Module: Component Tree Analyzer
 * Description: Maps the React component hierarchy and identifies orphan components.
 * Ensures the frontend architecture remains modular and clean.
 */

import * as fs from 'fs';
import * as path from 'path';

interface ComponentNode {
  id: string;
  name: string;
  path: string;
  children: string[];
  parents: string[];
  props: string[];
  isOrphan: boolean;
  depth: number;
}

interface ComponentReport {
  totalComponents: number;
  maxDepth: number;
  orphans: string[];
  tree: Record<string, ComponentNode>;
}

class ComponentTreeAnalyzer {
  private nodes: Record<string, ComponentNode> = {};
  private rootPath: string;

  constructor(rootPath: string) {
    this.rootPath = path.resolve(rootPath);
    console.log(`[ComponentTreeAnalyzer] Initialized for: ${this.rootPath}`);
  }

  /**
   * Reads a file and extracts component dependencies based on JSX usage.
   */
  private analyzeFile(filePath: string): ComponentNode {
    const content = fs.readFileSync(filePath, 'utf-8');
    const fileName = path.basename(filePath);
    const componentName = fileName.replace(/\.(tsx|ts)$/, '');
    
    const node: ComponentNode = {
      id: filePath,
      name: componentName,
      path: filePath,
      children: [],
      parents: [],
      props: [],
      isOrphan: true,
      depth: 0
    };

    // Regex to find JSX tags like <ComponentName ... /> or <ComponentName>
    const jsxRegex = /<([A-Z][a-zA-Z0-9]*)/g;
    let match;
    while ((match = jsxRegex.exec(content)) !== null) {
      const childName = match[1];
      if (childName !== componentName && !node.children.includes(childName)) {
        node.children.push(childName);
      }
    }

    // Extract props from interface or type definitions
    const propRegex = /interface\s+\w*Props\s*{([^}]+)}/g;
    const propMatch = propRegex.exec(content);
    if (propMatch) {
      const propLines = propMatch[1].split('\n');
      propLines.forEach(line => {
        const trimmed = line.trim();
        if (trimmed && trimmed.includes(':')) {
          node.props.push(trimmed.split(':')[0].trim());
        }
      });
    }

    return node;
  }

  /**
   * Recursively builds the component tree starting from a given directory.
   */
  public buildComponentTree(): void {
    const scanDir = (dir: string) => {
      const entries = fs.readdirSync(dir, { withFileTypes: true });
      
      for (const entry of entries) {
        const fullPath = path.join(dir, entry.name);
        if (entry.isDirectory() && entry.name !== 'node_modules') {
          scanDir(fullPath);
        } else if (entry.isFile() && /\.(tsx|ts)$/.test(entry.name)) {
          const node = this.analyzeFile(fullPath);
          this.nodes[node.name] = node;
        }
      }
    };

    try {
      scanDir(this.rootPath);
      this.resolveRelationships();
      this.calculateDepths();
      console.log(`[ComponentTreeAnalyzer] Successfully mapped ${Object.keys(this.nodes).length} components.`);
    } catch (error) {
      console.error(`[ComponentTreeAnalyzer] Error building tree: ${error}`);
    }
  }

  /**
   * Links parents to children and updates orphan status.
   */
  private resolveRelationships(): void {
    Object.values(this.nodes).forEach(node => {
      node.children.forEach(childName => {
        if (this.nodes[childName]) {
          this.nodes[childName].parents.push(node.name);
          this.nodes[childName].isOrphan = false;
        }
      });
    });
  }

  /**
   * Calculates the depth of each component in the hierarchy.
   */
  private calculateDepths(): void {
    const visited = new Set<string>();

    const getDepth = (name: string): number => {
      const node = this.nodes[name];
      if (!node || node.parents.length === 0) return 0;
      if (visited.has(name)) return 0; // Prevent infinite loops

      visited.add(name);
      const parentDepths = node.parents.map(p => getDepth(p));
      visited.delete(name);

      return Math.max(...parentDepths) + 1;
    };

    Object.keys(this.nodes).forEach(name => {
      this.nodes[name].depth = getDepth(name);
    });
  }

  /**
   * Identifies components that are not imported by any other component.
   */
  public findOrphanComponents(): string[] {
    return Object.values(this.nodes)
      .filter(node => node.isOrphan && node.name !== 'App' && node.name !== 'index')
      .map(node => node.name);
  }

  /**
   * Generates a comprehensive report of the frontend architecture.
   */
  public generateComponentReport(): ComponentReport {
    const orphans = this.findOrphanComponents();
    let maxDepth = 0;
    
    Object.values(this.nodes).forEach(n => {
      if (n.depth > maxDepth) maxDepth = n.depth;
    });

    return {
      totalComponents: Object.keys(this.nodes).length,
      maxDepth,
      orphans,
      tree: this.nodes
    };
  }

  /**
   * Validates that no forbidden strings exist in the components.
   */
  public securityAudit(): boolean {
    const forbidden = ["Chapman", "Lattimore", "Fagles", "Wilson"];
    let passed = true;

    Object.values(this.nodes).forEach(node => {
      const content = fs.readFileSync(node.path, 'utf-8');
      forbidden.forEach(term => {
        if (content.includes(term)) {
          console.error(`[SECURITY] Forbidden term '${term}' found in ${node.path}`);
          passed = false;
        }
      });
    });

    return passed;
  }

  /**
   * Exports the tree as a JSON file for the build pipeline.
   */
  public exportReport(outputPath: string): void {
    const report = this.generateComponentReport();
    fs.writeFileSync(outputPath, JSON.stringify(report, null, 2));
    console.log(`[ComponentTreeAnalyzer] Report exported to ${outputPath}`);
  }
}

/**
 * Main execution block for the frontend analysis.
 */
async function runFrontendAnalysis() {
  const analyzer = new ComponentTreeAnalyzer('./src');
  analyzer.buildComponentTree();
  
  if (!analyzer.securityAudit()) {
    console.error("Frontend analysis failed security audit. Build rejected.");
    process.exit(1);
  }

  const report = analyzer.generateComponentReport();
  console.log("--- LOGOS Component Analysis ---");
  console.log(`Total Components: ${report.totalComponents}`);
  console.log(`Max Tree Depth: ${report.maxDepth}`);
  console.log(`Orphan Components: ${report.orphans.join(', ')}`);
  
  analyzer.exportReport('./component_report.json');
}

// Execute analysis
runFrontendAnalysis().catch(err => {
  console.error("Fatal error during frontend analysis:", err);
  process.exit(1);
});
