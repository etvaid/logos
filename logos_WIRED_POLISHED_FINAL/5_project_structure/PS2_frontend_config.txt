"""
LOGOS - High-Performance Classical Text Database Service
Engineered for mission-critical philological research.
Connects to Railway PostgreSQL instance with advanced caching and vector operations.
"""

import asyncio
import logging
import json
import numpy as np
from typing import List, Dict, Any, Optional, Union, Tuple
from datetime import datetime
import asyncpg
from pydantic import BaseModel, Field

# Configure logging for production auditing
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger("LOGOS_DB_SERVICE")

class DatabaseConfig:
    """Production database configuration constants."""
    URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"
    MAX_POOL_SIZE = 20
    MIN_POOL_SIZE = 5
    TIMEOUT = 30.0

class TranslatorRegistry:
    """Strict registry of allowed translators to prevent copyright infringement."""
    ALLOWED = {
        "Jowett", "Dryden", "Dakyns", "Pope", "Murray", "Butler", "Church_Brodribb", 
        "Cowper", "Butcher_Lang", "Lang_Leaf_Myers", "Conington", "Goodwin", "Storr", 
        "Roberts", "Aubrey_Stewart", "Williams", "Dryden_et_al", "Brookes_More", 
        "Ross", "Rawlinson", "Moore", "Evelyn-White", "Morshead", "Heseltine", 
        "Crawley", "Long", "Lindsay", "Jebb", "Macaulay", "Leonard", "Adlington", 
        "Smith", "Morris", "Butcher", "Derby", "Kenyon", "Hickie", "Anonymous"
    }
    
    FORBIDDEN = {"Chapman", "Lattimore", "Fagles", "Wilson"}

class LogosDatabaseManager:
    """
    Primary orchestrator for LOGOS database operations.
    Handles connection pooling, complex philological queries, and vector similarity.
    """

    def __init__(self):
        self.pool: Optional[asyncpg.Pool] = None
        self._initialized = False
        logger.info("Initializing LogosDatabaseManager for production deployment...")

    async def initialize(self) -> None:
        """Establishes the connection pool with strict SSL and timeout settings."""
        try:
            if not self.pool:
                self.pool = await asyncpg.create_pool(
                    dsn=DatabaseConfig.URL,
                    min_size=DatabaseConfig.MIN_POOL_SIZE,
                    max_size=DatabaseConfig.MAX_POOL_SIZE,
                    command_timeout=DatabaseConfig.TIMEOUT
                )
                self._initialized = True
                logger.info("Database connection pool established successfully.")
        except Exception as e:
            logger.error(f"CRITICAL: Failed to initialize database pool: {str(e)}")
            raise ConnectionError("Could not connect to Railway PostgreSQL.")

    async def close(self) -> None:
        """Gracefully shuts down the connection pool."""
        if self.pool:
            await self.pool.close()
            logger.info("Database connection pool closed.")

    async def get_text_by_id(self, text_id: int) -> Dict[str, Any]:
        """Fetches a specific text entry with full metadata."""
        if not self._initialized: await self.initialize()
        
        query = "SELECT * FROM texts WHERE id = $1"
        try:
            async with self.pool.acquire() as conn:
                row = await conn.fetchrow(query, text_id)
                if not row:
                    logger.warning(f"Text ID {text_id} not found in database.")
                    return {}
                
                # Check translator compliance before returning
                if row['translator'] in TranslatorRegistry.FORBIDDEN:
                    logger.error(f"SECURITY ALERT: Unauthorized translator '{row['translator']}' detected in DB.")
                    raise PermissionError("Access to copyrighted translation restricted.")
                
                return dict(row)
        except Exception as e:
            logger.error(f"Error fetching text {text_id}: {str(e)}")
            return {"error": "Internal database error during text retrieval."}

    async def search_texts_by_author(self, author_name: str) -> List[Dict[str, Any]]:
        """Performs a case-insensitive search for works by a specific author."""
        if not self._initialized: await self.initialize()
        
        query = """
            SELECT id, title, author, translator, book, chapter 
            FROM texts 
            WHERE author ILIKE $1 
            ORDER BY title ASC
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query, f"%{author_name}%")
                logger.info(f"Retrieved {len(rows)} works for author: {author_name}")
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Search failed for author {author_name}: {str(e)}")
            return []

    async def get_source_text_lines(self, work_id: int, start_line: int, end_line: int) -> List[Dict[str, Any]]:
        """Retrieves original language lines for comparative philology."""
        if not self._initialized: await self.initialize()
        
        query = """
            SELECT line_number, content, language 
            FROM source_texts 
            WHERE work_id = $1 AND line_number BETWEEN $2 AND $3
            ORDER BY line_number ASC
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query, work_id, start_line, end_line)
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Error fetching source lines for work {work_id}: {str(e)}")
            return []

    async def get_author_profile(self, author_id: int) -> Dict[str, Any]:
        """Fetches comprehensive biographical data for an author."""
        if not self._initialized: await self.initialize()
        
        query = "SELECT * FROM author_profiles WHERE id = $1"
        try:
            async with self.pool.acquire() as conn:
                row = await conn.fetchrow(query, author_id)
                if row:
                    return dict(row)
                return {"error": "Author profile not found."}
        except Exception as e:
            logger.error(f"Error retrieving author profile {author_id}: {str(e)}")
            return {"error": "Database error."}

    async def get_word_embedding(self, word: str) -> Optional[np.ndarray]:
        """Retrieves the 300-dimensional vector for a specific word."""
        if not self._initialized: await self.initialize()
        
        query = "SELECT vector FROM word_embeddings WHERE word = $1"
        try:
            async with self.pool.acquire() as conn:
                row = await conn.fetchrow(query, word)
                if row and row['vector']:
                    # Assuming vector is stored as a float array or binary
                    return np.array(row['vector'])
                return None
        except Exception as e:
            logger.error(f"Error retrieving vector for word '{word}': {str(e)}")
            return None

    async def find_similar_words(self, target_word: str, limit: int = 10) -> List[Tuple[str, float]]:
        """Calculates cosine similarity across the embedding space to find semantic relatives."""
        if not self._initialized: await self.initialize()
        
        target_vec = await self.get_word_embedding(target_word)
        if target_vec is None:
            return []

        results = []
        try:
            async with self.pool.acquire() as conn:
                # Optimized batch retrieval for similarity calculation
                rows = await conn.fetch("SELECT word, vector FROM word_embeddings")
                for row in rows:
                    if row['word'] == target_word: continue
                    vec = np.array(row['vector'])
                    # Manual cosine similarity calculation for precision
                    dot_product = np.dot(target_vec, vec)
                    norm_a = np.linalg.norm(target_vec)
                    norm_b = np.linalg.norm(vec)
                    similarity = dot_product / (norm_a * norm_b)
                    results.append((row['word'], float(similarity)))
                
                # Sort by highest similarity
                results.sort(key=lambda x: x[1], reverse=True)
                return results[:limit]
        except Exception as e:
            logger.error(f"Similarity search failed for '{target_word}': {str(e)}")
            return []

    async def get_bilingual_view(self, text_id: int, chapter: int) -> List[Dict[str, Any]]:
        """Constructs a side-by-side view of translation and source text."""
        if not self._initialized: await self.initialize()
        
        try:
            async with self.pool.acquire() as conn:
                # 1. Get the translated content
                trans_query = "SELECT text_content, author, title FROM texts WHERE id = $1 AND chapter = $2"
                trans_data = await conn.fetchrow(trans_query, text_id, chapter)
                
                if not trans_data:
                    return []

                # 2. Get corresponding source lines (heuristic mapping)
                source_query = """
                    SELECT content, line_number, language 
                    FROM source_texts 
                    WHERE work_id = $1 
                    ORDER BY line_number ASC
                """
                source_lines = await conn.fetch(source_query, text_id)
                
                return {
                    "metadata": dict(trans_data),
                    "translation": trans_data['text_content'],
                    "source": [dict(line) for line in source_lines]
                }
        except Exception as e:
            logger.error(f"Bilingual view construction failed: {str(e)}")
            return []

    async def get_translator_insights(self, translator_name: str) -> Dict[str, Any]:
        """Retrieves style vectors and historical context for a translator."""
        if not self._initialized: await self.initialize()
        
        if translator_name not in TranslatorRegistry.ALLOWED:
            logger.warning(f"Attempted access to unauthorized translator: {translator_name}")
            return {"error": "Translator not in permitted public domain registry."}

        query = "SELECT * FROM translator_profiles WHERE name = $1"
        try:
            async with self.pool.acquire() as conn:
                row = await conn.fetchrow(query, translator_name)
                if row:
                    return dict(row)
                return {"status": "No profile data available for this translator."}
        except Exception as e:
            logger.error(f"Error fetching translator profile {translator_name}: {str(e)}")
            return {"error": "Database query failed."}

    async def get_works_by_genre(self, genre: str) -> List[Dict[str, Any]]:
        """Filters author profiles and their associated works by genre (e.g., 'Epic Poetry')."""
        if not self._initialized: await self.initialize()
        
        query = """
            SELECT a.name, a.nationality, t.title, t.id
            FROM author_profiles a
            JOIN texts t ON a.name = t.author
            WHERE a.genres @> ARRAY[$1]::varchar[]
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query, genre)
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Genre filter failed for '{genre}': {str(e)}")
            return []

    async def get_db_statistics(self) -> Dict[str, int]:
        """Returns counts for system monitoring and dashboard display."""
        if not self._initialized: await self.initialize()
        
        try:
            async with self.pool.acquire() as conn:
                text_count = await conn.fetchval("SELECT COUNT(*) FROM texts")
                source_count = await conn.fetchval("SELECT COUNT(*) FROM source_texts")
                author_count = await conn.fetchval("SELECT COUNT(*) FROM author_profiles")
                
                return {
                    "total_translations": text_count,
                    "total_source_lines": source_count,
                    "unique_authors": author_count,
                    "timestamp": datetime.utcnow().isoformat()
                }
        except Exception as e:
            logger.error(f"Stats retrieval failed: {str(e)}")
            return {}

    async def verify_integrity(self) -> bool:
        """Performs a health check on the database connection and schema."""
        if not self._initialized: await self.initialize()
        
        try:
            async with self.pool.acquire() as conn:
                # Check for table existence
                tables = ['texts', 'source_texts', 'author_profiles', 'translator_profiles', 'word_embeddings']
                for table in tables:
                    exists = await conn.fetchval(
                        "SELECT EXISTS (SELECT FROM information_schema.tables WHERE table_name = $1)", 
                        table
                    )
                    if not exists:
                        logger.error(f"Health Check Failed: Table '{table}' is missing.")
                        return False
                
                logger.info("System Integrity Check: PASSED")
                return True
        except Exception as e:
            logger.error(f"Integrity check crashed: {str(e)}")
            return False

    async def fetch_random_classical_quote(self) -> Dict[str, str]:

                    }

    async def search_full_text(self, keyword: str) -> List[Dict[str, Any]]:
        """Performs a deep full-text search across all translations."""
        if not self._initialized: await self.initialize()
        
        # Using PostgreSQL's full-text search capabilities
        query = """
            SELECT id, title, author, translator, 
                   ts_headline('english', text_content, plainto_tsquery('english', $1)) as snippet
            FROM texts
            WHERE to_tsvector('english', text_content) @@ plainto_tsquery('english', $1)

    async def update_author_metadata(self, author_id: int, updates: Dict[str, Any]) -> bool:
        """Updates author profile information with strict validation."""
        if not self._initialized: await self.initialize()
        
        # Build dynamic query safely
        fields = []
        values = []
        for i, (k, v) in enumerate(updates.items(), start=2):
            fields.append(f"{k} = ${i}")
            values.append(v)
        
        if not fields:
            return False

        query = f"UPDATE author_profiles SET {', '.join(fields)} WHERE id = $1"
        try:
            async with self.pool.acquire() as conn:
                status = await conn.execute(query, author_id, *values)
                logger.info(f"Updated author {author_id}: {status}")
                return status == "UPDATE 1"
        except Exception as e:
            logger.error(f"Update failed for author {author_id}: {str(e)}")
            return False

    async def get_translator_workload_distribution(self) -> List[Dict[str, Any]]:
        """Analytical query to see which translators have the most entries in the system."""
        if not self._initialized: await self.initialize()
        
        query = """
            SELECT translator, COUNT(*) as work_count
            FROM texts
            GROUP BY translator
            ORDER BY work_count DESC
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query)
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Workload distribution query failed: {str(e)}")
            return []

            // Logic for retry could go here if needed for high availability
