-- LOGOS CLASSICAL STUDIES PLATFORM
-- MIGRATION: 005_CREATE_FTS_INDEXES
-- DESCRIPTION: Full-Text Search configuration for English, Greek, and Latin.
-- AUTHOR: SENIOR ARCHITECT
-- DATE: 2024-05-22

-- 1. Create a custom configuration for Classical Greek (Simplified)
-- In a real production system, we would use a specialized dictionary like 'greek' 
-- or a custom Hunspell dictionary for polytonic Greek.
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_ts_config WHERE cfgname = 'classical_greek') THEN
        CREATE TEXT SEARCH CONFIGURATION public.classical_greek (COPY = pg_catalog.simple);
    END IF;
END $$;

-- 2. Add TSVector columns for efficient searching
ALTER TABLE texts ADD COLUMN IF NOT EXISTS search_vector tsvector;
ALTER TABLE source_texts ADD COLUMN IF NOT EXISTS search_vector tsvector;

-- 3. Function to update FTS vectors for English texts
CREATE OR REPLACE FUNCTION texts_fts_trigger() RETURNS trigger AS $$
BEGIN
  NEW.search_vector :=
    setweight(to_tsvector('english', coalesce(NEW.title, '')), 'A') ||
    setweight(to_tsvector('english', coalesce(NEW.author, '')), 'B') ||
    setweight(to_tsvector('english', coalesce(NEW.text_content, '')), 'D');
  RETURN NEW;
END
$$ LANGUAGE plpgsql;

-- 4. Function to update FTS vectors for Source texts (Multi-language)
CREATE OR REPLACE FUNCTION source_texts_fts_trigger() RETURNS trigger AS $$
BEGIN
  IF NEW.language = 'grc' THEN
    NEW.search_vector := to_tsvector('public.classical_greek', coalesce(NEW.content, ''));
  ELSIF NEW.language = 'lat' THEN
    NEW.search_vector := to_tsvector('english', coalesce(NEW.content, '')); -- Latin often uses English stemmer or simple
  ELSE
    NEW.search_vector := to_tsvector('english', coalesce(NEW.content, ''));
  END IF;
  RETURN NEW;
END
$$ LANGUAGE plpgsql;

-- 5. Create Triggers
DROP TRIGGER IF EXISTS trg_texts_fts ON texts;
CREATE TRIGGER trg_texts_fts BEFORE INSERT OR UPDATE ON texts
FOR EACH ROW EXECUTE FUNCTION texts_fts_trigger();

DROP TRIGGER IF EXISTS trg_source_texts_fts ON source_texts;
CREATE TRIGGER trg_source_texts_fts BEFORE INSERT OR UPDATE ON source_texts
FOR EACH ROW EXECUTE FUNCTION source_texts_fts_trigger();

-- 6. Create GIN Indexes for FTS
CREATE INDEX IF NOT EXISTS idx_texts_search ON texts USING GIN (search_vector);
CREATE INDEX IF NOT EXISTS idx_source_texts_search ON source_texts USING GIN (search_vector);

-- 7. Initial Population (Warning: Can take time on 6.6M rows)
-- In production, this should be run in batches.
UPDATE texts SET search_vector = 
    setweight(to_tsvector('english', coalesce(title, '')), 'A') ||
    setweight(to_tsvector('english', coalesce(author, '')), 'B') ||
    setweight(to_tsvector('english', coalesce(text_content, '')), 'D')
WHERE search_vector IS NULL;

-- 8. Ranking Function for Search Results
CREATE OR REPLACE FUNCTION search_classical_texts(query_str TEXT)
RETURNS TABLE (
    id INTEGER,
    title VARCHAR,
    author VARCHAR,
    rank REAL
) AS $$
BEGIN
    RETURN QUERY
    SELECT t.id, t.title, t.author, ts_rank_cd(t.search_vector, websearch_to_tsquery('english', query_str)) AS rank
    FROM texts t
    WHERE t.search_vector @@ websearch_to_tsquery('english', query_str)
    ORDER BY rank DESC
    LIMIT 50;
END;
$$ LANGUAGE plpgsql;

-- Comments
COMMENT ON COLUMN texts.search_vector IS 'Precomputed FTS vector for English translations';
COMMENT ON COLUMN source_texts.search_vector IS 'Precomputed FTS vector for original language source texts';

-- DESCRIPTION: Materialized views for complex analytics and research dashboards.

-- 1. View: Author Statistics
-- Aggregate data about authors and their surviving works
CREATE MATERIALIZED VIEW IF NOT EXISTS mv_author_analytics AS
SELECT 
    ap.id AS author_id,
    ap.name,
    ap.nationality,
    COUNT(DISTINCT t.id) AS total_translations,
    COUNT(DISTINCT st.id) AS total_source_lines,
    ap.genres,
    ap.birth_year,
    ap.death_year
FROM author_profiles ap
LEFT JOIN texts t ON t.author = ap.name
LEFT JOIN source_texts st ON st.work_id = t.id
GROUP BY ap.id, ap.name, ap.nationality, ap.genres, ap.birth_year, ap.death_year
WITH NO DATA;

CREATE UNIQUE INDEX idx_mv_author_id ON mv_author_analytics (author_id);

-- 2. View: Translator Comparison Metrics
-- Compare different translators of the same work
CREATE MATERIALIZED VIEW IF NOT EXISTS mv_translator_coverage AS
SELECT 
    tp.name AS translator_name,
    t.author,
    COUNT(t.id) AS works_translated,
    AVG(LENGTH(t.text_content)) AS avg_work_length,
    STRING_AGG(DISTINCT t.title, ', ') AS titles
FROM translator_profiles tp
JOIN texts t ON t.translator = tp.name
GROUP BY tp.name, t.author
WITH NO DATA;

CREATE INDEX idx_mv_translator_name ON mv_translator_coverage (translator_name);

-- 3. View: Recent Scholarly Activity
-- Aggregates bookmarks and annotations for a global "trending" feed
CREATE VIEW view_scholarly_activity AS
SELECT 
    'annotation' AS activity_type,
    a.id,
    u.username,
    u.institution,
    a.selected_text AS preview,
    a.created_at
FROM annotations a
JOIN users u ON a.user_id = u.id
WHERE a.is_private IS FALSE
UNION ALL

    'bookmark' AS activity_type,
    b.id,
    u.username,

    t.title AS preview,
    b.created_at

JOIN users u ON b.user_id = u.id
JOIN texts t ON b.text_id = t.id

FROM texts t
LEFT JOIN source_texts st ON st.work_id = t.id
GROUP BY t.id, t.title, t.author, t.translator
HAVING COUNT(st.id) = 0;

"""
LOGOS CLASSICAL STUDIES PLATFORM - MIGRATION ENGINE
--------------------------------------------------
A production-grade migration runner designed for high-concurrency 
environments. Handles sequential execution of SQL migrations, 
state management, and comprehensive logging.

Author: Senior Software Architect
Experience: 25+ years
"""

import os
import asyncio
import logging
import asyncpg
import datetime
from typing import List, Optional, Dict, Any
from pathlib import Path

# Configure logging to production standards
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - [%(filename)s:%(lineno)d] - %(message)s'
)
logger = logging.getLogger(__name__)

class MigrationManager:
    """
    Manages the lifecycle of database migrations for the LOGOS platform.
    
    Attributes:
        db_url (str): The PostgreSQL connection string.
        migrations_dir (Path): Directory containing .sql migration files.
        applied_migrations (List[str]): List of filenames already executed.
    """

    def __init__(self, db_url: str, migrations_dir: str = "backend/migrations"):
        """
        Initializes the MigrationManager with connection details.
        
        Args:
            db_url: PostgreSQL connection string.
            migrations_dir: Path to the directory containing SQL files.
        """
        self.db_url = db_url
        self.migrations_dir = Path(migrations_dir)
        self.conn: Optional[asyncpg.Connection] = None
        self.applied_migrations: List[str] = []
        
        if not self.migrations_dir.exists():
            logger.error(f"Migrations directory not found: {self.migrations_dir}")
            raise FileNotFoundError(f"Directory {self.migrations_dir} does not exist.")

    async def connect(self) -> None:
        """
        Establishes a connection to the Railway PostgreSQL database.
        Includes retry logic for production resilience.
        """
        retry_count = 0
        max_retries = 5
        while retry_count < max_retries:
            try:
                logger.info(f"Attempting to connect to database (Attempt {retry_count + 1})...")
                self.conn = await asyncpg.connect(self.db_url)
                logger.info("Successfully connected to Railway PostgreSQL.")
                return
            except Exception as e:
                retry_count += 1
                logger.warning(f"Connection attempt {retry_count} failed: {e}")
                await asyncio.sleep(2 ** retry_count)
        
        raise ConnectionError("Failed to connect to database after maximum retries.")

    async def disconnect(self) -> None:
        """Closes the database connection safely."""
        if self.conn:
            await self.conn.close()
            logger.info("Database connection closed.")

    async def ensure_migration_table(self) -> None:
        """
        Ensures the 'schema_migrations' table exists to track state.
        This table prevents duplicate execution of migration files.
        """
        if not self.conn:
            raise RuntimeError("Database connection not established.")

        query = """
        CREATE TABLE IF NOT EXISTS schema_migrations (
            id SERIAL PRIMARY KEY,
            filename VARCHAR(255) UNIQUE NOT NULL,
            applied_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
            execution_time_ms INTEGER,
            success BOOLEAN DEFAULT TRUE
        );
        """
        try:
            await self.conn.execute(query)
            logger.info("Schema migrations tracking table verified.")
        except Exception as e:
            logger.error(f"Error creating migration table: {e}")
            raise

    async def get_applied_migrations(self) -> List[str]:
        """
        Retrieves a list of migration filenames that have already been applied.
        
        Returns:
            List of strings representing filenames.
        """
        if not self.conn:
            return []
            
        rows = await self.conn.fetch("SELECT filename FROM schema_migrations WHERE success = TRUE")
        self.applied_migrations = [row['filename'] for row in rows]
        logger.info(f"Found {len(self.applied_migrations)} already applied migrations.")
        return self.applied_migrations

    async def get_pending_migrations(self) -> List[Path]:
        """
        Identifies SQL files in the migrations directory not yet applied.
        
        Returns:
            List of Path objects for pending migrations, sorted alphabetically.
        """
        all_files = sorted(list(self.migrations_dir.glob("*.sql")))
        applied = await self.get_applied_migrations()
        
        pending = [f for f in all_files if f.name not in applied]
        logger.info(f"Identified {len(pending)} pending migrations.")
        return pending

    async def execute_migration(self, file_path: Path) -> bool:
        """
        Executes a single SQL migration file within a transaction.
        
        Args:
            file_path: Path to the .sql file.
            
        Returns:
            bool: True if successful, False otherwise.
        """
        if not self.conn:
            return False

        filename = file_path.name
        logger.info(f"Executing migration: {filename}...")
        
        start_time = datetime.datetime.now()
        
        with open(file_path, 'r') as f:
            sql_content = f.read()

        # Execute migration within a transaction
        async with self.conn.transaction():
            try:
                # We use execute() because many migrations contain multiple statements
                await self.conn.execute(sql_content)
                
                end_time = datetime.datetime.now()
                duration = int((end_time - start_time).total_seconds() * 1000)
                
                await self.conn.execute(
                    "INSERT INTO schema_migrations (filename, execution_time_ms, success) VALUES ($1, $2, $3)",
                    filename, duration, True
                )
                
                logger.info(f"Successfully applied {filename} in {duration}ms.")
                return True
            except Exception as e:
                logger.error(f"Critical error applying migration {filename}: {e}")
                # Transaction will automatically rollback on exception
                return False

    async def run_all(self) -> Dict[str, Any]:
        """
        Orchestrates the execution of all pending migrations.
        
        Returns:
            Dictionary containing summary statistics of the run.
        """
        await self.connect()
        await self.ensure_migration_table()
        
        pending = await self.get_pending_migrations()
        results = {
            "total": len(pending),
            "applied": 0,
            "failed": 0,
            "details": []
        }

        if not pending:
            logger.info("No pending migrations to apply. System is up to date.")
            await self.disconnect()
            return results

        for migration_file in pending:
            success = await self.execute_migration(migration_file)
            if success:
                results["applied"] += 1
            else:
                results["failed"] += 1
                logger.error("Migration sequence halted due to error.")
                break # Stop execution on first failure to maintain integrity

        await self.disconnect()
        logger.info(f"Migration run complete. Applied: {results['applied']}, Failed: {results['failed']}")
        return results

    async def verify_database_state(self) -> None:
        """
        Performs a health check on the core tables after migrations.

                logger.info(f"Health Check: Table '{table}' exists with {count} rows.")

        await self.disconnect()
