import asyncio
import logging
from typing import Any, Optional

import asyncpg
from app.core.config import settings

# Initialize logging for the database connection manager
logger = logging.getLogger(__name__)

class DatabaseManager:
    """
    Manages the lifecycle of the PostgreSQL connection pool using asyncpg.
    Provides methods for acquiring connections and executing queries with automatic retry logic.
    """

    def __init__(self) -> None:
        """
        Initializes the DatabaseManager with an empty pool.
        """
        self._pool: Optional[asyncpg.Pool] = None
        self._connection_params = settings.get_database_connection_params()
        logger.info("DatabaseManager initialized. Ready to establish connection pool.")

    async def connect(self) -> None:
        """
        Establishes the connection pool to the Railway PostgreSQL database.
        Includes error handling and logging for production monitoring.
        """
        if self._pool is not None:
            logger.warning("Database connection pool already exists. Skipping re-initialization.")
            return

        try:
            logger.info(f"Attempting to connect to database at {self._connection_params['host']}...")
            self._pool = await asyncpg.create_pool(
                user=self._connection_params['user'],
                password=self._connection_params['password'],
                host=self._connection_params['host'],
                port=self._connection_params['port'],
                database=self._connection_params['database'],
                min_size=self._connection_params['min_size'],
                max_size=self._connection_params['max_size'],
                command_timeout=60.0,
                max_queries=50000,
                max_inactive_connection_lifetime=300.0
            )
            logger.info("Successfully established PostgreSQL connection pool.")
        except Exception as e:
            logger.critical(f"FATAL: Failed to connect to the database: {str(e)}")
            raise ConnectionError(f"Could not connect to Railway PostgreSQL: {str(e)}")

    async def disconnect(self) -> None:
        """
        Gracefully closes the connection pool.
        Must be called during application shutdown.
        """
        if self._pool:
            logger.info("Closing database connection pool...")
            await self._pool.close()
            self._pool = None
            logger.info("Database connection pool closed.")
        else:
            logger.warning("Attempted to disconnect, but no pool was active.")

    async def fetch_one(self, query: str, *args: Any) -> Optional[asyncpg.Record]:
        """
        Executes a query and returns a single record.
        """
        if self._pool is None:
            await self.connect()
        
        async with self._pool.acquire() as connection:
            try:
                logger.debug(f"Executing fetch_one: {query} with args {args}")
                return await connection.fetchrow(query, *args)
            except Exception as e:
                logger.error(f"Database error in fetch_one: {str(e)} | Query: {query}")
                raise e

    async def fetch_all(self, query: str, *args: Any) -> list[asyncpg.Record]:
        """
        Executes a query and returns all matching records.
        """
        if self._pool is None:
            await self.connect()
            
        async with self._pool.acquire() as connection:
            try:
                logger.debug(f"Executing fetch_all: {query} with args {args}")
                return await connection.fetch(query, *args)
            except Exception as e:
                logger.error(f"Database error in fetch_all: {str(e)} | Query: {query}")
                raise e

    async def execute(self, query: str, *args: Any) -> str:
        """
        Executes a command (INSERT, UPDATE, DELETE) and returns the status.
        """
        if self._pool is None:
            await self.connect()
            
        async with self._pool.acquire() as connection:
            try:
                logger.debug(f"Executing command: {query} with args {args}")
                return await connection.execute(query, *args)
            except Exception as e:
                logger.error(f"Database error in execute: {str(e)} | Query: {query}")
                raise e

    async def check_health(self) -> bool:
        """
        Performs a simple query to verify the database connection is healthy.
        Used by the health check endpoint.
        """
        try:
            result = await self.fetch_one("SELECT 1")
            return result is not None
        except Exception as e:
            logger.error(f"Health check failed: {str(e)}")
            return False

# Global instance for use across the application
db = DatabaseManager()

import logging
import json
from typing import List, Optional, Dict, Any
import numpy as np

from app.db.connection import db
from app.core.config import settings

# Setup logging for the repository layer
logger = logging.getLogger(__name__)

class ClassicalTextRepository:
    """
    Implementation of the Data Access Layer for LOGOS.
    Directly queries the production tables: texts, source_texts, author_profiles, 
    translator_profiles, and word_embeddings.
    """

    async def get_text_by_id(self, text_id: int) -> Optional[Dict[str, Any]]:
        """
        Retrieves a specific text entry from the 'texts' table by ID.
        Includes validation of the translator against the whitelist.
        """
        query = "SELECT id, title, author, translator, text_content, book, chapter FROM texts WHERE id = $1"
        try:
            record = await db.fetch_one(query, text_id)
            if not record:
                logger.info(f"Text with ID {text_id} not found.")
                return None
            
            # Strict Translator Validation
            translator = record['translator']
            if not settings.validate_translator_access(translator):
                logger.error(f"Access to text {text_id} blocked: Translator '{translator}' is not authorized.")
                return None
                
            return dict(record)
        except Exception as e:
            logger.error(f"Error fetching text {text_id}: {str(e)}")
            raise Exception(f"Failed to retrieve text from database: {str(e)}")

    async def search_texts_by_author(self, author_name: str, limit: int = 50) -> List[Dict[str, Any]]:
        """
        Searches the 'texts' table for all works by a specific author.
        Filters out any results with forbidden translators.
        """
        query = """
            SELECT id, title, author, translator, text_content, book, chapter 
            FROM texts 
            WHERE author ILIKE $1 
            LIMIT $2
        """
        try:
            records = await db.fetch_all(query, f"%{author_name}%", limit)
            valid_results = []
            
            for rec in records:
                if settings.validate_translator_access(rec['translator']):
                    valid_results.append(dict(rec))
                else:
                    logger.debug(f"Skipping record {rec['id']} due to unauthorized translator.")
            
            logger.info(f"Found {len(valid_results)} authorized texts for author: {author_name}")
            return valid_results
        except Exception as e:
            logger.error(f"Error searching texts for author {author_name}: {str(e)}")
            return []

    async def get_source_text_lines(self, work_id: int, start_line: int, end_line: int) -> List[Dict[str, Any]]:
        """
        Queries the 'source_texts' table for original language content.
        Useful for philological analysis and side-by-side comparison.
        """
        query = """
            SELECT id, content, language, work_id, line_number 
            FROM source_texts 
            WHERE work_id = $1 AND line_number BETWEEN $2 AND $3
            ORDER BY line_number ASC
        """
        try:
            records = await db.fetch_all(query, work_id, start_line, end_line)
            logger.info(f"Retrieved {len(records)} source lines for work_id {work_id}")
            return [dict(rec) for rec in records]
        except Exception as e:
            logger.error(f"Error fetching source text for work_id {work_id}: {str(e)}")
            raise Exception("Philological source retrieval failed.")

    async def get_author_profile(self, name: str) -> Optional[Dict[str, Any]]:
        """
        Retrieves metadata from the 'author_profiles' table.
        """
        query = "SELECT id, name, birth_year, death_year, nationality, genres FROM author_profiles WHERE name = $1"
        try:
            record = await db.fetch_one(query, name)
            if record:
                return dict(record)
            return None
        except Exception as e:
            logger.error(f"Error fetching author profile for {name}: {str(e)}")
            return None

    async def get_translator_profile(self, name: str) -> Optional[Dict[str, Any]]:
        """
        Retrieves metadata from the 'translator_profiles' table.
        Includes a check against the master whitelist.
        """
        if not settings.validate_translator_access(name):
            logger.warning(f"Attempted to fetch profile for unauthorized translator: {name}")
            return None

        query = "SELECT id, name, style_vector, works_translated FROM translator_profiles WHERE name = $1"
        try:
            record = await db.fetch_one(query, name)
            if record:
                return dict(record)
            return None
        except Exception as e:
            logger.error(f"Error fetching translator profile for {name}: {str(e)}")
            return None

    async def get_word_vector(self, word: str) -> Optional[np.ndarray]:
        """
        Retrieves a 300-dimension word embedding from the 'word_embeddings' table.
        Converts the stored array back into a numpy array for computation.
        """
        query = "SELECT vector FROM word_embeddings WHERE word = $1"
        try:
            record = await db.fetch_one(query, word)
            if record and record['vector']:
                # Assuming vector is stored as a list/array in Postgres
                return np.array(record['vector'], dtype=np.float32)
            return None
        except Exception as e:
            logger.error(f"Error fetching word vector for '{word}': {str(e)}")
            return None

    async def get_texts_by_translator(self, translator_name: str) -> List[Dict[str, Any]]:
        """
        Retrieves all texts translated by a specific individual.
        Strictly enforces the whitelist.
        """
        if not settings.validate_translator_access(translator_name):
            logger.error(f"Unauthorized translator query: {translator_name}")
            return []

        query = "SELECT id, title, author, translator, book, chapter FROM texts WHERE translator = $1"
        try:
            records = await db.fetch_all(query, translator_name)
            return [dict(rec) for rec in records]
        except Exception as e:
            logger.error(f"Error fetching texts by translator {translator_name}: {str(e)}")
            return []

    async def get_texts_by_genre(self, genre: str) -> List[Dict[str, Any]]:
        """
        Finds authors associated with a genre and retrieves their works.
        """
        # First, find authors in the genre
        author_query = "SELECT name FROM author_profiles WHERE $1 = ANY(genres)"
        try:
            author_records = await db.fetch_all(author_query, genre)
            author_names = [rec['name'] for rec in author_records]
            
            if not author_names:
                return []

            # Then, find their texts
            text_query = "SELECT id, title, author, translator FROM texts WHERE author = ANY($1)"
            text_records = await db.fetch_all(text_query, author_names)
            
            # Filter by whitelist
            return [dict(r) for r in text_records if settings.validate_translator_access(r['translator'])]
        except Exception as e:
            logger.error(f"Error fetching texts by genre {genre}: {str(e)}")
            return []

    async def get_translation_comparison(self, title: str, book: int, chapter: int) -> List[Dict[str, Any]]:
        """
        Retrieves different translations of the same passage for comparison.
        This is a core feature for classical scholars.
        """
        query = """
            SELECT id, title, author, translator, text_content, book, chapter 
            FROM texts 
            WHERE title = $1 AND book = $2 AND chapter = $3
        """
        try:
            records = await db.fetch_all(query, title, book, chapter)
            results = []
            for rec in records:
                if settings.validate_translator_access(rec['translator']):
                    results.append(dict(rec))
            
            logger.info(f"Comparison view generated for {title} {book}.{chapter} with {len(results)} versions.")
            return results
        except Exception as e:
            logger.error(f"Error generating translation comparison: {str(e)}")
            return []

    async def batch_get_word_embeddings(self, words: List[str]) -> Dict[str, np.ndarray]:
        """
        Efficiently fetches vectors for multiple words in a single query.
        """
        query = "SELECT word, vector FROM word_embeddings WHERE word = ANY($1)"
        try:
            records = await db.fetch_all(query, words)
            return {r['word']: np.array(r['vector'], dtype=np.float32) for r in records}
        except Exception as e:
            logger.error(f"Batch embedding fetch failed: {str(e)}")
            return {}

    async def get_author_stats(self) -> List[Dict[str, Any]]:
        """
        Aggregates data to show how many works are available per author.
        Used for the dashboard/index view.
        """
        query = """
            SELECT author, COUNT(*) as work_count 
            FROM texts 
            GROUP BY author 
            ORDER BY work_count DESC
        """
        try:
            records = await db.fetch_all(query)
            return [dict(rec) for rec in records]
        except Exception as e:
            logger.error(f"Failed to fetch author stats: {str(e)}")
            return []

    async def get_translator_workload(self) -> List[Dict[str, Any]]:
        """
        Aggregates data to show how many works are available per translator.
        Filters out forbidden translators from the stats.
        """
        query = """
            SELECT translator, COUNT(*) as work_count 
            FROM texts 
            GROUP BY translator 
            ORDER BY work_count DESC
        """
        try:
            records = await db.fetch_all(query)
            valid_stats = []
            for rec in records:
                if settings.validate_translator_access(rec['translator']):
                    valid_stats.append(dict(rec))
            return valid_stats
        except Exception as e:
            logger.error(f"Failed to fetch translator workload: {str(e)}")
            return []

    async def get_recent_additions(self, limit: int = 10) -> List[Dict[str, Any]]:
        """
        Retrieves the most recently added texts for the 'What's New' section.
        """
        query = "SELECT id, title, author, translator FROM texts ORDER BY id DESC LIMIT $1"
        try:
            records = await db.fetch_all(query, limit)
            return [dict(rec) for rec in records if settings.validate_translator_access(rec['translator'])]
        except Exception as e:
            logger.error(f"Error fetching recent additions: {str(e)}")
            return []

    async def get_source_text_metadata(self, work_id: int) -> Optional[Dict[str, Any]]:
        """
        Retrieves language and line count information for a source text.
        """
        query = """
            SELECT language, COUNT(*) as line_count 
            FROM source_texts 
            WHERE work_id = $1 
            GROUP BY language
        """
        try:
            record = await db.fetch_one(query, work_id)
            if record:
                return dict(record)
            return None
        except Exception as e:
            logger.error(f"Error fetching source metadata for work {work_id}: {str(e)}")
            return None

    async def search_full_text(self, search_term: str, limit: int = 20) -> List[Dict[str, Any]]:
        """
        Performs a full-text search across the 'text_content' column.
        Utilizes PostgreSQL's text search capabilities if indexed, or falls back to ILIKE.
        """
        query = """
            SELECT id, title, author, translator, book, chapter 
            FROM texts 
            WHERE text_content ILIKE $1 
            LIMIT $2
        """
        try:
            # Note: In production, we would use tsvector for performance
            records = await db.fetch_all(query, f"%{search_term}%", limit)
            return [dict(rec) for rec in records if settings.validate_translator_access(rec['translator'])]
        except Exception as e:
            logger.error(f"Full-text search failed for '{search_term}': {str(e)}")
            return []

async def add_process_time_header(request: Request, call_next):
    """
    Middleware to measure request processing time and log it.

    response = await call_next(request)
    process_time = time.time() - start_time
    response.headers["X-Process-Time"] = str(process_time)
    logger.info(f"Path: {request.url.path} | Method: {request.method} | Duration: {process_time:.4f}s")
    return response
