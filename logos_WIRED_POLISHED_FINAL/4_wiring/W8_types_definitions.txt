/**
 * Standard API Response wrapper for all successful requests.
 * @template T The type of the data being returned.
 */
export interface ApiResponse<T> {
  /** Indicates if the request was successful */
  success: boolean;
  /** The payload of the response */
  data: T;
  /** Server-side timestamp of the response in ISO 8601 format */
  timestamp: string;
  /** Optional message for the client (e.g., success notifications) */
  message?: string;
  /** Request ID for tracing in production logs */
  traceId: string;
}

/**
 * Standard API Error wrapper for failed requests.
 */
export interface ApiErrorResponse {
  /** Always false for error responses */
  success: false;
  /** Error details */
  error: {
    /** Internal error code for client-side handling */
    code: ErrorCode;
    /** Human-readable error message */
    message: string;
    /** Detailed technical information (only in non-production environments) */
    details?: Record<string, unknown>;
    /** Validation errors if the request was a 422 */
    validationErrors?: Array<{
      loc: (string | number)[];
      msg: string;
      type: string;
    }>;
  };
  /** Request ID for tracing in production logs */
  traceId: string;
}

/**
 * Makes all properties in T optional, including nested objects.
 */
export type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends (infer U)[]
    ? DeepPartial<U>[]
    : T[P] extends object
    ? DeepPartial<T[P]>
    : T[P];
};

/**
 * Removes the 'id' and 'createdAt' fields from a type for creation requests.
 */
export type CreateDTO<T> = Omit<T, 'id' | 'createdAt' | 'updatedAt'>;

/**
 * Ensures a type has an ID, used for components that require keys.
 */
export type Identifiable<T> = T & { id: string | number };

/**
 * Represents a result that could be either a success or a failure.
 * Functional programming pattern for error handling.
 */
export type Result<T, E = Error> = 
  | { success: true; value: T }
  | { success: false; error: E };

/**
 * Extracts the property types from a class or interface.
 */
export type PropertiesOf<T> = {
  [K in keyof T]: T[K] extends Function ? never : K;
}[keyof T];
