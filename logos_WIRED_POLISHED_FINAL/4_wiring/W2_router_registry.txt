class PerformanceMiddleware(BaseHTTPMiddleware):
    """
    Middleware to monitor request duration and log performance metrics.
    Critical for high-traffic classical scholarship applications.
    """
    async def dispatch(self, request: Request, call_next):
        start_time = time.time()
        try:
            logger.info(f"Incoming request: {request.method} {request.url.path}")
            response = await call_next(request)
            process_time = (time.time() - start_time) * 1000
            response.headers["X-Process-Time-MS"] = str(process_time)
            logger.info(f"Request completed: {request.method} {request.url.path} - Status: {response.status_code} - Time: {process_time:.2f}ms")
            return response
        except Exception as e:
            process_time = (time.time() - start_time) * 1000
            logger.error(f"Request failed: {request.method} {request.url.path} - Error: {str(e)} - Time: {process_time:.2f}ms")
            return JSONResponse(
                status_code=500,
                content={"detail": "Internal Server Error", "trace_id": str(time.time())}
            )

class DatabaseManager:
    """
    Singleton-style manager for the asyncpg connection pool.
    Ensures efficient connection reuse across all scholar requests.
    """
    _pool: Optional[asyncpg.Pool] = None

    @classmethod
    async def get_pool(cls) -> asyncpg.Pool:
        """
        Returns the existing pool or creates a new one if it doesn't exist.
        """
        if cls._pool is None:
            logger.info("Initializing new database pool in DatabaseManager...")
            try:
                cls._pool = await asyncpg.create_pool(
                    dsn=DATABASE_URL,
                    min_size=20,
                    max_size=100,
                    command_timeout=60.0
                )
                logger.info("Database pool initialized successfully.")
            except Exception as e:
                logger.error(f"Failed to create database pool: {str(e)}")
                raise HTTPException(
                    status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
                    detail="Database connection pool could not be established."
                )
        return cls._pool

    @classmethod
    async def close_pool(cls):
        """
        Gracefully closes the connection pool.
        """
        if cls._pool:
            await cls._pool.close()
            cls._pool = None
            logger.info("Database pool closed.")

async def get_db_connection() -> AsyncGenerator[asyncpg.Connection, None]:
    """
    Dependency that provides a database connection from the pool.
    Usage: async with get_db_connection() as conn: ...
    """
    pool = await DatabaseManager.get_pool()
    async with pool.acquire() as connection:
        try:
            yield connection
        except Exception as e:
            logger.error(f"Database connection error during request: {str(e)}")
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="A database error occurred while processing your request."
            )

async def get_redis_client() -> AsyncGenerator[aioredis.Redis, None]:

    """

    try:
        yield redis
    finally:
        await redis.close()

class SecurityManager:
    """
    Handles all security-related operations for LOGOS.
    """
    
    @staticmethod
    def verify_password(plain_password: str, hashed_password: str) -> bool:
        """
        Verifies a plain text password against its bcrypt hash.
        """
        try:
            return pwd_context.verify(plain_password, hashed_password)
        except Exception as e:
            logger.error(f"Password verification error: {str(e)}")
            return False

    @staticmethod
    def get_password_hash(password: str) -> str:
        """
        Generates a bcrypt hash for a plain text password.
        """
        try:
            return pwd_context.hash(password)
        except Exception as e:
            logger.error(f"Password hashing error: {str(e)}")
            raise RuntimeError("Could not securely hash password.")

    @staticmethod
    def create_access_token(data: Dict[str, Any], expires_delta: Optional[timedelta] = None) -> str:
        """
        Generates a signed JWT access token for a scholar.
        """
        to_encode = data.copy()
        if expires_delta:
            expire = datetime.utcnow() + expires_delta
        else:
            expire = datetime.utcnow() + timedelta(minutes=15)
        
        to_encode.update({"exp": expire, "iss": "logos-classical-studies"})
        try:
            encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
            return encoded_jwt
        except JWTError as e:
            logger.error(f"JWT encoding error: {str(e)}")
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="Could not generate security token."
            )

    @staticmethod
    async def get_current_user(token: str = Depends(oauth2_scheme)) -> Dict[str, Any]:
        """
        Decodes and validates a JWT token to retrieve the current scholar's identity.
        """
        credentials_exception = HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate scholar credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            username: str = payload.get("sub")
            if username is None:
                raise credentials_exception
            
            # In a real production system, we query the scholar's profile from the DB
            # For this context, we'll assume a schema where users are stored.
            return {"username": username, "role": payload.get("role", "scholar")}
            
        except JWTError as e:
            logger.warning(f"JWT validation failed: {str(e)}")
            raise credentials_exception

class ScholarAuthService:
    """
    High-level service for handling scholar registration and login.
    """
    def __init__(self, db_pool: asyncpg.Pool):
        self.db_pool = db_pool

    async def authenticate_scholar(self, username: str, password: str) -> Optional[Dict[str, Any]]:
        """
        Authenticates a scholar against the database.
        Queries the author_profiles or a dedicated users table.
        """
        logger.info(f"Attempting authentication for scholar: {username}")
        try:
            async with self.db_pool.acquire() as conn:
                # Assuming a users table exists for auth
                user = await conn.fetchrow(
                    "SELECT id, username, hashed_password, role FROM users WHERE username = $1",
                    username
                )
                
                if not user:
                    logger.warning(f"Scholar not found: {username}")
                    return None
                
                if not SecurityManager.verify_password(password, user['hashed_password']):
                    logger.warning(f"Invalid password for scholar: {username}")
                    return None
                
                return dict(user)
        except Exception as e:
            logger.error(f"Database error during authentication: {str(e)}")
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="Authentication service unavailable."
            )

    async def register_scholar(self, scholar_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        Registers a new classical scholar in the system.
        """
        hashed_pwd = SecurityManager.get_password_hash(scholar_data['password'])
        try:
            async with self.db_pool.acquire() as conn:
                # Check for existing user
                existing = await conn.fetchval("SELECT id FROM users WHERE username = $1", scholar_data['username'])
                if existing:
                    raise HTTPException(status_code=400, detail="Username already registered.")
                
                # Insert new scholar
                new_scholar_id = await conn.fetchval(
                    """INSERT INTO users (username, hashed_password, email, role) 
                       VALUES ($1, $2, $3, $4) RETURNING id""",
                    scholar_data['username'], hashed_pwd, scholar_data['email'], 'scholar'
                )
                logger.info(f"New scholar registered with ID: {new_scholar_id}")
                return {"id": new_scholar_id, "username": scholar_data['username']}
        except Exception as e:
            logger.error(f"Scholar registration failed: {str(e)}")
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="Could not complete scholar registration."
            )

"""
LOGOS - Rate Limiting Module
Implements Redis-backed request throttling to protect classical resources.
"""

import time
import logging
from typing import Optional
from fastapi import Request, HTTPException, status
from redis import asyncio as aioredis

logger = logging.getLogger(__name__)

class RateLimiter:
    """
    Implements a sliding window rate limiting algorithm using Redis.
    Essential for protecting the source_texts (6.6M rows) from scraping.
    """
    def __init__(
        self, 
        redis_client: aioredis.Redis, 
        requests_per_minute: int = 60,
        burst_limit: int = 10
    ):
        self.redis = redis_client
        self.requests_per_minute = requests_per_minute
        self.burst_limit = burst_limit

    async def is_rate_limited(self, identifier: str, limit_type: str = "default") -> bool:
        """
        Checks if the given identifier (IP or Scholar ID) has exceeded their limit.
        """
        current_time = int(time.time())
        window_start = current_time - 60
        key = f"rate_limit:{limit_type}:{identifier}"
        
        try:
            # Multi-exec for atomic operations
            async with self.redis.pipeline(transaction=True) as pipe:
                # Remove timestamps older than 60 seconds
                pipe.zremrangebyscore(key, 0, window_start)
                # Count remaining timestamps
                pipe.zcard(key)
                # Add current timestamp
                pipe.zadd(key, {str(current_time): current_time})
                # Set expiry on the key to clean up Redis
                pipe.expire(key, 60)
                
                results = await pipe.execute()
                request_count = results[1]
                
                if request_count >= self.requests_per_minute:
                    logger.warning(f"Rate limit exceeded for {identifier} ({limit_type}): {request_count} req/min")
                    return True
                
                return False
        except Exception as e:
            logger.error(f"Redis rate limiting error: {str(e)}")
            # Fail open in production to ensure scholar access, but log heavily
            return False

class RateLimitMiddleware:
    """
    FastAPI dependency to apply rate limits to specific routes.
    """
    def __init__(self, requests_per_minute: int = 60, limit_type: str = "api"):
        self.requests_per_minute = requests_per_minute
        self.limit_type = limit_type

    async def __call__(self, request: Request):
        # Identify the user by IP or Auth Token
        identifier = request.client.host
        auth_header = request.headers.get("Authorization")
        if auth_header:
            # Use part of the token as a more specific identifier
            identifier = auth_header[-10:]

        from .deps import get_redis_client
        async for redis in get_redis_client():
            limiter = RateLimiter(redis, self.requests_per_minute)
            if await limiter.is_rate_limited(identifier, self.limit_type):
                raise HTTPException(
                    status_code=status.HTTP_429_TOO_MANY_REQUESTS,
                    detail="Too many requests. Please slow down your scholarly research.",
                    headers={"Retry-After": "60"}
                )
            break

    book_max: Optional[int] = Field(None, ge=1)
    limit: int = Field(20, ge=1, le=100)

    @validator('translator')
    def validate_translator_name(cls, v):

            logger.error(f"Validation failed: Unauthorized translator '{v}' requested.")
            raise ValueError(f"Translator '{v}' is not supported. Please use an authorized public domain translator.")

    @root_validator
    def check_book_range(cls, values):
        b_min, b_max = values.get('book_min'), values.get('book_max')
        if b_min and b_max and b_min > b_max:
            raise ValueError("book_min cannot be greater than book_max")
        return values

    Validator for semantic similarity requests using 'word_embeddings'.
    """
    word: str = Field(..., pattern="^[a-zA-Zα-ωΑ-Ω]+$")
    top_n: int = Field(10, ge=1, le=50)
    min_similarity: float = Field(0.0, ge=-1.0, le=1.0)

    @validator('word')
    def sanitize_word(cls, v):
        return v.strip().lower()

class TranslatorProfileResponse(BaseModel):
    """
    Output model for 'translator_profiles'.
    """
    id: int
    name: str
    style_vector: List[float]
    works_translated: List[str]

    @validator('name')
    def ensure_authorized(cls, v):
        if v not in AUTHORIZED_TRANSLATORS:
            return f"{v} (Restricted/Archived)"
        return v

def validate_line_number(line: str) -> bool:
    """
    Utility to validate classical line numbering (e.g., '10.1.a', '55').
    """
    pattern = r'^\d+([.a-z\d]+)*$'
    if not re.match(pattern, line):
        logger.warning(f"Invalid line number format detected: {line}")
        return False
    return True

def parse_scholar_query(raw_query: str) -> Dict[str, str]:
    """
    Parses complex scholar queries like 'author:Plato justice' into structured data.
    """
    result = {"query": raw_query, "filters": {}}
    
    # Extract author filter
    author_match = re.search(r'author:(\w+)', raw_query)
    if author_match:
        result["filters"]["author"] = author_match.group(1)
        result["query"] = raw_query.replace(author_match.group(0), "").strip()
        
    # Extract translator filter
    trans_match = re.search(r'translator:(\w+)', raw_query)
    if trans_match:
        translator_name = trans_match.group(1)
        if translator_name in AUTHORIZED_TRANSLATORS:
            result["filters"]["translator"] = translator_name
        result["query"] = result["query"].replace(trans_match.group(0), "").strip()
        
    logger.info(f"Parsed scholarly query: {result}")
    return result
