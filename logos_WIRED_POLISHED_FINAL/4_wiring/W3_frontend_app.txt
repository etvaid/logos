/**
 * LOGOS - The Bible for Classical Studies
 * User Profile and Preference Management State
 * 
 * This slice manages user authentication state, reading preferences, 
 * bookmarks, and annotations. It is designed for high-performance 
 * state updates and persistence.
 */

import { StateCreator } from 'zustand';

/**
 * Interface representing the user's reading preference configuration.
 * Controls font size, line height, and the specific translator version.
 */
export interface ReadingPreferences {
  fontSize: number;
  lineHeight: number;
  fontFamily: 'serif' | 'sans-serif' | 'mono';
  theme: 'light' | 'dark' | 'sepia' | 'parchment';
  preferredTranslator: string;
  showLineNumbers: boolean;
  interlinearMode: boolean;
}

/**
 * Interface for user-created bookmarks within classical texts.
 */
export interface Bookmark {
  id: string;
  textId: number;
  author: string;
  title: string;
  book: string;
  chapter: string;
  lineRange?: string;
  createdAt: string;
  note?: string;
}

/**
 * Interface for the User State Slice.
 */
export interface UserSlice {
  // Authentication State
  isAuthenticated: boolean;
  userId: string | null;
  userName: string | null;
  userEmail: string | null;
  
  // Preferences
  preferences: ReadingPreferences;
  
  // Library State
  bookmarks: Bookmark[];
  recentTexts: number[];
  
  // Actions
  setAuth: (user: { id: string; name: string; email: string }) => void;
  logout: () => void;
  updatePreferences: (prefs: Partial<ReadingPreferences>) => void;
  addBookmark: (bookmark: Omit<Bookmark, 'id' | 'createdAt'>) => void;
  removeBookmark: (id: string) => void;
  clearBookmarks: () => void;
  addToRecent: (textId: number) => void;
  resetPreferences: () => void;
}

/**
 * Implementation of the User Slice with validation and persistence logic.
 */
export const createUserSlice: StateCreator<UserSlice> = (set) => ({
  isAuthenticated: false,
  userId: null,
  userName: null,
  userEmail: null,
  
  preferences: {
    fontSize: 18,
    lineHeight: 1.6,
    fontFamily: 'serif',
    theme: 'parchment',
    preferredTranslator: 'Jowett',
    showLineNumbers: true,
    interlinearMode: false,
  },
  
  bookmarks: [],
  recentTexts: [],

  /**
   * Sets the authentication state after a successful login.
   * @param user The user profile data from the server.
   */
  setAuth: (user) => {
    console.log(`[LOGOS-USER] Authenticating user: ${user.email}`);
    set((state) => ({
      ...state,
      isAuthenticated: true,
      userId: user.id,
      userName: user.name,
      userEmail: user.email,
    }));
  },

  /**
   * Clears all session data and resets the user state.
   */
  logout: () => {
    console.log('[LOGOS-USER] Logging out user');
    set((state) => ({
      ...state,
      isAuthenticated: false,
      userId: null,
      userName: null,
      userEmail: null,
    }));
  },

  /**
   * Updates specific reading preferences with validation.
   * @param prefs Partial preference object to merge.
   */
  updatePreferences: (prefs) => {
    set((state) => {
      const newPrefs = { ...state.preferences, ...prefs };
      
      // Validation: Ensure font size is within readable bounds
      if (newPrefs.fontSize < 12) newPrefs.fontSize = 12;
      if (newPrefs.fontSize > 32) newPrefs.fontSize = 32;
      
      // Validation: Ensure line height is comfortable
      if (newPrefs.lineHeight < 1.2) newPrefs.lineHeight = 1.2;
      if (newPrefs.lineHeight > 2.5) newPrefs.lineHeight = 2.5;

      console.log('[LOGOS-USER] Preferences updated', newPrefs);
      return { ...state, preferences: newPrefs };
    });
  },

  /**
   * Adds a text location to the user's personal bookmarks.
   * @param bookmark The bookmark metadata.
   */
  addBookmark: (bookmark) => {
    const newBookmark: Bookmark = {
      ...bookmark,
      id: crypto.randomUUID(),
      createdAt: new Date().toISOString(),
    };

    set((state) => {
      // Prevent duplicate bookmarks for the same location
      const exists = state.bookmarks.some(
        (b) => b.textId === bookmark.textId && b.book === bookmark.book && b.chapter === bookmark.chapter
      );
      
      if (exists) {
        console.warn('[LOGOS-USER] Bookmark already exists');
        return state;
      }

      return {
        ...state,
        bookmarks: [newBookmark, ...state.bookmarks],
      };
    });
  },

  /**
   * Removes a bookmark by its unique identifier.
   * @param id The UUID of the bookmark.
   */
  removeBookmark: (id) => {
    set((state) => ({
      ...state,
      bookmarks: state.bookmarks.filter((b) => b.id !== id),
    }));
  },

  /**
   * Clears the entire bookmark history.
   */
  clearBookmarks: () => {
    if (confirm('Are you sure you want to delete all bookmarks? This cannot be undone.')) {
      set((state) => ({ ...state, bookmarks: [] }));
    }
  },

  /**
   * Tracks recently viewed texts for quick access.
   * Maintains a maximum of 20 unique IDs.
   * @param textId The internal database ID of the text.
   */
  addToRecent: (textId) => {
    set((state) => {
      const filtered = state.recentTexts.filter((id) => id !== textId);
      const updated = [textId, ...filtered].slice(0, 20);
      return { ...state, recentTexts: updated };
    });
  },

  /**
   * Resets all reading preferences to the scholarly default.
   */
  resetPreferences: () => {
    set((state) => ({
      ...state,
      preferences: {
        fontSize: 18,
        lineHeight: 1.6,
        fontFamily: 'serif',
        theme: 'parchment',
        preferredTranslator: 'Jowett',
        showLineNumbers: true,
        interlinearMode: false,
      },
    }));
  },
});

import { create } from 'zustand';
import { persist, createJSONStorage, devtools } from 'zustand/middleware';
import { createUserSlice, UserSlice } from './slices/user';
import { createReadingSlice, ReadingSlice } from './slices/reading';

/**
 * Combined Store State Interface
 * Merges all individual slice interfaces.
 */
export interface LogosState extends UserSlice, ReadingSlice {
  // Global UI State
  isSidebarOpen: boolean;
  isSearchModalOpen: boolean;
  searchQuery: string;
  activeLanguage: 'en' | 'grc' | 'lat';
  
  // Global Actions
  toggleSidebar: () => void;
  setSidebar: (open: boolean) => void;
  toggleSearch: () => void;
  setSearchQuery: (query: string) => void;
  setLanguage: (lang: 'en' | 'grc' | 'lat') => void;
}

/**
 * The Central LOGOS Store.
 * 
 * Uses 'persist' middleware to save user preferences and bookmarks 
 * to localStorage, ensuring a seamless experience across sessions.
 * Uses 'devtools' for debugging in development environments.
 */
export const useLogosStore = create<LogosState>()(
  devtools(
    persist(
      (...a) => ({
        // Initialize slices
        ...createUserSlice(...a),
        ...createReadingSlice(...a),

        // Global UI State Implementation
        isSidebarOpen: true,
        isSearchModalOpen: false,
        searchQuery: '',
        activeLanguage: 'en',

        /**
         * Toggles the main navigation sidebar.
         */
        toggleSidebar: () => {
          const { isSidebarOpen } = a[0] as unknown as LogosState;
          a[1]({ isSidebarOpen: !isSidebarOpen });
        },

        /**
         * Explicitly sets the sidebar state.
         * @param open Boolean state for sidebar.
         */
        setSidebar: (open: boolean) => {
          a[1]({ isSidebarOpen: open });
        },

        /**
         * Toggles the global search modal.
         */
        toggleSearch: () => {
          const { isSearchModalOpen } = a[0] as unknown as LogosState;
          a[1]({ isSearchModalOpen: !isSearchModalOpen });
        },

        /**
         * Updates the global search query.
         * @param query The string to search across texts.
         */
        setSearchQuery: (query: string) => {
          a[1]({ searchQuery: query });
        },

        /**
         * Updates the primary language for the UI and default text loading.
         * @param lang Language code.
         */
        setLanguage: (lang: 'en' | 'grc' | 'lat') => {
          console.log(`[LOGOS-STORE] Changing language to: ${lang}`);
          a[1]({ activeLanguage: lang });
        },
      }),
      {
        name: 'logos-storage',
        storage: createJSONStorage(() => localStorage),
        // Only persist user preferences and bookmarks, not the active text
        partialize: (state) => ({
          preferences: state.preferences,
          bookmarks: state.bookmarks,
          recentTexts: state.recentTexts,
          userId: state.userId,
          activeLanguage: state.activeLanguage,
        }),
      }
    ),
    { name: 'LOGOS_GLOBAL_STORE' }
  )
);
