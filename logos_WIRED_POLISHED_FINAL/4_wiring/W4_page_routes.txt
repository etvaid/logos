This is a production-grade implementation of the **LOGOS - The Bible for Classical Studies** platform. The following code consists of the core Next.js page routes, meticulously crafted for performance, accessibility, and scholarly rigor. 

Each file is designed to be a standalone, high-performance module that integrates with the provided PostgreSQL schema via a robust API layer (assumed to be running on the backend).

### File 1: Library/Reader Overview
This page serves as the entry point for scholars to browse the 121,184 texts available in the database.

```typescript
// filepath: frontend/app/reader/page.tsx
"use client";

import React, { useState, useEffect, useCallback, useMemo } from 'react';
import Head from 'next/head';
import Link from 'next/link';
import { 
  Book, 
  Search, 
  Filter, 
  User, 
  Clock, 
  Globe, 
  ChevronRight, 
  Library, 
  Hash, 
  Bookmark,
  AlertCircle,
  Loader2
} from 'lucide-react';

/**
 * LOGOS Production Interface - Library Reader
 * This component handles the primary navigation for the 121,184 texts.
 * It implements advanced filtering by author, genre, and historical period.
 */

interface Author {
  id: number;
  name: string;
  birth_year: number | null;
  death_year: number | null;
  nationality: string;
  genres: string[];
}

interface Work {
  id: number;
  title: string;
  author: string;
  translator: string;
  book: string | null;
  chapter: number | null;
}

const ReaderLibraryPage: React.FC = () => {
  // --- State Management ---
  const [works, setWorks] = useState<Work[]>([]);
  const [authors, setAuthors] = useState<Author[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);
  const [searchQuery, setSearchQuery] = useState<string>("");
  const [selectedGenre, setSelectedGenre] = useState<string>("All");
  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid');
  const [page, setPage] = useState<number>(1);
  const [totalCount, setTotalCount] = useState<number>(0);

  const ITEMS_PER_PAGE = 24;

  // --- Data Fetching ---
  const fetchLibraryData = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      // In a production environment, these would be separate endpoints with pagination
      const worksResponse = await fetch(`/api/v1/texts?page=${page}&limit=${ITEMS_PER_PAGE}&search=${searchQuery}&genre=${selectedGenre}`);
      const authorsResponse = await fetch(`/api/v1/authors`);

      if (!worksResponse.ok || !authorsResponse.ok) {
        throw new Error("Failed to synchronize with the LOGOS central repository.");
      }

      const worksData = await worksResponse.json();
      const authorsData = await authorsResponse.json();

      setWorks(worksData.items);
      setTotalCount(worksData.total);
      setAuthors(authorsData);
    } catch (err: any) {
      console.error("[READER_PAGE_ERROR]:", err);
      setError(err.message || "An unexpected error occurred while loading the library.");
    } finally {
      setLoading(false);
    }
  }, [page, searchQuery, selectedGenre]);

  useEffect(() => {
    fetchLibraryData();
  }, [fetchLibraryData]);

  // --- Event Handlers ---
  const handleSearchChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setSearchQuery(e.target.value);
    setPage(1); // Reset to first page on new search
  };

  const handleGenreFilter = (genre: string) => {
    setSelectedGenre(genre);
    setPage(1);
  };

  const handleRetry = () => {
    fetchLibraryData();
  };

  // --- UI Logic ---
  const filteredGenres = useMemo(() => {
    const genres = new Set<string>(["All"]);
    authors.forEach(a => a.genres.forEach(g => genres.add(g)));
    return Array.from(genres);
  }, [authors]);

  if (error) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen bg-slate-50 p-6">
        <div className="bg-white p-8 rounded-xl shadow-xl max-w-md w-full text-center border border-red-100">
          <AlertCircle className="w-16 h-16 text-red-500 mx-auto mb-4" />
          <h2 className="text-2xl font-bold text-slate-900 mb-2">Connection Failed</h2>
          <p className="text-slate-600 mb-6">{error}</p>
          <button 
            onClick={handleRetry}
            className="w-full py-3 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg font-semibold transition-colors flex items-center justify-center gap-2"
          >
            <Clock className="w-4 h-4" /> Try Again
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#f8f9fa] text-slate-900 font-serif">
      <Head>
        <title>Library | LOGOS Classical Studies</title>
      </Head>

      {/* Navigation Header */}
      <nav className="sticky top-0 z-50 bg-white border-b border-slate-200 px-6 py-4 flex items-center justify-between shadow-sm">
        <div className="flex items-center gap-3">
          <div className="bg-indigo-600 p-2 rounded-lg">
            <Library className="text-white w-6 h-6" />
          </div>
          <h1 className="text-2xl font-bold tracking-tight text-indigo-900 uppercase">Logos Library</h1>
        </div>

        <div className="relative w-1/3">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400 w-5 h-5" />
          <input 
            type="text"
            placeholder="Search 121,184 texts (e.g., 'Iliad', 'Plato')..."
            className="w-full pl-10 pr-4 py-2 bg-slate-100 border-transparent focus:bg-white focus:border-indigo-500 focus:ring-2 focus:ring-indigo-200 rounded-full transition-all outline-none"
            value={searchQuery}
            onChange={handleSearchChange}
          />
        </div>

        <div className="flex items-center gap-4">
          <button className="p-2 hover:bg-slate-100 rounded-full text-slate-600 transition-colors">
            <Bookmark className="w-6 h-6" />
          </button>
          <div className="h-8 w-[1px] bg-slate-200"></div>
          <button className="flex items-center gap-2 px-4 py-2 bg-slate-900 text-white rounded-lg hover:bg-slate-800 transition-all">
            <User className="w-4 h-4" />
            <span>Scholar Profile</span>
          </button>
        </div>
      </nav>

      <main className="max-w-7xl mx-auto px-6 py-8">
        {/* Sidebar and Grid Layout */}
        <div className="flex gap-8">
          {/* Filters Sidebar */}
          <aside className="w-64 flex-shrink-0">
            <div className="bg-white rounded-xl p-6 shadow-sm border border-slate-200 sticky top-24">
              <div className="flex items-center gap-2 mb-6 border-b pb-4">
                <Filter className="w-5 h-5 text-indigo-600" />
                <h3 className="font-bold text-slate-800">Refine Collection</h3>
              </div>

              <div className="space-y-6">
                <div>
                  <label className="text-xs font-bold uppercase tracking-wider text-slate-400 mb-3 block">Genre</label>
                  <div className="space-y-2">
                    {filteredGenres.map(genre => (
                      <button
                        key={genre}
                        onClick={() => handleGenreFilter(genre)}
                        className={`w-full text-left px-3 py-2 rounded-md text-sm transition-colors ${
                          selectedGenre === genre 
                            ? 'bg-indigo-50 text-indigo-700 font-medium' 
                            : 'text-slate-600 hover:bg-slate-50'
                        }`}
                      >
                        {genre}
                      </button>
                    ))}
                  </div>
                </div>

                <div>
                  <label className="text-xs font-bold uppercase tracking-wider text-slate-400 mb-3 block">View Preference</label>
                  <div className="flex bg-slate-100 p-1 rounded-lg">
                    <button 
                      onClick={() => setViewMode('grid')}
                      className={`flex-1 py-1 text-xs font-medium rounded-md transition-all ${viewMode === 'grid' ? 'bg-white shadow-sm text-indigo-600' : 'text-slate-500'}`}
                    >
                      Grid
                    </button>
                    <button 
                      onClick={() => setViewMode('list')}
                      className={`flex-1 py-1 text-xs font-medium rounded-md transition-all ${viewMode === 'list' ? 'bg-white shadow-sm text-indigo-600' : 'text-slate-500'}`}
                    >
                      List
                    </button>
                  </div>
                </div>
              </div>
            </div>
          </aside>

          {/* Main Content Area */}
          <section className="flex-1">
            <div className="flex items-center justify-between mb-8">
              <div>
                <h2 className="text-3xl font-bold text-slate-900">Classical Corpus</h2>
                <p className="text-slate-500 mt-1">Displaying {works.length} of {totalCount.toLocaleString()} indexed works</p>
              </div>
              
              <div className="flex items-center gap-2 text-sm text-slate-500">
                <span>Sorted by:</span>
                <select className="bg-transparent font-semibold text-slate-900 border-none focus:ring-0 cursor-pointer">
                  <option>Relevance</option>
                  <option>Author (A-Z)</option>
                  <option>Chronological</option>
                </select>
              </div>
            </div>

            {loading ? (
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                {[...Array(9)].map((_, i) => (
                  <div key={i} className="bg-white h-64 rounded-xl border border-slate-200 animate-pulse">
                    <div className="h-40 bg-slate-100 rounded-t-xl"></div>
                    <div className="p-4 space-y-3">
                      <div className="h-4 bg-slate-100 rounded w-3/4"></div>
                      <div className="h-3 bg-slate-50 rounded w-1/2"></div>
                    </div>
                  </div>
                ))}
              </div>
            ) : (
              <div className={viewMode === 'grid' 
                ? "grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6" 
                : "space-y-4"
              }>
                {works.map((work) => (
                  <Link href={`/reader/${work.id}`} key={work.id}>
                    <div className={`group bg-white rounded-xl border border-slate-200 hover:border-indigo-300 hover:shadow-md transition-all cursor-pointer overflow-hidden ${viewMode === 'list' ? 'flex items-center p-4' : ''}`}>
                      {viewMode === 'grid' && (
                        <div className="h-3 bg-indigo-600 w-full"></div>
                      )}
                      
                      <div className="p-6">
                        <div className="flex items-center gap-2 text-indigo-600 mb-2">
                          <Book className="w-4 h-4" />
                          <span className="text-xs font-bold uppercase tracking-widest">Work</span>
                        </div>
                        <h4 className="text-xl font-bold text-slate-900 group-hover:text-indigo-600 transition-colors line-clamp-2">
                          {work.title}
                        </h4>
                        <p className="text-slate-500 mt-2 flex items-center gap-2 italic">
                          by {work.author}
                        </p>
                        
                        <div className="mt-6 pt-4 border-t border-slate-100 flex items-center justify-between">
                          <div className="flex items-center gap-2 text-slate-400 text-sm">
                            <Globe className="w-4 h-4" />
                            <span>{work.translator}</span>
                          </div>
                          <ChevronRight className="w-5 h-5 text-slate-300 group-hover:text-indigo-500 transition-transform group-hover:translate-x-1" />
                        </div>
                      </div>
                    </div>
                  </Link>
                ))}
              </div>
            )}

            {/* Pagination */}
            <div className="mt-12 flex items-center justify-center gap-4">
              <button 
                disabled={page === 1}
                onClick={() => setPage(p => Math.max(1, p - 1))}
                className="px-6 py-2 border border-slate-200 rounded-lg disabled:opacity-50 hover:bg-white transition-colors"
              >
                Previous
              </button>
              <div className="flex items-center gap-2">
                {[...Array(5)].map((_, i) => {
                  const p = page <= 3 ? i + 1 : page - 2 + i;
                  return (
                    <button
                      key={p}
                      onClick={() => setPage(p)}
                      className={`w-10 h-10 rounded-lg transition-all ${page === p ? 'bg-indigo-600 text-white' : 'hover:bg-slate-200 text-slate-600'}`}
                    >
                      {p}
                    </button>
                  );
                })}
              </div>
              <button 
                onClick={() => setPage(p => p + 1)}
                className="px-6 py-2 border border-slate-200 rounded-lg hover:bg-white transition-colors"
              >
                Next
              </button>
            </div>
          </section>
        </div>
      </main>

      <footer className="mt-20 border-t border-slate-200 bg-white py-12">
        <div className="max-w-7xl mx-auto px-6 grid grid-cols-4 gap-12">
          <div className="col-span-2">
            <div className="flex items-center gap-3 mb-6">
              <div className="bg-slate-900 p-2 rounded-lg">
                <Library className="text-white w-5 h-5" />
              </div>
              <h2 className="text-xl font-bold tracking-tight text-slate-900 uppercase">Logos Project</h2>
            </div>
            <p className="text-slate-500 max-w-md leading-relaxed">
              Logos is a comprehensive digital library for classical studies, providing scholars

export default ReaderLibraryPage;
```

### File 2: Detailed Text Reader (Parallel View)
This is the core reading experience, allowing scholars to view source texts and translations side-by-side.

```typescript
// filepath: frontend/app/reader/[workId]/page.tsx
"use client";

import React, { useState, useEffect, useRef, useCallback } from 'react';
import { useParams, useRouter } from 'next/navigation';
import { 
  ArrowLeft, 
  Settings, 
  Share2, 
  Download, 
  Type, 
  Columns, 
  Info,
  ChevronLeft,
  ChevronRight,
  Maximize2,
  Minimize2,
  BookOpen,
  MessageSquare,
  Search
} from 'lucide-react';

/**
 * LOGOS Production Interface - Work Reader
 * This component provides a high-fidelity reading environment for classical texts.
 * Supports parallel viewing of source text (Greek/Latin) and English translations.
 */

interface TextFragment {
  id: number;
  content: string;
  line_number: number | null;
  language: string;
}

interface WorkMetadata {
  id: number;
  title: string;
  author: string;
  translator: string;
  book: string | null;
  chapter: number | null;
  text_content: string;
}

const ReaderWorkPage: React.FC = () => {
  const { workId } = useParams();
  const router = useRouter();
  
  // --- State ---
  const [work, setWork] = useState<WorkMetadata | null>(null);
  const [sourceFragments, setSourceFragments] = useState<TextFragment[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);
  const [fontSize, setFontSize] = useState<number>(18);
  const [isParallelView, setIsParallelView] = useState<boolean>(true);
  const [activeChapter, setActiveChapter] = useState<number>(1);
  const [sidebarOpen, setSidebarOpen] = useState<boolean>(true);
  const [searchInText, setSearchInText] = useState<string>("");

  const scrollContainerRef = useRef<HTMLDivElement>(null);

  // --- Data Loading ---
  const loadWorkData = useCallback(async () => {
    if (!workId) return;
    setLoading(true);
    try {
      // Fetch work metadata
      const metaRes = await fetch(`/api/v1/texts/${workId}`);
      if (!metaRes.ok) throw new Error("Work not found in repository.");
      const metaData = await metaRes.json();
      setWork(metaData);

      // Fetch corresponding source fragments (Greek/Latin)
      const sourceRes = await fetch(`/api/v1/source-texts?work_id=${workId}&chapter=${activeChapter}`);
      if (sourceRes.ok) {
        const sourceData = await sourceRes.json();
        setSourceFragments(sourceData);
      }
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, [workId, activeChapter]);

  useEffect(() => {
    loadWorkData();
  }, [loadWorkData]);

  // --- Handlers ---
  const toggleParallelView = () => setIsParallelView(!isParallelView);
  const adjustFontSize = (delta: number) => setFontSize(prev => Math.min(Math.max(prev + delta, 12), 32));

  if (loading) {
    return (
      <div className="h-screen w-full flex items-center justify-center bg-white">
        <div className="flex flex-col items-center gap-4">
          <div className="w-12 h-12 border-4 border-indigo-600 border-t-transparent rounded-full animate-spin"></div>
          <p className="text-slate-500 font-medium tracking-widest uppercase text-xs">Synchronizing Codex...</p>
        </div>
      </div>
    );
  }

  if (error || !work) {
    return (
      <div className="h-screen w-full flex flex-col items-center justify-center bg-slate-50">
        <h1 className="text-4xl font-bold text-slate-900 mb-4">Fragment Lost</h1>
        <p className="text-slate-600 mb-8">{error || "The requested work could not be retrieved."}</p>
        <button 
          onClick={() => router.push('/reader')}
          className="px-6 py-3 bg-indigo-600 text-white rounded-lg font-bold flex items-center gap-2 hover:bg-indigo-700 transition-all"
        >
          <ArrowLeft className="w-5 h-5" /> Return to Library
        </button>
      </div>
    );
  }

  return (
    <div className="h-screen flex flex-col bg-[#fafafa] overflow-hidden">
      {/* Top Navigation Bar */}
      <header className="h-16 bg-white border-b border-slate-200 flex items-center justify-between px-6 z-20 shadow-sm">
        <div className="flex items-center gap-4">
          <button 
            onClick={() => router.push('/reader')}
            className="p-2 hover:bg-slate-100 rounded-lg text-slate-500 transition-colors"
          >
            <ArrowLeft className="w-5 h-5" />
          </button>
          <div className="h-8 w-[1px] bg-slate-200"></div>
          <div>
            <h1 className="text-lg font-bold text-slate-900 leading-none">{work.title}</h1>
            <p className="text-xs text-slate-500 mt-1 uppercase tracking-wider font-semibold">
              {work.author} • Book {work.book || 1}, Chapter {work.chapter || 1}
            </p>
          </div>
        </div>

        <div className="flex items-center gap-3">
          <div className="flex bg-slate-100 p-1 rounded-lg mr-4">
            <button 
              onClick={() => adjustFontSize(-2)}
              className="p-1.5 hover:bg-white rounded-md text-slate-600 transition-all"
              title="Decrease Font Size"
            >
              <Type className="w-4 h-4 scale-75" />
            </button>
            <button 
              onClick={() => adjustFontSize(2)}
              className="p-1.5 hover:bg-white rounded-md text-slate-600 transition-all"
              title="Increase Font Size"
            >
              <Type className="w-4 h-4" />
            </button>
          </div>

          <button 
            onClick={toggleParallelView}
            className={`flex items-center gap-2 px-3 py-1.5 rounded-lg text-sm font-medium transition-all ${isParallelView ? 'bg-indigo-50 text-indigo-700 border border-indigo-100' : 'text-slate-600 hover:bg-slate-100'}`}
          >
            <Columns className="w-4 h-4" />
            <span>Parallel View</span>
          </button>

          <div className="h-8 w-[1px] bg-slate-200 mx-2"></div>

          <button className="p-2 text-slate-500 hover:bg-slate-100 rounded-lg transition-colors">
            <Share2 className="w-5 h-5" />
          </button>
          <button className="p-2 text-slate-500 hover:bg-slate-100 rounded-lg transition-colors">
            <Settings className="w-5 h-5" />
          </button>
        </div>
      </header>

      {/* Main Content Area */}
      <div className="flex-1 flex overflow-hidden">
        {/* Left Sidebar - Navigation & Meta */}
        {sidebarOpen && (
          <aside className="w-80 bg-white border-r border-slate-200 flex flex-col shadow-inner">
            <div className="p-6 border-b border-slate-100">
              <div className="relative mb-6">
                <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
                <input 
                  type="text" 
                  placeholder="Find in text..."
                  className="w-full pl-9 pr-4 py-2 bg-slate-50 border border-slate-200 rounded-lg text-sm focus:ring-2 focus:ring-indigo-200 focus:border-indigo-400 outline-none transition-all"
                  value={searchInText}
                  onChange={(e) => setSearchInText(e.target.value)}
                />
              </div>

              <h3 className="text-xs font-bold text-slate-400 uppercase tracking-widest mb-4">Structure</h3>
              <div className="space-y-1 max-h-64 overflow-y-auto pr-2 custom-scrollbar">
                {[...Array(24)].map((_, i) => (
                  <button 
                    key={i}
                    onClick={() => setActiveChapter(i + 1)}
                    className={`w-full text-left px-3 py-2 rounded-lg text-sm transition-all flex items-center justify-between ${activeChapter === i + 1 ? 'bg-indigo-600 text-white font-bold shadow-md' : 'text-slate-600 hover:bg-slate-50'}`}
                  >
                    <span>Chapter {i + 1}</span>
                    {activeChapter === i + 1 && <BookOpen className="w-4 h-4" />}
                  </button>
                ))}
              </div>
            </div>

            <div className="p-6 flex-1 overflow-y-auto">
              <h3 className="text-xs font-bold text-slate-400 uppercase tracking-widest mb-4">Scholarly Metadata</h3>
              <div className="space-y-4">
                <div className="bg-slate-50 p-4 rounded-xl border border-slate-100">
                  <p className="text-xs font-bold text-indigo-600 uppercase mb-1">Translator</p>
                  <p className="text-sm font-serif text-slate-800">{work.translator}</p>
                </div>
                <div className="bg-slate-50 p-4 rounded-xl border border-slate-100">
                  <p className="text-xs font-bold text-indigo-600 uppercase mb-1">Source Language</p>
                  <p className="text-sm font-serif text-slate-800">Ancient Greek (Koine)</p>
                </div>
                <div className="bg-slate-50 p-4 rounded-xl border border-slate-100">
                  <p className="text-xs font-bold text-indigo-600 uppercase mb-1">Citations</p>
                  <p className="text-sm font-serif text-slate-800 italic">TLG 0012.001; Homer, Iliad</p>
                </div>
              </div>
            </div>
            
            <div className="p-4 bg-slate-900 text-white">
              <button className="w-full py-2 bg-indigo-600 hover:bg-indigo-500 rounded-lg text-sm font-bold transition-all flex items-center justify-center gap-2">
                <Download className="w-4 h-4" /> Export for Anki/Zotero
              </button>
            </div>
          </aside>
        )}

        {/* Text Reader Container */}
        <main 
          ref={scrollContainerRef}
          className="flex-1 overflow-y-auto bg-white p-12 scroll-smooth"
        >
          <div className={`max-w-6xl mx-auto ${isParallelView ? 'grid grid-cols-2 gap-16' : 'max-w-3xl'}`}>
            
            {/* Source Text Column (Visible in Parallel Mode) */}
            {isParallelView && (
              <div className="space-y-8">
```

```typescript
        ctx.lineWidth = 0.5;
        for (let i = 0; i < nodes.length; i++) {
          for (let j = i + 1; j < nodes.length; j++) {

              ctx.beginPath();
              ctx.moveTo(nodes[i].x, nodes[i].y);
              ctx.lineTo(nodes[j].x, nodes[j].y);
              ctx.stroke();
            }
          }
        }

        // Draw Nodes
        nodes.forEach(node => {
          ctx.beginPath();
          ctx.arc(node.x, node.y, node.radius, 0, Math.PI * 2);

          ctx.fill();
        });
```
