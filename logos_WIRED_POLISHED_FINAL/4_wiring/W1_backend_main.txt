import logging
import sys
import json

from datetime import datetime
from typing import Any, Dict

class LogosJsonFormatter(logging.Formatter):
    """
    Custom JSON formatter for LOGOS production logging.
    Enables structured logging for ingestion into ELK/Grafana.
    """
    
    def __init__(self, service_name: str = "logos-backend"):
        super().__init__()
        self.service_name = service_name

    def format(self, record: logging.LogRecord) -> str:
        """Formats the log record into a structured JSON string."""
        log_data: Dict[str, Any] = {
            "timestamp": datetime.utcfromtimestamp(record.created).isoformat(),
            "level": record.levelname,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno,
            "service": self.service_name,
            "process_id": record.process,
            "thread_id": record.thread
        }
        
        if record.exc_info:
            log_data["exception"] = self.formatException(record.exc_info)
            
        if hasattr(record, "request_id"):
            log_data["request_id"] = record.request_id
            
        return json.dumps(log_data)

def setup_logging() -> None:
    """
    Configures the global logging system for the LOGOS application.
    Sets up stream handlers for stdout and file handlers for persistent storage.
    """
    logger = logging.getLogger()
    logger.setLevel(logging.INFO)
    
    # Remove existing handlers to prevent duplicates
    for handler in logger.handlers[:]:
        logger.removeHandler(handler)
    
    # Console Handler (JSON for production)
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(LogosJsonFormatter())
    logger.addHandler(console_handler)
    
    # Error File Handler
    try:
        error_file_handler = logging.FileHandler("logs/error.log")
        error_file_handler.setLevel(logging.ERROR)
        error_file_handler.setFormatter(LogosJsonFormatter())
        logger.addHandler(error_file_handler)
    except Exception as e:
        print(f"Failed to initialize file logging: {str(e)}")
        
    # LOGOS Specific Log Levels
    logging.getLogger("uvicorn.access").setLevel(logging.WARNING)
    logging.getLogger("asyncpg").setLevel(logging.INFO)
    
    logger.info("LOGOS Logging System Initialized Successfully")

class LogosBaseException(Exception):
    """Base exception for all LOGOS system errors."""

        self.code = code
        super().__init__(self.message)

    """Raised when the connection to Railway PostgreSQL fails."""
    def __init__(self, message: str = "Critical failure connecting to the database pool."):
        super().__init__(message, code="DB_CONNECTION_FAILURE")

class CopyrightViolationError(LogosBaseException):
    """Raised when a forbidden translator (Chapman, Lattimore, etc.) is requested."""
    def __init__(self, translator_name: str):
        self.translator_name = translator_name
        message = f"Copyright Violation: Translator '{translator_name}' is not in the public domain whitelist."
        super().__init__(message, code="COPYRIGHT_VIOLATION")

class TextNotFoundError(LogosBaseException):
    """Raised when a requested classical text or source fragment is missing."""
    def __init__(self, text_id: Any):
        message = f"Classical text with ID {text_id} could not be found in the corpus."
        super().__init__(message, code="TEXT_NOT_FOUND")

class TranslatorNotAllowedError(LogosBaseException):
    """Raised when a translator is requested that is not in the 38-member whitelist."""
    def __init__(self, name: str):
        message = f"Translator '{name}' is not authorized for use in the LOGOS production system."
        super().__init__(message, code="TRANSLATOR_UNAUTHORIZED")

class AuthenticationError(LogosBaseException):
    """Raised when authentication fails for restricted scholar features."""
    def __init__(self, message: str = "Authentication failed or token expired."):
        super().__init__(message, code="AUTH_FAILED")

async def logos_exception_handler(request: Request, exc: LogosBaseException) -> JSONResponse:
    """Global handler for LogosBaseException and its subclasses."""
    logger.error(f"LOGOS Error {exc.code}: {exc.message} at {request.url}")
    
    status_code = status.HTTP_500_INTERNAL_SERVER_ERROR
    if isinstance(exc, CopyrightViolationError):
        status_code = status.HTTP_403_FORBIDDEN
    elif isinstance(exc, TextNotFoundError):
        status_code = status.HTTP_404_NOT_FOUND
    elif isinstance(exc, (TranslatorNotAllowedError, AuthenticationError)):
        status_code = status.HTTP_401_UNAUTHORIZED
        
    return JSONResponse(
        status_code=status_code,
        content={
            "success": False,
            "error_code": exc.code,
            "message": exc.message,
            "path": request.url.path
        }
    )

async def generic_exception_handler(request: Request, exc: Exception) -> JSONResponse:
    """Global handler for unhandled generic exceptions."""
    logger.critical(f"UNHANDLED EXCEPTION: {str(exc)}", exc_info=True)
    return JSONResponse(
        status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
        content={
            "success": False,
            "error_code": "UNHANDLED_SYSTEM_ERROR",
            "message": "An unexpected error occurred in the LOGOS system. Our engineers have been notified.",
            "path": request.url.path
        }
    )

import time
import uuid
import logging
from typing import Callable, Awaitable
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.types import ASGIApp

logger = logging.getLogger(__name__)

class LogosRequestIDMiddleware(BaseHTTPMiddleware):
    """
    Middleware that assigns a unique Request ID to every incoming call.
    Essential for tracing requests through the LOGOS production pipeline.
    """
    async def dispatch(self, request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response:
        request_id = str(uuid.uuid4())
        request.state.request_id = request_id
        
        # Add request ID to logging context
        start_time = time.time()
        
        try:
            response = await call_next(request)
            process_time = (time.time() - start_time) * 1000
            response.headers["X-Request-ID"] = request_id
            response.headers["X-Process-Time-MS"] = f"{process_time:.2f}"
            
            logger.info(
                f"Request {request.method} {request.url.path} completed",
                extra={
                    "request_id": request_id,
                    "method": request.method,
                    "path": request.url.path,
                    "status_code": response.status_code,
                    "duration_ms": process_time
                }
            )
            return response
            
        except Exception as e:
            logger.error(
                f"Request {request.method} {request.url.path} failed",
                exc_info=True,
                extra={"request_id": request_id}
            )
            raise

class LogosSecurityMiddleware(BaseHTTPMiddleware):
    """
    Enforces security headers and prevents common web vulnerabilities.
    Protects the integrity of the LOGOS classical corpus data.
    """
    async def dispatch(self, request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response:
        response = await call_next(request)
        
        # Security Headers
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["X-XSS-Protection"] = "1; mode=block"
        response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
        response.headers["Content-Security-Policy"] = "default-src 'self'; script-src 'self' 'unsafe-inline'; object-src 'none';"
        response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
        
        return response

class LogosRateLimiter:
    """
    In-memory rate limiter for the LOGOS API.
    Prevents scraping of the 6.6 million source text rows.
    """
    def __init__(self, requests_per_minute: int = 100):
        self.requests_per_minute = requests_per_minute
        self.request_history: dict[str, list[float]] = {}

    def is_allowed(self, client_ip: str) -> bool:
        """Checks if a client IP has exceeded the rate limit."""
        current_time = time.time()
        if client_ip not in self.request_history:
            self.request_history[client_ip] = [current_time]
            return True
            
        # Clean old history
        self.request_history[client_ip] = [
            t for t in self.request_history[client_ip] 
            if current_time - t < 60
        ]
        
        if len(self.request_history[client_ip]) < self.requests_per_minute:
            self.request_history[client_ip].append(current_time)
            return True
            
        logger.warning(f"Rate limit exceeded for IP: {client_ip}")
        return False

class LogosPerformanceMiddleware(BaseHTTPMiddleware):
    """
    Monitors system performance and logs slow queries/responses.
    Critical for maintaining the $10M system's high-availability standards.
    """
    async def dispatch(self, request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response:
        start_time = time.perf_counter()
        
        response = await call_next(request)
        
        end_time = time.perf_counter()
        duration = (end_time - start_time) * 1000
        
        if duration > 500:  # Log responses slower than 500ms
            logger.warning(
                f"SLOW_RESPONSE: {request.method} {request.url.path} took {duration:.2f}ms",
                extra={
                    "path": request.url.path,
                    "duration_ms": duration,
                    "slow_threshold": 500
                }
            )
            
        return response

class SystemDiagnostics:

    async def check_embedding_vectors(db_conn: asyncpg.Connection) -> Dict[str, Any]:
        """Validates that word embeddings have the correct 300-dimension structure."""
        try:
            sample = await db_conn.fetchrow("SELECT word, vector FROM word_embeddings LIMIT 1")
            if not sample:
                return {"status": "empty", "message": "No embeddings found"}
                
            vector = sample['vector']
            # Assuming vector is stored as an array or binary
            dim = len(vector)
            return {
                "status": "ok",
                "dimensions": dim,
                "valid": dim == 300,
                "sample_word": sample['word']
