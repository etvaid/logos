            logger.error("Unbalanced parentheses detected in SQL query.")

            logger.error(f"Integrity check FAILED for {table_name}. Missing: {missing}")

import re
import logging

# Configure high-fidelity logging

    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'

class InjectionScanner:
    """
    Scans and sanitizes SQL queries and user inputs to prevent malicious attacks.
    """

    def __init__(self):
        # Patterns commonly used in SQL injection attacks
        self.malicious_patterns = [
            r"(--)",                     # SQL comments
            r"(/\*.*?\*/)",              # Multi-line comments
            r";",                        # Query chaining
            r"\bUNION\b",                # Union based attacks
            r"\bDROP\b",                 # Data destruction
            r"\bDELETE\b",               # Data deletion
            r"\bUPDATE\b",               # Data modification
            r"\bINSERT\b",               # Data insertion
            r"\bTRUNCATE\b",             # Table clearing
            r"\bALTER\b",                # Schema modification
            r"\bEXEC\b",                 # Command execution
            r"\bXP_CMD_SHELL\b",         # System shell access
            r"OR\s+['\"]?\d+['\"]?\s*=\s*['\"]?\d+['\"]?", # Tautology (e.g., OR 1=1)
            r"WAITFOR\s+DELAY",          # Time-based attacks
            r"SLEEP\(",                  # Time-based attacks
            r"PG_SLEEP"                  # Postgres specific time attack
        ]
        self.compiled_patterns = [re.compile(p, re.IGNORECASE) for p in self.malicious_patterns]
        logger.info("InjectionScanner initialized with security patterns.")

    def scan_string(self, input_str: str) -> Tuple[bool, List[str]]:
        """
        Scans a single string for any injection patterns.
        
        Args:
            input_str (str): The string to scan.
            
        Returns:
            Tuple[bool, List[str]]: (is_safe, found_threats)
        """
        if not input_str:
            return True, []

        found_threats = []
        for pattern in self.compiled_patterns:
            if pattern.search(input_str):
                threat = pattern.pattern
                found_threats.append(threat)
                logger.warning(f"SECURITY ALERT: Injection pattern detected: {threat}")

        is_safe = len(found_threats) == 0
        return is_safe, found_threats

    def scan_query_parameters(self, params: Dict[str, Any]) -> Tuple[bool, Dict[str, List[str]]]:
        """
        Scans all parameters in a query dictionary.
        
        Args:
            params (Dict): Dictionary of query parameters.
            
        Returns:
            Tuple[bool, Dict]: (is_safe, mapping_of_param_to_threats)
        """
        all_safe = True
        threat_map = {}

        for key, value in params.items():
            if isinstance(value, str):
                is_safe, threats = self.scan_string(value)
                if not is_safe:
                    all_safe = False
                    threat_map[key] = threats
                    logger.warning(f"Security threat in parameter '{key}': {threats}")
            elif isinstance(value, (list, tuple)):
                # Recursively check lists of strings
                for item in value:
                    if isinstance(item, str):
                        is_safe, threats = self.scan_string(item)
                        if not is_safe:
                            all_safe = False
                            threat_map[f"{key}_item"] = threats

        return all_safe, threat_map

    def sanitize_search_term(self, term: str) -> str:
        """
        Strictly sanitizes a search term for use in a LIKE clause.
        
        Args:
            term (str): The search term.
            
        Returns:
            str: The sanitized search term.
        """
        if not term:
            return ""
            
        # Remove characters that can break SQL or are used in injection
        sanitized = re.sub(r"[;'\"]", "", term)
        # Escape percentage and underscore for LIKE
        sanitized = sanitized.replace("%", "\\%").replace("_", "\\_")
        
        logger.debug(f"Term sanitized for search: {term} -> {sanitized}")
        return sanitized

    def check_tautology(self, sql: str) -> bool:
        """
        Checks for tautological expressions like '1=1'.
        """
        pattern = re.compile(r"(\w+)\s*=\s*\1", re.IGNORECASE)
        if pattern.search(sql):
            logger.warning("Potential tautology injection detected.")
            return False
        return True

    def validate_identifier(self, identifier: str) -> bool:
        """
        Validates that a table or column name identifier is alphanumeric and safe.
        """
        if not identifier:
            return False
        # Identifiers should only contain alphanumeric and underscores
        if not re.match(r"^[a-zA-Z0-9_]+$", identifier):
            logger.error(f"Invalid identifier detected: {identifier}")
            return False
        return True

    def protect_against_xss(self, content: str) -> str:
        """
        While primarily focused on SQL, LOGOS content must also be safe for rendering.
        """
        if not content:
            return ""
        # Basic HTML escaping
        safe_content = content.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
        return safe_content

    def analyze_risk_level(self, sql: str) -> str:
        """
        Categorizes the risk level of a query.
        """
        threat_count = 0
        for pattern in self.compiled_patterns:
            if pattern.search(sql):
                threat_count += 1
        
        if threat_count == 0:
            return "LOW"
        elif threat_count < 2:
            return "MEDIUM"
        else:
            return "HIGH"

    def verify_no_forbidden_keywords(self, sql: str) -> bool:
        """
        Ensures scholars are not using administrative keywords.
        """
        forbidden = ["GRANT", "REVOKE", "CREATE", "OWNER", "SUPERUSER", "REPLICATION"]
        sql_upper = sql.upper()
        for word in forbidden:
            if word in sql_upper:
                logger.error(f"Forbidden administrative keyword used: {word}")
                return False
        return True

    def check_hex_encoding(self, sql: str) -> bool:
        """
        Checks for obfuscated SQL using hex encoding.
        """
        if "0x" in sql.lower():
            logger.warning("Hex encoding detected in query, possible obfuscation attack.")
            return False
        return True

    def check_char_encoding(self, sql: str) -> bool:
        """
        Checks for obfuscated SQL using CHAR() functions.
        """
        if "CHAR(" in sql.upper() or "CHR(" in sql.upper():
            logger.warning("CHAR/CHR function detected, possible obfuscation attack.")
            return False
        return True

    def is_complex_injection_attempt(self, sql: str) -> bool:
        """
        Runs multiple checks to determine if a query is a complex injection attempt.
        """
        checks = [
            self.scan_string(sql)[0],
            self.check_tautology(sql),
            self.verify_no_forbidden_keywords(sql),
            self.check_hex_encoding(sql),
            self.check_char_encoding(sql)
        ]
        
        is_safe = all(checks)
        if not is_safe:
            logger.critical(f"Complex injection attempt blocked: {sql}")
        return is_safe

    def sanitize_translator_list(self, translators: List[str]) -> List[str]:
        """
        Ensures a list of translators only contains allowed names.
        """
        from backend.verification.pass2.query_validator import ALLOWED_TRANSLATORS
        sanitized = [t for t in translators if t in ALLOWED_TRANSLATORS]
        if len(sanitized) != len(translators):
            logger.warning(f"Some translators were removed as they are not in the allowed list.")
        return sanitized

    def generate_security_report(self, sql: str) -> Dict[str, Any]:
        """
        Generates a full security report for a given query.
        """
        is_safe, threats = self.scan_string(sql)
        risk = self.analyze_risk_level(sql)
        
        report = {
            "query_preview": sql[:100],
            "is_safe": is_safe and self.is_complex_injection_attempt(sql),
            "threats_found": threats,
            "risk_level": risk,
            "recommendation": "Block" if risk != "LOW" else "Allow"
        }
        
        logger.info(f"Security report generated. Risk: {risk}")
        return report

    async def connect(self) -> None:
        """

        if self.pool is None:

    async def get_explain_plan(self, sql: str) -> List[Dict[str, Any]]:
        """
        Runs EXPLAIN (FORMAT JSON) on the provided query.
        """
        try:
            if not self.pool: await self.connect()
            async with self.pool.acquire() as conn:
                explain_query = f"EXPLAIN (FORMAT JSON) {sql}"
                result = await conn.fetchval(explain_query)
                # asyncpg returns the result as a string for JSON format
                plan = json.loads(result)
                logger.info("Execution plan retrieved successfully.")
                return plan
        except Exception as e:
            logger.error(f"Failed to get explain plan: {str(e)}")
            return []

    async def analyze_query_cost(self, sql: str) -> Dict[str, Any]:
        """
        Extracts cost and plan information from the query.
        """
        plan_data = await self.get_explain_plan(sql)
        if not plan_data:
            return {"error": "Could not analyze query."}

        root_plan = plan_data[0]['Plan']
        total_cost = root_plan.get('Total Cost', 0)
        plan_rows = root_plan.get('Plan Rows', 0)
        
        logger.info(f"Query Cost: {total_cost}, Estimated Rows: {plan_rows}")
        
        return {
            "total_cost": total_cost,
            "estimated_rows": plan_rows,
            "node_type": root_plan.get('Node Type'),
            "is_expensive": total_cost > 10000
        }

    async def check_for_seq_scans(self, sql: str) -> bool:
        """
        Checks if the query plan involves a Sequential Scan on large tables.
        """
        plan_data = await self.get_explain_plan(sql)
        if not plan_data:
            return False

        def has_seq_scan(node):
            if node.get('Node Type') == 'Seq Scan':
                # Sequential scans on small tables are fine, but source_texts is 6M rows
                if node.get('Relation Name') in ['source_texts', 'texts']:
                    logger.warning(f"Sequential Scan detected on large table: {node.get('Relation Name')}")
                    return True
            
            for child in node.get('Plans', []):
                if has_seq_scan(child):
                    return True
            return False

        return has_seq_scan(plan_data[0]['Plan'])

    async def measure_actual_execution_time(self, sql: str) -> float:
        """
        Runs EXPLAIN ANALYZE to measure real-world performance.
        WARNING: This actually executes the query.
        """
        try:
            if not self.pool: await self.connect()
            async with self.pool.acquire() as conn:
                # We use a transaction to ensure we don't commit anything (though LOGOS is read-only for scholars)
                async with conn.transaction():
                    query = f"EXPLAIN ANALYZE {sql}"
                    rows = await conn.fetch(query)
                    # Parse the last row for execution time
                    last_row = rows[-1][0]
                    if "Execution Time" in last_row:
                        time_str = last_row.split(":")[1].strip().split(" ")[0]
                        execution_time = float(time_str)
                        logger.info(f"Actual Execution Time: {execution_time}ms")
                        return execution_time
            return 0.0
        except Exception as e:
            logger.error(f"Error measuring execution time: {str(e)}")
            return -1.0
