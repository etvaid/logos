/**
 * LOGOS - The Bible for Classical Studies
 * Database Service Layer
 * 
 * This service handles all direct interactions with the backend API,
 * which in turn queries the Railway PostgreSQL database.
 * 
 * SCHEMA REFERENCE:
 * - texts: [id, title, author, translator, text_content, book, chapter]
 * - author_profiles: [id, name, birth_year, death_year, nationality, genres]
 */

import axios, { AxiosResponse } from 'axios';

// Authorized Translator Whitelist
export const AUTHORIZED_TRANSLATORS = [
    "Jowett", "Dryden", "Dakyns", "Pope", "Murray", "Butler", "Church_Brodribb", 
    "Cowper", "Butcher_Lang", "Lang_Leaf_Myers", "Conington", "Goodwin", "Storr", 
    "Roberts", "Aubrey_Stewart", "Williams", "Dryden_et_al", "Brookes_More", 
    "Ross", "Rawlinson", "Moore", "Evelyn-White", "Morshead", "Heseltine", 
    "Crawley", "Long", "Lindsay", "Jebb", "Macaulay", "Leonard", "Adlington", 
    "Smith", "Morris", "Butcher", "Derby", "Kenyon", "Hickie", "Anonymous"
];

export interface TextRecord {
    id: number;
    title: string;
    author: string;
    translator: string;
    text_content: string;
    book?: number;
    chapter?: number;
}

export interface AuthorProfile {
    id: number;
    name: string;
    birth_year: number;
    death_year: number;
    nationality: string;
    genres: string[];
}

class DatabaseService {
    private static instance: DatabaseService;
    private readonly baseUrl: string = "/api";

    private constructor() {}

    public static getInstance(): DatabaseService {
        if (!DatabaseService.instance) {
            DatabaseService.instance = new DatabaseService();
        }
        return DatabaseService.instance;
    }

    /**
     * Fetches a specific text from the 'texts' table.
     * Includes strict validation for translator licensing.
     */
    public async getText(title: string, translator: string): Promise<TextRecord[]> {
        try {
            console.info(`LOGOS_DB_SERVICE: Querying text [${title}] by [${translator}]`);

            // Security Check: Prevent unauthorized translator access at the service level
            if (!AUTHORIZED_TRANSLATORS.includes(translator)) {
                console.error(`LOGOS_DB_SERVICE_ERR: Unauthorized translator requested: ${translator}`);
                throw new Error("LICENSING_ERROR: Translator not available in public domain.");
            }

            const response: AxiosResponse<TextRecord[]> = await axios.get(`${this.baseUrl}/texts`, {
                params: { title, translator }
            });

            if (!response.data || response.data.length === 0) {
                console.warn(`LOGOS_DB_SERVICE: No results for ${title}`);
                return [];
            }

            return response.data;
        } catch (error) {
            console.error("LOGOS_DB_SERVICE_ERR: Failed to fetch text record", error);
            throw error;
        }
    }

    /**
     * Fetches author metadata from 'author_profiles'.
     */
    public async getAuthorProfile(name: string): Promise<AuthorProfile | null> {
        try {
            console.info(`LOGOS_DB_SERVICE: Fetching profile for author: ${name}`);
            const response: AxiosResponse<AuthorProfile> = await axios.get(`${this.baseUrl}/authors/${name}`);
            
            if (!response.data) {
                return null;
            }

            return response.data;
        } catch (error) {
            console.error(`LOGOS_DB_SERVICE_ERR: Profile fetch failed for ${name}`, error);
            return null;
        }
    }

    /**
     * Performs a semantic search using word embeddings.
     */
    public async semanticSearch(word: string): Promise<any> {
        try {
            console.info(`LOGOS_DB_SERVICE: Performing vector search for: ${word}`);
            const response = await axios.get(`${this.baseUrl}/embeddings/${word}`);
            return response.data;
        } catch (error) {
            console.error("LOGOS_DB_SERVICE_ERR: Semantic search failed", error);
            throw error;
        }
    }

    /**
     * Retrieves the intertextual connectome graph.
     */
    public async getConnectomeData(): Promise<any> {
        try {
            console.info("LOGOS_DB_SERVICE: Fetching full intertextual graph.");
            const response = await axios.get(`${this.baseUrl}/graph/network`);
            return response.data;
        } catch (error) {
            console.error("LOGOS_DB_SERVICE_ERR: Connectome retrieval failed", error);
            throw error;
        }
    }

    /**
     * Logs user interactions for analytical scholarship tracking.
     */
    public async logInteraction(action: string, metadata: object): Promise<void> {
        try {
            await axios.post(`${this.baseUrl}/logs`, {
                timestamp: new Date().toISOString(),
                action,
                ...metadata
            });
        } catch (error) {
            // Non-blocking error
            console.error("LOGOS_DB_SERVICE_ERR: Failed to log interaction", error);
        }
    }
}

export const dbService = DatabaseService.getInstance();

"""
LOGOS - The Bible for Classical Studies
Production Database Query Engine (Python/asyncpg)

This module provides high-performance, asynchronous access to the 
Railway PostgreSQL database. It handles the mission-critical 
classical texts and embeddings.
"""

import asyncio
import asyncpg
import logging
import json
from typing import List, Dict, Any, Optional

# Configure production logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"

class LogosQueryEngine:
    """
    World-class query engine for the LOGOS system.
    Handles connection pooling and complex classical text queries.
    """

    def __init__(self):
        self.pool: Optional[asyncpg.Pool] = None
        self.authorized_translators = [
            "Jowett", "Dryden", "Dakyns", "Pope", "Murray", "Butler", "Church_Brodribb", 
            "Cowper", "Butcher_Lang", "Lang_Leaf_Myers", "Conington", "Goodwin", "Storr", 
            "Roberts", "Aubrey_Stewart", "Williams", "Dryden_et_al", "Brookes_More", 
            "Ross", "Rawlinson", "Moore", "Evelyn-White", "Morshead", "Heseltine", 
            "Crawley", "Long", "Lindsay", "Jebb", "Macaulay", "Leonard", "Adlington", 
            "Smith", "Morris", "Butcher", "Derby", "Kenyon", "Hickie", "Anonymous"
        ]

    async def initialize(self):
        """Initializes the database connection pool."""
        try:
            logger.info("LOGOS_DB_ENGINE: Initializing connection pool...")
            self.pool = await asyncpg.create_pool(DATABASE_URL, min_size=5, max_size=20)
            logger.info("LOGOS_DB_ENGINE: Connection pool established.")
        except Exception as e:
            logger.error(f"LOGOS_DB_ENGINE_CRITICAL: Failed to connect to Railway DB: {e}")
            raise

    async def get_text_content(self, title: str, translator: str, book: int = 1) -> List[Dict[str, Any]]:
        """
        Retrieves classical text content from the 'texts' table.
        Strictly enforces translator authorization.
        """
        if translator not in self.authorized_translators:
            logger.warning(f"LOGOS_DB_ENGINE: Unauthorized translator attempt: {translator}")
            return []

        if not self.pool:
            await self.initialize()

        try:
            async with self.pool.acquire() as connection:
                query = """
                    SELECT id, title, author, translator, text_content, book, chapter
                    FROM texts
                    WHERE title = $1 AND translator = $2 AND book = $3
                    ORDER BY chapter ASC;
                """
                rows = await connection.fetch(query, title, translator, book)
                
                results = [dict(row) for row in rows]
                logger.info(f"LOGOS_DB_ENGINE: Successfully retrieved {len(results)} rows for {title}")
                return results
        except Exception as e:
            logger.error(f"LOGOS_DB_ENGINE_ERR: Query failed for {title}: {e}")
            return []

    async def search_texts(self, search_term: str, limit: int = 50) -> List[Dict[str, Any]]:
        """
        Performs full-text search using PostgreSQL tsvector on the 'text_content' column.
        """
        if not self.pool:
            await self.initialize()

        try:
            async with self.pool.acquire() as connection:
                # Using websearch_to_tsquery for scholar-friendly search syntax
                query = """
                    SELECT id, title, author, translator, 
                           ts_headline('english', text_content, websearch_to_tsquery('english', $1)) as snippet,
                           ts_rank(to_tsvector('english', text_content), websearch_to_tsquery('english', $1)) as rank
                    FROM texts
                    WHERE to_tsvector('english', text_content) @@ websearch_to_tsquery('english', $1)
                    ORDER BY rank DESC
                    LIMIT $2;
                """
                rows = await connection.fetch(query, search_term, limit)
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"LOGOS_DB_ENGINE_ERR: Search failed for '{search_term}': {e}")
            return []

    async def get_word_vector(self, word: str) -> Optional[List[float]]:
        """
        Retrieves a 300-dimension vector from the 'word_embeddings' table.
        """
        if not self.pool:
            await self.initialize()

        try:
            async with self.pool.acquire() as connection:
                row = await connection.fetchrow(
                    "SELECT vector FROM word_embeddings WHERE word = $1", word
                )
                if row:
                    # Vectors are stored as numpy arrays in DB, cast to list for JSON
                    return list(row['vector'])
                return None
        except Exception as e:
            logger.error(f"LOGOS_DB_ENGINE_ERR: Vector retrieval failed for {word}: {e}")
            return None

    async def get_author_timeline(self) -> List[Dict[str, Any]]:
        """
        Queries 'author_profiles' for the CHRONOS visualization.
        """
        if not self.pool:
            await self.initialize()

        try:
            async with self.pool.acquire() as connection:
                query = """
                    SELECT id, name, birth_year, death_year, nationality, genres
                    FROM author_profiles
                    ORDER BY birth_year ASC;
                """
                rows = await connection.fetch(query)
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"LOGOS_DB_ENGINE_ERR: Timeline query failed: {e}")
            return []

    async def close(self):
        """Gracefully closes the connection pool."""
        if self.pool:
            await self.pool.close()
            logger.info("LOGOS_DB_ENGINE: Connection pool closed.")

# Singleton instance for production use
