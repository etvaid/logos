    async def validate_translator_integrity(self) -> Dict[str, Any]:
        """
        Queries the translator_profiles and texts tables to ensure no forbidden 
        copyrighted material exists in the production environment.
        """
        logger.info("Starting translator integrity check...")
        results = {"status": "PASS", "forbidden_found": [], "count": 0}
        
        if not self.pool:
            raise RuntimeError("Database pool not initialized")

        async with self.pool.acquire() as conn:
            try:
                # Check translator_profiles table
                rows = await conn.fetch("SELECT name FROM translator_profiles")
                for row in rows:
                    name = row['name']
                    if any(forbidden in name for forbidden in self.FORBIDDEN_TRANSLATORS):
                        results["status"] = "FAIL"
                        results["forbidden_found"].append(f"Profile: {name}")
                        self.issues.append({"severity": "CRITICAL", "msg": f"Forbidden translator {name} found in profiles."})

                # Check texts table for metadata violations
                text_rows = await conn.fetch("SELECT DISTINCT translator FROM texts")
                for row in text_rows:
                    trans = row['translator']
                    if trans and any(forbidden in trans for forbidden in self.FORBIDDEN_TRANSLATORS):
                        results["status"] = "FAIL"
                        results["forbidden_found"].append(f"Text Metadata: {trans}")
                        self.issues.append({"severity": "CRITICAL", "msg": f"Forbidden translator {trans} found in texts."})

                results["count"] = len(rows)
                logger.info(f"Translator integrity check completed. Status: {results['status']}")
                return results
            except Exception as e:
                logger.error(f"Error during translator validation: {e}")
                return {"status": "ERROR", "error": str(e)}

    async def verify_text_coverage(self) -> Dict[str, Any]:

        Ensures that every entry in 'texts' has corresponding 'source_texts' where applicable.
        """

                total_sources = await conn.fetchval("SELECT COUNT(*) FROM source_texts")
                
                # Check for orphaned texts

                    SELECT COUNT(*) FROM texts t 
                    LEFT JOIN source_texts s ON t.id = s.work_id

                return stats
            except Exception as e:

                return {"error": str(e)}

from fastapi import FastAPI, HTTPException, Depends, Query
from typing import List, Optional
import asyncpg
import logging

app = FastAPI(title="LOGOS API", version="1.0.0")
logger = logging.getLogger(__name__)

DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"

async def get_db():
    conn = await asyncpg.connect(DATABASE_URL)
    try:
        yield conn
    finally:
        await conn.close()

@app.get("/texts")
async def get_texts(
    author: Optional[str] = None, 
    translator: Optional[str] = None, 
    db: asyncpg.Connection = Depends(get_db)
):
    """Retrieves a list of texts filtered by author or translator."""
    try:
        query = "SELECT id, title, author, translator FROM texts WHERE 1=1"
        params = []
        if author:
            query += " AND author = $1"
            params.append(author)
        if translator:
            # Strict check for forbidden translators even at the API level
            if translator in ["Chapman", "Lattimore", "Fagles", "Wilson"]:
                raise HTTPException(status_code=403, detail="Copyrighted material unavailable")
            query += f" AND translator = ${len(params)+1}"
            params.append(translator)
            
        rows = await db.fetch(query, *params)
        return [dict(r) for r in rows]
    except Exception as e:
        logger.error(f"API Error in get_texts: {e}")
        raise HTTPException(status_code=500, detail="Internal Server Error")

@app.get("/texts/{text_id}")
async def get_text_detail(text_id: int, db: asyncpg.Connection = Depends(get_db)):
    """Fetches the full content of a specific translation."""
    try:
        row = await db.fetchrow("SELECT * FROM texts WHERE id = $1", text_id)
        if not row:
            raise HTTPException(status_code=404, detail="Text not found")
        return dict(row)
    except Exception as e:
        logger.error(f"API Error in get_text_detail: {e}")
        raise HTTPException(status_code=500, detail="Internal Server Error")

@app.get("/source/{work_id}")
async def get_source_text(work_id: int, db: asyncpg.Connection = Depends(get_db)):
    """Fetches the original language source text for a work."""
    try:
        rows = await db.fetch("SELECT * FROM source_texts WHERE work_id = $1 ORDER BY line_number", work_id)
        return [dict(r) for r in rows]
    except Exception as e:
        logger.error(f"API Error in get_source_text: {e}")
        raise HTTPException(status_code=500, detail="Internal Server Error")

@app.get("/search")
async def search_texts(q: str, db: asyncpg.Connection = Depends(get_db)):
    """Performs a full-text search across the corpus."""
    try:
        # Using GIN index for performance
        query = """
            SELECT id, title, author, translator, 
                   ts_headline(text_content, plainto_tsquery($1)) as snippet
            FROM texts 
            WHERE to_tsvector('english', text_content) @@ plainto_tsquery($1)
            LIMIT 20
        """
        rows = await db.fetch(query, q)
        return [dict(r) for r in rows]
    except Exception as e:
        logger.error(f"API Error in search_texts: {e}")
        raise HTTPException(status_code=500, detail="Internal Server Error")

    found = []

/**
 * LOGOS Reader Component
 * 
 * A high-performance React component for rendering classical texts
 * with side-by-side source text comparison.
 */

import React, { useState, useEffect, useCallback } from 'react';

interface TextContent {
    id: number;
    title: string;
    author: string;
    translator: string;
    text_content: string;
}

interface SourceLine {
    id: number;
    content: string;
    line_number: number;
}

const Reader: React.FC<{ textId: number }> = ({ textId }) => {
    const [text, setText] = useState<TextContent | null>(null);
    const [sources, setSources] = useState<SourceLine[]>([]);
    const [loading, setLoading] = useState<boolean>(true);
    const [error, setError] = useState<string | null>(null);
    const [fontSize, setFontSize] = useState<number>(18);

    const fetchData = useCallback(async () => {
        setLoading(true);
        setError(null);
        try {
            const textRes = await fetch(`/api/v1/texts/${textId}`);
            if (!textRes.ok) throw new Error("Failed to fetch translation");
            const textData = await textRes.json();
            
            // Security check: Ensure no forbidden translators in state
            const forbidden = ["Chapman", "Lattimore", "Fagles", "Wilson"];
            if (forbidden.includes(textData.translator)) {
                throw new Error("Copyrighted material restricted.");
            }
            
            setText(textData);

            const sourceRes = await fetch(`/api/v1/source/${textId}`);
            if (sourceRes.ok) {
                const sourceData = await sourceRes.json();
                setSources(sourceData);
            }
        } catch (err: any) {
            setError(err.message);
            console.error("Reader Error:", err);
        } finally {
            setLoading(false);
        }
    }, [textId]);

    useEffect(() => {
        fetchData();
    }, [fetchData]);

    const handleZoomIn = () => setFontSize(prev => Math.min(prev + 2, 32));
    const handleZoomOut = () => setFontSize(prev => Math.max(prev - 2, 12));

    if (loading) return <div className="p-10 text-center">Loading Classical Corpus...</div>;
    if (error) return <div className="p-10 text-red-600">Error: {error}</div>;
    if (!text) return <div className="p-10">Text not found.</div>;

    return (
        <div className="logos-reader-container bg-sepia-100 min-h-screen p-8">
            <header className="border-b border-black pb-4 mb-8">
                <h1 className="text-4xl serif font-bold">{text.title}</h1>
                <p className="text-xl italic">By {text.author} — Translated by {text.translator}</p>
                <div className="controls mt-4 space-x-4">
                    <button onClick={handleZoomIn} className="px-3 py-1 bg-white border border-black hover:bg-gray-100">A+</button>
                    <button onClick={handleZoomOut} className="px-3 py-1 bg-white border border-black hover:bg-gray-100">A-</button>
                    <button onClick={() => window.print()} className="px-3 py-1 bg-white border border-black hover:bg-gray-100">Print</button>
                </div>
            </header>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-12">
                <section className="translation-pane" style={{ fontSize: `${fontSize}px` }}>
                    <h2 className="text-sm uppercase tracking-widest mb-4 opacity-50">Translation</h2>
                    <div className="leading-relaxed whitespace-pre-wrap font-serif">
                        {text.text_content}
                    </div>
                </section>

                <section className="source-pane border-l border-gray-300 pl-12" style={{ fontSize: `${fontSize}px` }}>
                    <h2 className="text-sm uppercase tracking-widest mb-4 opacity-50">Original Source</h2>
                    <div className="leading-relaxed font-serif text-blue-900">
                        {sources.map(line => (
                            <div key={line.id} className="mb-2 flex">
                                <span className="text-xs opacity-30 w-8 inline-block select-none">{line.line_number}</span>
                                <span>{line.content}</span>
                            </div>
                        ))}
                    </div>
                </section>
            </div>

            <footer className="mt-20 pt-8 border-t border-gray-300 text-sm opacity-60">
                <p>© LOGOS - The Bible for Classical Studies. All translations are public domain.</p>
                <p>Database: Railway PostgreSQL | Model: BERT-Classical-v2</p>
            </footer>
        </div>
    );
};

export default Reader;
