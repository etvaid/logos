        try:
            if self.pool:
                await self.pool.close()
                self._is_connected = False
                logger.info("Database pool closed successfully.")
        except Exception as e:
            logger.error(f"Error during database disconnection: {str(e)}")

    async def fetch_text_by_id(self, text_id: int) -> Dict[str, Any]:
        """
        Retrieves a specific text record by its ID.
        :param text_id: The unique identifier for the text.
        :return: A dictionary containing the text metadata and content.
        """
        if not self._is_connected:
            await self.connect()
        
        try:
            async with self.pool.acquire() as conn:
                query = "SELECT id, title, author, translator, text_content, book, chapter FROM texts WHERE id = $1"
                row = await conn.fetchrow(query, text_id)
                if row:
                    logger.info(f"Successfully fetched text ID {text_id}: {row['title']}")
                    return dict(row)
                logger.warning(f"Text ID {text_id} not found in database.")
                return {}
        except Exception as e:
            logger.error(f"Error fetching text {text_id}: {str(e)}")
            return {}

    async def search_corpus(self, search_term: str, limit: int = 50) -> List[Dict[str, Any]]:
        """
        Performs a full-text search across the classical corpus.
        :param search_term: The string to search for.
        :param limit: Maximum number of results to return.
        """
        if not self._is_connected:
            await self.connect()

        try:
            async with self.pool.acquire() as conn:
                query = """
                    SELECT id, title, author, text_content 
                    FROM texts 
                    WHERE text_content ILIKE $1 
                    LIMIT $2
                """
                rows = await conn.fetch(query, f"%{search_term}%", limit)
                logger.info(f"Search for '{search_term}' returned {len(rows)} results.")
                return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Search failed for term '{search_term}': {str(e)}")
            return []

    async def get_translator_works(self, translator_name: str) -> List[Dict[str, Any]]:
        """
        Fetches all works associated with a specific authorized translator.
        Authorized translators include: Jowett, Dryden, Pope, etc.

        valid_translators = [

        ]

    async def get_semantic_neighbors(self, word: str, threshold: float = 0.7) -> List[Dict[str, Any]]:
        """
        Calculates semantic similarity between words using vector embeddings.
        Uses 300-dimension numpy arrays stored in the word_embeddings table.
        """
        try:
            async with self.pool.acquire() as conn:
                # Fetch target vector
                target_row = await conn.fetchrow("SELECT vector FROM word_embeddings WHERE word = $1", word)
                if not target_row:
                    return []
                
                target_vec = np.frombuffer(target_row['vector'], dtype=np.float32)
                
                # In a real production environment, we'd use pgvector. 
                # Here we simulate the logic for the E2E test state.
                query = "SELECT word, vector FROM word_embeddings LIMIT 1000"
                all_words = await conn.fetch(query)
                
                results = []
                for row in all_words:
                    vec = np.frombuffer(row['vector'], dtype=np.float32)
                    similarity = np.dot(target_vec, vec) / (np.linalg.norm(target_vec) * np.linalg.norm(vec))
                    if similarity >= threshold:
                        results.append({"word": row['word'], "similarity": float(similarity)})
                
                return sorted(results, key=lambda x: x['similarity'], reverse=True)
        except Exception as e:
            logger.error(f"Semantic search failed for word '{word}': {str(e)}")
            return []

    async def get_source_text_lines(self, work_id: int, start_line: int, end_line: int) -> List[Dict[str, Any]]:

                query = """
                    SELECT name, birth_year, death_year, nationality 
                    FROM author_profiles 
                    WHERE birth_year >= $1 AND death_year <= $2

                """

    async def get_corpus_statistics(self) -> Dict[str, Any]:
        """Returns metadata about the database size and content distribution."""
        try:
            async with self.pool.acquire() as conn:
                text_count = await conn.fetchval("SELECT COUNT(*) FROM texts")
                source_count = await conn.fetchval("SELECT COUNT(*) FROM source_texts")

        try:

                return dict(row) if row else {}
