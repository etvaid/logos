        return health_report

    async def check_translator_profile_consistency(self) -> List[str]:
        """
        Ensures that every translator mentioned in 'texts' has a corresponding entry 
        in 'translator_profiles'.
        """
        if not self.pool:
            await self.connect()
            
        missing_profiles = []
        try:
            async with self.pool.acquire() as conn:
                logger.info("Checking consistency between 'texts' and 'translator_profiles'...")
                
                query = """
                    SELECT DISTINCT t.translator 
                    FROM texts t
                    LEFT JOIN translator_profiles tp ON t.translator = tp.name
                    WHERE tp.name IS NULL AND t.translator IS NOT NULL
                """
                results = await conn.fetch(query)
                
                for row in results:
                    missing_profiles.append(row['translator'])
                    logger.warning(f"Missing translator profile for: {row['translator']}")
                    
                return missing_profiles
        except Exception as e:
            logger.error(f"Error checking translator profile consistency: {str(e)}")
            return ["ERROR: " + str(e)]

    async def run_full_diagnostic_suite(self) -> str:
        """
        Executes the full suite of diagnostic tests and returns a JSON summary.
        This is the primary entry point for the production health check script.
        """
        try:
            await self.connect()
            
            health = await self.get_system_health_metrics()
            consistency = await self.check_translator_profile_consistency()
            
            report = {
                "system": "LOGOS Classical Bible",
                "environment": "Production",
                "health_metrics": health,
                "missing_translator_profiles": consistency,
                "timestamp": "2023-10-27T10:00:00Z"
            }
            
            final_json = json.dumps(report, indent=4)
            logger.info("Full diagnostic suite completed successfully.")
            return final_json
            
        except Exception as e:
            logger.error(f"Fatal error in diagnostic suite: {str(e)}")
            return json.dumps({"status": "CRITICAL_FAILURE", "error": str(e)})
        finally:
            await self.disconnect()

    async def validate_line_number_sequence(self, work_id: int) -> bool:
        """
        Validates that the line_number sequence in source_texts for a specific work
        is continuous and has no gaps.
        """
        if not self.pool:
            await self.connect()
            
        try:
            async with self.pool.acquire() as conn:
                logger.info(f"Validating line sequence for work_id: {work_id}")
                
                lines = await conn.fetch(
                    "SELECT line_number FROM source_texts WHERE work_id = $1 ORDER BY line_number",
                    work_id
                )
                
                if not lines:
                    logger.warning(f"No lines found for work_id: {work_id}")
                    return True
                
                expected = 1
                for row in lines:
                    if row['line_number'] != expected:
                        logger.error(f"Line gap detected in work {work_id}: Expected {expected}, got {row['line_number']}")
                        return False
                    expected += 1
                    
                return True
        except Exception as e:
            logger.error(f"Error validating line sequence: {str(e)}")
            return False

    async def audit_text_lengths(self) -> Dict[str, int]:
        """
        Provides a breakdown of text lengths across the database to identify 
        potentially truncated entries.
        """
        if not self.pool:
            await self.connect()
            
        distribution = {"short": 0, "medium": 0, "long": 0, "total": 0}
        
        try:
            async with self.pool.acquire() as conn:
                logger.info("Auditing text content length distribution...")
                
                rows = await conn.fetch("SELECT LENGTH(text_content) as len FROM texts")
                
                for row in rows:
                    length = row['len'] or 0
                    distribution["total"] += 1
                    if length < 500:
                        distribution["short"] += 1
                    elif length < 5000:
                        distribution["medium"] += 1
                    else:
                        distribution["long"] += 1
                        
                logger.info(f"Text length audit complete: {distribution}")
                return distribution
        except Exception as e:
            logger.error(f"Error auditing text lengths: {str(e)}")
            return {"error": 0}

    async def verify_language_tags(self) -> Dict[str, List[str]]:
        """
        Verifies that all language tags in source_texts are valid ISO codes 
        supported by the LOGOS search engine.
        """
        if not self.pool:
            await self.connect()
            
        valid_codes = {"grc", "lat", "eng", "heb"}
        violations = {"invalid_tags": []}
        
        try:
            async with self.pool.acquire() as conn:
                logger.info("Verifying language tags in source_texts...")
                
                tags = await conn.fetch("SELECT DISTINCT language FROM source_texts")
                for row in tags:
                    lang = row['language']
                    if lang not in valid_codes:
                        violations["invalid_tags"].append(lang)
                        logger.error(f"Invalid language tag found: {lang}")
                        
                return violations
        except Exception as e:
            logger.error(f"Error verifying language tags: {str(e)}")
            return {"error": [str(e)]}

    async def check_duplicate_entries(self) -> Dict[str, int]:
        """
        Identifies duplicate titles or content in the 'texts' table to prevent
        redundant search results.
        """
        if not self.pool:
            await self.connect()
            
        duplicates = {}
        try:
            async with self.pool.acquire() as conn:
                logger.info("Checking for duplicate entries in 'texts' table...")
                
                query = """
                    SELECT title, COUNT(*) 
                    FROM texts 
                    GROUP BY title 
                    HAVING COUNT(*) > 1
                """
                results = await conn.fetch(query)
                
                for row in results:
                    duplicates[row['title']] = row['count']
                    logger.warning(f"Duplicate found: '{row['title']}' appears {row['count']} times.")
                    
                return duplicates
        except Exception as e:
            logger.error(f"Error checking for duplicates: {str(e)}")
            return {"error": 0}

    async def analyze_embedding_coverage(self) -> float:
        """
        Calculates the percentage of unique words in 'texts' that have corresponding
        vectors in 'word_embeddings'.
        """
        if not self.pool:
            await self.connect()
            
        try:
            async with self.pool.acquire() as conn:
                logger.info("Analyzing embedding coverage across corpus...")
                
                # This is an expensive query in production, usually run on a read-replica
                coverage_query = """
                    WITH corpus_words AS (
                        SELECT DISTINCT lower(regexp_split_to_table(text_content, '\s+')) as word
                        FROM texts
                        LIMIT 10000
                    )
                    SELECT 
                        (SELECT COUNT(*) FROM corpus_words cw JOIN word_embeddings we ON cw.word = we.word)::float / 
                        (SELECT COUNT(*) FROM corpus_words)::float as coverage
                """
                result = await conn.fetchval(coverage_query)
                
                coverage_pct = (result or 0.0) * 100
                logger.info(f"Embedding coverage analysis: {coverage_pct:.2f}%")
                return coverage_pct
        except Exception as e:
            logger.error(f"Error analyzing embedding coverage: {str(e)}")
            return 0.0
