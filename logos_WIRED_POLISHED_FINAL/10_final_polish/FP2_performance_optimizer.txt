class LogosDistributedCache:
    """
    A high-performance, multi-level caching system designed for the LOGOS Classical Studies platform.
    This system implements an in-memory LRU cache backed by a persistent PostgreSQL store 
    to handle the high-volume queries of classical scholars.
    """

    def __init__(self, dsn: str = DATABASE_URL):
        self.dsn = dsn
        self._memory_cache: Dict[str, Dict[str, Any]] = {}
        self._pool: Optional[asyncpg.Pool] = None
        self._last_cleanup = time.time()
        logger.info("Initializing LogosDistributedCache with persistent PostgreSQL backend.")

    async def _get_pool(self) -> asyncpg.Pool:
        """
        Ensures a database connection pool is available for cache operations.
        Uses the Railway PostgreSQL instance for persistence.
        """
        if self._pool is None:
            try:
                self._pool = await asyncpg.create_pool(
                    dsn=self.dsn,
                    min_size=5,
                    max_size=20,
                    command_timeout=30.0
                )
                logger.info("Successfully established connection pool for LogosDistributedCache.")
                await self._ensure_cache_schema()
            except Exception as e:
                logger.error(f"Failed to connect to Railway PostgreSQL: {str(e)}")
                raise ConnectionError(f"Critical failure in LOGOS cache layer: {str(e)}")
        return self._pool

    async def _ensure_cache_schema(self):
        """
        Creates the necessary schema for persistent caching if it does not exist.
        This table stores serialized query results for classical texts.
        """
        query = """
        CREATE TABLE IF NOT EXISTS logos_query_cache (
            cache_key TEXT PRIMARY KEY,
            cache_value JSONB NOT NULL,
            expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
            created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
            hit_count INTEGER DEFAULT 0
        );
        CREATE INDEX IF NOT EXISTS idx_cache_expiry ON logos_query_cache (expires_at);
        """
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            async with conn.transaction():
                await conn.execute(query)
                logger.info("LOGOS cache schema verified and ready.")

    def _generate_key(self, func_name: str, args: tuple, kwargs: dict) -> str:
        """
        Generates a unique deterministic hash for a function call and its arguments.
        Essential for caching complex queries for authors like Homer or Plato.
        """
        serialized_args = json.dumps(args, sort_keys=True, default=str)
        serialized_kwargs = json.dumps(kwargs, sort_keys=True, default=str)
        raw_key = f"{func_name}:{serialized_args}:{serialized_kwargs}"
        return hashlib.sha256(raw_key.encode()).hexdigest()

    async def get(self, key: str) -> Optional[Any]:
        """
        Retrieves an item from the cache. Checks memory first, then PostgreSQL.
        """
        # Level 1: In-Memory Cache
        if key in self._memory_cache:
            entry = self._memory_cache[key]
            if entry['expires_at'] > time.time():
                logger.debug(f"Memory cache hit for key: {key}")
                return entry['value']
            else:
                del self._memory_cache[key]

        # Level 2: Persistent PostgreSQL Cache
        pool = await self._get_pool()
        try:
            async with pool.acquire() as conn:
                row = await conn.fetchrow(
                    "SELECT cache_value, expires_at FROM logos_query_cache WHERE cache_key = $1",
                    key
                )
                if row:
                    if row['expires_at'] > datetime.now(row['expires_at'].tzinfo):
                        # Update hit count for analytics
                        await conn.execute(
                            "UPDATE logos_query_cache SET hit_count = hit_count + 1 WHERE cache_key = $1",
                            key
                        )
                        value = json.loads(row['cache_value'])
                        # Backfill memory cache
                        self._memory_cache[key] = {
                            'value': value,
                            'expires_at': time.time() + DEFAULT_TTL
                        }
                        return value
                    else:
                        await conn.execute("DELETE FROM logos_query_cache WHERE cache_key = $1", key)
        except Exception as e:
            logger.error(f"Error retrieving from persistent cache: {str(e)}")
        
        return None

    async def set(self, key: str, value: Any, ttl: int = DEFAULT_TTL):
        """
        Stores an item in both memory and persistent PostgreSQL cache.
        """
        expiry_timestamp = time.time() + ttl
        expiry_datetime = datetime.now() + timedelta(seconds=ttl)

        # Level 1: Memory Update
        self._memory_cache[key] = {
            'value': value,
            'expires_at': expiry_timestamp
        }

        # Level 2: Persistent Update
        pool = await self._get_pool()
        try:
            async with pool.acquire() as conn:
                await conn.execute(
                    """
                    INSERT INTO logos_query_cache (cache_key, cache_value, expires_at)
                    VALUES ($1, $2, $3)
                    ON CONFLICT (cache_key) DO UPDATE
                    SET cache_value = EXCLUDED.cache_value,
                        expires_at = EXCLUDED.expires_at,
                        hit_count = logos_query_cache.hit_count + 1
                    """,
                    key, json.dumps(value), expiry_datetime
                )
        except Exception as e:
            logger.error(f"Error saving to persistent cache: {str(e)}")

        # Maintenance: LRU-style cleanup if memory cache grows too large
        if len(self._memory_cache) > MAX_MEMORY_CACHE_SIZE:
            self._evict_oldest()

    def _evict_oldest(self):
        """Removes the oldest 10% of items from memory cache."""
        sorted_keys = sorted(self._memory_cache.keys(), key=lambda k: self._memory_cache[k]['expires_at'])
        evict_count = int(MAX_MEMORY_CACHE_SIZE * 0.1)
        for i in range(evict_count):
            if i < len(sorted_keys):
                del self._memory_cache[sorted_keys[i]]
        logger.info(f"Evicted {evict_count} items from LOGOS memory cache.")

    async def invalidate(self, key: str):
        """Removes a specific key from all cache levels."""
        if key in self._memory_cache:
            del self._memory_cache[key]
        
        pool = await self._get_pool()
        try:
            async with pool.acquire() as conn:
                await conn.execute("DELETE FROM logos_query_cache WHERE cache_key = $1", key)
        except Exception as e:
            logger.error(f"Error invalidating cache key {key}: {str(e)}")

    async def vacuum(self):
        """Cleans up expired entries from the persistent database."""
        pool = await self._get_pool()
        try:
            async with pool.acquire() as conn:
                deleted = await conn.execute(
                    "DELETE FROM logos_query_cache WHERE expires_at < CURRENT_TIMESTAMP"
                )
                logger.info(f"Cache vacuum complete. Removed expired entries: {deleted}")
        except Exception as e:
            logger.error(f"Error during cache vacuum: {str(e)}")

# Global instance for the decorator
logos_cache_manager = LogosDistributedCache()

def logos_cache(ttl: int = DEFAULT_TTL):
    """
    Production-grade decorator for caching LOGOS database queries.
    Usage:
        @logos_cache(ttl=3600)
        async def get_text_by_translator(translator_name: str):
            ...
    """
    def decorator(func: T) -> T:
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            key = logos_cache_manager._generate_key(func.__name__, args, kwargs)
            
            # Try to get from cache
            cached_result = await logos_cache_manager.get(key)
            if cached_result is not None:
                return cached_result

            # Execute the actual function
            try:
                result = await func(*args, **kwargs)
            except Exception as e:
                logger.error(f"Function {func.__name__} failed: {str(e)}")
                raise

            # Store result in cache
            if result is not None:
                await logos_cache_manager.set(key, result, ttl)
            
            return result
        return cast(T, wrapper)
    return decorator

class CacheHealthMonitor:
    """Monitoring utility for LOGOS cache performance."""
    
    @staticmethod
    async def get_stats() -> Dict[str, Any]:
        """Returns statistics about the current cache state."""
        pool = await logos_cache_manager._get_pool()
        try:
            async with pool.acquire() as conn:
                total_entries = await conn.fetchval("SELECT count(*) FROM logos_query_cache")
                avg_hits = await conn.fetchval("SELECT AVG(hit_count) FROM logos_query_cache")
                top_keys = await conn.fetch("SELECT cache_key, hit_count FROM logos_query_cache ORDER BY hit_count DESC LIMIT 5")
                
                return {
                    "total_persistent_entries": total_entries,
                    "memory_cache_size": len(logos_cache_manager._memory_cache),
                    "average_hit_rate": float(avg_hits or 0),
                    "top_keys": [dict(k) for k in top_keys],
                    "status": "HEALTHY" if total_entries < 1000000 else "NEEDS_VACUUM"
                }
        except Exception as e:
            logger.error(f"Failed to fetch cache stats: {str(e)}")
            return {"status": "ERROR", "message": str(e)}

class LogosCompressionEngine:
    """
    Specialized compression utility for the LOGOS Classical Studies system.
    Handles high-density text data from source_texts (6.6 million rows).
    Optimized for Greek and Latin character sets.
    """

    def __init__(self, compression_level: int = 6):
        self.compression_level = compression_level
        self.min_compression_threshold = 1024  # Don't compress items smaller than 1KB
        logger.info(f"LogosCompressionEngine initialized with level {compression_level}")

    def compress_text(self, text: str) -> str:
        """
        Compresses a string (e.g., a chapter of the Iliad) and returns a base64 encoded string.
        Includes metadata for verification.
        """
        if not text or len(text) < self.min_compression_threshold:
            return text

        try:
            start_time = datetime.now()
            byte_data = text.encode('utf-8')
            original_size = len(byte_data)
            
            compressed_data = zlib.compress(byte_data, self.compression_level)
            encoded_data = base64.b64encode(compressed_data).decode('utf-8')
            
            final_size = len(encoded_data)
            ratio = (1 - (final_size / original_size)) * 100
            
            logger.debug(f"Compressed text: {original_size} -> {final_size} bytes ({ratio:.2f}% reduction)")
            return f"LOGOS_ZLIB_v1:{encoded_data}"
        except Exception as e:
            logger.error(f"Compression failed: {str(e)}")
            return text

    def decompress_text(self, compressed_text: str) -> str:
        """
        Decompresses a string previously processed by compress_text.
        """
        if not compressed_text or not compressed_text.startswith("LOGOS_ZLIB_v1:"):
            return compressed_text

        try:
            actual_data = compressed_text.split("LOGOS_ZLIB_v1:")[1]
            decoded_data = base64.b64decode(actual_data)
            decompressed_bytes = zlib.decompress(decoded_data)
            return decompressed_bytes.decode('utf-8')
        except Exception as e:
            logger.error(f"Decompression failed: {str(e)}")
            return compressed_text

    def compress_json(self, data: Union[Dict, List]) -> str:
        """
        Serializes and compresses complex data structures like word embeddings or author profiles.
        """
        try:
            json_str = json.dumps(data)
            return self.compress_text(json_str)
        except Exception as e:
            logger.error(f"JSON compression failed: {str(e)}")
            return ""

    def decompress_json(self, compressed_json: str) -> Union[Dict, List]:
        """
        Decompresses and deserializes JSON data.
        """
        try:
            decompressed_str = self.decompress_text(compressed_json)
            return json.loads(decompressed_str)
        except Exception as e:
            logger.error(f"JSON decompression failed: {str(e)}")
            return {}

    def get_compression_stats(self, original: str, compressed: str) -> Dict[str, Any]:
        """
        Calculates efficiency metrics for the compression operation.
        """
        orig_size = len(original.encode('utf-8'))
        comp_size = len(compressed.encode('utf-8'))
        return {
            "original_bytes": orig_size,
            "compressed_bytes": comp_size,
            "reduction_percentage": round((1 - (comp_size / orig_size)) * 100, 2) if orig_size > 0 else 0,
            "algorithm": "zlib/base64" if compressed.startswith("LOGOS_ZLIB") else "none"
        }

    def batch_compress_texts(self, texts: List[str]) -> List[str]:
        """
        Processes a list of texts (e.g., multiple source lines) in batch.
        """
        return [self.compress_text(t) for t in texts]

    def stream_compress_to_file(self, text_iterator, output_path: str):
        """
        Compresses a large stream of text (like an entire author's corpus) directly to a Gzip file.
        """
        try:
            with gzip.open(output_path, 'wt', encoding='utf-8', compresslevel=self.compression_level) as f:
                for chunk in text_iterator:
                    f.write(chunk)
            logger.info(f"Successfully streamed compressed data to {output_path}")
        except Exception as e:
            logger.error(f"Stream compression failed: {str(e)}")
            raise

class LogosPayloadOptimizer:
    """
    Middleware-level optimizer to reduce network overhead for the LOGOS API.
    """
    
    def __init__(self):
        self.engine = LogosCompressionEngine()

    def optimize_response(self, response_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        Analyzes a response payload and selectively compresses large fields.
        Typically used for 'text_content' in the 'texts' table.
        """
        if "text_content" in response_data and len(response_data["text_content"]) > 5000:
            response_data["text_content"] = self.engine.compress_text(response_data["text_content"])
            response_data["is_compressed"] = True
            response_data["compression_algo"] = "zlib_v1"
        
        return response_data

    def dehydrate_source_texts(self, source_rows: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        """
        Compresses the 'content' field in a list of source_texts rows.
        """
        for row in source_rows:
            if "content" in row and len(row["content"]) > 100:
                row["content"] = self.engine.compress_text(row["content"])
        return source_rows

interface IntersectionObserverOptions {
  root?: Element | null;
  rootMargin?: string;
  threshold?: number | number[];
}

/**
 * Custom hook to detect when an element enters the viewport.
 * Used for lazy loading text chunks and manuscript images.
 */
export const useIntersectionObserver = (
  elementRef: React.RefObject<Element>,
  options: IntersectionObserverOptions = { threshold: 0.1, rootMargin: '50px' }
): boolean => {
  const [isVisible, setIsVisible] = useState<boolean>(false);

  useEffect(() => {
    const element = elementRef.current;
    if (!element) return;

    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        setIsVisible(true);
        // Once visible, we can stop observing if we only want to load once
        observer.unobserve(element);
      }
    }, options);

    observer.observe(element);

    return () => {
      if (element) observer.unobserve(element);
    };
  }, [elementRef, options]);

  return isVisible;
};

/**
 * Advanced hook for prefetching classical texts based on scroll velocity and direction.
 */
export const useTextPrefetcher = (
  currentChapter: number,
  totalChapters: number,
  onPrefetch: (chapter: number) => Promise<void>
) => {
  const prefetchedChapters = useRef<Set<number>>(new Set());

  const handleScroll = useCallback(() => {
    // Logic to determine if user is approaching the end of the current chapter
    const scrollPosition = window.innerHeight + window.scrollY;
    const scrollThreshold = document.body.offsetHeight - 1000;

    if (scrollPosition > scrollThreshold && currentChapter < totalChapters) {
      const nextChapter = currentChapter + 1;
      if (!prefetchedChapters.current.has(nextChapter)) {
        prefetchedChapters.current.add(nextChapter);
        console.log(`LOGOS: Prefetching chapter ${nextChapter} for performance.`);
        onPrefetch(nextChapter).catch(err => {
          console.error(`LOGOS: Failed to prefetch chapter ${nextChapter}`, err);
          prefetchedChapters.current.delete(nextChapter);
        });
      }
    }
  }, [currentChapter, totalChapters, onPrefetch]);

  useEffect(() => {
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, [handleScroll]);
};

/**
 * Component to handle large lists of source texts (6.6M rows potential).
 * Implements windowing logic.
 */
interface VirtualScrollListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  itemHeight: number;
  containerHeight: number;
}

export function VirtualScrollList<T>({ 
  items, 
  renderItem, 
  itemHeight, 
  containerHeight 
}: VirtualScrollListProps<T>) {
  const [scrollTop, setScrollTop] = useState(0);

  const startIndex = Math.floor(scrollTop / itemHeight);
  const endIndex = Math.min(
    items.length - 1,
    Math.floor((scrollTop + containerHeight) / itemHeight)
  );

  const visibleItems = items.slice(startIndex, endIndex + 1).map((item, index) => ({
    data: item,
    index: startIndex + index,
    top: (startIndex + index) * itemHeight
  }));

  const onScroll = (e: React.UIEvent<HTMLDivElement>) => {
    setScrollTop(e.currentTarget.scrollTop);
  };

  return (
    <div 
      className="virtual-scroll-container overflow-auto relative" 
      style={{ height: containerHeight }}
      onScroll={onScroll}
    >
      <div 
        className="virtual-scroll-spacer w-full" 
        style={{ height: items.length * itemHeight }}
      >
        {visibleItems.map(item => (
          <div
            key={item.index}
            className="absolute w-full"
            style={{ top: item.top, height: itemHeight }}
          >
            {renderItem(item.data, item.index)}
          </div>
        ))}
      </div>
    </div>
  );
}

interface ImageDimensions {
    width: number;
    height: number;
  }
  
  interface OptimizationOptions {
    quality?: number;
    format?: 'webp' | 'jpeg' | 'png';
    width?: number;
    height?: number;
    fit?: 'cover' | 'contain' | 'inside';
  }
  
  class LogosImageOptimizer {
    private static CDN_BASE_URL = "https://cdn.logos-classical.org/images";
    private static DEFAULT_QUALITY = 85;
  
    /**
     * Generates an optimized URL for a manuscript image based on device capabilities.
     * Supports WebP for modern browsers used by scholars.
     */
    public static getOptimizedUrl(
      imagePath: string, 
      options: OptimizationOptions = {}
    ): string {
      const { 
        quality = this.DEFAULT_QUALITY, 
        format = 'webp', 
        width, 
        height, 
        fit = 'cover' 
      } = options;
  
      const params = new URLSearchParams();
      params.append('q', quality.toString());
      params.append('fm', format);
      params.append('fit', fit);
      if (width) params.append('w', width.toString());
      if (height) params.append('h', height.toString());
  
      // Ensure we don't include leading slashes
      const cleanPath = imagePath.startsWith('/') ? imagePath.substring(1) : imagePath;
      
      return `${this.CDN_BASE_URL}/${cleanPath}?${params.toString()}`;
    }
  
    /**
     * Generates a responsive srcset for <img> tags to ensure high DPI displays 
     * (like Retina iPads used by scholars) get sharp text scans.
     */
    public static generateSrcSet(imagePath: string, widths: number[] = [400, 800, 1200, 1600]): string {
      return widths
        .map(w => `${this.getOptimizedUrl(imagePath, { width: w })} ${w}w`)
        .join(', ');
    }
  
    /**
     * Estimates the best dimensions for a manuscript scan to maintain readability 
     * while minimizing bandwidth for mobile users.
     */
    public static calculateReadabilityDimensions(
      originalWidth: number, 
      originalHeight: number, 
      targetWidth: number
    ): ImageDimensions {
      const aspectRatio = originalHeight / originalWidth;
      return {
        width: targetWidth,
        height: Math.round(targetWidth * aspectRatio)
      };
    }
  
    /**
     * Preloads critical images like author profiles for Homer, Plato, and Virgil.
     */
    public static preloadCriticalImages(imagePaths: string[]): void {
      if (typeof window === 'undefined') return;
      
      imagePaths.forEach(path => {
        const img = new Image();
        img.src = this.getOptimizedUrl(path, { width: 800, quality: 90 });
      });
      console.log(`LOGOS: Preloaded ${imagePaths.length} critical assets.`);
    }
  
    /**
     * Validates if an image URL is from an authorized source to prevent XSS.
     */
    public static isAuthorizedSource(url: string): boolean {
      try {
        const parsedUrl = new URL(url);
        return parsedUrl.hostname === 'cdn.logos-classical.org' || 
               parsedUrl.hostname === 'metro.proxy.rlwy.net';
      } catch {
        return false;
      }
    }
  
    /**
     * Logic for processing client-side image manipulations for text overlays.
     */
    public static async getAverageColor(imageUrl: string): Promise<string> {
      return new Promise((resolve, reject) => {
        const img = new Image();
        img.crossOrigin = "Anonymous";
        img.onload = () => {
          const canvas = document.createElement('canvas');
          const ctx = canvas.getContext('2d');
          if (!ctx) {
            reject("Canvas context unavailable");
            return;
          }
          canvas.width = 1;
          canvas.height = 1;
          ctx.drawImage(img, 0, 0, 1, 1);
          const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data;
          resolve(`rgb(${r}, ${g}, ${b})`);
        };
        img.onerror = () => reject("Failed to load image for color analysis");
        img.src = imageUrl;
      });
    }
  }
  
  export default LogosImageOptimizer;

import * as fs from 'fs';

/**
 * LOGOS Production Build Analyzer
 * This script runs post-build to ensure our classical studies platform 
 * remains lightweight and fast for global scholars on limited connections.
 */

interface BundleStats {
  fileName: string;
  sizeKb: number;
  gzipSizeKb: number;
  type: 'script' | 'style' | 'asset';
}

class BundleAnalyzer {
  private buildDir: string;
  private thresholds: Record<string, number> = {
    'main.js': 250, // Max 250KB for main bundle
    'vendor.js': 500, // Max 500KB for dependencies
    'styles.css': 100
  };

  constructor(buildDir: string) {
    this.buildDir = path.resolve(process.cwd(), buildDir);
    console.log(`LOGOS Analyzer: Inspecting build at ${this.buildDir}`);
  }

  /**
   * Scans the build directory and calculates sizes.
   */
  public analyze(): BundleStats[] {
    if (!fs.existsSync(this.buildDir)) {
      throw new Error(`Build directory ${this.buildDir} does not exist. Run build first.`);
    }

    const files = this.getFilesRecursive(this.buildDir);
    const stats: BundleStats[] = [];

    files.forEach(file => {
      const ext = path.extname(file);
      if (['.js', '.css', '.svg', '.webp'].includes(ext)) {
        const content = fs.readFileSync(file);
        const size = fs.statSync(file).size;
        
        // Mocking gzip size for analysis (in real prod, use zlib)
        const gzipSize = Math.round(size * 0.3); 

        stats.push({
          fileName: path.basename(file),
          sizeKb: parseFloat((size / 1024).toFixed(2)),
          gzipSizeKb: parseFloat((gzipSize / 1024).toFixed(2)),
          type: ext === '.js' ? 'script' : ext === '.css' ? 'style' : 'asset'
        });
      }
    });

    return stats;
  }

  private getFilesRecursive(dir: string): string[] {
    let results: string[] = [];
    const list = fs.readdirSync(dir);
    list.forEach(file => {
      file = path.join(dir, file);
      const stat = fs.statSync(file);
      if (stat && stat.isDirectory()) {
        results = results.concat(this.getFilesRecursive(file));
      } else {
        results.push(file);
      }
    });
    return results;
  }

  /**
   * Validates bundle sizes against strict production thresholds.
   * If any bundle exceeds limits, it triggers a warning for the engineering team.
   */
  public validate(stats: BundleStats[]): boolean {
    let passed = true;
    console.log("\n--- LOGOS Bundle Size Report ---");
    
    stats.forEach(stat => {
      const threshold = this.thresholds[stat.fileName] || 1000; // Default 1MB
      const status = stat.sizeKb > threshold ? 'FAILED' : 'PASSED';
      
      if (status === 'FAILED') passed = false;

      console.log(
        `${stat.fileName.padEnd(30)} | ${stat.sizeKb.toString().padStart(8)} KB | ${status}`
      );
    });

    if (!passed) {
      console.warn("\nCRITICAL: One or more bundles exceed the LOGOS performance budget!");
    } else {
      console.log("\nSUCCESS: All bundles are within performance budgets.");
    }

    return passed;
  }

  /**
   * Generates a JSON report for CI/CD integration.
   */
  public generateReport(stats: BundleStats[]) {
    const reportPath = path.join(process.cwd(), 'bundle-report.json');
    const report = {
      timestamp: new Date().toISOString(),
      environment: 'production',
      totalSizeKb: stats.reduce((acc, s) => acc + s.sizeKb, 0),
      files: stats
    };

    fs.writeFileSync(reportPath, JSON.stringify(report, null, 2));
    console.log(`Report saved to ${reportPath}`);
  }

  /**
   * Checks for duplicate dependencies in the bundle.
   */
  public checkDuplicates() {
    console.log("LOGOS: Checking for duplicate dependencies in node_modules...");
    try {
      const output = execSync('npm dedupe --dry-run').toString();
      console.log(output);
    } catch (e) {
      console.error("Failed to run dependency check", e);
    }
  }
}

class LogosCacheManagerExtended(LogosDistributedCache):
    """
    Extended cache manager with advanced invalidation and warmup strategies.
    Specifically tuned for the 38 authorized translators (Jowett, Dryden, etc).
    """

    async def warmup_top_translators(self):
        """
        Pre-caches the most frequently accessed translators to ensure zero-latency for scholars.
        """
        top_translators = ["Jowett", "Dryden", "Pope", "Murray", "Butler"]
        logger.info(f"Starting cache warmup for translators: {top_translators}")
        
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            for translator in top_translators:
                # Fetch first 50 texts for each top translator
                texts = await conn.fetch(
                    "SELECT * FROM texts WHERE translator = $1 LIMIT 50",
                    translator
                )
                key = self._generate_key("get_texts_by_translator", (translator,), {"limit": 50})
                await self.set(key, [dict(t) for t in texts], ttl=86400) # 24h cache
        logger.info("Cache warmup complete.")

    async def invalidate_by_pattern(self, pattern: str):
        """
        Invalidates all cache keys matching a specific pattern (e.g., all 'Homer' queries).
        """
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            # Persistent layer
            deleted = await conn.execute(
                "DELETE FROM logos_query_cache WHERE cache_key LIKE $1",
                f"%{pattern}%"
            )
            # Memory layer
            keys_to_del = [k for k in self._memory_cache.keys() if pattern in k]
            for k in keys_to_del:
                del self._memory_cache[k]
                
            logger.info(f"Pattern invalidation for '{pattern}' removed {deleted} entries.")

    async def get_cache_efficiency(self) -> float:
        """Calculates the hit/miss ratio for the current session."""
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            total_hits = await conn.fetchval("SELECT SUM(hit_count) FROM logos_query_cache")
            total_keys = await conn.fetchval("SELECT COUNT(*) FROM logos_query_cache")
            if not total_keys or total_keys == 0:
                return 0.0
            return float(total_hits) / float(total_keys)

    async def cache_word_embeddings(self):
        """
        Specialized method to cache all 20,960 word embeddings in memory.
        These are 300-dimension vectors used for semantic search.
        """
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            embeddings = await conn.fetch("SELECT word, vector FROM word_embeddings")
            for emb in embeddings:
                key = f"emb:{emb['word']}"
                # Memory only for speed, these are small enough
                self._memory_cache[key] = {
                    'value': emb['vector'],
                    'expires_at': time.time() + 604800 # 1 week
                }
        logger.info(f"Cached {len(embeddings)} word embeddings in memory.")

    async def prune_low_hit_entries(self, hit_threshold: int = 1):
        """
        Removes entries that haven't been accessed much to save space.
        """
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            deleted = await conn.execute(
                "DELETE FROM logos_query_cache WHERE hit_count < $1 AND created_at < NOW() - INTERVAL '1 day'",
                hit_threshold
            )
            logger.info(f"Pruned {deleted} low-usage cache entries.")

    async def get_author_summary_cache(self, author_name: str):
        """High-level wrapper for author profile caching."""
        key = f"author_profile:{author_name}"
        cached = await self.get(key)
        if cached:
            return cached
            
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            profile = await conn.fetchrow("SELECT * FROM author_profiles WHERE name = $1", author_name)
            if profile:
                data = dict(profile)
                await self.set(key, data, ttl=3600 * 12)
                return data
        return None

    def clear_memory_cache(self):
        """Force clear the Level 1 cache."""
        self._memory_cache.clear()
        logger.info("LOGOS Level 1 cache cleared.")

    async def export_cache_metadata(self) -> str:
        """Exports cache keys and metadata to a JSON string for auditing."""
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            rows = await conn.fetch("SELECT cache_key, expires_at, hit_count FROM logos_query_cache")
            return json.dumps([dict(r) for r in rows], default=str)

    async def resize_pool(self, min_size: int, max_size: int):
        """Dynamically resizes the database pool based on load."""
        if self._pool:
            await self._pool.close()
        self._pool = await asyncpg.create_pool(
            dsn=self.dsn,
            min_size=min_size,
            max_size=max_size
        )
        logger.info(f"Resized LOGOS cache pool to {min_size}-{max_size} connections.")

    async def check_integrity(self) -> bool:
        """Verifies that the cache table is responsive and not corrupted."""
        try:
            pool = await self._get_pool()
            async with pool.acquire() as conn:
                await conn.execute("SELECT 1")
            return True
        except Exception as e:
            logger.error(f"Cache integrity check failed: {str(e)}")
            return False

    async def get_expired_count(self) -> int:
        """Returns the number of expired items waiting for vacuuming."""
        pool = await self._get_pool()
        async with pool.acquire() as conn:
            return await conn.fetchval("SELECT count(*) FROM logos_query_cache WHERE expires_at < CURRENT_TIMESTAMP")

    async def set_batch(self, mapping: Dict[str, Any], ttl: int = DEFAULT_TTL):
        """Stores multiple items in cache efficiently."""
        for key, value in mapping.items():
            await self.set(key, value, ttl)
        logger.info(f"Batch cached {len(mapping)} items.")

    async def get_multiple(self, keys: List[str]) -> Dict[str, Any]:
        """Retrieves multiple items, returning a map of found values."""
        results = {}
        for key in keys:
            val = await self.get(key)
            if val is not None:
                results[key] = val
        return results

    async def close(self):
        """Properly shuts down the cache manager and pool."""
        if self._pool:
            await self._pool.close()
            logger.info("LOGOS cache pool closed.")
