class LogosDatabaseSeeder:
    """
    Production-grade database seeder for the LOGOS platform.
    Handles complex data relationships and batch inserts for high performance.
    """

    def __init__(self, dsn: str):
        """
        Initializes the seeder with the database connection string.
        
        Args:
            dsn (str): The PostgreSQL connection string.
        """
        self.dsn = dsn
        self.pool: Optional[asyncpg.Pool] = None
        logger.info("Initializing LogosDatabaseSeeder with production DSN.")

    async def connect(self) -> None:
        """
        Establishes a connection pool to the Railway PostgreSQL instance.
        Implements retry logic for robustness.
        """
        retry_count = 0
        max_retries = 5
        while retry_count < max_retries:
            try:
                logger.info(f"Attempting to connect to database (Attempt {retry_count + 1})...")
                self.pool = await asyncpg.create_pool(self.dsn, min_size=5, max_size=20)
                logger.info("Successfully established database connection pool.")
                return
            except Exception as e:
                retry_count += 1
                logger.error(f"Connection failed: {str(e)}. Retrying in 5 seconds...")
                await asyncio.sleep(5)
        
        logger.critical("Failed to connect to the database after multiple attempts.")
        sys.exit(1)

    async def close(self) -> None:
        """
        Gracefully closes the database connection pool.
        """
        if self.pool:
            await self.pool.close()
            logger.info("Database connection pool closed.")

            # Generate a 300-dimension vector

            embedding_data.append((word, vector))

        query = """
            INSERT INTO word_embeddings (word, vector)
            VALUES ($1, $2)

        try:

                    logger.warning(f"Translator count mismatch! Expected {len(APPROVED_TRANSLATORS)}, found {translator_count}")
                else:
                    logger.info("Translator count verified successfully.")
        except Exception as e:
            logger.error(f"Verification failed: {str(e)}")

    async def run_all(self) -> None:
        """
        Orchestrates the entire seeding process in the correct order.
        """
        try:
            await self.connect()
            
            # Execute seeding steps
            logger.info("Starting master seeding process...")
            await self.seed_author_profiles()
            await self.seed_translator_profiles()
            await self.seed_texts()
            await self.seed_word_embeddings()
            
            # Verify
            await self.verify_seeding()
            
            logger.info("Master seeding process completed successfully.")
        except Exception as e:
            logger.critical(f"Seeding process failed: {str(e)}")
        finally:
            await self.close()

    cp main.py "$BUILD_DIR/"

    if [ ! -d "frontend" ]; then

    cd frontend
    log "Installing production dependencies..."

    npm run build

    local attempt=1
    
    while [ $attempt -le $max_attempts ]; do
        status=$(railway status --json | python3 -c "import sys, json; data=json.load(sys.stdin); print(data.get('status', 'unknown'))" 2>/dev/null || echo "deploying")

#!/bin/bash

# Version: 1.0.0

set -e

# Configuration
DB_URL="postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"
BACKUP_DIR="./backups"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/logos_prod_$TIMESTAMP.sql"
RETENTION_DAYS=30

# UI Helpers
PURPLE='\033[0;35m'
NC='\033[0m'

log() {
    echo -e "${PURPLE}[BACKUP] $(date +'%Y-%m-%d %H:%M:%S') - $1${NC}"
}

initialize_backup() {
    log "Initializing backup process..."
    mkdir -p "$BACKUP_DIR"
    
    if ! command -v pg_dump &> /dev/null; then
        echo "Error: pg_dump is not installed. Please install postgresql-client."
        exit 1
    fi
}

perform_backup() {
    log "Executing pg_dump for production database..."
    
    # Exporting schema and data
    pg_dump "$DB_URL" --no-owner --no-privileges > "$BACKUP_FILE"
    
    if [ -f "$BACKUP_FILE" ]; then
        local size=$(du -h "$BACKUP_FILE" | cut -f1)
        log "Backup created successfully: $BACKUP_FILE (Size: $size)"
    else
        echo "Error: Backup file was not created."
        exit 1
    fi
}

compress_backup() {
    log "Compressing backup file..."
    gzip "$BACKUP_FILE"
    log "Compression complete: $BACKUP_FILE.gz"
}

rotate_backups() {
