import time
import functools
import logging
import asyncio
import statistics
from typing import Dict, List, Any, Callable
from datetime import datetime
import asyncpg

# LOGOS Production DB
DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"

logger = logging.getLogger(__name__)

class PerformanceInterceptor:
    """
    Advanced performance monitoring interceptor for the LOGOS backend.
    Tracks execution times of heavy database operations and complex text analysis.
    """

    def __init__(self):
        self.execution_times: Dict[str, List[float]] = {}
        self.slow_query_threshold = 0.8  # seconds
        self.history_limit = 1000
        logger.info("PerformanceInterceptor initialized.")

    def track_time(self, operation_name: str):
        """
        Decorator to measure the execution time of an asynchronous function.
        Usage: @performance_interceptor.track_time("fetch_homer_iliad")
        """
        def decorator(func: Callable):
            @functools.wraps(func)
            async def wrapper(*args, **kwargs):
                start_time = time.perf_counter()
                try:
                    result = await func(*args, **kwargs)
                    return result
                finally:
                    duration = time.perf_counter() - start_time
                    self._record_duration(operation_name, duration)
                    if duration > self.slow_query_threshold:
                        logger.warning(f"SLOW OPERATION: {operation_name} took {duration:.4f}s")
            return wrapper
        return decorator

    def _record_duration(self, name: str, duration: float):
        """Internal helper to store duration metrics."""
        if name not in self.execution_times:
            self.execution_times[name] = []
        
        self.execution_times[name].append(duration)
        
        # Keep history within limits
        if len(self.execution_times[name]) > self.history_limit:
            self.execution_times[name].pop(0)

    def get_statistics(self, operation_name: str) -> Dict[str, Any]:
        """Calculates performance statistics for a specific operation."""
        durations = self.execution_times.get(operation_name, [])
        if not durations:
            return {"error": "No data for operation"}

        return {
            "operation": operation_name,
            "count": len(durations),
            "min": min(durations),
            "max": max(durations),
            "mean": statistics.mean(durations),
            "p95": sorted(durations)[int(len(durations) * 0.95) - 1] if len(durations) >= 20 else "N/A",
            "last_updated": datetime.utcnow().isoformat()
        }

    async def analyze_db_query_plan(self, query: str, params: List[Any] = None):
        """
        Performs an EXPLAIN ANALYZE on a query to diagnose performance bottlenecks.
        Only runs in a safe, controlled manner.
        """
        conn = None
        try:
            conn = await asyncpg.connect(DATABASE_URL)
            explain_query = f"EXPLAIN ANALYZE {query}"
            plan = await conn.fetch(explain_query, *(params or []))
            
            logger.info("--- QUERY PLAN ANALYSIS ---")
            for row in plan:
                logger.info(row[0])
            return plan
        except Exception as e:
            logger.error(f"Failed to analyze query plan: {e}")
            return None
        finally:
            if conn:
                await conn.close()

    def clear_metrics(self):
        """Resets all collected performance metrics."""
        self.execution_times.clear()
        logger.info("Performance metrics cleared.")

    async def monitor_event_loop_lag(self):
        """Monitors the asyncio event loop for blocking operations."""
        while True:
            start = time.perf_counter()
            await asyncio.sleep(1)
            lag = time.perf_counter() - start - 1
            if lag > 0.1:
                logger.warning(f"EVENT LOOP LAG DETECTED: {lag:.4f}s. Possible CPU blocking.")
            await asyncio.sleep(5)

    def log_summary_report(self):

        """Checks for table bloat in the PostgreSQL database."""
        query = """
        SELECT relname AS table_name, 
               pg_size_pretty(pg_total_relation_size(relid)) AS total_size,
               pg_size_pretty(pg_relation_size(relid)) AS internal_size
        FROM pg_catalog.pg_statio_user_tables 
        ORDER BY pg_total_relation_size(relid) DESC;
        """
        conn = None
        try:
            conn = await asyncpg.connect(DATABASE_URL)
            results = await conn.fetch(query)
            for row in results:
                logger.info(f"Table {row['table_name']}: {row['total_size']}")
        except Exception as e:
            logger.error(f"Bloat check failed: {e}")
        finally:
            if conn:
                await conn.close()

/**
 * LOGOS Frontend Analytics and Monitoring System
 * Designed for high-precision tracking of scholar interactions and performance vitals.
 */

interface AnalyticsEvent {
    category: string;
    action: string;
    label?: string;
    value?: number;
    metadata?: Record<string, any>;
    timestamp: string;
}

interface WebVitals {
    id: string;
    name: string;
    value: number;
    rating: 'good' | 'needs-improvement' | 'poor';
    delta: number;
}

class LogosAnalytics {
    private endpoint: string = "/api/v1/analytics/collect";
    private sessionID: string;
    private buffer: AnalyticsEvent[] = [];
    private maxBufferSize: number = 10;
    private flushInterval: number = 5000;

    constructor() {
        this.sessionID = this.generateUUID();
        this.initializePerformanceObservers();
        this.startFlushCycle();
        console.log("LOGOS Analytics initialized for session: " + this.sessionID);
    }

    private generateUUID(): string {
        return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
            const r = Math.random() * 16 | 0, v = c === 'x' ? r : (r & 0x3 | 0x8);
            return v.toString(16);
        });
    }

    /**
     * Tracks a scholar's interaction with the classical library.
     */
    public trackInteraction(action: string, label: string, metadata?: Record<string, any>): void {
        const event: AnalyticsEvent = {
            category: 'scholar_interaction',
            action,
            label,
            metadata,
            timestamp: new Date().toISOString()
        };
        
        console.log(`[Analytics] ${action}: ${label}`, metadata);
        this.enqueueEvent(event);
    }

    /**
     * Tracks when a scholar switches between translators (e.g., Jowett to Dryden).
     */
    public trackTranslationSwitch(work: string, fromTranslator: string, toTranslator: string): void {
        this.trackInteraction('translation_switch', work, {
            from: fromTranslator,
            to: toTranslator
        });
    }

    /**
     * Tracks search queries performed by scholars.
     */
    public trackSearch(query: string, resultCount: number, timeMs: number): void {
        this.trackInteraction('search_performed', query, {
            results: resultCount,
            latency: timeMs
        });
    }

    /**
     * Captures Core Web Vitals for performance monitoring.
     */
    private initializePerformanceObservers(): void {
        if ('PerformanceObserver' in window) {
            // Track Largest Contentful Paint (LCP)
            const lcpObserver = new PerformanceObserver((entryList) => {
                const entries = entryList.getEntries();
                const lastEntry = entries[entries.length - 1];
                this.reportVital('LCP', lastEntry.startTime);
            });
            lcpObserver.observe({ type: 'largest-contentful-paint', buffered: true });

            // Track First Input Delay (FID)
            const fidObserver = new PerformanceObserver((entryList) => {
                entryList.getEntries().forEach(entry => {
                    const delay = (entry as any).processingStart - entry.startTime;
                    this.reportVital('FID', delay);
                });
            });
            fidObserver.observe({ type: 'first-input', buffered: true });

            // Track Cumulative Layout Shift (CLS)
            const clsObserver = new PerformanceObserver((entryList) => {
                let clsValue = 0;
                entryList.getEntries().forEach((entry: any) => {
                    if (!entry.hadRecentInput) {
                        clsValue += entry.value;
                    }
                });
                this.reportVital('CLS', clsValue);
            });
            clsObserver.observe({ type: 'layout-shift', buffered: true });
        }
    }

    private reportVital(name: string, value: number): void {
        let rating: 'good' | 'needs-improvement' | 'poor' = 'good';
        
        if (name === 'LCP' && value > 2500) rating = 'poor';
        if (name === 'FID' && value > 100) rating = 'poor';
        if (name === 'CLS' && value > 0.1) rating = 'poor';

        this.trackInteraction('web_vital', name, { value, rating });
    }

    /**
     * Enqueues an event for batch processing.
     */
    private enqueueEvent(event: AnalyticsEvent): void {
        this.buffer.push(event);
        if (this.buffer.length >= this.maxBufferSize) {
            this.flush();
        }
    }

    /**
     * Sends the buffered events to the backend.
     */
    private async flush(): Promise<void> {
        if (this.buffer.length === 0) return;

        const payload = {
            sessionID: this.sessionID,
            events: [...this.buffer],
            userAgent: navigator.userAgent
        };

        this.buffer = [];

        try {
            const response = await fetch(this.endpoint, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                console.error("Failed to flush analytics data.");
            }
        } catch (error) {
            console.error("Analytics network error:", error);
        }
    }

    private startFlushCycle(): void {
        setInterval(() => this.flush(), this.flushInterval);
    }

    /**
     * Tracks errors occurring in the React component tree.
     */
    public trackError(error: Error, componentStack?: string): void {
        this.trackInteraction('error_boundary', error.name, {
            message: error.message,
            stack: componentStack || error.stack,
            url: window.location.href
        });
    }

    /**
     * Tracks the loading time of a specific classical text.
     */
    public trackTextLoadTime(author: string, work: string, durationMs: number): void {
        this.trackInteraction('text_load_performance', `${author} - ${work}`, {
            duration: durationMs
        });
    }
}

export const analytics = new LogosAnalytics();
