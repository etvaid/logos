class HistoricalPeriod(BaseModel):
    """
    Represents a defined historical period in Classical studies.
    Used for filtering and categorizing texts and authors.
    """
    id: str = Field(..., description="Unique identifier for the period (e.g., 'archaic')")
    name: str = Field(..., description="Display name of the period")
    start_year: int = Field(..., description="Start year of the period (negative for BCE)")
    end_year: int = Field(..., description="End year of the period (negative for BCE)")
    description: str = Field(..., description="Detailed description of the historical context")
    key_authors: List[str] = Field(default_factory=list, description="List of major authors in this period")
    cultural_context: Optional[str] = Field(None, description="Brief summary of cultural developments")

    @validator('end_year')
    def validate_years(cls, v, values):
        """Ensures that the end year is chronologically after the start year."""
        if 'start_year' in values and v <= values['start_year']:
            raise ValueError(f"End year ({v}) must be greater than start year ({values['start_year']})")
        return v

    def to_dict(self) -> Dict[str, Any]:
        """Converts the model to a dictionary representation for serialization."""
        return {
            "id": self.id,
            "name": self.name,
            "start_year": self.start_year,
            "end_year": self.end_year,
            "description": self.description,
            "key_authors": self.key_authors,
            "cultural_context": self.cultural_context
        }

class TimelineEvent(BaseModel):
    """
    Represents a specific event or data point on a historical timeline.
    """
    event_id: str = Field(..., description="Unique identifier for the event")
    year: int = Field(..., description="Year of the event")
    title: str = Field(..., description="Title of the event or work")
    author_id: Optional[int] = Field(None, description="Associated author ID")
    category: str = Field(..., description="Type of event: 'work_published', 'author_birth', 'historical_event'")
    importance_score: float = Field(default=0.5, ge=0.0, le=1.0)
    description: Optional[str] = None
    metadata: Dict[str, Any] = Field(default_factory=dict)

    def is_bce(self) -> bool:
        """Returns True if the event occurred before the Common Era."""
        return self.year < 0

class WordDriftPoint(BaseModel):
    """
    Represents the semantic state of a word at a specific point in time.
    """
    period_id: str = Field(..., description="The historical period ID")
    period_name: str = Field(..., description="The name of the period")
    mean_year: float = Field(..., description="The average year of usage in this period")
    frequency: int = Field(..., description="Number of occurrences in this period")
    context_keywords: List[str] = Field(default_factory=list, description="Top co-occurring words")
    vector_magnitude: float = Field(..., description="Magnitude of the semantic vector")
    drift_distance: float = Field(default=0.0, description="Cosine distance from the earliest known usage")

class WordDriftAnalysis(BaseModel):
    """
    Full analysis of semantic drift for a specific word across multiple periods.
    """
    word: str = Field(..., description="The word being analyzed")
    language: str = Field(default="greek", description="Language of the word")
    total_occurrences: int = Field(..., description="Total count across all periods")
    drift_path: List[WordDriftPoint] = Field(..., description="Chronological path of semantic change")
    overall_instability: float = Field(..., description="Measure of how much the meaning changed over time")
    most_stable_period: Optional[str] = None
    most_volatile_period: Optional[str] = None

class PeriodVocabulary(BaseModel):

    def calculate_cosine_similarity(self, vec_a: np.ndarray, vec_b: np.ndarray) -> float:
        """
        Calculates the cosine similarity between two word vectors.
        
        Args:
            vec_a (np.ndarray): First vector.
            vec_b (np.ndarray): Second vector.
            
        Returns:
            float: Similarity score between 0.0 and 1.0.
        """
        try:
            if vec_a is None or vec_b is None:
                return 0.0
            
            dot_product = np.dot(vec_a, vec_b)
            norm_a = np.linalg.norm(vec_a)
            norm_b = np.linalg.norm(vec_b)
            
            if norm_a == 0 or norm_b == 0:
                return 0.0
                
            return float(dot_product / (norm_a * norm_b))
        except Exception as e:
            logger.error("Error calculating cosine similarity: %s", str(e))
            return 0.0

            for i, (p_id, period) in enumerate(self.periods.items()):
                # Simulate finding the word in this period
                # In production, this would query the 'word_embeddings' table

                # Mocking drift logic for demonstration

                simulated_vector = np.random.rand(300)

                dist = 1.0 - self.calculate_cosine_similarity(base_vector, simulated_vector)
                
                point = WordDriftPoint(

    async def generate_period_vocabulary_stats(self, period_id: str, word_counts: List[Dict[str, Any]]) -> PeriodVocabulary:
        """
        Generates statistical metrics for a period's vocabulary.
        
        Args:
            period_id (str): The ID of the period.
            word_counts (List[Dict]): List of word frequency data.
            
        Returns:
            PeriodVocabulary: Statistics about the period's language usage.
        """
        try:
            logger.info("Generating vocabulary stats for period: %s", period_id)
            
            if not word_counts:
                return PeriodVocabulary(
                    period_id=period_id,
                    unique_word_count=0,
                    total_word_count=0,
                    top_terms=[],
                    lexical_density=0.0,
                    hapax_legomena_count=0
                )

            unique_words = len(word_counts)
            total_words = sum(item['count'] for item in word_counts)
            hapax_count = sum(1 for item in word_counts if item['count'] == 1)
            
            # Sort for top terms
            sorted_terms = sorted(word_counts, key=lambda x: x['count'], reverse=True)
            top_terms = sorted_terms[:50]
            
            density = unique_words / total_words if total_words > 0 else 0.0
            
            return PeriodVocabulary(
                period_id=period_id,
                unique_word_count=unique_words,
                total_word_count=total_words,
                top_terms=top_terms,
                lexical_density=density,
                hapax_legomena_count=hapax_count
            )
            
        except Exception as e:
            logger.error("Error generating vocab stats for %s: %s", period_id, str(e))
            raise

    async def compare_vocabulary_overlap(self, vocab_a: List[str], vocab_b: List[str]) -> Dict[str, Any]:
        """
        Compares two sets of vocabulary to find overlap and differences.
        
        Args:
            vocab_a (List[str]): List of words from first period.
            vocab_b (List[str]): List of words from second period.
            
        Returns:
            Dict[str, Any]: Comparison results.
        """
        try:
            set_a = set(vocab_a)
            set_b = set(vocab_b)
            
            intersection = set_a.intersection(set_b)
            union = set_a.union(set_b)
            
            only_a = set_a - set_b
            only_b = set_b - set_a
            
            similarity = len(intersection) / len(union) if union else 0.0
            
            return {
                "shared_count": len(intersection),
                "unique_a_count": len(only_a),
                "unique_b_count": len(only_b),
                "similarity_score": similarity,
                "overlap_percentage": (len(intersection) / len(set_a)) * 100 if set_a else 0
            }
        except Exception as e:
            logger.error("Error comparing vocabulary overlap: %s", str(e))
            return {"error": str(e)}

    async def get_timeline_events(self, start_year: int, end_year: int) -> TimelineResponse:
        """
        Queries the database for historical events, author births/deaths, 
        and work publications within a specific timeframe.
        
        Args:
            start_year (int): Beginning of the range.
            end_year (int): End of the range.
            
        Returns:
            TimelineResponse: Collection of events.
        """
        if not self.pool: await self.connect()
        
        try:
            events = []
            
            # 1. Fetch author birth/death events from author_profiles
            author_query = """
                SELECT id, name, birth_year, death_year 
                FROM author_profiles 
                WHERE (birth_year BETWEEN $1 AND $2) OR (death_year BETWEEN $1 AND $2)
            """
            authors = await self.pool.fetch(author_query, start_year, end_year)
            
            for auth in authors:
                if auth['birth_year'] and start_year <= auth['birth_year'] <= end_year:
                    events.append(TimelineEvent(
                        event_id=f"birth_{auth['id']}",
                        year=auth['birth_year'],
                        title=f"Birth of {auth['name']}",
                        author_id=auth['id'],
                        category="author_birth",
                        importance_score=0.7,
                        description=f"The birth of the author {auth['name']}."
                    ))
                
                if auth['death_year'] and start_year <= auth['death_year'] <= end_year:
                    events.append(TimelineEvent(
                        event_id=f"death_{auth['id']}",
                        year=auth['death_year'],
                        title=f"Death of {auth['name']}",
                        author_id=auth['id'],
                        category="author_death",
                        importance_score=0.7,
                        description=f"The death of the author {auth['name']}."
                    ))

            # 2. Fetch works/texts from texts table
            # Since texts table doesn't have a year, we'd typically join with 
            # an author_works table or use author dates as a proxy.
            # For this implementation, we will use author dates to place works.
            
            # Sort events chronologically
            events.sort(key=lambda x: x.year)
            
            return TimelineResponse(
                start_year=start_year,
                end_year=end_year,
                event_count=len(events),
                events=events,
                metadata={"query_time": str(datetime.utcnow())}
            )
            
        except Exception as e:
            logger.error("Error fetching timeline events: %s", str(e))
            raise

    async def get_word_drift(self, word: str) -> Dict[str, Any]:
        """
        Calculates the semantic drift of a word by querying the word_embeddings table.
        
        Args:
            word (str): The word to analyze.
            
        Returns:
            Dict: Analysis results.
        """
        if not self.pool: await self.connect()
        
        try:
            # Query the real word_embeddings table
            query = "SELECT word, vector FROM word_embeddings WHERE word = $1"
            row = await self.pool.fetchrow(query, word)
            
            if not row:
                logger.warning("Word '%s' not found in embeddings table", word)
                return {"error": "Word not found", "word": word}
            
            # In a production system, we'd have multiple vectors for the same word
            # tagged with different periods. Here we use the analyzer to simulate 
            # drift based on the core vector.
            
            # Convert binary/array vector to numpy
            vector = np.array(row['vector'])
            
            # Pass to analyzer for temporal breakdown
            analysis = await self.analyzer.analyze_semantic_drift(word, [{"word": word, "vector": vector}])
            
            return analysis.dict()
            
        except Exception as e:
            logger.error("Error in get_word_drift for %s: %s", word, str(e))
            raise

    async def get_authors_in_period(self, period_id: str) -> PeriodAuthorList:
        """
        Retrieves all authors who were active during a specific historical period.
        
        Args:
            period_id (str): The period ID (e.g., 'classical').
            
        Returns:
            PeriodAuthorList: List of authors.
        """
        if not self.pool: await self.connect()
        
        period = self.analyzer.get_period_by_id(period_id)
        if not period:
            raise ValueError(f"Invalid period ID: {period_id}")
            
        try:
            query = """
                SELECT id, name, birth_year, death_year, nationality, genres
                FROM author_profiles
                WHERE (birth_year BETWEEN $1 AND $2)
                   OR (death_year BETWEEN $1 AND $2)
                   OR (birth_year <= $1 AND death_year >= $2)
            """
            rows = await self.pool.fetch(query, period.start_year, period.end_year)
            
            authors = []
            for row in rows:
                authors.append({
                    "id": row['id'],
                    "name": row['name'],
                    "dates": f"{row['birth_year']} to {row['death_year']}",
                    "nationality": row['nationality'],
                    "genres": row['genres']
                })
                
            return PeriodAuthorList(
                period_id=period_id,
                period_name=period.name,
                authors=authors,
                total_count=len(authors)
            )
        except Exception as e:
            logger.error("Error fetching authors for period %s: %s", period_id, str(e))
            raise

    async def get_period_vocabulary(self, period_id: str) -> Dict[str, Any]:
        """
        Aggregates vocabulary usage for a specific period.
        
        Args:
            period_id (str): The period ID.
            
        Returns:
            Dict: Vocabulary statistics.
        """
        if not self.pool: await self.connect()
        
        period = self.analyzer.get_period_by_id(period_id)
        if not period:
            raise ValueError(f"Invalid period ID: {period_id}")

        try:
            # This query joins texts with author_profiles to filter by period dates
            # and then aggregates word counts from the text_content.
            # (Simplified for performance)
            query = """
                SELECT t.text_content
                FROM texts t
                JOIN author_profiles a ON t.author = a.name
                WHERE a.birth_year BETWEEN $1 AND $2
                LIMIT 100
            """
            rows = await self.pool.fetch(query, period.start_year, period.end_year)
            
            # Process text content to get word counts
            word_counts = {}
            for row in rows:
                words = row['text_content'].lower().split()
                for w in words:
                    # Clean word
                    w = "".join(filter(str.isalpha, w))
                    if len(w) > 3:
                        word_counts[w] = word_counts.get(w, 0) + 1
            
            formatted_counts = [{"word": k, "count": v} for k, v in word_counts.items()]
            stats = await self.analyzer.generate_period_vocabulary_stats(period_id, formatted_counts)
            
            return stats.dict()
            
        except Exception as e:
            logger.error("Error getting vocabulary for period %s: %s", period_id, str(e))
            raise

    async def compare_periods(self, period_a_id: str, period_b_id: str) -> Dict[str, Any]:
        """
        Performs a comparative analysis between two historical periods.
        
        Args:
            period_a_id (str): First period.
            period_b_id (str): Second period.
            
        Returns:
            Dict: Comparison metrics.
        """
        try:
            vocab_a = await self.get_period_vocabulary(period_a_id)
            vocab_b = await self.get_period_vocabulary(period_b_id)
            
            words_a = [item['word'] for item in vocab_a['top_terms']]
            words_b = [item['word'] for item in vocab_b['top_terms']]
            
            overlap_data = await self.analyzer.compare_vocabulary_overlap(words_a, words_b)
            
            return {
                "period_a": self.analyzer.get_period_by_id(period_a_id).dict(),
                "period_b": self.analyzer.get_period_by_id(period_b_id).dict(),
                "vocabulary_comparison": overlap_data,
                "similarity_index": overlap_data['similarity_score'],
                "generated_at": str(datetime.utcnow())
            }
        except Exception as e:
            logger.error("Error comparing periods %s and %s: %s", period_a_id, period_b_id, str(e))
            raise

    async def get_author_historical_context(self, author_name: str) -> AuthorHistoricalContext:
        """
        Retrieves the historical context for a specific author.
        
        Args:
            author_name (str): Name of the author.
            
        Returns:
            AuthorHistoricalContext: Contextual data.
        """
        if not self.pool: await self.connect()
        
        try:
            # 1. Get author profile
            query = "SELECT * FROM author_profiles WHERE name = $1"
            author = await self.pool.fetchrow(query, author_name)
            
            if not author:
                raise ValueError(f"Author not found: {author_name}")
                
            # 2. Determine period
            period = self.analyzer.find_period_for_year(author['birth_year'] or 0)
            if not period:
                # Default to classical if not found
                period = self.analyzer.get_period_by_id("classical")
            
            # 3. Find contemporaries
            contemp_query = """
                SELECT name FROM author_profiles 
                WHERE birth_year BETWEEN $1 AND $2 
                AND name != $3 
                LIMIT 5
            """
            contemps = await self.pool.fetch(contemp_query, 
                                             author['birth_year'] - 20, 
                                             author['birth_year'] + 20, 
                                             author_name)
            
            return AuthorHistoricalContext(
                author_id=author['id'],
                author_name=author['name'],
                period=period,
                contemporaries=[c['name'] for c in contemps],
                influence_score=0.85 # Simulated
            )
            
        except Exception as e:
            logger.error("Error fetching context for author %s: %s", author_name, str(e))
            raise

    async def get_semantic_clusters_for_period(self, period_id: str) -> List[SemanticCluster]:
        """
        Retrieves semantic clusters for a period.
        
        Args:
            period_id (str): Period ID.
            
        Returns:
            List[SemanticCluster]: List of clusters.
        """
        return await self.analyzer.identify_semantic_clusters(period_id)

    async def run_custom_query(self, sql: str, params: List[Any] = None) -> List[Dict[str, Any]]:
        """
        Executes a custom SQL query against the database (internal use).
        
        Args:
            sql (str): SQL query string.
            params (List): Query parameters.
            
        Returns:
            List[Dict]: Query results.
        """
        if not self.pool: await self.connect()
        try:
            if params:
                rows = await self.pool.fetch(sql, *params)
            else:
                rows = await self.pool.fetch(sql)
            return [dict(r) for r in rows]
        except Exception as e:
            logger.error("Custom query failed: %s", str(e))
            return []
