class TextMetadata(BaseModel):

"""

Corpus API SQL Queries

This module contains the raw SQL queries and query building logic for the LOGOS corpus.
It uses optimized PostgreSQL syntax for large-scale classical text retrieval.

Author: Senior Software Architect
Version: 1.0.0

import logging
from typing import Optional, List, Dict, Any

logger = logging.getLogger(__name__)

# --- Texts Table Queries ---

GET_TEXTS_PAGINATED = """
    SELECT id, title, author, translator, book, chapter
    FROM texts
    WHERE ($1::text IS NULL OR author = $1)
      AND ($2::text IS NULL OR translator = $2)
      AND ($3::text IS NULL OR title ILIKE $3)
    ORDER BY author, title, book, chapter
    LIMIT $4 OFFSET $5;
"""

GET_TEXT_COUNT = """
    SELECT COUNT(*) 
    FROM texts
    WHERE ($1::text IS NULL OR author = $1)
      AND ($2::text IS NULL OR translator = $2)
      AND ($3::text IS NULL OR title ILIKE $3);
"""

GET_TEXT_BY_ID = """
    SELECT id, title, author, translator, text_content, book, chapter
    FROM texts
    WHERE id = $1;
"""

# --- Search Queries ---

FULL_TEXT_SEARCH = """
    SELECT 
        id as text_id, 
        title, 
        author, 
        ts_headline('english', text_content, plainto_tsquery('english', $1)) as snippet,
        ts_rank(to_tsvector('english', text_content), plainto_tsquery('english', $1)) as rank
    FROM texts
    WHERE to_tsvector('english', text_content) @@ plainto_tsquery('english', $1)
    ORDER BY rank DESC
    LIMIT $2 OFFSET $3;
"""

SEARCH_COUNT = """
    SELECT COUNT(*)
    FROM texts
    WHERE to_tsvector('english', text_content) @@ plainto_tsquery('english', $1);
"""

# --- Author Profile Queries ---

GET_ALL_AUTHORS = """
    SELECT id, name, birth_year, death_year, nationality, genres
    FROM author_profiles
    ORDER BY name ASC;
"""

GET_AUTHOR_BY_NAME = """
    SELECT id, name, birth_year, death_year, nationality, genres
    FROM author_profiles
    WHERE name = $1;
"""

GET_WORKS_BY_AUTHOR = """
    SELECT DISTINCT title
    FROM texts
    WHERE author = $1
    ORDER BY title ASC;
"""

# --- Source Text Queries ---

GET_SOURCE_PASSAGE = """
    SELECT id, content, language, work_id, line_number
    FROM source_texts
    WHERE work_id = (SELECT id FROM texts WHERE title = $1 AND author = $2 LIMIT 1)
      AND ($3::int IS NULL OR line_number::int >= $3)
      AND ($4::int IS NULL OR line_number::int <= $4)
    ORDER BY line_number::int ASC;
"""

# --- Comparison Queries ---

GET_MULTIPLE_TEXTS = """
    SELECT id, title, author, translator, text_content, book, chapter
    FROM texts
    WHERE id = ANY($1::int[]);
"""

# --- Helper Logic for Dynamic Queries ---

def build_search_query(query_str: str, limit: int, offset: int) -> tuple:
    """
    Constructs a safe search query with proper parameters.
    """
    try:
        # Sanitize query string for PostgreSQL tsquery
        sanitized_query = ' & '.join(query_str.split())
        return FULL_TEXT_SEARCH, (sanitized_query, limit, offset)
    except Exception as e:
        logger.error(f"Error building search query: {e}")
        raise ValueError("Invalid search parameters")


def build_text_filter_query(
    author: Optional[str] = None,
    translator: Optional[str] = None,
    title_search: Optional[str] = None,
    limit: int = 50,
    offset: int = 0
) -> tuple:
    """
    Helper to prepare the paginated text query parameters.
    """
    try:
        title_param = f"%{title_search}%" if title_search else None
        return GET_TEXTS_PAGINATED, (author, translator, title_param, limit, offset)
    except Exception as e:
        logger.error(f"Error building filter query: {e}")
        raise ValueError("Invalid filter parameters")

"""
LOGOS - The Bible for Classical Studies
Corpus API Service Layer

This module implements the business logic for classical text retrieval, 
full-text search, and cross-reference analysis. It communicates directly 
with the Railway PostgreSQL database using asyncpg.

Author: Senior Software Architect
Version: 1.0.0
"""

import logging
import asyncpg
import json
import time
from typing import List, Optional, Dict, Any
from fastapi import HTTPException

from .models import (
    TextMetadata, TextFull, AuthorProfile, SearchResult, 
    SearchResponse, PaginatedResponse, Language
)
from . import queries

# Configure logging for production monitoring
logger = logging.getLogger(__name__)

DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"

class CorpusService:
    """
    High-performance service for handling all classical corpus operations.
    Implements connection pooling and optimized data mapping.
    """

    def __init__(self):
        self.pool: Optional[asyncpg.Pool] = None

    async def initialize(self):
        """Initializes the database connection pool."""
        try:
            if not self.pool:
                self.pool = await asyncpg.create_pool(
                    DATABASE_URL,
                    min_size=5,
                    max_size=20,
                    command_timeout=60
                )
                logger.info("Database connection pool established successfully.")
        except Exception as e:
            logger.critical(f"Failed to initialize database pool: {str(e)}")
            raise

    async def close(self):
        """Closes the connection pool."""
        if self.pool:
            await self.pool.close()
            logger.info("Database connection pool closed.")

    async def get_texts(
        self, 
        page: int = 1, 
        page_size: int = 50, 
        author: Optional[str] = None,
        translator: Optional[str] = None,
        query: Optional[str] = None
    ) -> PaginatedResponse:
        """
        Retrieves a paginated list of texts with optional filtering.
        """
        try:
            offset = (page - 1) * page_size
            sql, params = queries.build_text_filter_query(author, translator, query, page_size, offset)
            
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(sql, *params)
                total_count = await conn.fetchval(queries.GET_TEXT_COUNT, author, translator, f"%{query}%" if query else None)

            items = [TextMetadata(**dict(row)) for row in rows]
            
            return PaginatedResponse(
                items=items,
                total_count=total_count,
                page=page,
                page_size=page_size,
                has_more=(offset + page_size) < total_count
            )
        except Exception as e:
            logger.error(f"Error in get_texts: {str(e)}")
            raise HTTPException(status_code=500, detail="Internal database error during text retrieval")

    async def get_text_by_id(self, text_id: int) -> TextFull:
        """
        Retrieves a single text by its primary key.
        """
        try:
            async with self.pool.acquire() as conn:
                row = await conn.fetchrow(queries.GET_TEXT_BY_ID, text_id)
                
            if not row:
                logger.warning(f"Text ID {text_id} not found.")
                raise HTTPException(status_code=404, detail=f"Text with ID {text_id} not found")
            
            return TextFull(**dict(row))
        except HTTPException:
            raise
        except Exception as e:
            logger.error(f"Error in get_text_by_id: {str(e)}")
            raise HTTPException(status_code=500, detail="Failed to retrieve text content")

    async def search_corpus(self, query: str, page: int = 1, page_size: int = 20) -> SearchResponse:
        """
        Performs full-text search across the entire corpus content.
        """
        start_time = time.time()
        try:
            offset = (page - 1) * page_size
            sql, params = queries.build_search_query(query, page_size, offset)
            
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(sql, *params)
                total_hits = await conn.fetchval(queries.SEARCH_COUNT, ' & '.join(query.split()))

            results = [SearchResult(**dict(row)) for row in rows]
            query_time = (time.time() - start_time) * 1000

            return SearchResponse(
                total_hits=total_hits or 0,
                results=results,
                query_time_ms=round(query_time, 2)
            )
        except Exception as e:
            logger.error(f"Search error for query '{query}': {str(e)}")
            raise HTTPException(status_code=500, detail="Full-text search failed")

    async def list_authors(self) -> List[AuthorProfile]:
        """
        Returns all author profiles from the database.
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(queries.GET_ALL_AUTHORS)
            
            profiles = []
            for row in rows:
                data = dict(row)
                # Ensure genres is a list (asyncpg handles JSONB/Array automatically usually)
                if isinstance(data['genres'], str):
                    data['genres'] = json.loads(data['genres'])
                profiles.append(AuthorProfile(**data))
            return profiles
        except Exception as e:
            logger.error(f"Error listing authors: {str(e)}")
            raise HTTPException(status_code=500, detail="Failed to retrieve author profiles")

    async def get_works_by_author(self, author_name: str) -> List[str]:
        """
        Retrieves a unique list of work titles for a specific author.
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(queries.GET_WORKS_BY_AUTHOR, author_name)
            
            if not rows:
                logger.info(f"No works found for author: {author_name}")
                return []
            
            return [row['title'] for row in rows]
        except Exception as e:
            logger.error(f"Error fetching works for {author_name}: {str(e)}")
            raise HTTPException(status_code=500, detail="Failed to retrieve works for author")

    async def get_passage(
        self, 
        work_title: str, 
        author: str, 
        start_line: Optional[int] = None, 
        end_line: Optional[int] = None
    ) -> List[Dict[str, Any]]:
        """
        Retrieves specific lines from the source_texts table.
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(
                    queries.GET_SOURCE_PASSAGE, 
                    work_title, 
                    author, 
                    start_line, 
                    end_line
                )
            
            return [dict(row) for row in rows]
        except Exception as e:
            logger.error(f"Error retrieving passage: {str(e)}")
            raise HTTPException(status_code=500, detail="Failed to retrieve specific passage")

    async def compare_texts(self, text_ids: List[int]) -> List[TextFull]:
        """
        Retrieves multiple texts side-by-side for comparison.
        """
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(queries.GET_MULTIPLE_TEXTS, text_ids)
            
            if len(rows) < len(text_ids):
                logger.warning("Some requested IDs for comparison were not found.")
            
            return [TextFull(**dict(row)) for row in rows]
        except Exception as e:
            logger.error(f"Comparison error: {str(e)}")
            raise HTTPException(status_code=500, detail="Failed to perform text comparison")
