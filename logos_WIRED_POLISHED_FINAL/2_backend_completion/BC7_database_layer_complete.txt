class DatabaseConfigurationError(Exception):
    """Raised when database configuration is invalid or missing."""
    pass

class DatabaseConnectionError(Exception):
    """Raised when a connection to the database cannot be established."""
    pass

class DatabasePool:
    """
    High-performance Connection Pool Manager for LOGOS.
    Manages the lifecycle of asyncpg connections to the Railway PostgreSQL instance.
    
    This class implements a singleton-like pattern to ensure only one pool exists
    per application instance, managing resources efficiently for thousands of scholars.
    """

    def __init__(self):
        self._pool: Optional[Pool] = None
        self._dsn: str = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"
        self._min_size: int = 10
        self._max_size: int = 50
        self._max_queries: int = 50000
        self._max_inactive_connection_lifetime: float = 300.0
        self._connect_timeout: float = 30.0
        self._is_initialized: bool = False
        self._last_health_check: float = 0.0

    async def initialize(self) -> None:
        """
        Initializes the connection pool with optimized production settings.
        
        Raises:
            DatabaseConnectionError: If the pool cannot be created.
        """
        if self._is_initialized:
            logger.warning("Database pool is already initialized.")
            return

        logger.info("Initializing LOGOS Production Database Pool...")
        try:
            self._pool = await asyncpg.create_pool(
                dsn=self._dsn,
                min_size=self._min_size,
                max_size=self._max_size,
                max_queries=self._max_queries,
                max_inactive_connection_lifetime=self._max_inactive_connection_lifetime,
                command_timeout=60.0,
                init=self._setup_connection,
                timeout=self._connect_timeout
            )
            self._is_initialized = True
            self._last_health_check = time.time()
            logger.info(f"Successfully connected to Railway PostgreSQL at {self._dsn.split('@')[1]}")
        except Exception as e:
            logger.error(f"Failed to initialize database pool: {str(e)}")
            raise DatabaseConnectionError(f"Critical failure in database initialization: {e}")

    async def _setup_connection(self, conn: Connection) -> None:
        """
        Configures individual connections upon creation.
        Sets up JSON/JSONB encoders and custom type handlers.
        """
        try:
            # Add custom encoders/decoders for complex types if necessary
            # For example, handling numpy-like arrays for word_embeddings
            await conn.set_type_codec(
                'json',
                encoder=json.dumps,
                decoder=json.loads,
                schema='pg_catalog'
            )
            logger.debug("Connection codec configuration complete.")
        except Exception as e:
            logger.error(f"Error setting up connection codecs: {e}")

    @asynccontextmanager
    async def get_connection(self) -> AsyncGenerator[Connection, None]:
        """
        Context manager to acquire and release a connection from the pool.
        
        Yields:
            asyncpg.Connection: A live database connection.
            
        Raises:
            DatabaseConnectionError: If the pool is not initialized or connection fails.
        """
        if not self._pool:
            logger.info("Pool not initialized. Attempting lazy initialization...")
            await self.initialize()

        connection = await self._pool.acquire()
        try:
            logger.debug("Database connection acquired from pool.")
            yield connection
        except Exception as e:
            logger.error(f"Error during connection usage: {e}")
            raise
        finally:
            await self._pool.release(connection)
            logger.debug("Database connection returned to pool.")

    async def execute(self, query: str, *args: Any, timeout: float = 30.0) -> str:
        """
        Executes a command (INSERT, UPDATE, DELETE, etc).
        
        Args:
            query: SQL query string.
            args: Query parameters.
            timeout: Maximum execution time.
            
        Returns:
            str: The status message from the database.
        """
        async with self.get_connection() as conn:
            try:
                start_time = time.time()
                result = await conn.execute(query, *args, timeout=timeout)
                duration = time.time() - start_time
                logger.info(f"Execution successful ({duration:.3f}s): {query[:100]}...")
                return result
            except asyncpg.InternalClientError as e:
                logger.error(f"Internal Client Error: {e}")
                raise
            except Exception as e:
                logger.error(f"Query execution failed: {e} | Query: {query}")
                raise

    async def fetch(self, query: str, *args: Any, timeout: float = 30.0) -> List[asyncpg.Record]:
        """
        Fetches multiple rows from the database.
        """
        async with self.get_connection() as conn:
            try:
                start_time = time.time()
                results = await conn.fetch(query, *args, timeout=timeout)
                duration = time.time() - start_time
                logger.info(f"Fetch successful: {len(results)} rows returned ({duration:.3f}s)")
                return results
            except Exception as e:
                logger.error(f"Fetch failed: {e}")
                raise

    async def fetchrow(self, query: str, *args: Any, timeout: float = 30.0) -> Optional[asyncpg.Record]:
        """
        Fetches a single row from the database.
        """
        async with self.get_connection() as conn:
            try:
                result = await conn.fetchrow(query, *args, timeout=timeout)
                return result
            except Exception as e:
                logger.error(f"Fetchrow failed: {e}")
                raise

    async def fetchval(self, query: str, *args: Any, column: int = 0, timeout: float = 30.0) -> Any:
        """
        Fetches a single value from the first row of the result.
        """
        async with self.get_connection() as conn:
            try:
                return await conn.fetchval(query, *args, column=column, timeout=timeout)
            except Exception as e:
                logger.error(f"Fetchval failed: {e}")
                raise

    async def check_health(self) -> Dict[str, Any]:
        """
        Performs a health check on the database connection.
        Used by monitoring systems to ensure the $10M system is operational.
        """
        if not self._pool:
            return {"status": "uninitialized", "healthy": False}

        try:
            start_time = time.time()
            # Simple heartbeat query
            val = await self.fetchval("SELECT 1")
            latency = time.time() - start_time
            
            if val != 1:
                return {"status": "degraded", "healthy": False, "error": "Unexpected heartbeat value"}

            stats = {
                "status": "healthy",
                "healthy": True,
                "latency_ms": latency * 1000,
                "pool_size": self._pool.get_size(),
                "pool_idle": self._pool.get_idle_size(),
                "pool_max": self._max_size,
                "timestamp": time.time()
            }
            self._last_health_check = time.time()
            return stats
        except Exception as e:
            logger.critical(f"Database health check failed: {e}")
            return {"status": "unhealthy", "healthy": False, "error": str(e)}

    async def close_pool(self) -> None:
        """
        Gracefully shuts down the connection pool.
        Must be called on application shutdown.
        """
        if self._pool:
            logger.info("Closing database pool...")
            try:
                await asyncio.wait_for(self._pool.close(), timeout=10.0)
                self._is_initialized = False
                logger.info("Database pool closed successfully.")
            except asyncio.TimeoutError:
                logger.warning("Timed out waiting for pool to close. Forcing termination.")
                self._pool.terminate()
            except Exception as e:
                logger.error(f"Error closing database pool: {e}")

    def get_pool_status(self) -> str:
        """Returns a string representation of the current pool status."""
        if not self._pool:
            return "Pool: NOT_INITIALIZED"
        return f"Pool: {self._pool.get_size()} total, {self._pool.get_idle_size()} idle"

class BaseQuery:
    """Base class for all query modules providing shared utilities."""
    
    @staticmethod
    def _format_record(record: Any) -> Dict[str, Any]:
        """Converts an asyncpg.Record to a standard dictionary."""
        if record is None:
            return {}
        return dict(record)

    @staticmethod
    def _format_records(records: List[Any]) -> List[Dict[str, Any]]:
        """Converts a list of asyncpg.Records to a list of dictionaries."""
        return [dict(r) for r in records]

class TextQueries(BaseQuery):
    """
    Handles all operations related to the 'texts' table (121,184 rows).
    Supports advanced filtering by author, translator, and full-text search.
    """

    ALLOWED_TRANSLATORS = {
        "Jowett", "Dryden", "Dakyns", "Pope", "Murray", "Butler", "Church_Brodribb", 
        "Cowper", "Butcher_Lang", "Lang_Leaf_Myers", "Conington", "Goodwin", "Storr", 
        "Roberts", "Aubrey_Stewart", "Williams", "Dryden_et_al", "Brookes_More", 
        "Ross", "Rawlinson", "Moore", "Evelyn-White", "Morshead", "Heseltine", 
        "Crawley", "Long", "Lindsay", "Jebb", "Macaulay", "Leonard", "Adlington", 
        "Smith", "Morris", "Butcher", "Derby", "Kenyon", "Hickie", "Anonymous"
    }

    async def get_text_by_id(self, text_id: int) -> Optional[Dict[str, Any]]:
        """Fetches a specific text entry by its primary key."""
        query = "SELECT * FROM texts WHERE id = $1"
        record = await db_manager.fetchrow(query, text_id)
        return self._format_record(record) if record else None

    async def search_texts(self, search_term: str, limit: int = 50, offset: int = 0) -> List[Dict[str, Any]]:
        """
        Performs a full-text search on text_content and title.
        Uses PostgreSQL tsvector for optimized searching.
        """
        query = """
            SELECT id, title, author, translator, book, chapter,
                   ts_rank(to_tsvector('english', text_content), websearch_to_tsquery('english', $1)) as rank
            FROM texts
            WHERE to_tsvector('english', text_content) @@ websearch_to_tsquery('english', $1)
            OR title ILIKE $2
            ORDER BY rank DESC
            LIMIT $3 OFFSET $4
        """
        records = await db_manager.fetch(query, search_term, f"%{search_term}%", limit, offset)
        return self._format_records(records)

    async def get_by_author(self, author_name: str, limit: int = 20) -> List[Dict[str, Any]]:
        """Fetches texts written by a specific author."""
        query = "SELECT * FROM texts WHERE author = $1 ORDER BY id LIMIT $2"
        records = await db_manager.fetch(query, author_name, limit)
        return self._format_records(records)

    async def get_by_translator(self, translator_name: str, limit: int = 20) -> List[Dict[str, Any]]:
        """
        Fetches texts translated by a specific scholar.
        Validates against the ALLOWED_TRANSLATORS list to prevent SQL injection and invalid lookups.
        """
        if translator_name not in self.ALLOWED_TRANSLATORS:
            logger.warning(f"Invalid translator requested: {translator_name}")
            return []
            
        query = "SELECT * FROM texts WHERE translator = $1 ORDER BY id LIMIT $2"
        records = await db_manager.fetch(query, translator_name, limit)
        return self._format_records(records)

    async def get_chapters(self, title: str, author: str) -> List[int]:
        """Returns a list of available chapters for a specific work."""
        query = "SELECT DISTINCT chapter FROM texts WHERE title = $1 AND author = $2 ORDER BY chapter"
        records = await db_manager.fetch(query, title, author)
        return [r['chapter'] for r in records]

    async def get_full_work(self, title: str, author: str, translator: str) -> List[Dict[str, Any]]:
        """Fetches all segments of a work in order."""
        query = """
            SELECT * FROM texts 
            WHERE title = $1 AND author = $2 AND translator = $3 
            ORDER BY chapter, id
        """
        records = await db_manager.fetch(query, title, author, translator)
        return self._format_records(records)

class EmbeddingQueries(BaseQuery):
    """
    Handles semantic operations using 'word_embeddings' (20,960 rows).
    """

    async def get_word_vector(self, word: str) -> Optional[List[float]]:
        """Retrieves the 300-dimension vector for a specific word."""
        query = "SELECT vector FROM word_embeddings WHERE word = $1"
        val = await db_manager.fetchval(query, word.lower())
        return val

    async def find_similar_words(self, word: str, limit: int = 10) -> List[Dict[str, Any]]:
        """
        Finds semantically similar words using cosine similarity.
        Assumes the vector is stored as a type that supports the <=> operator (pgvector)
        or calculates it manually using dot product.
        """
        # Manual cosine similarity if pgvector is not installed: (A · B) / (||A|| ||B||)
        # For simplicity in this production layer, we use the dot product approach
        target_vector = await self.get_word_vector(word)
        if not target_vector:
            return []

        query = """
            SELECT word, 
                   (vector <#> $1) * -1 as similarity
            FROM word_embeddings
            WHERE word != $2
            ORDER BY vector <#> $1
            LIMIT $3
        """
        records = await db_manager.fetch(query, target_vector, word.lower(), limit)
        return self._format_records(records)

import logging
import functools
from typing import Any, Callable, Optional, TypeVar, cast
from .connection import db_manager

logger = logging.getLogger(__name__)

F = TypeVar('F', bound=Callable[..., Any])

class TransactionError(Exception):
    """Raised when a transaction fails to commit or rollback."""
    pass

class Transaction:
    """
    Managed transaction context for LOGOS.
    Ensures ACID compliance for multi-step database operations.
    """

    def __init__(self):
        self._connection = None
        self._transaction = None
        self._savepoints = []

    async def __aenter__(self):
        """Starts a new transaction."""
        self._connection = await db_manager._pool.acquire()
        self._transaction = self._connection.transaction()
        await self._transaction.start()
        logger.debug("Transaction started.")
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Commits or rolls back the transaction based on exception state."""
        try:
            if exc_type is None:
                await self._transaction.commit()
                logger.debug("Transaction committed successfully.")
            else:
                await self._transaction.rollback()
                logger.error(f"Transaction rolled back due to error: {exc_val}")
        finally:
            await db_manager._pool.release(self._connection)
            self._connection = None
            self._transaction = None

    async def create_savepoint(self, name: str) -> None:
        """Creates a named savepoint within the current transaction."""
        if not self._connection:
            raise TransactionError("No active connection for savepoint.")
        await self._connection.execute(f"SAVEPOINT {name}")
        self._savepoints.append(name)
        logger.debug(f"Savepoint '{name}' created.")

    async def rollback_to_savepoint(self, name: str) -> None:
        """Rolls back the transaction to a specific savepoint."""
        if name not in self._savepoints:
            raise TransactionError(f"Savepoint '{name}' does not exist.")
        await self._connection.execute(f"ROLLBACK TO SAVEPOINT {name}")
        logger.info(f"Rolled back to savepoint '{name}'.")

    async def execute(self, query: str, *args: Any) -> str:
        """Executes a query within the transaction context."""
        if not self._connection:
            raise TransactionError("Transaction is not active.")
        return await self._connection.execute(query, *args)

    async def fetch(self, query: str, *args: Any) -> List[Any]:
        """Fetches results within the transaction context."""
        if not self._connection:
            raise TransactionError("Transaction is not active.")
        return await self._connection.fetch(query, *args)

def transactional(func: F) -> F:
    """
    Decorator to wrap a function in a database transaction.
    Automatically handles begin, commit, and rollback.
    """
    @functools.wraps(func)
    async def wrapper(*args: Any, **kwargs: Any) -> Any:
        async with Transaction() as tx:
            # Inject the transaction object if the function accepts it
            kwargs['tx'] = tx
            try:
                result = await func(*args, **kwargs)
                return result
            except Exception as e:
                logger.error(f"Error in transactional function {func.__name__}: {e}")
                raise

    return cast(F, wrapper)

class TransactionManager:
    """Utility class for complex transaction logic."""

    @staticmethod
    async def run_in_transaction(coro: Callable[..., Any], *args: Any, **kwargs: Any) -> Any:
        """Runs a coroutine inside a managed transaction."""
        async with Transaction() as tx:
            return await coro(tx, *args, **kwargs)

    @staticmethod
    async def batch_update_texts(updates: List[Dict[str, Any]]) -> bool:
        """
        Example of a complex transactional operation: 
        Updating multiple text records atomically.
        """
        async with Transaction() as tx:
            try:
                for update in updates:
                    query = "UPDATE texts SET text_content = $1 WHERE id = $2"
                    await tx.execute(query, update['content'], update['id'])
                return True
            except Exception as e:
                logger.error(f"Batch update failed: {e}")
                return False

import logging
import time
from typing import List, Dict, Any, Optional
from datetime import datetime

from .connection import db_manager

logger = logging.getLogger(__name__)

class Migration:
    """Represents a single database schema migration."""
    
    def __init__(self, version: int, description: str, up_script: str, down_script: str):
        self.version = version
        self.description = description
        self.up_script = up_script
        self.down_script = down_script

class MigrationManager:
    """
    Handles database schema evolution for the LOGOS system.
    Ensures that all 5 core tables are correctly indexed and structured.
    """

    def __init__(self):
        self.migrations_table = "schema_migrations"
        self._migration_list: List[Migration] = []
        self._register_default_migrations()

    def _register_default_migrations(self):
        """Registers the initial core migrations for the system."""
        
        # Migration 1: Create Migration Tracking Table
        self._migration_list.append(Migration(
            1,
            "Initial setup - migrations tracking",
            f"""
            CREATE TABLE IF NOT EXISTS {self.migrations_table} (
                version INTEGER PRIMARY KEY,
                description TEXT NOT NULL,
                applied_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
            );
            """,
            f"DROP TABLE IF EXISTS {self.migrations_table};"
        ))

        # Migration 2: Optimize Texts Table
        self._migration_list.append(Migration(
            2,
            "Add full-text search index to texts content",
            """
            CREATE INDEX IF NOT EXISTS idx_texts_content_gin 
            ON texts USING GIN (to_tsvector('english', text_content));
            CREATE INDEX IF NOT EXISTS idx_texts_author ON texts(author);
            CREATE INDEX IF NOT EXISTS idx_texts_translator ON texts(translator);
            """,
            "DROP INDEX IF EXISTS idx_texts_content_gin;"
        ))

        # Migration 3: Optimize Source Texts
        self._migration_list.append(Migration(
            3,
            "Index source_texts for high-speed retrieval",
            """
            CREATE INDEX IF NOT EXISTS idx_source_work_line 
            ON source_texts(work_id, line_number);
            """,
            "DROP INDEX IF EXISTS idx_source_work_line;"
        ))

        # Migration 4: Vector Extensions
        self._migration_list.append(Migration(
            4,
            "Setup vector search capabilities",
            """
            CREATE EXTENSION IF NOT EXISTS vector;
            -- Assuming pgvector is available, otherwise this will fail gracefully in the runner
            """,
            "DROP EXTENSION IF EXISTS vector;"
        ))

    async def get_current_version(self) -> int:
        """Returns the current migration version from the database."""
        query = f"SELECT MAX(version) FROM {self.migrations_table}"
        try:
            version = await db_manager.fetchval(query)
            return version if version else 0
        except Exception:
            # Table might not exist yet
            return 0

    async def run_migrations(self):
        """Executes all pending migrations in order."""
        logger.info("Starting database migrations...")
        
        # Ensure migration table exists
        if await self.get_current_version() == 0:
            logger.info("Initializing migration tracking...")
            await db_manager.execute(self._migration_list[0].up_script)
            await db_manager.execute(
                f"INSERT INTO {self.migrations_table} (version, description) VALUES ($1, $2)",
                1, "Initial setup"
            )

        current_version = await self.get_current_version()
        
        for migration in self._migration_list:
            if migration.version > current_version:
                logger.info(f"Applying migration v{migration.version}: {migration.description}")
                try:
                    # Run migration script
                    await db_manager.execute(migration.up_script)
                    
                    # Record success
                    await db_manager.execute(
                        f"INSERT INTO {self.migrations_table} (version, description) VALUES ($1, $2)",
                        migration.version, migration.description
                    )
                    logger.info(f"Successfully applied migration v{migration.version}")
                except Exception as e:
                    logger.error(f"Failed to apply migration v{migration.version}: {e}")
                    raise

        logger.info("Migrations completed successfully.")

    async def rollback_last_migration(self):
        """Rolls back the most recently applied migration."""
        current_version = await self.get_current_version()
        if current_version <= 1:
            logger.warning("Cannot rollback initial migration.")
            return

        migration = next((m for m in self._migration_list if m.version == current_version), None)
        if migration:
            logger.info(f"Rolling back migration v{current_version}: {migration.description}")
            try:
                await db_manager.execute(migration.down_script)
                await db_manager.execute(
                    f"DELETE FROM {self.migrations_table} WHERE version = $1",
                    current_version
                )
                logger.info(f"Rollback of v{current_version} complete.")
            except Exception as e:
                logger.error(f"Rollback failed: {e}")
                raise

class DatabaseUtils:
    """Utility functions for database maintenance and diagnostics."""

    @staticmethod
    async def get_table_sizes() -> Dict[str, int]:
        """Returns row counts for all core tables."""
        tables = ['texts', 'source_texts', 'author_profiles', 'translator_profiles', 'word_embeddings']
        results = {}
        for table in tables:
            count = await db_manager.fetchval(f"SELECT count(*) FROM {table}")
            results[table] = count
        return results

    @staticmethod
    async def vacuum_analyze() -> None:
        """Performs a VACUUM ANALYZE to optimize query plans."""
        logger.info("Starting database optimization (VACUUM ANALYZE)...")
        # Vacuum cannot run inside a transaction block in some Postgres configs
        async with db_manager.get_connection() as conn:
            await conn.execute("VACUUM ANALYZE")
        logger.info("Database optimization complete.")

    @staticmethod
    async def check_index_usage() -> List[Dict[str, Any]]:
        """Returns statistics about index usage to identify bottlenecks."""
        query = """
            SELECT 
                relname as table_name, 
                indexrelname as index_name, 
                idx_scan as times_used
            FROM pg_stat_user_indexes
            ORDER BY idx_scan DESC;
        """
        records = await db_manager.fetch(query)
        return [dict(r) for r in records]

    @staticmethod
    async def get_active_connections() -> List[Dict[str, Any]]:
        """Returns currently active database connections/queries."""
        query = """
            SELECT pid, state, query, query_start
            FROM pg_stat_activity
            WHERE state != 'idle'
        """
        records = await db_manager.fetch(query)
        return [dict(r) for r in records]

import logging
from typing import List, Dict, Any
from .queries import text_queries, embedding_queries

logger = logging.getLogger(__name__)

class ClassicalSearchEngine:
    """
    Advanced search engine combining full-text search and semantic embeddings.
    Designed specifically for Classical Scholars using LOGOS.
    """

    def __init__(self):
        self.min_relevance_score = 0.1

    async def semantic_search(self, query: str, limit: int = 20) -> List[Dict[str, Any]]:
        """
        Performs a hybrid search:
        1. Finds similar words via embeddings.
        2. Searches texts for those related concepts.
        """
        words = query.lower().split()
        expanded_terms = [query]
        
        for word in words:
            similar = await embedding_queries.find_similar_words(word, limit=3)
            expanded_terms.extend([s['word'] for s in similar])

        # Dedup terms
        expanded_terms = list(set(expanded_terms))
        logger.info(f"Expanded search terms: {expanded_terms}")

        results = []
        for term in expanded_terms:
            hits = await text_queries.search_texts(term, limit=limit // 2)
            results.extend(hits)

        # Sort by rank and remove duplicates
        seen_ids = set()
        unique_results = []
        # Sort by rank if available, otherwise by title
        sorted_results = sorted(results, key=lambda x: x.get('rank', 0), reverse=True)

        for res in sorted_results:
            if res['id'] not in seen_ids:
                unique_results.append(res)
                seen_ids.add(res['id'])

        return unique_results[:limit]

    async def filtered_search(self, 
                               author: Optional[str] = None, 
                               translator: Optional[str] = None, 
                               genre: Optional[str] = None) -> List[Dict[str, Any]]:
        """
        Performs a highly filtered search for specific scholarly research.
        """
        base_query = "SELECT * FROM texts t"
        joins = []
        conditions = []
        params = []
        param_idx = 1

        if author:
            conditions.append(f"t.author = ${param_idx}")
            params.append(author)
            param_idx += 1
        
        if translator:
            conditions.append(f"t.translator = ${param_idx}")
            params.append(translator)
            param_idx += 1

        if genre:
            joins.append("JOIN author_profiles ap ON t.author = ap.name")
            conditions.append(f"${param_idx} = ANY(ap.genres)")
            params.append(genre)
            param_idx += 1

        sql = base_query
        if joins:
            sql += " " + " ".join(joins)
        if conditions:
            sql += " WHERE " + " AND ".join(conditions)
        
        sql += " LIMIT 50"
        
        records = await db_manager.fetch(sql, *params)
        return [dict(r) for r in records]
