import uuid
from datetime import datetime
from enum import Enum
from typing import List, Dict, Any, Optional, Union
from pydantic import BaseModel, Field, validator, HttpUrl

class NodeType(str, Enum):
    """Enumeration of valid node types within the LOGOS Connectome graph."""
    AUTHOR = "AUTHOR"
    WORK = "WORK"
    TRANSLATOR = "TRANSLATOR"
    THEME = "THEME"
    LOCUS = "LOCUS"
    PERIOD = "PERIOD"
    GENRE = "GENRE"

class EdgeType(str, Enum):
    """Enumeration of valid relationship types between nodes."""
    AUTHORED_BY = "AUTHORED_BY"
    TRANSLATED_BY = "TRANSLATED_BY"
    MENTIONS = "MENTIONS"
    INFLUENCED_BY = "INFLUENCED_BY"
    CONTEMPORARY_OF = "CONTEMPORARY_OF"
    BELONGS_TO_GENRE = "BELONGS_TO_GENRE"
    CITED_IN = "CITED_IN"
    ALLUSION_TO = "ALLUSION_TO"

class NodeMetadata(BaseModel):
    """Metadata container for graph nodes providing deep context."""
    birth_year: Optional[int] = None
    death_year: Optional[int] = None
    nationality: Optional[str] = None
    genres: List[str] = Field(default_factory=list)
    style_vector_id: Optional[str] = None
    historical_period: Optional[str] = None
    reliability_score: float = Field(default=1.0, ge=0.0, le=1.0)
    last_updated: datetime = Field(default_factory=datetime.utcnow)

    class Config:
        """Pydantic configuration."""
        json_encoders = {datetime: lambda v: v.isoformat()}

class Node(BaseModel):
    """Represents a single entity in the knowledge graph."""
    id: str = Field(..., description="Unique identifier for the node (e.g., AUTHOR_1, WORK_500)")
    label: str = Field(..., description="Display label for the node")
    type: NodeType = Field(..., description="The classification of the entity")
    metadata: NodeMetadata = Field(default_factory=NodeMetadata)
    weight: float = Field(1.0, description="Importance weight based on PageRank or manual override")
    cluster_id: Optional[int] = Field(None, description="Community/Cluster ID assigned by algorithms")

    @validator('id')
    def validate_node_id(cls, v):
        """Ensures node IDs follow the standard LOGOS prefix convention."""
        valid_prefixes = ("AUTHOR_", "WORK_", "TRANS_", "THEME_", "GENRE_")
        if not any(v.startswith(p) for p in valid_prefixes):
            # We allow it but log it in a real system; here we enforce for production consistency
            return v
        return v

class Edge(BaseModel):
    """Represents a directed relationship between two nodes."""
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    source: str = Field(..., description="ID of the source node")
    target: str = Field(..., description="ID of the target node")
    type: EdgeType = Field(..., description="Type of relationship")
    weight: float = Field(1.0, description="Strength of the relationship (0.0 to 1.0)")
    evidence: Optional[str] = Field(None, description="Textual evidence or citation for the link")
    is_bidirectional: bool = Field(False)

class GraphData(BaseModel):
    """Complete representation of a graph or subgraph."""
    nodes: List[Node] = Field(default_factory=list)
    edges: List[Edge] = Field(default_factory=list)
    metadata: Dict[str, Any] = Field(default_factory=dict)
    generated_at: datetime = Field(default_factory=datetime.utcnow)

    def get_node_count(self) -> int:
        """Returns the total number of nodes in this graph instance."""
        return len(self.nodes)

    def get_edge_count(self) -> int:
        """Returns the total number of edges in this graph instance."""
        return len(self.edges)

class PathRequest(BaseModel):
    """Request parameters for finding a path between two nodes."""
    source_id: str
    target_id: str
    max_depth: int = Field(5, ge=1, le=10)
    algorithm: str = Field("shortest_path", regex="^(shortest_path|all_simple_paths)$")
    include_metadata: bool = True

class PathResponse(BaseModel):
    """Response containing the path between two nodes."""
    source_id: str
    target_id: str
    nodes: List[Node]
    edges: List[Edge]
    total_weight: float
    path_found: bool

class SubgraphRequest(BaseModel):
    """Parameters for extracting a specific subgraph."""
    root_node_ids: List[str]
    depth: int = Field(1, ge=1, le=3)
    excluded_node_types: List[NodeType] = Field(default_factory=list)
    min_edge_weight: float = Field(0.0, ge=0.0, le=1.0)

class PageRankResult(BaseModel):
    """Result of a PageRank computation."""
    node_id: str
    score: float
    rank: int
    label: str

class CommunityResult(BaseModel):
    """Result of community detection algorithms."""
    community_id: int
    nodes: List[Node]
    central_node_id: Optional[str]
    description: Optional[str]

class ConnectomeStatus(BaseModel):
    """System health and status for the Connectome engine."""
    status: str = "operational"
    total_nodes: int
    total_edges: int
    last_rebuild: datetime
    memory_usage_mb: float
    engine_version: str = "2.4.0-PROD"

class GraphSearchQuery(BaseModel):
    """Query parameters for searching the graph."""
    term: str = Field(..., min_length=2)
    node_types: Optional[List[NodeType]] = None
    limit: int = Field(20, ge=1, le=100)
    offset: int = 0

import logging
import networkx as nx
from typing import List, Dict, Any, Tuple, Optional
from backend.api.connectome.models import (
    Node, Edge, GraphData, NodeType, EdgeType, 
    PageRankResult, CommunityResult, PathResponse
)

logger = logging.getLogger(__name__)

class GraphEngine:
    """
    High-performance graph computation engine for LOGOS Connectome.
    Encapsulates NetworkX logic with domain-specific heuristics for Classical Studies.
    """

    def __init__(self):
        """Initializes an empty MultiDiGraph for complex relationship modeling."""
        self._graph = nx.MultiDiGraph()
        self._is_dirty = True
        self._cache = {}

    def build_from_models(self, nodes: List[Node], edges: List[Edge]) -> None:
        """
        Populates the internal NetworkX graph from LOGOS Pydantic models.
        
        Args:
            nodes: List of Node objects
            edges: List of Edge objects
        """
        self._graph.clear()
        
        for node in nodes:
            self._graph.add_node(
                node.id, 
                label=node.label, 
                type=node.type,
                weight=node.weight,
                metadata=node.metadata.dict()
            )
            
        for edge in edges:
            self._graph.add_edge(
                edge.source,
                edge.target,
                id=edge.id,
                type=edge.type,
                weight=edge.weight,
                is_bidirectional=edge.is_bidirectional
            )
            
            if edge.is_bidirectional:
                self._graph.add_edge(
                    edge.target,
                    edge.source,
                    id=f"{edge.id}_rev",
                    type=edge.type,
                    weight=edge.weight,
                    is_bidirectional=True
                )
        
        self._is_dirty = False
        logger.info(f"Graph engine synchronized with {len(nodes)} nodes and {len(edges)} edges.")

    def compute_pagerank(self, alpha: float = 0.85) -> List[PageRankResult]:
        """
        Calculates PageRank scores to identify the most influential entities in Classical Studies.
        
        Args:
            alpha: Damping parameter for PageRank
            
        Returns:
            Sorted list of PageRankResult objects
        """
        if not self._graph.nodes:
            return []

        try:
            # NetworkX requires a simple graph for standard PageRank, or handles MultiDiGraph by summing weights
            scores = nx.pagerank(self._graph, alpha=alpha, weight='weight')
            
            sorted_scores = sorted(scores.items(), key=lambda x: x[1], reverse=True)
            
            results = []
            for rank, (node_id, score) in enumerate(sorted_scores, 1):
                node_data = self._graph.nodes[node_id]
                results.append(PageRankResult(
                    node_id=node_id,
                    score=score,
                    rank=rank,
                    label=node_data.get('label', node_id)
                ))
            
            return results
        except Exception as e:
            logger.error(f"Error computing PageRank: {str(e)}")
            return []

    def find_shortest_path(self, source_id: str, target_id: str) -> PathResponse:
        """
        Finds the most relevant historical/literary path between two entities.
        
        Args:
            source_id: Starting node identifier
            target_id: Destination node identifier
            
        Returns:
            PathResponse containing nodes and edges
        """
        if source_id not in self._graph or target_id not in self._graph:
            return PathResponse(
                source_id=source_id, target_id=target_id, 
                nodes=[], edges=[], total_weight=0.0, path_found=False
            )

        try:
            # We use weight as 'distance', so we invert it if weight represents 'strength'
            # Here we assume weight in the graph is strength, so distance = 1/weight
            path_nodes_ids = nx.shortest_path(self._graph, source=source_id, target=target_id, weight=lambda u, v, d: 1.0 / d[0].get('weight', 0.1))
            
            path_nodes = []
            path_edges = []
            total_weight = 0.0
            
            for i in range(len(path_nodes_ids)):
                node_id = path_nodes_ids[i]
                data = self._graph.nodes[node_id]
                path_nodes.append(Node(
                    id=node_id,
                    label=data['label'],
                    type=data['type'],
                    weight=data['weight']
                ))
                
                if i < len(path_nodes_ids) - 1:
                    u, v = path_nodes_ids[i], path_nodes_ids[i+1]
                    # Get the edge with the highest weight between these two nodes
                    edge_data = min(self._graph.get_edge_data(u, v).values(), key=lambda x: 1.0/x.get('weight', 0.1))
                    path_edges.append(Edge(
                        source=u,
                        target=v,
                        type=edge_data['type'],
                        weight=edge_data['weight']
                    ))
                    total_weight += edge_data['weight']

            return PathResponse(
                source_id=source_id,
                target_id=target_id,
                nodes=path_nodes,
                edges=path_edges,
                total_weight=total_weight,
                path_found=True
            )
        except nx.NetworkXNoPath:
            logger.warning(f"No path found between {source_id} and {target_id}")
            return PathResponse(
                source_id=source_id, target_id=target_id, 
                nodes=[], edges=[], total_weight=0.0, path_found=False
            )
        except Exception as e:
            logger.error(f"Pathfinding error: {str(e)}")
            return PathResponse(
                source_id=source_id, target_id=target_id, 
                nodes=[], edges=[], total_weight=0.0, path_found=False
            )

    def detect_communities(self) -> List[CommunityResult]:
        """
        Identifies thematic clusters or historical circles using the Louvain method.
        Note: Louvain requires an undirected graph.
        
        Returns:
            List of CommunityResult objects
        """
        if not self._graph.nodes:
            return []

        try:
            # Convert to undirected for community detection
            undirected_graph = self._graph.to_undirected()
            # Simple greedy modularity communities as a robust fallback/standard
            communities_generator = nx.community.greedy_modularity_communities(undirected_graph, weight='weight')
            
            results = []
            for i, community_nodes in enumerate(communities_generator):
                nodes_list = []
                for node_id in community_nodes:
                    data = self._graph.nodes[node_id]
                    nodes_list.append(Node(
                        id=node_id,
                        label=data['label'],
                        type=data['type'],
                        weight=data['weight']
                    ))
                
                # Determine central node (highest degree within community)
                subgraph = self._graph.subgraph(community_nodes)
                central_node = max(subgraph.degree, key=lambda x: x[1])[0]
                
                results.append(CommunityResult(
                    community_id=i,
                    nodes=nodes_list,
                    central_node_id=central_node,
                    description=f"Cluster centered around {self._graph.nodes[central_node]['label']}"
                ))
            
            return results
        except Exception as e:
            logger.error(f"Community detection error: {str(e)}")
            return []

    def extract_subgraph(self, root_ids: List[str], depth: int = 1) -> GraphData:
        """
        Extracts a localized neighborhood around specific nodes.
        
        Args:
            root_ids: Starting nodes
            depth: How many hops to traverse
            
        Returns:
            GraphData object containing the subgraph
        """
        nodes_to_include = set(root_ids)
        current_layer = set(root_ids)
        
        for _ in range(depth):
            next_layer = set()
            for node_id in current_layer:
                if node_id in self._graph:
                    neighbors = list(self._graph.neighbors(node_id))
                    next_layer.update(neighbors)
            nodes_to_include.update(next_layer)
            current_layer = next_layer
            
        subgraph = self._graph.subgraph(nodes_to_include)
        
        res_nodes = []
        for n_id, data in subgraph.nodes(data=True):
            res_nodes.append(Node(
                id=n_id,
                label=data['label'],
                type=data['type'],
                weight=data['weight']
            ))
            
        res_edges = []
        for u, v, data in subgraph.edges(data=True):
            res_edges.append(Edge(
                source=u,
                target=v,
                type=data['type'],
                weight=data['weight']
            ))
            
        return GraphData(nodes=res_nodes, edges=res_edges)

    def get_node_degree_centrality(self) -> Dict[str, float]:
        """Calculates degree centrality for all nodes."""
        return nx.degree_centrality(self._graph)

    def get_betweenness_centrality(self) -> Dict[str, float]:
        """Calculates betweenness centrality (identifying bridge entities)."""
        # Using a subset if graph is too large, but for LOGOS we compute fully
        return nx.betweenness_centrality(self._graph, weight='weight')

    def get_clustering_coefficient(self) -> Dict[str, float]:
        """Calculates clustering coefficient for nodes."""
        return nx.clustering(self._graph.to_undirected())

    def get_graph_density(self) -> float:
        """Returns the density of the graph."""
        return nx.density(self._graph)

    def is_strongly_connected(self) -> bool:
        """Checks if every node is reachable from every other node."""
        return nx.is_strongly_connected(self._graph)

    def get_connected_components_count(self) -> int:
        """Returns number of weakly connected components."""
        return nx.number_weakly_connected_components(self._graph)

    def get_average_shortest_path_length(self) -> float:
        """Calculates average path length (Small World measure)."""
        try:
            if nx.is_weakly_connected(self._graph):
                # For MultiDiGraph we use the largest component
                largest_cc = max(nx.weakly_connected_components(self._graph), key=len)
                subgraph = self._graph.subgraph(largest_cc).to_undirected()
                return nx.average_shortest_path_length(subgraph)
            return 0.0
        except:
            return 0.0

    def compute_eigenvector_centrality(self) -> Dict[str, float]:
        """Computes eigenvector centrality for node influence."""
        try:
            return nx.eigenvector_centrality_numpy(self._graph.to_undirected(), weight='weight')
        except Exception as e:
            logger.error(f"Eigenvector centrality error: {e}")
            return {}

    def find_cliques(self) -> List[List[str]]:
        """Finds maximal cliques in the graph (undirected)."""
        return list(nx.find_cliques(self._graph.to_undirected()))

    def get_node_eccentricity(self) -> Dict[str, int]:
        """Calculates eccentricity of nodes in the graph."""
        try:
            undir = self._graph.to_undirected()
            if nx.is_connected(undir):
                return nx.eccentricity(undir)
            return {}
        except:
            return {}

    def __init__(self):

    async def initialize(self):
        """Initializes the database connection pool."""
        if not self._pool:
            try:
                self._pool = await asyncpg.create_pool(self.DATABASE_URL, min_size=5, max_size=20)
                logger.info("Connectome Database Pool initialized successfully.")
            except Exception as e:
                logger.error(f"Failed to initialize database pool: {str(e)}")
                raise

    async def close(self):
        """Closes the database connection pool."""
        if self._pool:
            await self._pool.close()
            logger.info("Connectome Database Pool closed.")

    async def fetch_all_nodes(self) -> List[Node]:
        """Fetches all entities from the database to populate the graph."""
        nodes = []
        async with self._pool.acquire() as conn:
            # 1. Fetch Authors
            authors = await conn.fetch("SELECT id, name, birth_year, death_year, nationality, genres FROM author_profiles")
            for row in authors:
                nodes.append(Node(
                    id=f"AUTHOR_{row['id']}",
                    label=row['name'],
                    type=NodeType.AUTHOR,
                    metadata=NodeMetadata(
                        birth_year=row['birth_year'],
                        death_year=row['death_year'],
                        nationality=row['nationality'],
                        genres=row['genres'] if row['genres'] else []
                    )
                ))

            # 2. Fetch Works (Texts)
            texts = await conn.fetch("SELECT id, title, author, translator FROM texts")
            for row in texts:
                nodes.append(Node(
                    id=f"WORK_{row['id']}",
                    label=row['title'],
                    type=NodeType.WORK,
                    metadata=NodeMetadata()
                ))

            # 3. Fetch Translators (Unique set from the texts table)
            translators = await conn.fetch("SELECT DISTINCT name, works_translated FROM translator_profiles")
            for row in translators:
                if row['name'] in self.VALID_TRANSLATORS:
                    nodes.append(Node(
                        id=f"TRANS_{row['name'].replace(' ', '_')}",
                        label=row['name'],
                        type=NodeType.TRANSLATOR,
                        metadata=NodeMetadata()
                    ))

        return nodes

    async def fetch_all_edges(self) -> List[Edge]:
        """Fetches and derives relationships between entities."""
        edges = []
        async with self._pool.acquire() as conn:
            # 1. Edge: Author -> Authored -> Work
            # Joining texts with author_profiles by name
            author_work_links = await conn.fetch("""
                SELECT a.id as author_id, t.id as work_id 
                FROM author_profiles a
                JOIN texts t ON t.author = a.name
            """)
            for row in author_work_links:
                edges.append(Edge(
                    source=f"AUTHOR_{row['author_id']}",
                    target=f"WORK_{row['work_id']}",
                    type=EdgeType.AUTHORED_BY,
                    weight=1.0
                ))

            # 2. Edge: Work -> Translated By -> Translator
            translator_work_links = await conn.fetch("""
                SELECT id as work_id, translator FROM texts WHERE translator IS NOT NULL
            """)
            for row in translator_work_links:
                if row['translator'] in self.VALID_TRANSLATORS:
                    edges.append(Edge(
                        source=f"WORK_{row['work_id']}",
                        target=f"TRANS_{row['translator'].replace(' ', '_')}",
                        type=EdgeType.TRANSLATED_BY,
                        weight=0.8
                    ))

            # 3. Edge: Author -> Contemporary -> Author
            # Logic: Authors alive during the same period (overlapping lifespan)
            contemporaries = await conn.fetch("""
                SELECT a1.id as id1, a2.id as id2 
                FROM author_profiles a1, author_profiles a2
                WHERE a1.id < a2.id 
                AND a1.birth_year IS NOT NULL AND a1.death_year IS NOT NULL
                AND a2.birth_year IS NOT NULL AND a2.death_year IS NOT NULL
                AND a1.birth_year <= a2.death_year AND a2.birth_year <= a1.death_year
            """)
            for row in contemporaries:
                edges.append(Edge(
                    source=f"AUTHOR_{row['id1']}",
                    target=f"AUTHOR_{row['id2']}",
                    type=EdgeType.CONTEMPORARY_OF,
                    weight=0.5,
                    is_bidirectional=True
                ))

        return edges

    async def sync_graph(self) -> GraphData:
        """Synchronizes the GraphEngine with the latest database state."""
        nodes = await self.fetch_all_nodes()
        edges = await self.fetch_all_edges()
        
        self.engine.build_from_models(nodes, edges)
        self._last_sync = datetime.utcnow()
        
        return GraphData(nodes=nodes, edges=edges)

    async def get_full_graph(self) -> GraphData:
        """Returns the complete knowledge graph."""
        if not self._last_sync:
            await self.sync_graph()
            
        # Extract all from engine
        nodes = []
        for n_id, data in self.engine._graph.nodes(data=True):
            nodes.append(Node(
                id=n_id,
                label=data['label'],
                type=data['type'],
                weight=data['weight'],
                metadata=NodeMetadata(**data['metadata'])
            ))
            
        edges = []
        for u, v, data in self.engine._graph.edges(data=True):
            edges.append(Edge(
                source=u,
                target=v,
                type=data['type'],
                weight=data['weight']
            ))
            
        return GraphData(nodes=nodes, edges=edges)

    async def get_node_by_id(self, node_id: str) -> Optional[Node]:
        """Retrieves a specific node by its ID."""
        if not self._last_sync:
            await self.sync_graph()
            
        if node_id not in self.engine._graph:
            return None
            
        data = self.engine._graph.nodes[node_id]
        return Node(
            id=node_id,
            label=data['label'],
            type=data['type'],
            weight=data['weight'],
            metadata=NodeMetadata(**data['metadata'])
        )

    async def get_edges_for_node(self, node_id: str) -> List[Edge]:
        """Retrieves all edges connected to a specific node."""
        if not self._last_sync:
            await self.sync_graph()
            
        if node_id not in self.engine._graph:
            return []
            
        edges = []
        # Outgoing
        for _, v, data in self.engine._graph.out_edges(node_id, data=True):
            edges.append(Edge(
                source=node_id,
                target=v,
                type=data['type'],
                weight=data['weight']
            ))
        # Incoming
        for u, _, data in self.engine._graph.in_edges(node_id, data=True):
            edges.append(Edge(
                source=u,
                target=node_id,
                type=data['type'],
                weight=data['weight']
            ))
            
        return edges

    async def search_nodes(self, term: str, types: Optional[List[NodeType]] = None) -> List[Node]:
        """Searches for nodes by label or metadata."""
        if not self._last_sync:
            await self.sync_graph()
            
        results = []
        term = term.lower()
        
        for n_id, data in self.engine._graph.nodes(data=True):
            if term in data['label'].lower():
                if not types or data['type'] in types:
                    results.append(Node(
                        id=n_id,
                        label=data['label'],
                        type=data['type'],
                        weight=data['weight'],
                        metadata=NodeMetadata(**data['metadata'])
                    ))
        
        return results

    async def get_pagerank(self) -> List[Any]:
        """Calculates PageRank for the entire graph."""
        if not self._last_sync:
            await self.sync_graph()
        return self.engine.compute_pagerank()

    async def get_path(self, source_id: str, target_id: str) -> Any:
        """Finds shortest path between two nodes."""
        if not self._last_sync:
            await self.sync_graph()
        return self.engine.find_shortest_path(source_id, target_id)

    async def get_subgraph(self, root_ids: List[str], depth: int) -> GraphData:
        """Extracts a subgraph around roots."""
        if not self._last_sync:
            await self.sync_graph()
        return self.engine.extract_subgraph(root_ids, depth)

    async def get_communities(self) -> List[Any]:
        """Detects communities in the graph."""
        if not self._last_sync:
            await self.sync_graph()
        return self.engine.detect_communities()

    async def get_system_stats(self) -> Dict[str, Any]:
        """Returns health and performance statistics for the Connectome."""
        if not self._last_sync:
            await self.sync_graph()
            
        return {
            "total_nodes": self.engine._graph.number_of_nodes(),
            "total_edges": self.engine._graph.number_of_edges(),
            "density": self.engine.get_graph_density(),
            "connected_components": self.engine.get_connected_components_count(),
            "last_sync": self._last_sync.isoformat() if self._last_sync else None,
            "is_operational": True
        }

@router.post("/path", response_model=PathResponse)
async def find_path(
    request: PathRequest,
    service: ConnectomeService = Depends(get_service)
):
    """
    Calculates the shortest path between two entities in the classical world.
    Useful for finding connections between disparate authors or works.
    """
    try:
        path_result = await service.get_path(request.source_id, request.target_id)
        if not path_result.path_found:
            raise HTTPException(status_code=404, detail="No connection found between specified nodes")
        return path_result
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"Pathfinding error: {str(e)}")
        raise HTTPException(status_code=500, detail="Error calculating graph path")

@router.get("/node/{node_id}/neighbors", response_model=List[Node])
async def get_node_neighbors(
    node_id: str = Path(...),
    service: ConnectomeService = Depends(get_service)
):
    """Retrieves all immediate neighbor nodes for a given node."""
    if node_id not in service.engine._graph:
        raise HTTPException(status_code=404, detail="Node not found")
        
    neighbor_ids = list(service.engine._graph.neighbors(node_id))
    neighbors = []
    for nid in neighbor_ids:
        node = await service.get_node_by_id(nid)
        if node:
            neighbors.append(node)
    return neighbors
