import React, { useState, useEffect, useRef, useCallback, ChangeEvent, KeyboardEvent } from 'react';
import { Search, X, History, BookOpen, User, Loader2, ArrowRight, Star } from 'lucide-react';
import { debounce } from 'lodash';

/**
 * LOGOS SearchBar Component
 * 
 * A high-performance search input with real-time suggestions, 
 * keyboard navigation, and integrated search history.
 * 
 * @version 1.0.0
 * @author Senior Software Architect
 */

interface Suggestion {
  id: string;
  text: string;
  type: 'author' | 'work' | 'text' | 'translator';
  metadata?: string;
}

interface SearchBarProps {
  onSearch: (query: string) => void;
  initialValue?: string;
  placeholder?: string;
  isLoading?: boolean;
}

export const SearchBar: React.FC<SearchBarProps> = ({
  onSearch,
  initialValue = '',
  placeholder = 'Search the Classical Corpus (e.g., "Homer", "virtue in Republic")...',
  isLoading = false
}) => {
  const [query, setQuery] = useState<string>(initialValue);
  const [suggestions, setSuggestions] = useState<Suggestion[]>([]);
  const [showSuggestions, setShowSuggestions] = useState<boolean>(false);
  const [selectedIndex, setSelectedIndex] = useState<number>(-1);
  const [isFetchingSuggestions, setIsFetchingSuggestions] = useState<boolean>(false);
  const [searchHistory, setSearchHistory] = useState<string[]>([]);
  
  const inputRef = useRef<HTMLInputElement>(null);
  const suggestionRef = useRef<HTMLDivElement>(null);

  /**
   * Loads search history from local storage on mount
   */
  useEffect(() => {
    try {
      const savedHistory = localStorage.getItem('logos_search_history');
      if (savedHistory) {
        setSearchHistory(JSON.parse(savedHistory).slice(0, 5));
      }
    } catch (error) {
      console.error('[SearchBar] Failed to load search history:', error);
    }
  }, []);

  /**
   * Fetches suggestions from the LOGOS API based on the texts and author_profiles tables.
   * Implementation uses a debounced approach to prevent API flooding.
   */
  const fetchSuggestions = useCallback(
    debounce(async (searchQuery: string) => {
      if (searchQuery.length < 3) {
        setSuggestions([]);
        return;
      }

      setIsFetchingSuggestions(true);
      try {
        // In a production environment, this calls our FastAPI backend
        // which queries the 'texts' and 'author_profiles' tables.
        const response = await fetch(`/api/v1/search/suggest?q=${encodeURIComponent(searchQuery)}`);
        if (!response.ok) throw new Error('Network response was not ok');
        
        const data = await response.json();
        setSuggestions(data.suggestions);
      } catch (error) {
        console.error('[SearchBar] Suggestion fetch error:', error);
        // Fallback to empty suggestions on error to maintain UI stability
        setSuggestions([]);
      } finally {
        setIsFetchingSuggestions(false);
      }
    }, 300),
    []
  );

  useEffect(() => {
    if (query.length >= 3) {
      fetchSuggestions(query);
    } else {
      setSuggestions([]);
    }
  }, [query, fetchSuggestions]);

  /**
   * Handles the main search execution
   */
  const executeSearch = (searchQuery: string) => {
    const trimmedQuery = searchQuery.trim();
    if (!trimmedQuery) return;

    // Update history
    const updatedHistory = [trimmedQuery, ...searchHistory.filter(h => h !== trimmedQuery)].slice(0, 10);
    setSearchHistory(updatedHistory);
    localStorage.setItem('logos_search_history', JSON.stringify(updatedHistory));

    setShowSuggestions(false);
    onSearch(trimmedQuery);
  };

  /**
   * Keyboard navigation logic for accessibility and power users
   */
  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setSelectedIndex(prev => (prev < suggestions.length - 1 ? prev + 1 : prev));
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setSelectedIndex(prev => (prev > -1 ? prev - 1 : prev));
    } else if (e.key === 'Enter') {
      if (selectedIndex >= 0 && suggestions[selectedIndex]) {
        setQuery(suggestions[selectedIndex].text);
        executeSearch(suggestions[selectedIndex].text);
      } else {
        executeSearch(query);
      }
    } else if (e.key === 'Escape') {
      setShowSuggestions(false);
    }
  };

  const handleInputChange = (e: ChangeEvent<HTMLInputElement>) => {
    setQuery(e.target.value);
    setShowSuggestions(true);
    setSelectedIndex(-1);
  };

  const clearSearch = () => {
    setQuery('');
    setSuggestions([]);
    inputRef.current?.focus();
  };

  return (
    <div className="relative w-full max-w-4xl mx-auto z-50">
      <div className={`relative flex items-center w-full bg-white border-2 rounded-xl transition-all duration-200 shadow-sm
        ${showSuggestions ? 'border-blue-500 ring-4 ring-blue-50/50' : 'border-slate-200 hover:border-slate-300'}`}>
        
        <div className="pl-4 text-slate-400">
          {isLoading ? (
            <Loader2 className="w-5 h-5 animate-spin text-blue-500" />
          ) : (
            <Search className="w-5 h-5" />
          )}
        </div>

        <input
          ref={inputRef}
          type="text"
          value={query}
          onChange={handleInputChange}
          onKeyDown={handleKeyDown}
          onFocus={() => setShowSuggestions(true)}
          placeholder={placeholder}
          className="w-full py-4 px-4 text-lg text-slate-800 bg-transparent focus:outline-none placeholder:text-slate-400"
          aria-label="Search classical texts"
          autoComplete="off"
        />

        {query && (
          <button
            onClick={clearSearch}
            className="p-2 mr-1 text-slate-400 hover:text-slate-600 hover:bg-slate-100 rounded-full transition-colors"
            title="Clear search"
          >
            <X className="w-5 h-5" />
          </button>
        )}

        <button
          onClick={() => executeSearch(query)}
          className="mr-2 px-6 py-2 bg-blue-600 hover:bg-blue-700 text-white font-semibold rounded-lg transition-all active:scale-95 shadow-md"
        >
          Search
        </button>
      </div>

      {/* Suggestion Dropdown */}
      {showSuggestions && (query.length > 0 || searchHistory.length > 0) && (
        <div 
          ref={suggestionRef}
          className="absolute top-full left-0 right-0 mt-2 bg-white border border-slate-200 rounded-xl shadow-2xl overflow-hidden animate-in fade-in slide-in-from-top-2 duration-200"
        >
          {/* History Section */}
          {query.length === 0 && searchHistory.length > 0 && (
            <div className="p-2">
              <div className="px-3 py-2 text-xs font-bold text-slate-500 uppercase tracking-wider flex items-center">
                <History className="w-3 h-3 mr-2" /> Recent Searches
              </div>
              {searchHistory.map((item, idx) => (
                <div
                  key={`history-${idx}`}
                  onClick={() => {
                    setQuery(item);
                    executeSearch(item);
                  }}
                  className="flex items-center px-4 py-3 hover:bg-slate-50 cursor-pointer rounded-lg transition-colors group"
                >
                  <History className="w-4 h-4 mr-3 text-slate-400" />
                  <span className="text-slate-700 flex-grow">{item}</span>
                  <ArrowRight className="w-4 h-4 text-slate-300 opacity-0 group-hover:opacity-100 transition-opacity" />
                </div>
              ))}
            </div>
          )}

          {/* Real-time Suggestions */}
          {query.length > 0 && (
            <div className="p-2">
              {isFetchingSuggestions && suggestions.length === 0 ? (
                <div className="flex items-center justify-center py-8 text-slate-500">
                  <Loader2 className="w-5 h-5 animate-spin mr-2" />
                  <span>Scanning the archives...</span>
                </div>
              ) : suggestions.length > 0 ? (
                <>
                  <div className="px-3 py-2 text-xs font-bold text-slate-500 uppercase tracking-wider">
                    Suggested Results
                  </div>
                  {suggestions.map((item, idx) => (
                    <div
                      key={item.id}

                        executeSearch(item.text);
                      }}

   * High-performance text highlighting using regex.
   * Wraps matches in a styled mark tag.
   */
  const highlightText = (text: string, query: string) => {
    if (!query.trim()) return text;
    
    // Escape special characters for regex
    const escapedQuery = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    const terms = escapedQuery.split(/\s+/).filter(t => t.length > 2);
    
    if (terms.length === 0) return text;
    
    const regex = new RegExp(`(${terms.join('|')})`, 'gi');
    const parts = text.split(regex);
    
    return (
      <>
        {parts.map((part, i) => 
          regex.test(part) ? (
            <mark key={i} className="bg-yellow-200 text-slate-900 rounded-sm px-0.5 font-medium">
              {part}
            </mark>
          ) : (
            <span key={i}>{part}</span>
          )
        )}
      </>
    );
  };

   * Generates a CSV export of the current results

  const handleExport = async () => {

      const csvData = results.map(r => [
        `"${r.title}"`,
        `"${r.author}"`,
        `"${r.translator}"`,
        `"${r.book || ''} ${r.chapter || ''}"`,
        `"${r.text_content.replace(/"/g, '""')}"`
      ].join(','));
      
      const blob = new Blob([[headers.join(','), ...csvData].join('\n')], { type: 'text/csv' });

      a.href = url;
      a.download = `logos_search_${new Date().toISOString().split('T')[0]}.csv`;
      a.click();

import { useState, useCallback, useRef, useEffect } from 'react';
import { SearchResultItem } from '../SearchResults';

/**
 * useSearch Custom Hook
 * 
 * The central engine for the LOGOS search system.
 * Handles API communication, state management for results,
 * facets, pagination, and aborting stale requests.
 */

interface SearchFilters {
  authors: string[];
  translators: string[];
  genres: string[];
  dateRange: [number, number];
  language: string;
}

interface SearchState {
  results: SearchResultItem[];
  totalCount: number;
  facets: {
    authors: any[];
    works: any[];
    centuries: any[];
  };
  isLoading: boolean;
  error: string | null;
}

interface SearchParams {
  query: string;
  page: number;
  pageSize: number;
  filters: SearchFilters;
}

export const useSearch = () => {
  const [state, setState] = useState<SearchState>({
    results: [],
    totalCount: 0,
    facets: {
      authors: [],
      works: [],
      centuries: []
    },
    isLoading: false,
    error: null
  });

  // Use a ref to store the current AbortController to cancel previous requests
  const abortControllerRef = useRef<AbortController | null>(null);

  /**
   * Primary search function.
   * Calls the backend API which interfaces with the Railway PostgreSQL DB.
   */
  const executeSearch = useCallback(async (params: SearchParams) => {
    const { query, page, pageSize, filters } = params;

    if (!query.trim()) {
      setState(prev => ({ ...prev, results: [], totalCount: 0 }));
      return;
    }

    // Cancel any ongoing requests
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
    }

    // Create new abort controller for this request
    const controller = new AbortController();
    abortControllerRef.current = controller;

    setState(prev => ({ ...prev, isLoading: true, error: null }));

    try {
      // Construction of query parameters for the API
      const queryParams = new URLSearchParams({
        q: query,
        page: page.toString(),
        page_size: pageSize.toString(),
        authors: filters.authors.join(','),
        translators: filters.translators.join(','),
        genres: filters.genres.join(','),
        start_year: filters.dateRange[0].toString(),
        end_year: filters.dateRange[1].toString(),
        language: filters.language
      });

      console.log(`[useSearch] Executing search for: "${query}" (Page ${page})`);

      // In production, this targets the Python/FastAPI backend
      const response = await fetch(`/api/v1/search?${queryParams.toString()}`, {
        signal: controller.signal,
        headers: {
          'Accept': 'application/json',
          'X-Client-ID': 'LOGOS-Frontend-v1'
        }
      });

      if (!response.ok) {
        const errorData = await response.json().catch(() => ({}));
        throw new Error(errorData.detail || `Search failed with status ${response.status}`);
      }

      const data = await response.json();

      // Update state with results and facets
      setState({
        results: data.results,
        totalCount: data.total_count,
        facets: {
          authors: data.facets?.authors || [],
          works: data.facets?.works || [],
          centuries: data.facets?.centuries || []
        },
        isLoading: false,
        error: null
      });

      // Persist to local history
      saveToLocalHistory(query, data.total_count, filters);

    } catch (err: any) {
      // Don't update state if the request was intentionally aborted
      if (err.name === 'AbortError') {
        console.log('[useSearch] Search request aborted');
        return;
      }

      console.error('[useSearch] Fatal search error:', err);
      setState(prev => ({
        ...prev,
        isLoading: false,
        error: 'The Great Library is temporarily unreachable. Please check your connection and try again.'
      }));
    }
  }, []);

  /**
   * Internal helper to persist search history
   */
  const saveToLocalHistory = (query: string, count: number, filters: SearchFilters) => {
    try {
      const historyStr = localStorage.getItem('logos_full_history');
      const history = historyStr ? JSON.parse(historyStr) : [];
      
      const newEntry = {
        id: Math.random().toString(36).substr(2, 9),
        query,
        timestamp: Date.now(),
        resultsCount: count,
        filters
      };

      // Keep only unique queries in the top list
      const filteredHistory = history.filter((h: any) => h.query !== query);
      const updatedHistory = [newEntry, ...filteredHistory].slice(0, 50);
      
      localStorage.setItem('logos_full_history', JSON.stringify(updatedHistory));
    } catch (e) {
      console.warn('[useSearch] Failed to save history:', e);
    }
  };

  /**
   * Cleanup on unmount
   */
  useEffect(() => {
    return () => {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
    };
  }, []);

  /**
   * Specialized function for semantic vector search
   * Uses the 'word_embeddings' table for conceptual matching.
   */
  const executeSemanticSearch = useCallback(async (concept: string) => {
    setState(prev => ({ ...prev, isLoading: true, error: null }));
    
    try {
      const response = await fetch(`/api/v1/search/semantic?concept=${encodeURIComponent(concept)}`);
      if (!response.ok) throw new Error('Semantic search failed');
      
      const data = await response.json();
      setState({
        results: data.results,
        totalCount: data.total_count,
        facets: data.facets,
        isLoading: false,
        error: null
      });
    } catch (err: any) {
      setState(prev => ({ ...prev, isLoading: false, error: err.message }));
    }
  }, []);

  return {
    ...state,
    executeSearch,
    executeSemanticSearch,
    clearError: () => setState(prev => ({ ...prev, error: null }))
  };
};

export default useSearch;
