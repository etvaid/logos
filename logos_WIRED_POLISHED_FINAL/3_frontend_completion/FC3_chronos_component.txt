logger = logging.getLogger(__name__)

class ChronosService:
    """

    """

    DATABASE_URL = "postgresql://postgres:JKLqDvdTtmRjGnOgDvGFLqLKVkcjQLFs@metro.proxy.rlwy.net:58888/railway"
    
    # Strictly enforced list of public domain translators to avoid legal risk

    def __init__(self):

    async def connect(self) -> None:

                    self.DATABASE_URL,
                    min_size=5,
                    max_size=20,
                    command_timeout=60
                )
                logger.info("Connected to Railway PostgreSQL database successfully.")
        except Exception as e:
            logger.error(f"Failed to connect to database: {str(e)}")
            raise ConnectionError(f"Database connection failed: {e}")

    async def get_author_lifespans(self) -> List[Dict[str, Any]]:
        """
        Retrieves birth and death years for all authors to populate the timeline.
        Handles BCE dates by storing them as negative integers.
        """
        if not self.pool:
            await self.connect()
            
        query = """
            SELECT id, name, birth_year, death_year, nationality, genres
            FROM author_profiles
            WHERE birth_year IS NOT NULL AND death_year IS NOT NULL
            ORDER BY birth_year ASC;
        """
        try:
            async with self.pool.acquire() as connection:
                rows = await connection.fetch(query)
                results = []
                for row in rows:
                    results.append({
                        "id": row["id"],
                        "name": row["name"],
                        "start": row["birth_year"],
                        "end": row["death_year"],
                        "nationality": row["nationality"],
                        "genres": row["genres"].split(",") if row["genres"] else []
                    })
                logger.info(f"Retrieved {len(results)} author lifespans.")
                return results
        except Exception as e:
            logger.error(f"Error fetching author lifespans: {str(e)}")
            return []

    async def get_semantic_drift(self, word: str) -> List[Dict[str, Any]]:
        """
        Calculates the semantic drift of a word across different historical periods.
        Uses word embeddings (300-dim) and cross-references with text timestamps.
        """
        if not self.pool:
            await self.connect()

        # This query joins word embeddings with the text content and author dates
        # to see how word usage contexts shift over centuries.
        query = """
            SELECT 
                a.birth_year as period_marker,
                we.vector,
                t.title,
                t.author
            FROM word_embeddings we
            JOIN texts t ON t.text_content ILIKE '%' || we.word || '%'
            JOIN author_profiles a ON t.author = a.name
            WHERE we.word = $1
            ORDER BY a.birth_year ASC;
        """
        try:
            async with self.pool.acquire() as connection:
                rows = await connection.fetch(query, word)
                drift_data = []
                for row in rows:
                    # Convert binary/array vector to list for JSON serialization
                    vector = row["vector"]
                    if isinstance(vector, bytes):
                        vector = np.frombuffer(vector, dtype=np.float32).tolist()
                    
                    drift_data.append({
                        "year": row["period_marker"],
                        "vector_sample": vector[:5],  # Sample for drift visualization
                        "source_title": row["title"],
                        "author": row["author"]
                    })
                logger.info(f"Calculated semantic drift for '{word}' across {len(drift_data)} points.")
                return drift_data
        except Exception as e:
            logger.error(f"Error calculating semantic drift: {str(e)}")
            return []

    async def get_works_timeline(self, author_name: Optional[str] = None) -> List[Dict[str, Any]]:
        """
        Returns a chronological list of works, filtered by author if provided.
        Only includes works by authorized translators.
        """
        if not self.pool:
            await self.connect()

        base_query = """
            SELECT t.id, t.title, t.author, t.translator, a.birth_year
            FROM texts t
            JOIN author_profiles a ON t.author = a.name
            WHERE t.translator = ANY($1)
        """
        params = [self.ALLOWED_TRANSLATORS]
        
        if author_name:
            base_query += " AND t.author = $2"
            params.append(author_name)
            
        base_query += " ORDER BY a.birth_year ASC, t.title ASC"

        try:
            async with self.pool.acquire() as connection:
                rows = await connection.fetch(base_query, *params)
                works = []
                for row in rows:
                    works.append({
                        "id": row["id"],
                        "title": row["title"],
                        "author": row["author"],
                        "translator": row["translator"],
                        "estimated_date": row["birth_year"] + 30  # Estimated peak of career
                    })
                return works
        except Exception as e:
            logger.error(f"Error fetching works timeline: {str(e)}")
            return []

    async def get_period_statistics(self) -> Dict[str, Any]:
        """
        Generates comparative statistics across major historical periods:
        Archaic, Classical, Hellenistic, and Roman.
        """
        if not self.pool:
            await self.connect()

        periods = {
            "Archaic": (-800, -480),
            "Classical": (-480, -323),
            "Hellenistic": (-323, -31),
            "Roman": (-31, 476)
        }
        
        stats = {}
        try:
            async with self.pool.acquire() as connection:
                for name, (start, end) in periods.items():
                    count_query = "SELECT COUNT(*) FROM author_profiles WHERE birth_year BETWEEN $1 AND $2"
                    work_query = """
                        SELECT COUNT(t.id) 
                        FROM texts t 
                        JOIN author_profiles a ON t.author = a.name 
                        WHERE a.birth_year BETWEEN $1 AND $2
                    """
                    
                    authors_count = await connection.fetchval(count_query, start, end)
                    works_count = await connection.fetchval(work_query, start, end)
                    
                    stats[name] = {
                        "author_count": authors_count,
                        "works_count": works_count,
                        "range": [start, end]
                    }
                return stats
        except Exception as e:
            logger.error(f"Error generating period statistics: {str(e)}")
            return {}

    async def disconnect(self) -> None:
        """Gracefully closes the database connection pool."""
        if self.pool:
            await self.pool.close()
            logger.info("Database connection pool closed.")

# Usage Example (Internal Logic)
async def main():
    service = ChronosService()
    await service.connect()
    authors = await service.get_author_lifespans()
    print(f"Loaded {len(authors)} authors.")
    await service.disconnect()

if __name__ == "__main__":
    asyncio.run(main())

import React, { useEffect, useRef, useState } from 'react';
import { motion } from 'framer-motion';
import { Activity, Wind, Info, RefreshCw } from 'lucide-react';

/**
 * DriftAnimation visualizes semantic shifts in word meanings over time.
 * It uses a particle system to represent the transformation of word embeddings
 * across different historical periods.
 */

interface DriftPoint {
    year: number;
    vector: number[];
    label: string;
}

const DriftAnimation: React.FC<{ word: string }> = ({ word }) => {
    const canvasRef = useRef<HTMLCanvasElement>(null);
    const [driftData, setDriftData] = useState<DriftPoint[]>([]);
    const [isPlaying, setIsPlaying] = useState(true);
    const [currentTime, setCurrentTime] = useState(0);

    // Simulation parameters
    const PARTICLE_COUNT = 150;
    const particles = useRef<any[]>([]);

    useEffect(() => {
        // Initialize particles
        const p = [];
        for (let i = 0; i < PARTICLE_COUNT; i++) {
            p.push({
                x: Math.random() * 800,
                y: Math.random() * 400,
                vx: (Math.random() - 0.5) * 2,
                vy: (Math.random() - 0.5) * 2,
                size: Math.random() * 3 + 1,
                color: `hsla(${Math.random() * 60 + 20}, 70%, 50%, ${Math.random() * 0.5 + 0.2})`
            });
        }
        particles.current = p;

        // Fetch drift data from backend
        const fetchDrift = async () => {
            try {
                const response = await fetch(`/api/chronos/drift?word=${word}`);
                const data = await response.json();
                setDriftData(data);
            } catch (e) {
                console.error("Semantic Drift Fetch Failed", e);
            }
        };
        fetchDrift();
    }, [word]);

    useEffect(() => {
        if (!canvasRef.current || !isPlaying) return;

        const canvas = canvasRef.current;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        let animationFrameId: number;

        const render = () => {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw background grid
            ctx.strokeStyle = 'rgba(214, 211, 209, 0.2)';
            ctx.lineWidth = 1;
            for(let i = 0; i < canvas.width; i += 50) {
                ctx.beginPath();
                ctx.moveTo(i, 0);
                ctx.lineTo(i, canvas.height);
                ctx.stroke();
            }

            // Update and draw particles
            particles.current.forEach(p => {
                // Drift logic based on "currentTime" (simulating centuries)
                const driftFactor = Math.sin(currentTime / 100) * 2;
                p.x += p.vx + driftFactor;
                p.y += p.vy;

                // Bounce
                if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
                if (p.y < 0 || p.y > canvas.height) p.vy *= -1;

                ctx.fillStyle = p.color;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();

                // Draw connections
                particles.current.forEach(p2 => {
                    const dist = Math.hypot(p.x - p2.x, p.y - p2.y);
                    if (dist < 60) {
                        ctx.strokeStyle = `rgba(120, 53, 15, ${0.1 * (1 - dist/60)})`;
                        ctx.beginPath();
                        ctx.moveTo(p.x, p.y);
                        ctx.lineTo(p2.x, p2.y);
                        ctx.stroke();
                    }
                });
            });

            // Draw central "Word Nucleus"
            ctx.fillStyle = '#78350f';
            ctx.shadowBlur = 20;
            ctx.shadowColor = 'rgba(120, 53, 15, 0.5)';
            ctx.font = 'bold 32px serif';
            ctx.textAlign = 'center';
            ctx.fillText(word.toUpperCase(), canvas.width / 2, canvas.height / 2);
            ctx.shadowBlur = 0;

            setCurrentTime(prev => prev + 1);
            animationFrameId = requestAnimationFrame(render);
        };

        render();
        return () => cancelAnimationFrame(animationFrameId);
    }, [isPlaying, word, currentTime]);

    return (
        <div className="bg-stone-100 p-8 rounded-2xl border border-stone-300 shadow-inner">
            <div className="flex items-center justify-between mb-6">
                <div>
                    <h3 className="text-2xl font-serif font-bold text-stone-800 flex items-center gap-2">
                        <Wind className="text-amber-600" />
                        Semantic Drift: <span className="italic text-amber-800">"{word}"</span>
                    </h3>
                    <p className="text-sm text-stone-500 font-sans">Visualizing vector space transformation from 800 BCE to 400 CE</p>
                </div>
                <div className="flex gap-2">
                    <button 
                        onClick={() => setIsPlaying(!isPlaying)}
                        className="p-3 bg-white rounded-full shadow-md hover:shadow-lg transition-all text-stone-700"
                    >
                        {isPlaying ? <Activity size={20} /> : <RefreshCw size={20} />}
                    </button>
                    <button className="p-3 bg-white rounded-full shadow-md hover:shadow-lg transition-all text-stone-700">
                        <Info size={20} />
                    </button>
                </div>
            </div>

            <canvas 
                ref={canvasRef}

### File 7: Timeline Hook
# filepath: frontend/components/Chronos/hooks/useTimeline.ts

```ts
import { useMemo, useCallback } from 'react';

/**
 * useTimeline Hook
 * Provides mathematical utilities for scaling, positioning, and grid generation
 * for temporal data. Handles the conversion between years (BCE/CE) and pixel coordinates.
 */

interface TimelineUtils {
    calculateXPosition: (year: number) => number;
    scaleWidth: (years: number) => number;
    getTicks: () => number[];
    formatYear: (year: number) => string;
}

export const useTimeline = (
    viewRange: [number, number],
    containerWidth: number
): TimelineUtils => {
    
    const [startYear, endYear] = viewRange;
    const totalYears = endYear - startYear;

    /**
     * Converts a specific year into its corresponding X coordinate on the SVG canvas.
     */
    const calculateXPosition = useCallback((year: number): number => {
        const relativePosition = (year - startYear) / totalYears;
        return relativePosition * containerWidth;
    }, [startYear, totalYears, containerWidth]);

    /**
     * Calculates the pixel width for a duration of years.
     */
    const scaleWidth = useCallback((years: number): number => {
        return (years / totalYears) * containerWidth;
    }, [totalYears, containerWidth]);

    /**
     * Generates an array of year markers (ticks) based on the current zoom level.
     * Adjusts density automatically to prevent visual clutter.
     */
    const getTicks = useCallback((): number[] => {
        const ticks: number[] = [];
        
        // Determine interval based on zoom
        let interval = 100;
        if (totalYears < 200) interval = 25;
        if (totalYears < 50) interval = 10;
        if (totalYears > 1500) interval = 250;

        const firstTick = Math.ceil(startYear / interval) * interval;
        
        for (let t = firstTick; t <= endYear; t += interval) {
            ticks.push(t);
        }
        
        return ticks;
    }, [startYear, endYear, totalYears]);

    /**
     * Formats a numeric year into a scholar-friendly string (BCE/CE).
     */
    const formatYear = useCallback((year: number): string => {
        if (year === 0) return "1 CE";
        const absYear = Math.abs(year);
        return year < 0 ? `${absYear} BCE` : `${absYear} CE`;
    }, []);

    return {
        calculateXPosition,
        scaleWidth,
        getTicks,
        formatYear
    };
};
```
