    async def find_nearest_neighbors(self, word: str, limit: int = 10) -> List[Dict[str, Any]]:
        """
        Finds the most semantically similar words using cosine similarity.
        Calculated directly in-database for performance.
        """
        if not self.pool:
            await self.connect()

        target_vector = await self.get_word_vector(word)
        if not target_vector:
            logger.warning(f"Word '{word}' not found in embeddings.")
            return []

        # SQL Implementation of Cosine Similarity: (A · B) / (||A|| * ||B||)
        # We assume vectors are normalized for performance, or calculate on the fly
        query = """
            SELECT word, vector,
            (
                SELECT SUM(a*b) 
                FROM UNNEST(vector) WITH ORDINALITY AS t1(a, i)
                JOIN UNNEST($1::float8[]) WITH ORDINALITY AS t2(b, j) ON t1.i = t2.j
            ) / (
                SQRT((SELECT SUM(pow(a, 2)) FROM UNNEST(vector) AS a)) * 
                SQRT((SELECT SUM(pow(b, 2)) FROM UNNEST($1::float8[]) AS b))
            ) as similarity
            FROM word_embeddings
            WHERE word != $2
            ORDER BY similarity DESC
            LIMIT $3
        """
        
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query, target_vector, word, limit)
                results = []
                for row in rows:
                    results.append({
                        "word": row['word'],
                        "similarity": float(row['similarity']),
                        "vector": list(row['vector'])
                    })
                return results
        except Exception as e:
            logger.error(f"Error finding neighbors for '{word}': {str(e)}")
            return []

        """
        Constructs a recursive etymological tree.
        Note: This uses the source_texts and texts tables to find root correlations.
        """
        if not self.pool:
            await self.connect()

        # Complex query to find original source word correlations
        query = """
            SELECT DISTINCT 
                st.content as source_word, 
                st.language,
                t.title as work_title
            FROM source_texts st
            JOIN texts t ON st.work_id = t.id
            WHERE t.text_content ILIKE $1
            LIMIT 5
        """
        
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch(query, f"% {word} %")

interface ClusterNode {
  word: string;
  similarity: number;
  weight?: number;
}

interface ClusterViewProps {
  data: ClusterNode[];
  onNodeClick: (word: string) => void;
}

export const ClusterView: React.FC<ClusterViewProps> = ({ data, onNodeClick }) => {
  // Sort by similarity to create a hierarchical visual flow
  const sortedData = useMemo(() => {
    return [...data].sort((a, b) => b.similarity - a.similarity);
  }, [data]);

  const clusters = useMemo(() => {
    // Group words into logical semantic buckets for visualization
    const groups: Record<string, ClusterNode[]> = {
      'High Correlation (>0.8)': [],
      'Strongly Related (0.7-0.8)': [],
      'Associated (0.6-0.7)': [],
      'Peripheral (<0.6)': []
    };

    sortedData.forEach(item => {
      if (item.similarity >= 0.8) groups['High Correlation (>0.8)'].push(item);
      else if (item.similarity >= 0.7) groups['Strongly Related (0.7-0.8)'].push(item);
      else if (item.similarity >= 0.6) groups['Associated (0.6-0.7)'].push(item);
      else groups['Peripheral (<0.6)'].push(item);
    });

    return Object.entries(groups).filter(([_, items]) => items.length > 0);
  }, [sortedData]);

  return (
    <div className="p-8 h-full overflow-y-auto bg-slate-950 scrollbar-hide">
      <div className="max-w-6xl mx-auto">
        <div className="mb-10">
          <h2 className="text-3xl font-light text-slate-200">Semantic Distribution</h2>
          <p className="text-slate-400 mt-2">Clustered by vector proximity in LOGOS high-dimensional space.</p>
        </div>

        <div className="space-y-12">
          {clusters.map(([title, items], idx) => (
            <motion.section 
              key={title}
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: idx * 0.1 }}
            >
              <div className="flex items-center mb-4">
                <div className="h-px flex-1 bg-slate-800"></div>
                <h3 className="mx-4 text-xs font-bold text-blue-500 uppercase tracking-[0.2em]">{title}</h3>
                <div className="h-px flex-1 bg-slate-800"></div>
              </div>

              <div className="flex flex-wrap gap-3">
                {items.map((node) => (
                  <motion.button
                    key={node.word}
                    whileHover={{ scale: 1.05, backgroundColor: 'rgba(59, 130, 246, 0.2)' }}
                    whileTap={{ scale: 0.95 }}
                    onClick={() => onNodeClick(node.word)}
                    className="px-4 py-2 rounded-full border border-slate-800 bg-slate-900/50 flex items-center space-x-3 transition-colors hover:border-blue-500/50"
                  >
                    <span className="text-slate-200 font-medium capitalize">{node.word}</span>
                    <div className="h-4 w-px bg-slate-700"></div>
                    <span className="text-xs text-blue-400 font-mono">

          ))}

        {/* Empty State */}
