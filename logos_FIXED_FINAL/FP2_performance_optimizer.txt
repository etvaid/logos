        """
        Generates a unique deterministic hash for a function call and its arguments.
        """
        try:
            serialized_args = json.dumps(args, sort_keys=True, default=str)
            serialized_kwargs = json.dumps(kwargs, sort_keys=True, default=str)
            raw_key = f"{func_name}:{serialized_args}:{serialized_kwargs}"
            return hashlib.sha256(raw_key.encode()).hexdigest()
        except (TypeError, ValueError) as e:
            logger.warning(f"Key generation serialization failed: {e}. Using fallback.")

            if entry['expires_at'] > time.time():
                logger.debug(f"Memory cache hit for key: {key}")
                return entry['value']

                del self._memory_cache[key]

                        self._memory_cache[key] = {
                            'value': value,
                            'expires_at': time.time() + DEFAULT_TTL
                        }

                    else:
                        await conn.execute("DELETE FROM logos_query_cache WHERE cache_key = $1", key)

            logger.error(f"Error retrieving from persistent cache: {str(e)}")

        """
        Stores an item in both memory and persistent PostgreSQL cache.
        """
        if value is None:
            return

        expiry_timestamp = time.time() + ttl
        expiry_datetime = datetime.now() + timedelta(seconds=ttl)

        # Level 1: Memory Update
        self._memory_cache[key] = {
            'value': value,
            'expires_at': expiry_timestamp
        }

        # Level 2: Persistent Update
        try:
            pool = await self._get_pool()
            async with pool.acquire() as conn:
                await conn.execute(
                    """
                    INSERT INTO logos_query_cache (cache_key, cache_value, expires_at)
                    VALUES ($1, $2, $3)
                    ON CONFLICT (cache_key) DO UPDATE
                    SET cache_value = EXCLUDED.cache_value,
                        expires_at = EXCLUDED.expires_at,
                        hit_count = logos_query_cache.hit_count + 1
                    """,
                    key, json.dumps(value), expiry_datetime
                )
        except Exception as e:
            logger.error(f"Error saving to persistent cache: {str(e)}")

        if len(self._memory_cache) > MAX_MEMORY_CACHE_SIZE:
            self._evict_oldest()

    def _evict_oldest(self):

        if not self._memory_cache:
            return
        sorted_keys = sorted(self._memory_cache.keys(), key=lambda k: self._memory_cache[k]['expires_at'])
        evict_count = max(1, int(len(sorted_keys) * 0.1))
        for i in range(min(evict_count, len(sorted_keys))):
            del self._memory_cache[sorted_keys[i]]
        logger.info(f"Evicted {evict_count} items from LOGOS memory cache.")

def logos_cache(ttl: int = DEFAULT_TTL):
    """
    Production-grade decorator for caching LOGOS database queries.
    """
    def decorator(func: T) -> T:
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            key = logos_cache_manager._generate_key(func.__name__, args, kwargs)
            
            cached_result = await logos_cache_manager.get(key)
            if cached_result is not None:
                return cached_result

            try:
                result = await func(*args, **kwargs)
            except Exception as e:
                logger.error(f"Function {func.__name__} failed: {str(e)}")
                raise

            if result is not None:
                await logos_cache_manager.set(key, result, ttl)
            
            return result
        return cast(T, wrapper)
    return decorator
