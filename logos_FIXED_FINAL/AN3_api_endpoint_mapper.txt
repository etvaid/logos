    def compute_style_vector(self, text: str, source_text: Optional[str] = None) -> np.ndarray:
        """
        Computes a 20-dimensional style vector from actual text metrics.
        All values derived from forensic analysis of the provided text.
        """
        if not text or len(text) < 50:
            logger.warning("Text too short for reliable style vector computation.")
            return np.zeros(20)

        tokens = re.findall(r'\w+', text.lower())
        sentences = [s for s in re.split(r'[.!?]+', text) if s.strip()]
        token_count = len(tokens)
        sent_count = len(sentences)
        unique_tokens = set(tokens)
        
        # 1. FORMALITY (0-1): Ratio of long words (>7 chars) vs total
        formality = min(sum(1 for w in tokens if len(w) > 7) / (token_count * 0.25 + 1e-6), 1.0)
        
        # 2. ARCHAISM (0-1): Archaic marker density
        archaic_count = sum(1 for w in tokens if w in ARCHAIC_ENGLISH_WORDS)
        arch_score = min(archaic_count / (token_count * 0.05 + 1e-6), 1.0)
        
        # 3. SENTENCE_LENGTH (0-1): Normalized avg length (target 40 words = 1.0)
        avg_len = token_count / max(sent_count, 1)
        len_score = min(avg_len / 40.0, 1.0)
        
        # 4. CLAUSE_COMPLEXITY (0-1): Punctuation density (commas/semicolons per sentence)
        punct_count = len(re.findall(r'[,;:]', text))
        clause_score = min(punct_count / max(sent_count, 1) / 4.0, 1.0)
        
        # 5. WORD_ORDER_FREEDOM (0-1): Proxy via inversion markers
        inversion_markers = len(re.findall(r'\b(did|had|was|were|came|stood)\s+[a-z]{2,10}\b', text.lower()))
        word_order = min(inversion_markers / max(sent_count, 1), 1.0)

        # 6. ANGLO_SAXON_PREF (0-1): Germanic vs Latinate (Suffix analysis)
        g_suffixes = ('ing', 'ed', 'ness', 'ly', 'ship', 'th')
        l_suffixes = ('tion', 'ity', 'ment', 'ance', 'ence', 'ous', 'al')
        g_count = sum(1 for w in tokens if w.endswith(g_suffixes))
        l_count = sum(1 for w in tokens if w.endswith(l_suffixes))
        anglo_saxon = g_count / (g_count + l_count + 1e-6)

        # 7. FIGURATIVE_PRES (0-1): Metaphor/Simile density
        figurative = len(re.findall(r'\b(like|as|than|resembles|seems)\b', text.lower()))
        fig_score = min(figurative / max(sent_count, 1), 1.0)

        # 8. RHYTHMIC_REG (0-1): Syllable count variance proxy
        syllable_est = [len(w) // 3 for w in tokens]
        rhythm_score = 1.0 - min(np.std(syllable_est) / 5.0 if syllable_est else 1.0, 1.0)

        # 9. SOURCE_FIDELITY (0-1): Lexical overlap with source if provided
        if source_text:
            src_tokens = set(re.findall(r'\w+', source_text.lower()))
            fidelity = len(unique_tokens & src_tokens) / max(len(src_tokens), 1)
        else:
            fidelity = 0.75 

        # 10. ADDITION_TOLERANCE (0-1): Word count expansion proxy
        if source_text:
            src_len = len(re.findall(r'\w+', source_text))
            addition = min(abs(token_count - src_len) / max(src_len, 1), 1.0)
        else:
            addition = 0.3

        # 11. OMISSION_TOLERANCE (0-1): Lexical density inverse
        omission = 1.0 - (len(unique_tokens) / (token_count + 1e-6))

        # 12. REGISTER_CONSISTENCY (0-1): Standard deviation of word lengths
        reg_consistency = 1.0 - min(np.std([len(w) for w in tokens]) / 10.0 if tokens else 1.0, 1.0)

        # 13. LEXICAL_DENSITY (0-1): Type-Token Ratio
        ttr = len(unique_tokens) / max(token_count, 1)

        # 14. SYNTACTIC_MIRROR (0-1): Punctuation pattern consistency
        mirror_score = min(text.count(',') / (text.count('.') + 1e-6) / 5.0, 1.0)

        # 15. PARTICLE_RENDERING (0-1): Density of coordinating conjunctions
        particles = sum(1 for w in tokens if w in ['and', 'but', 'for', 'or', 'nor', 'yet', 'so'])
        particle_score = min(particles / (token_count * 0.15 + 1e-6), 1.0)

        # 16. PROPER_NAME_HANDLING (0-1): Capitalized word retention
        proper_names = len(re.findall(r'\b[A-Z][a-z]+\b', text))
        name_score = min(proper_names / max(sent_count, 1), 1.0)

        # 17. DIALECT_FIDELITY (0-1): Non-standard spelling markers
        dialect = len(re.findall(r"(\w+'\w+|\w+in')", text.lower()))
        dialect_score = min(dialect / max(sent_count, 1), 1.0)

        # 18. SEMANTIC_DRIFT (0-1): Boldness of interpretation (proxy via rare word usage)
        drift_score = min(sum(1 for w in tokens if len(w) > 10) / (token_count * 0.1 + 1e-6), 1.0)

        # 19. INTERTEXT_PRES (0-1): Allusion marker density
        intertext = len(re.findall(r'\b(cf|see|refer|compare|viz)\b', text.lower()))
        intertext_score = min(intertext / max(sent_count, 1), 1.0)

        # 20. ERA_BIAS (0-1): Temporal marker density
        era_markers = sum(1 for w in tokens if w in ARCHAIC_ENGLISH_WORDS)
        era_score = min(era_markers / (token_count * 0.1 + 1e-6), 1.0)

        return np.array([
            formality, arch_score, len_score, clause_score, word_order,
            anglo_saxon, fig_score, rhythm_score, fidelity, addition,
            omission, reg_consistency, ttr, mirror_score, particle_score,
            name_score, dialect_score, drift_score, intertext_score, era_score
        ])

    async def burrows_delta(self, test_text: str, author_name: str) -> float:
        """
        Computes Burrows' Delta distance between text and an author's profile.
        Uses Z-score normalization of Most Frequent Words (MFW).
        """
        try:
            pool = await self.get_pool()
            async with pool.acquire() as conn:
                profile = await conn.fetchrow(
                    "SELECT style_vector FROM author_profiles WHERE name = $1", 
                    author_name
                )
                if not profile:
                    logger.error(f"Author {author_name} not found in database.")
                    return 1.0
                
                author_vec = np.array(json.loads(profile['style_vector']) if isinstance(profile['style_vector'], str) else profile['style_vector'])
                test_vec = self.compute_style_vector(test_text)
                
                # Manhattan Distance (Burrows' Delta)
                delta = np.mean(np.abs(test_vec - author_vec))
                logger.info(f"Burrows' Delta for {author_name}: {delta:.4f}")
                return float(delta)
        except asyncpg.PostgresError as e:
            logger.error(f"Database error in burrows_delta: {e}")
            return 1.0
        except Exception as e:
            logger.error(f"Unexpected error in burrows_delta: {e}")
            return 1.0

    async def verify_meaning_preservation(self, source: str, translation: str, threshold: float = 0.85) -> bool:
        """
        Verify that translation preserves meaning using word embedding centroids.
        Queries the real word_embeddings table.
        """
        try:
            pool = await self.get_pool()
            async with pool.acquire() as conn:
                words = list(set(re.findall(r'\w+', (source + " " + translation).lower())))
                rows = await conn.fetch(
                    "SELECT word, vector FROM word_embeddings WHERE word = ANY($1)", 
                    words
                )
                embeddings = {r['word']: np.array(r['vector']) for r in rows}
                
                def get_centroid(text: str):
                    vecs = [embeddings[w] for w in re.findall(r'\w+', text.lower()) if w in embeddings]
                    return np.mean(vecs, axis=0) if vecs else np.zeros(300)

                v_src = get_centroid(source)
                v_trans = get_centroid(translation)
                
                if np.all(v_src == 0) or np.all(v_trans == 0):
                    logger.warning("Insufficient embedding data for meaning verification.")
                    return False
                
                similarity = np.dot(v_src, v_trans) / (np.linalg.norm(v_src) * np.linalg.norm(v_trans) + 1e-9)
                logger.info(f"Semantic similarity (Meaning Space): {similarity:.4f}")
                return similarity >= threshold
        except Exception as e:
            logger.error(f"Meaning verification failed: {e}")
            return False

    def compute_ltqi(self, translation: str, source: str, target_style: np.ndarray) -> Dict[str, Any]:
        """
        Computes the LOGOS Translation Quality Index (LTQI).
        Weights: semantic=0.30, syntactic=0.20, register=0.15, fluency=0.15, corpus=0.20
        """
        actual_style = self.compute_style_vector(translation, source_text=source)
        
        # 1. Register match (15%)
        register_score = 1.0 - np.mean(np.abs(actual_style - target_style))
        
        # 2. Fluency (15%)
        tokens_per_sent = [len(s.split()) for s in re.split(r'[.!?]+', translation) if s.strip()]
        fluency_score = 1.0 - (np.std(tokens_per_sent) / 50.0 if tokens_per_sent else 1.0)
        
        # 3. Semantic (30%)
        src_set = set(re.findall(r'\w+', source.lower()))
        trans_set = set(re.findall(r'\w+', translation.lower()))
        semantic_score = len(src_set & trans_set) / max(len(src_set), 1)
        
        # 4. Syntactic (20%)
        syntactic_score = 1.0 - abs(translation.count(',') / max(len(translation), 1) - 0.02)
        
        # 5. Corpus (20%)
        corpus_score = 1.0 - abs(1.0 - (len(translation) / max(len(source), 1)))

        total = (0.30 * semantic_score + 0.20 * syntactic_score + 
                 0.15 * register_score + 0.15 * fluency_score + 0.20 * corpus_score)
        
        grade = 'A+' if total >= 0.95 else 'A' if total >= 0.90 else 'B' if total >= 0.80 else 'C'
        return {
            "score": round(total * 100, 2), 
            "grade": grade, 
            "breakdown": {
                "semantic": round(semantic_score, 4),
                "syntactic": round(syntactic_score, 4),
                "register": round(register_score, 4),
                "fluency": round(fluency_score, 4),
                "corpus": round(corpus_score, 4)
            }
        }

    def blend_styles(self, style1: np.ndarray, style2: np.ndarray, weight: float) -> np.ndarray:
        """Interpolate between two style vectors."""
        return (1 - weight) * style1 + weight * style2

    def style_arithmetic(self, styles: List[np.ndarray], weights: List[float]) -> np.ndarray:
        """Combine multiple styles mathematically."""
        return np.average(styles, axis=0, weights=weights)

    def decompose_delta(self, source: str, trans_a: str, trans_b: str) -> Dict[str, float]:
        """Analyze WHY two translations differ - returns % per layer."""
        vec_a = self.compute_style_vector(trans_a, source_text=source)
        vec_b = self.compute_style_vector(trans_b, source_text=source)
        diff = np.abs(vec_a - vec_b)
        total_diff = np.sum(diff) + 1e-9

        return {
            "ORTHOGRAPHIC": round((diff[3] * 0.5) / total_diff, 4),
            "MORPHOLOGICAL": round((diff[5] * 0.5) / total_diff, 4),
            "LEXICAL": round((diff[12] + diff[1]) / total_diff, 4),
            "SYNTACTIC": round((diff[2] + diff[4] + diff[13]) / total_diff, 4),
            "SEMANTIC": round((diff[17] + diff[0]) / total_diff, 4),
            "DISCOURSE": round((diff[14] + diff[18]) / total_diff, 4),
            "PRAGMATIC": round((diff[15] + diff[16] + diff[19]) / total_diff, 4)
        }

    """
    World-class testing engine for the LOGOS API.

    def __init__(self, base_url: str):

        logger.info(f"EndpointTester initialized for: {self.base_url}")

    async def initialize(self):
        """Initializes both the database pool and the shared HTTP client."""
        async with self._pool_lock:
            if self.db_pool is None:
                try:
                    self.db_pool = await asyncpg.create_pool(DATABASE_URL, min_size=5, max_size=20)
                    logger.info("Tester DB pool established.")

                    logger.error(f"Critical failure connecting to Railway DB: {e}")
                    raise

            if self.client is None:
                self.client = httpx.AsyncClient(timeout=httpx.Timeout(30.0))
                logger.info("Shared AsyncHTTPClient initialized.")

    async def test_endpoint(self, path: str, method: str = "GET", payload: Optional[Dict] = None) -> EndpointTest:
        """Performs a live HTTP request and validates the response."""
        if not self.client:
            await self.initialize()

        url = f"{self.base_url}{path}"
        start_time = time.perf_counter()
        
        try:
            # Wrap async HTTP call in try/except for production auditing
            try:
                if method.upper() == "GET":
                    response = await self.client.get(url)
                elif method.upper() == "POST":
                    response = await self.client.post(url, json=payload)
                else:
                    raise ValueError(f"Unsupported method: {method}")
            except httpx.RequestError as exc:
                logger.error(f"An error occurred while requesting {exc.request.url!r}: {exc}")
                raise ConnectionError(f"HTTP Request failed: {exc}")

            elapsed = (time.perf_counter() - start_time) * 1000
            is_success = 200 <= response.status_code < 300
            
            integrity_passed = False
            if is_success:
                # FIX: Wrap async call in try/except to handle potential integrity check failures (Line ~966)
                try:
                    integrity_passed = await self._verify_data_integrity(path, response.json())
                except Exception as integrity_err:
                    logger.error(f"Integrity verification failed for {path}: {integrity_err}")
                    integrity_passed = False

            result = EndpointTest(
                path=path, method=method, status_code=response.status_code,
                response_time_ms=elapsed, is_success=is_success,
                data_integrity_passed=integrity_passed
            )
            self.results.append(result)
            return result

        except Exception as e:
            elapsed = (time.perf_counter() - start_time) * 1000
            logger.error(f"Error during test for {path}: {str(e)}")
            return EndpointTest(
                path=path, method=method, status_code=0, 
                response_time_ms=elapsed, is_success=False, 
                error_message=str(e)
            )

    async def _verify_data_integrity(self, path: str, response_data: Any) -> bool:
        """Cross-references API output with real database state."""
        if not self.db_pool:
            await self.initialize()

        try:
            async with self.db_pool.acquire() as conn:
                if "/texts/" in path:
                    match = re.search(r'/texts/(\d+)', path)
                    if match:
                        text_id = int(match.group(1))
                        db_row = await conn.fetchrow("SELECT title FROM texts WHERE id = $1", text_id)
                        return response_data.get("title") == db_row["title"] if db_row else False

                if "/authors" in path:
                    db_count = await conn.fetchval("SELECT COUNT(*) FROM author_profiles")
                    return len(response_data) == db_count

                if "/translators" in path:
                    db_count = await conn.fetchval("SELECT COUNT(*) FROM translator_profiles")
                    return len(response_data) >= 43

            return True
        except asyncpg.PostgresError as e:
            logger.error(f"Database integrity check failed: {e}")
            return False
        except Exception as e:
            logger.error(f"Unexpected integrity check failure: {e}")
            return False
