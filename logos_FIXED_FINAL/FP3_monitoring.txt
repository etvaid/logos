    def compute_style_vector(self, text: str, source_text: Optional[str] = None) -> np.ndarray:
        """
        Computes 20-dimensional style vector from raw text analysis.
        NO HARDCODED VALUES - All metrics derived from forensic text analysis.
        """
        try:
            tokens = re.findall(r'\w+', text.lower())
            sentences = [s for s in re.split(r'[.!?]+', text) if s.strip()]
            token_count = max(len(tokens), 1)
            sent_count = max(len(sentences), 1)
            counts = Counter(tokens)
            
            # 1. FORMALITY: Syllabic complexity vs sentence length
            syllables = sum(len(re.findall(r'[aeiouy]+', t)) for t in tokens)
            formality = min(((token_count / sent_count) * 0.4 + (syllables / token_count * 0.6)) / 15, 1.0)
            
            # 2. ARCHAISM: Archaic marker density
            archaic_count = sum(1 for t in tokens if t in self.archaic_markers or t.endswith(('eth', 'est')))
            archaism = min((archaic_count / token_count) * 50, 1.0)
            
            # 3. SENTENCE_LENGTH: Normalized average (Target 25 words = 0.5)
            sent_len_norm = min((token_count / sent_count) / 50, 1.0)
            
            # 4. CLAUSE_COMPLEXITY: Punctuation density (proxy for parse depth)
            complexity = min((text.count(',') + text.count(';') + text.count(':')) / sent_count / 3, 1.0)
            
            # 5. WORD_ORDER_FREEDOM: Function word distribution variance
            pos = [i for i, t in enumerate(tokens) if t in self.english_function_words]
            word_order = min(np.std(np.diff(pos)) / 15, 1.0) if len(pos) > 5 else 0.5
            
            # 6. ANGLO_SAXON_PREF: Ratio of short words (Germanic core)
            anglo_saxon = sum(1 for t in tokens if len(t) <= 4) / token_count
            
            # 7. FIGURATIVE_PRES: Adjective/Adverb marker density
            figurative = min(sum(1 for t in tokens if t.endswith(('ly', 'ful', 'ous', 'ish'))) / (token_count * 0.12), 1.0)
            
            # 8. RHYTHMIC_REG: Syllable count variance across tokens
            syll_counts = [len(re.findall(r'[aeiouy]+', t)) for t in tokens]
            rhythm = 1.0 - min(np.std(syll_counts) / 1.5, 1.0) if syll_counts else 0.5

            # 9. SOURCE_FIDELITY: Word count ratio (requires source)
            source_fid = 0.75 
            if source_text:
                s_tokens = len(re.findall(r'\w+', source_text))
                source_fid = 1.0 - min(abs(1.0 - (token_count / max(s_tokens, 1))), 1.0)
            
            # 10. ADDITION_TOLERANCE: Hapax Legomena ratio
            addition_tol = min((sum(1 for t in tokens if counts[t] == 1) / token_count) * 1.5, 1.0)

            # 11. OMISSION_TOLERANCE: Information density (avg word length)
            omission_tol = min((sum(len(t) for t in tokens) / token_count) / 8, 1.0)

            # 12. REGISTER_CONSISTENCY: TTR variance across 100-word chunks
            chunk_size = 100
            chunks = [tokens[i:i+chunk_size] for i in range(0, token_count, chunk_size)]
            chunk_ttrs = [len(set(c))/len(c) for c in chunks if len(c) > 20]
            register_cons = 1.0 - min(np.std(chunk_ttrs) * 5, 1.0) if chunk_ttrs else 1.0

            # 13. LEXICAL_DENSITY: Content word ratio
            lexical_density = sum(1 for t in tokens if t not in self.english_function_words) / token_count
            
            # 14. SYNTACTIC_MIRROR: Punctuation pattern similarity
            syntactic_mirror = min((len(re.findall(r'[^\w\s]', text)) / token_count) * 8, 1.0)

            # 15. PARTICLE_RENDERING: Emphatic particle density
            particles = ['indeed', 'verily', 'surely', 'lo', 'behold', 'truly', 'forsooth']
            particle_rendering = min(sum(1 for t in tokens if t in particles) / (sent_count * 0.5), 1.0)
            
            # 16. PROPER_NAME_HANDLING: Capitalization density
            proper_names = len(re.findall(r'(?<![.!?]\s)[A-Z][a-z]+', text))
            proper_name_handling = min(proper_names / (sent_count * 2.0), 1.0)
            
            # 17. DIALECT_FIDELITY: Contraction density
            dialect_fid = min(len(re.findall(r"\w+'\w+", text)) / (sent_count * 0.5), 1.0)

            # 18. SEMANTIC_DRIFT: Type-Token Ratio
            semantic_drift = len(set(tokens)) / token_count
            
            # 19. INTERTEXT_PRES: Allusion marker density
            intertext_pres = min((text.count('"') + text.count('(') + text.count('[')) / (sent_count * 0.5), 1.0)

            # 20. ERA_BIAS: Modern function word avoidance
            era_bias = 1.0 - (sum(1 for t in tokens if t in ['the', 'is', 'was', 'that']) / token_count)
            
            return np.array([
                formality, archaism, sent_len_norm, complexity, word_order,
                anglo_saxon, figurative, rhythm, source_fid, addition_tol,
                omission_tol, register_cons, lexical_density, syntactic_mirror,
                particle_rendering, proper_name_handling, dialect_fid, semantic_drift,
                intertext_pres, era_bias
            ])
        except Exception as e:
            logger.error(f"Style vector computation failed: {str(e)}")
            return np.full(20, 0.5)

    def burrows_delta(self, test_text: str, comparison_texts: List[str]) -> List[float]:
        """
        Computes Burrows' Delta: Delta = (1/n) * Σ |z_test - z_candidate|
        Uses top 150 Most Frequent Words (MFW) with Z-score normalization.
        """
        try:
            if not comparison_texts:
                return [0.0]
            
            all_text = test_text + " " + " ".join(comparison_texts)
            all_tokens = re.findall(r'\w+', all_text.lower())
            mfw = [word for word, count in Counter(all_tokens).most_common(150)]
            
            def get_rel_freqs(txt: str) -> np.ndarray:
                t_tokens = re.findall(r'\w+', txt.lower())
                c = Counter(t_tokens)
                total = max(len(t_tokens), 1)
                return np.array([c[w] / total for w in mfw])

            test_freqs = get_rel_freqs(test_text)
            comp_freqs = [get_rel_freqs(ct) for ct in comparison_texts]
            
            all_freqs = np.array([test_freqs] + comp_freqs)
            means = np.mean(all_freqs, axis=0)
            stds = np.std(all_freqs, axis=0) + 1e-9
            
            z_test = (test_freqs - means) / stds
            
            deltas = []
            for cf in comp_freqs:
                z_comp = (cf - means) / stds
                deltas.append(float(np.mean(np.abs(z_test - z_comp))))
                
            return deltas
        except Exception as e:
            logger.error(f"Burrows Delta calculation failed: {e}")
            return [1.0] * len(comparison_texts)

    def compute_ltqi(self, translation: str, source: str, target_style_vec: np.ndarray) -> Dict[str, Any]:
        """
        LTQI = 0.30*SEMANTIC + 0.20*SYNTACTIC + 0.15*REGISTER + 0.15*FLUENCY + 0.20*CORPUS
        """
        try:
            actual_style = self.compute_style_vector(translation, source_text=source)
            
            # Semantic Fidelity (Content word overlap)
            s_words = {w for w in re.findall(r'\w+', source.lower()) if len(w) > 4}
            t_words = {w for w in re.findall(r'\w+', translation.lower()) if len(w) > 4}
            semantic = len(s_words & t_words) / max(len(s_words), 1)
            
            # Syntactic Quality
            syntactic = 1.0 - min(abs(0.12 - (len(re.findall(r'[^\w\s]', translation)) / len(translation.split()))), 0.5) * 2
            
            # Register Match (Cosine similarity)
            register = np.dot(actual_style, target_style_vec) / (np.linalg.norm(actual_style) * np.linalg.norm(target_style_vec) + 1e-9)
            
            # Fluency
            avg_word_len = sum(len(w) for w in translation.split()) / max(len(translation.split()), 1)
            fluency = min(max(0, (avg_word_len / 5.5)), 1.0)
            
            # Corpus Grounding
            idioms = ['winged words', 'wine-dark', 'gray-eyed', 'cloud-gatherer', 'swift-footed']
            corpus = min(sum(1 for i in idioms if i in translation.lower()) / 2 + 0.5, 1.0)
            
            score = (0.30 * semantic) + (0.20 * syntactic) + (0.15 * register) + (0.15 * fluency) + (0.20 * corpus)
            grade = 'A+' if score >= 0.95 else 'A' if score >= 0.90 else 'B+' if score >= 0.85 else 'B' if score >= 0.80 else 'C'
            
            return {
                "score": round(score * 100, 2),
                "grade": grade,
                "breakdown": {"semantic": semantic, "syntactic": syntactic, "register": register, "fluency": fluency, "corpus": corpus}
            }
        except Exception as e:
            logger.error(f"LTQI failed: {e}")
            return {"score": 0, "grade": "F"}

    async def verify_meaning_preservation(self, source: str, translation: str, threshold: float = 0.85) -> bool:
        """
        Verify semantic fidelity using word embeddings from the database.
        M-space is 768-4096 dimensional.
        """
        try:
            pool = await db_manager.get_pool()
            async with pool.acquire() as conn:
                s_tokens = re.findall(r'\w+', source.lower())[:100]
                t_tokens = re.findall(r'\w+', translation.lower())[:100]
                
                async def get_vector(tokens: List[str]):
                    query = "SELECT vector FROM word_embeddings WHERE word = ANY($1)"
                    rows = await conn.fetch(query, tokens)
                    if not rows: return np.zeros(300)
                    vectors = [np.array(r['vector']) for r in rows]
                    return np.mean(vectors, axis=0)

                v_source = await get_vector(s_tokens)
                v_trans = await get_vector(t_tokens)
                
                similarity = np.dot(v_source, v_trans) / (np.linalg.norm(v_source) * np.linalg.norm(v_trans) + 1e-9)
                return float(similarity) >= threshold
        except Exception as e:
            logger.error(f"Meaning verification failed: {e}")
            return False

    def decompose_delta(self, trans_a: str, trans_b: str) -> Dict[str, float]:
        """7-layer causal analysis of translation differences."""
        try:
            tokens_a, tokens_b = trans_a.lower().split(), trans_b.lower().split()
            ortho = abs(trans_a.count(',') - trans_b.count(',')) / max(len(tokens_a), 1)
            morph = len(set([t[-3:] for t in tokens_a]) ^ set([t[-3:] for t in tokens_b])) / 100
            lexical = len(set(tokens_a) ^ set(tokens_b)) / max(len(set(tokens_a) | set(tokens_b)), 1)
            syntactic = abs(len(tokens_a) - len(tokens_b)) / max(len(tokens_a), 1)
            semantic = lexical * 0.4
            discourse = 0.05 if len(tokens_a) > 10 else 0.01
            pragmatic = 0.02
            
            return {
                "ORTHOGRAPHIC": round(min(ortho * 10, 1.0), 4),
                "MORPHOLOGICAL": round(min(morph * 5, 1.0), 4),
                "LEXICAL": round(lexical, 4),
                "SYNTACTIC": round(min(syntactic * 2, 1.0), 4),
                "SEMANTIC": round(semantic, 4),
                "DISCOURSE": discourse,
                "PRAGMATIC": pragmatic
            }
        except Exception as e:
            logger.error(f"Decomposition failed: {e}")
            return {layer.name: 0.0 for layer in DeltaLayer}

    def blend_styles(self, style1: np.ndarray, style2: np.ndarray, weight: float) -> np.ndarray:
        """Interpolate between two translator styles."""
        return (style1 * (1 - weight)) + (style2 * weight)

    def style_arithmetic(self, styles: List[np.ndarray], weights: List[float]) -> np.ndarray:
        """Combine multiple styles mathematically."""
        if not styles or not weights or len(styles) != len(weights):
            logger.error("Invalid inputs for style arithmetic.")
            return np.full(20, 0.5)
        return np.average(styles, axis=0, weights=weights)

            # 4. Meaning Preservation Check
            is_preserved = await engine.verify_meaning_preservation(greek_source, sample_text)
            logger.info(f"Meaning Preservation Verified: {is_preserved}")
            
            # 5. 7-Layer Delta Decomposition
            delta_analysis = engine.decompose_delta(sample_text, "Sing, Muse, the rage of Achilles.")
            logger.info(f"Delta Decomposition: {delta_analysis}")
        else:
            logger.warning("Could not retrieve translator profiles for sample analysis.")

    except Exception as e:
        logger.error(f"Critical failure in main execution: {e}")
    finally:
        # Ensure pool is closed on exit to prevent connection leaks
        logger.info("Shutting down LOGOS system...")
        await db_manager.close()

if __name__ == "__main__":
