/**
 * Analyzes React Hook usage patterns for potential infinite loops or missing dependencies.
 */
export async function checkHookUsage(source: string): Promise<HookUsage[]> {
  const foundHooks: HookUsage[] = [];
  const hookRegex = /(use[A-Z][a-zA-Z]+)\s*\(([^)]*)\)/g;
  let match;

  while ((match = hookRegex.exec(source)) !== null) {
    const hookName = match[1];
    const args = match[2];
    const lineLocation = source.substring(0, match.index).split('\n').length;
    const issues: string[] = [];
    
    const hasDependencyArray = args.includes('[');
    const dependencyMatch = args.match(/\[(.*?)\]/);
    const dependencyCount = dependencyMatch ? dependencyMatch[1].split(',').filter(x => x.trim()).length : 0;

    if (hookName === 'useEffect' && !hasDependencyArray) {
      issues.push(`Line ${lineLocation}: useEffect missing dependency array (Infinite Loop Risk).`);
    }
    if ((hookName === 'useMemo' || hookName === 'useCallback') && !hasDependencyArray) {
      issues.push(`Line ${lineLocation}: ${hookName} requires a dependency array for stable memoization.`);
    }
    if (hookName === 'useState' && args.includes('any')) {
      issues.push(`Line ${lineLocation}: useState should use concrete types instead of 'any'.`);
    }

    foundHooks.push({
      hookName,
      isCorrectlyUsed: issues.length === 0,
      hasDependencyArray,
      dependencyCount,
      lineLocation,
      potentialIssues: issues
    });
  }
  return foundHooks;
}

/**
 * Validates accessibility standards including ARIA roles and keyboard navigation.
 */
export async function checkAccessibility(source: string): Promise<AccessibilityMetric> {
  const hasAriaLabel = /aria-label=/g.test(source) || /aria-labelledby=/g.test(source);
  const roleMatch = source.match(/role=["']([^"']+)["']/);
  const roleAssigned = roleMatch ? roleMatch[1] : null;
  
  const interactiveTags = ['button', 'input', 'select', 'textarea', 'a'];
  let isKeyboardNavigable = true;
  
  interactiveTags.forEach(tag => {
    const tagRegex = new RegExp(`<${tag}`, 'g');
    if (tagRegex.test(source)) {
      const hasFocusLogic = source.includes('tabIndex') || source.includes('onKeyDown') || source.includes('onKeyUp');
      if (!hasFocusLogic && tag === 'button') {
        isKeyboardNavigable = false;
      }
    }
  });

  const imgTags = source.match(/<img/g) || [];
  const altTexts = source.match(/alt=["'][^"']*["']/g) || [];
  const altTextPresent = imgTags.length === altTexts.length;

  return {
    hasAriaLabel,
    roleAssigned,
    isKeyboardNavigable,
    contrastRatioCompliant: true,
    altTextPresent
  };
}

/**
 * Calculates a responsiveness score based on Tailwind CSS breakpoint usage and layout patterns.
 */
export async function checkResponsiveness(source: string): Promise<number> {
  const prefixes = ['sm:', 'md:', 'lg:', 'xl:', '2xl:'];
  let found = 0;
  prefixes.forEach(p => { if (source.includes(p)) found++; });
  
  let score = (found / prefixes.length) * 50;
  
  if (source.includes('flex')) score += 15;
  if (source.includes('grid')) score += 15;
  if (source.includes('max-w-') || source.includes('container')) score += 10;
  if (source.includes('aspect-')) score += 10;
  
  return Math.min(score, 100);
}

          analyzePropTypes(source),
          analyzeStateFlow(source)
        ]);
      } catch (innerErr) {
        logger.error(`Concurrent analysis failed for ${fileName}:`, innerErr);
        throw new Error(`Sub-system analysis failure: ${(innerErr as Error).message}`);
      }
      
      const fileReport = {
        fileName,
        isProductionReady: compAnalysis.isProductionReady,
        errorCount: compAnalysis.detectedErrors.length,
        strictTypes: propAnalysis.every(p => p.isStrict),
        stateNodes: stateAnalysis.length,
        timestamp: new Date().toISOString()
      };
      
      auditResults.push(fileReport);
      logger.info(`Audit ${fileReport.isProductionReady ? 'PASSED' : 'REJECTED'} for ${fileName}`);
      
    } catch (err) {
      logger.error(`Audit failed for ${fileName} due to internal error:`, err);
      auditResults.push({ 
        fileName, 
        error: (err as Error).message,
        status: "FAILED_AUDIT"
      });
    }
  }
  
  logger.info(`Global Audit Complete. Processed ${auditResults.length} files.`);
  return auditResults;
}
