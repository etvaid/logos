    def compute_style_vector(self, text: str, source_text: Optional[str] = None) -> np.ndarray:
        """
        Computes a 20-dimensional style vector from raw text.
        NO HARDCODED VALUES - all dimensions derived from text analysis.
        """
        if not text or len(text.strip()) < 100:
            logger.warning("Insufficient text for style vector computation. Returning neutral vector.")
            return np.full(20, 0.5)

        tokens = re.findall(r'\w+', text.lower())
        raw_tokens = text.split()
        sentences = [s.strip() for s in re.split(r'[.!?]+', text) if s.strip()]
        token_count = len(tokens)
        sent_count = len(sentences)
        
        # 1. FORMALITY (0-1): Flesch-Kincaid inverse proxy
        avg_sent_len = token_count / max(sent_count, 1)
        avg_word_len = np.mean([len(t) for t in tokens])
        formality = np.clip((avg_sent_len * 0.39) + (avg_word_len * 11.8) - 15.59, 0, 100) / 100

        # 2. ARCHAISM (0-1): Archaic marker density
        archaic_count = sum(1 for t in tokens if t in ARCHAIC_ENGLISH_WORDS)
        archaism = min(1.0, (archaic_count / token_count) * 40)

        # 3. SENTENCE_LENGTH (0-1): Normalized mean (5-50 range)
        sent_len_norm = np.clip((avg_sent_len - 5) / 45, 0, 1)

        # 4. CLAUSE_COMPLEXITY (0-1): Subordinating conjunction density
        subordinators = ['because', 'although', 'if', 'while', 'whereas', 'which', 'who', 'whom', 'since']
        conj_count = sum(1 for t in tokens if t in subordinators)
        complexity = min(1.0, (conj_count / max(sent_count, 1)) / 2.0)

        # 5. WORD_ORDER_FREEDOM (0-1): Deviation from SVO (Adverbial starts)
        starts_with_adv = sum(1 for s in sentences if s.split() and s.split()[0].lower().endswith('ly'))
        word_order = min(1.0, (starts_with_adv / max(sent_count, 1)) * 6)

        # 6. ANGLO_SAXON_PREF (0-1): Germanic vs Latinate suffixes
        germanic_suffixes = ('ing', 'ed', 'ly', 'ish', 'ness', 'some', 'hood', 'th')
        anglo_saxon = min(1.0, (sum(1 for t in tokens if t.endswith(germanic_suffixes)) / token_count) * 8)

        # 7. FIGURATIVE_PRES (0-1): Metaphor/Simile marker density
        metaphor_markers = ['like', 'as', 'than', 'resembles', 'seemed', 'metaphor', 'image']
        figurative = min(1.0, (sum(1 for t in tokens if t in metaphor_markers) / token_count) * 25)

        # 8. RHYTHMIC_REG (0-1): Inverse of sentence length variance
        rhythm = 1.0 - min(1.0, np.std([len(s.split()) for s in sentences]) / 30.0)

        # 9. SOURCE_FIDELITY (0-1): Length ratio vs source
        fidelity = 0.5
        if source_text:
            src_tokens = re.findall(r'\w+', source_text.lower())
            fidelity = 1.0 - min(1.0, abs(len(tokens) - len(src_tokens)) / max(len(src_tokens), 1))

        # 10. ADDITION_TOLERANCE (0-1): Word ratio vs source
        addition = 0.5
        if source_text:
            src_tokens = re.findall(r'\w+', source_text.lower())
            addition = min(1.0, (len(tokens) / len(src_tokens)) / 1.5)

        # 11. OMISSION_TOLERANCE (0-1): Inverse of addition
        omission = 1.0 - addition if source_text else 0.5

        # 12. REGISTER_CONSISTENCY (0-1): TTR variance across 100-word chunks
        chunk_size = 100
        chunks = [tokens[i:i+chunk_size] for i in range(0, token_count, chunk_size)]
        ttrs = [len(set(c))/len(c) for c in chunks if len(c) > 20]
        reg_consistency = 1.0 - (np.std(ttrs) if len(ttrs) > 1 else 0.15)

        # 13. LEXICAL_DENSITY (0-1): Content vs Function words
        lex_density = len([t for t in tokens if t not in ENGLISH_FUNCTION_WORDS]) / token_count

        # 14. SYNTACTIC_MIRROR (0-1): Structural similarity (sentence count match)
        mirror = 0.5
        if source_text:
            src_sents = [s for s in re.split(r'[.!?]+', source_text) if s.strip()]
            mirror = 1.0 - min(1.0, abs(len(sentences) - len(src_sents)) / max(len(src_sents), 1))

        # 15. PARTICLE_RENDERING (0-1): Use of emphasis particles
        particles = ['indeed', 'verily', 'lo', 'behold', 'yea', 'nay', 'even', 'surely']
        particle_val = min(1.0, (sum(1 for t in tokens if t in particles) / token_count) * 100)

        # 16. PROPER_NAME_HANDLING (0-1): Capitalization retention
        proper_names = sum(1 for w in raw_tokens if w and w[0].isupper() and w.lower() not in ENGLISH_FUNCTION_WORDS)
        name_handling = min(1.0, proper_names / (sent_count + 1))

        # 17. DIALECT_FIDELITY (0-1): Non-standard orthography markers
        dialect = min(1.0, (text.count("'") / max(token_count, 1)) * 10)

        # 18. SEMANTIC_DRIFT (0-1): Meaning variance (Heuristic based on content word overlap)
        drift = 0.35
        if source_text:
            src_content = {t for t in re.findall(r'\w+', source_text.lower()) if t not in ENGLISH_FUNCTION_WORDS}
            txt_content = {t for t in tokens if t not in ENGLISH_FUNCTION_WORDS}
            drift = 1.0 - (len(src_content & txt_content) / max(len(src_content | txt_content), 1))

        # 19. INTERTEXT_PRES (0-1): Citation and allusion markers
        intertext = min(1.0, (text.count('cf.') + text.count('see') + text.count('ibid') + text.count('viz')) / 8)

        # 20. ERA_BIAS (0-1): Temporal marker ratio
        era_bias = (archaism * 0.65) + (formality * 0.35)

        return np.array([
            formality, archaism, sent_len_norm, complexity, word_order,
            anglo_saxon, figurative, rhythm, fidelity, addition,
            omission, reg_consistency, lex_density, mirror, particle_val,
            name_handling, dialect, drift, intertext, era_bias
        ])

    async def burrows_delta(self, text: str, candidate_name: str) -> float:
        """
        Computes Burrows' Delta authorship distance.
        Delta = (1/n) * Σ |z_test - z_candidate|
        """
        if not text or not candidate_name:
            logger.error("Missing text or candidate name for Burrows' Delta")
            raise ValueError("Text and candidate name required for Burrows' Delta")

        logger.info(f"Computing Burrows' Delta for candidate: {candidate_name}")
        try:
            tokens = re.findall(r'\w+', text.lower())
            text_counts = Counter(tokens)
            total_tokens = len(tokens)
            
            async with self.pool.acquire() as conn:
                # Fetch global MFW stats for normalization
                mfw_query = """
                    SELECT word, AVG(freq) as mean_freq, STDDEV(freq) as std_dev 
                    FROM (
                        SELECT word, (count(*)::float / total_words) * 1000 as freq
                        FROM (SELECT unnest(string_to_array(lower(text_content), ' ')) as word, 
                                     cardinality(string_to_array(text_content, ' ')) as total_words 
                              FROM texts LIMIT 1000) sub
                        GROUP BY word, total_words
                    ) stats
                    GROUP BY word ORDER BY mean_freq DESC LIMIT $1
                """
                mfw_rows = await conn.fetch(mfw_query, self.mfw_count)
                
                # Fetch candidate-specific frequencies
                cand_query = """
                    SELECT word, (count(*)::float / SUM(total_words)) * 1000 as freq
                    FROM (SELECT unnest(string_to_array(lower(text_content), ' ')) as word, 
                                 cardinality(string_to_array(text_content, ' ')) as total_words 
                          FROM texts WHERE translator = $1 OR author = $1) sub
                    GROUP BY word
                """
                cand_rows = await conn.fetch(cand_query, candidate_name)
                cand_freqs = {r['word']: r['freq'] for r in cand_rows}
                
                if not mfw_rows:
                    logger.warning("No MFW stats found in database. Using default distance.")
                    return 3.0

                delta_sum = 0
                valid_words = 0
                for row in mfw_rows:
                    word = row['word']
                    mean = row['mean_freq']
                    std = row['std_dev'] if row['std_dev'] and row['std_dev'] > 0 else 0.001
                    
                    text_freq = (text_counts.get(word, 0) / total_tokens) * 1000
                    z_test = (text_freq - mean) / std
                    
                    cand_freq = cand_freqs.get(word, mean)
                    z_cand = (cand_freq - mean) / std
                    
                    delta_sum += abs(z_test - z_cand)
                    valid_words += 1
                
                delta = delta_sum / max(valid_words, 1)
                logger.info(f"Burrows' Delta result for {candidate_name}: {delta:.4f}")
                return delta
        except Exception as e:
            logger.error(f"Burrows' Delta calculation failed: {e}")
            return 5.0

    async def verify_meaning_preservation(self, source: str, translation: str, threshold: float = 0.85) -> bool:
        """
        Verifies semantic fidelity using word embedding centroids from the database.
        Ensures that style changes do not corrupt the underlying meaning.
        """
        if not source or not translation:
            logger.warning("Empty source or translation for meaning verification")
            return False

        try:
            async with self.pool.acquire() as conn:
                source_tokens = re.findall(r'\w+', source.lower())
                trans_tokens = re.findall(r'\w+', translation.lower())
                all_words = list(set(source_tokens + trans_tokens))
                
                rows = await conn.fetch(
                    "SELECT word, vector FROM word_embeddings WHERE word = ANY($1)", all_words
                )
                embedding_map = {row['word']: np.array(row['vector']) for row in rows}
                
                def get_centroid(tkns: List[str]):
                    vecs = [embedding_map[t] for t in tkns if t in embedding_map]
                    return np.mean(vecs, axis=0) if vecs else np.zeros(300)

                c_src = get_centroid(source_tokens)
                c_trans = get_centroid(trans_tokens)
                
                if np.all(c_src == 0) or np.all(c_trans == 0):
                    logger.warning("Insufficient embedding data for semantic verification. Defaulting to True.")
                    return True 

                sim = 1 - distance.cosine(c_src, c_trans)
                logger.info(f"Semantic similarity (Meaning Invariance): {sim:.4f}")
                return sim >= threshold
        except Exception as e:
            logger.error(f"Meaning verification failed: {e}")
            return False

    async def compute_ltqi(self, translation: str, source: str, target_style: Dict[str, float]) -> Dict[str, Any]:
        """
        LOGOS Translation Quality Index (LTQI).
        Weights: semantic=0.30, syntactic=0.20, register=0.15, fluency=0.15, corpus=0.20
        """
        logger.info("Computing LTQI Score...")
        try:
            # 1. SEMANTIC FIDELITY (30%)
            semantic = 0.95 if await self.verify_meaning_preservation(source, translation) else 0.65
            
            # 2. SYNTACTIC QUALITY (20%)
            sentences = [s for s in re.split(r'[.!?]+', translation) if s.strip()]
            valid_sentences = sum(1 for s in sentences if s[0].isupper() and (s.endswith('.') or s.endswith('!') or s.endswith('?')))
            syntactic = valid_sentences / max(len(sentences), 1)
            
            # 3. REGISTER APPROPRIATENESS (15%)
            actual_vec = self.compute_style_vector(translation, source)
            target_vec = np.array([target_style.get(k, 0.5) for k in [
                "formality", "archaism", "sent_len", "complexity", "word_order",
                "anglo_saxon", "figurative", "rhythm", "fidelity", "addition",
                "omission", "reg_consistency", "lex_density", "mirror", "particle",
                "name_handling", "dialect", "drift", "intertext", "era_bias"
            ]])
            register = 1.0 - distance.cosine(actual_vec, target_vec)
            
            # 4. FLUENCY (15%)
            tokens = re.findall(r'\w+', translation.lower())
            ttr = len(set(tokens)) / max(len(tokens), 1)
            fluency = np.clip(ttr * 1.5, 0, 1)
            
            # 5. CORPUS GROUNDING (20%)
            async with self.pool.acquire() as conn:
                sample_phrase = " ".join(tokens[:2]) if len(tokens) >= 2 else tokens[0]
                count = await conn.fetchval("SELECT count(*) FROM texts WHERE text_content ILIKE $1", f"%{sample_phrase}%")
                corpus_score = min(1.0, count / 20.0)
            
            ltqi = (0.30 * semantic + 0.20 * syntactic + 0.15 * register + 0.15 * fluency + 0.20 * corpus_score)
            grade = 'A+' if ltqi >= 0.95 else 'A' if ltqi >= 0.90 else 'B+' if ltqi >= 0.85 else 'B' if ltqi >= 0.80 else 'C'
            
            return {
                "score": round(ltqi * 100, 2),
                "grade": grade,
                "breakdown": {
                    "semantic": round(semantic, 3), "syntactic": round(syntactic, 3),
                    "register": round(register, 3), "fluency": round(fluency, 3), "corpus": round(corpus_score, 3)
                }
            }
        except Exception as e:
            logger.error(f"LTQI computation failed: {e}")
            raise

    def blend_styles(self, style1: np.ndarray, style2: np.ndarray, weight: float) -> np.ndarray:
        """Interpolates between two translator styles using a weighted average."""
        if not (0 <= weight <= 1):
            logger.error(f"Invalid weight for style blending: {weight}")
            weight = 0.5
        logger.info(f"Blending styles with weight: {weight}")
        return np.clip(style1 * weight + style2 * (1 - weight), 0, 1)

    def style_arithmetic(self, styles: List[np.ndarray], weights: List[float]) -> np.ndarray:
        """Performs mathematical operations on multiple style vectors."""
        if len(styles) != len(weights):
            logger.error("Style arithmetic dimension mismatch")
            raise ValueError("Number of styles must match number of weights")
        
        result = np.zeros(20)
        for style, weight in zip(styles, weights):
            result += style * weight
        return np.clip(result, 0, 1)

    def decompose_delta(self, source: str, trans_a: str, trans_b: str) -> Dict[str, float]:
        """
        7-layer causal analysis of WHY two translations differ.
        Returns percentage contribution for each layer.
        """
        logger.info("Decomposing translation delta across 7 layers...")
        
        # Layer 1: Orthographic (Punctuation and Case)
        punct_a = len(re.findall(r'[^\w\s]', trans_a))
        punct_b = len(re.findall(r'[^\w\s]', trans_b))
        ortho_delta = abs(punct_a - punct_b) / max(len(trans_a), 1)

        # Layer 2: Morphological (Suffixes and Word Forms)
        tokens_a = trans_a.lower().split()
        tokens_b = trans_b.lower().split()
        morph_delta = sum(1 for a, b in zip(tokens_a, tokens_b) if a != b and a[:4] == b[:4]) / max(len(tokens_a), 1)

        # Layer 3: Lexical (Vocabulary Choice)
        vocab_a = set(tokens_a)
        vocab_b = set(tokens_b)
        lex_delta = len(vocab_a ^ vocab_b) / max(len(vocab_a | vocab_b), 1)

        # Layer 4: Syntactic (Sentence Structure)
        sent_a = [s for s in re.split(r'[.!?]+', trans_a) if s.strip()]
        sent_b = [s for s in re.split(r'[.!?]+', trans_b) if s.strip()]
        syn_delta = abs(len(sent_a) - len(sent_b)) / max(len(sent_a), 1)

        # Layer 5: Semantic (Content Meaning)
        content_a = {t for t in tokens_a if t not in ENGLISH_FUNCTION_WORDS}
        content_b = {t for t in tokens_b if t not in ENGLISH_FUNCTION_WORDS}
        sem_delta = len(content_a ^ content_b) / max(len(content_a | content_b), 1)

        # Layer 6: Discourse (Cohesion Markers)
        connectives = ['however', 'therefore', 'but', 'and', 'then', 'yet', 'moreover', 'consequently']
        conn_a = {t for t in tokens_a if t in connectives}
        conn_b = {t for t in tokens_b if t in connectives}
        disc_delta = len(conn_a ^ conn_b) / max(len(connectives), 1)

        # Layer 7: Pragmatic (Style and Tone)
        vec_a = self.compute_style_vector(trans_a, source)
        vec_b = self.compute_style_vector(trans_b, source)
        prag_delta = distance.euclidean(vec_a, vec_b) / np.sqrt(20)

        total = ortho_delta + morph_delta + lex_delta + syn_delta + sem_delta + disc_delta + prag_delta
        if total == 0: return {layer.name: 0.0 for layer in DeltaLayer}

        return {
            "ORTHOGRAPHIC": round(ortho_delta / total, 4),
            "MORPHOLOGICAL": round(morph_delta / total, 4),
            "LEXICAL": round(lex_delta / total, 4),
            "SYNTACTIC": round(syn_delta / total, 4),
            "SEMANTIC": round(sem_delta / total, 4),
            "DISCOURSE": round(disc_delta / total, 4),
            "PRAGMATIC": round(prag_delta / total, 4)
        }

    async def translate_in_style(self, text: str, target_style_vector: np.ndarray) -> str:
        """
        Renders a translation by adjusting lexical features to match a target style vector.
        This implementation performs rule-based lexical substitution and structural adjustment.
        """
        logger.info("Rendering style-aware translation...")
        if not text:
            return ""
            
        tokens = text.split()
        
        # 1. Archaism Adjustment (Dim 1)
        if target_style_vector[1] > 0.7:
            archaic_map = {
                "you": "thou", "your": "thy", "yours": "thine", "are": "art", 
                "have": "hast", "has": "hath", "does": "doth", "know": "wot"
            }
            tokens = [archaic_map.get(t.lower(), t) for t in tokens]
        
        # 2. Formality Adjustment (Dim 0)
        if target_style_vector[0] > 0.8:
            formal_map = {
                "anger": "wrath", "sing": "chant", "man": "mortal", "sea": "deep",
                "war": "strife", "god": "deity", "home": "abode", "quick": "fleet"
            }
            tokens = [formal_map.get(t.lower(), t) for t in tokens]
            
        # 3. Anglo-Saxon vs Latinate (Dim 5)
        if target_style_vector[5] < 0.3: # Latinate preference
            latinate_map = {
                "fight": "combat", "help": "assist", "build": "construct", "show": "demonstrate"
            }
            tokens = [latinate_map.get(t.lower(), t) for t in tokens]

        # 4. Particle Rendering (Dim 14)
        if target_style_vector[14] > 0.6:
            if len(tokens) > 5:
                tokens.insert(0, "Verily,")
                tokens.insert(len(tokens)//2, "indeed")

        result = " ".join(tokens)
        
        # 5. Sentence Length Adjustment (Dim 2)
        if target_style_vector[2] < 0.3: # Short sentences
            result = result.replace(", ", ". ").replace("; ", ". ")
            
        logger.info(f"Style-aware translation rendered. Length: {len(result)}")
        return result

class DependencyGraph:
    """Manages the directed graph of dependencies and DB synchronization."""

    def __init__(self, pool: asyncpg.Pool):
        self.nodes: Dict[str, Any] = {}
        self.pool = pool

    async def initialize(self):
        """Initializes database connections and verifies schema."""
        logger.info("Initializing DependencyGraph...")
        try:
            async with self.pool.acquire() as conn:
                version = await conn.fetchval("SELECT version()")
                logger.info(f"Connected to Railway PostgreSQL: {version}")
                await self.sync_with_database()
        except asyncpg.PostgresError as e:
            logger.error(f"Failed to initialize DependencyGraph: {e}")
            raise

    async def sync_with_database(self):
        """Cross-references identified tables with the actual PostgreSQL schema."""
        try:
            async with self.pool.acquire() as conn:
                rows = await conn.fetch("SELECT table_name FROM information_schema.tables WHERE table_schema = 'public'")
                existing_tables = {row['table_name'] for row in rows}
                required_tables = {
                    'texts', 'source_texts', 'author_profiles', 
                    'translator_profiles', 'word_embeddings', 
                    'style_analyses', 'translation_vectors'
                }
                missing = required_tables - existing_tables
                if missing:
                    logger.warning(f"Missing tables in database: {missing}")
                else:
                    logger.info(f"All {len(required_tables)} required tables verified.")
                return existing_tables
        except Exception as e:
            logger.error(f"Database sync failed: {e}")
            raise

        logger.info(f"LTQI for Pope: {ltqi_pope['score']} (Grade: {ltqi_pope['grade']})")

        # 5. Authorship Attribution (Burrows' Delta)
        # Lower delta means higher likelihood of authorship match
        dist_pope = await engine.burrows_delta(trans_pope, "Pope")
        dist_wilson = await engine.burrows_delta(trans_pope, "Wilson")

        logger.info(f"Burrows' Delta (Pope-to-Wilson): {dist_wilson:.4f}")

    finally:

            logger.info("Database Pool Closed. Pipeline Shutdown.")

if __name__ == "__main__":
