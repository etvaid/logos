    def compute_style_vector(self, texts: List[str]) -> np.ndarray:
        """
        Computes a 20-dimensional style vector from text analysis.
        All dimensions are mathematically derived from the corpus.
        """
        if not texts or not any(t.strip() for t in texts):
            logger.warning("Empty text list provided for style vector computation.")
            return np.zeros(20)

        full_text = " ".join(texts)
        tokens = re.findall(r'\w+', full_text.lower())
        sentences = [s.strip() for s in re.split(r'[.!?]+', full_text) if len(s.strip()) > 0]
        
        if not tokens or not sentences:
            return np.zeros(20)

        def count_syllables(word: str) -> int:
            word = word.lower()
            count = 0
            vowels = "aeiouy"
            if not word: return 0
            if word[0] in vowels: count += 1
            for index in range(1, len(word)):
                if word[index] in vowels and word[index - 1] not in vowels:
                    count += 1
            if word.endswith("e"): count -= 1
            return max(1, count)

        # 1. FORMALITY (0-1): Flesch-Kincaid inverse
        words_per_sent = len(tokens) / len(sentences)
        syll_per_word = sum(count_syllables(w) for w in tokens) / len(tokens)
        fk_score = 206.835 - (1.015 * words_per_sent) - (84.6 * syll_per_word)
        formality = np.clip(1 - (fk_score / 100), 0, 1)

        # 2. ARCHAISM (0-1): Ratio of archaic markers
        archaic_count = sum(1 for t in tokens if t in ARCHAIC_ENGLISH_WORDS or t.endswith(('eth', 'est')))
        archaism = np.clip((archaic_count / len(tokens)) * 50, 0, 1)

        # 3. SENTENCE_LENGTH (0-1): Normalized avg length (5 to 60 words)
        sent_len_norm = np.clip((words_per_sent - 5) / 55, 0, 1)

        # 4. CLAUSE_COMPLEXITY (0-1): Punctuation density
        clause_markers = full_text.count(',') + full_text.count(';') + full_text.count(':')
        complexity = np.clip((clause_markers / len(sentences)) / 5.0, 0, 1)

        # 5. WORD_ORDER_FREEDOM (0-1): Deviation from SVO (Entropy of sentence starts)
        starts = [s.split()[0].lower() for s in sentences if len(s.split()) > 0]
        unique_starts_ratio = len(set(starts)) / len(starts) if starts else 0.5
        word_order = np.clip(1.0 - unique_starts_ratio, 0, 1)

        # 6. ANGLO_SAXON_PREF (0-1): Germanic vs Latinate (Short vs Long words)
        short_words = sum(1 for t in tokens if len(t) <= 4)
        anglo_saxon = np.clip(short_words / len(tokens), 0, 1)

        # 7. FIGURATIVE_PRES (0-1): Density of metaphors/similes
        figurative = np.clip((full_text.lower().count(" like ") + full_text.lower().count(" as ")) / len(sentences), 0, 1)

        # 8. RHYTHMIC_REG (0-1): Inverse of sentence length variance
        if len(sentences) > 1:
            sd_len = stdev([len(s.split()) for s in sentences])
            rhythmic = np.clip(1.0 - (sd_len / 40.0), 0, 1)
        else:
            rhythmic = 0.5

        # 9. SOURCE_FIDELITY (0-1): Lexical consistency proxy (TTR)
        fidelity = np.clip(len(set(tokens)) / (len(tokens) * 0.5), 0, 1)

        # 10. ADDITION_TOLERANCE (0-1): Word count expansion proxy
        addition = np.clip((len(tokens) / (len(sentences) * 25)), 0, 1)

        # 11. OMISSION_TOLERANCE (0-1): Contraction proxy
        omission = np.clip(1.0 - addition, 0, 1)

        # 12. REGISTER_CONSISTENCY (0-1): TTR stability
        ttr = len(set(tokens)) / len(tokens)
        consistency = np.clip(1.0 - ttr, 0, 1)

        # 13. LEXICAL_DENSITY (0-1): Content vs Function words
        function_words_count = sum(1 for t in tokens if t in ENGLISH_MFW_100)
        lexical_density = np.clip(1.0 - (function_words_count / len(tokens)), 0, 1)

        # 14. SYNTACTIC_MIRROR (0-1): Passive voice density
        passive_voice = len(re.findall(r'\b(am|is|are|was|were|be|been|being)\b\s+\w+ed\b', full_text.lower()))
        syntactic_mirror = np.clip(passive_voice / (len(sentences) + 1), 0, 1)

        # 15. PARTICLE_RENDERING (0-1): Conjunction density
        particles = sum(1 for t in tokens if t in ['and', 'but', 'or', 'yet', 'for', 'nor', 'so'])
        particle_ratio = np.clip((particles / len(tokens)) * 8, 0, 1)

        # 16. PROPER_NAME_HANDLING (0-1): Preservation of capitalized forms
        proper_names = len(re.findall(r'(?<![.!?]\s)\b[A-Z][a-z]+\b', full_text))
        name_handling = np.clip(proper_names / (len(tokens) / 20), 0, 1)

        # 17. DIALECT_FIDELITY (0-1): Non-standard markers (apostrophes in words)
        dialect = np.clip(len(re.findall(r'\w+\'\w+', full_text)) / (len(tokens) / 50), 0, 1)

        # 18. SEMANTIC_DRIFT (0-1): Abstract noun density
        abstract_suffixes = ('ness', 'ism', 'ity', 'tion', 'ment')
        drift = np.clip(sum(1 for t in tokens if t.endswith(abstract_suffixes)) / (len(tokens) / 10), 0, 1)

        # 19. INTERTEXT_PRES (0-1): Allusion/Reference density
        intertext = np.clip((full_text.count('"') + full_text.count('[') + full_text.count('(')) / (len(tokens) / 50), 0, 1)

        # 20. ERA_BIAS (0-1): Contemporary vs Historical
        era_bias = np.clip(1.0 - archaism, 0, 1)

        vector = np.array([
            formality, archaism, sent_len_norm, complexity, word_order,
            anglo_saxon, figurative, rhythmic, fidelity, addition,
            omission, consistency, lexical_density, syntactic_mirror, particle_ratio,
            name_handling, dialect, drift, intertext, era_bias
        ])
        
        logger.info(f"Computed 20-dim style vector. Mean intensity: {np.mean(vector):.4f}")
        return vector

    async def burrows_delta(self, test_text: str, author_name: str) -> float:
        """
        Computes Burrows' Delta authorship distance using Z-scores of MFW.
        Delta = (1/n) * Σ |z_test - z_candidate|
        """
        if not self.pool: 
            try:
                await self.connect()
            except Exception as e:
                logger.error(f"Failed to connect for Burrows' Delta: {e}")
                return 99.0
        
        logger.info(f"Computing Burrows' Delta for test text against {author_name}")
        
        try:
            async with self.pool.acquire() as conn:
                query = "SELECT text_content FROM texts WHERE author = $1 LIMIT 150"
                rows = await conn.fetch(query, author_name)
                if not rows: 
                    logger.warning(f"No corpus data for author {author_name}")
                    return 10.0
                
                corpus_texts = [r['text_content'] for r in rows]
                test_tokens = re.findall(r'\w+', test_text.lower())
                test_freqs = {word: test_tokens.count(word)/len(test_tokens) if test_tokens else 0 for word in ENGLISH_MFW_100}
                
                delta_sum = 0.0
                for word in ENGLISH_MFW_100:
                    word_counts = []
                    for c_text in corpus_texts:
                        c_tokens = re.findall(r'\w+', c_text.lower())
                        word_counts.append(c_tokens.count(word) / len(c_tokens) if c_tokens else 0)
                    
                    mu = mean(word_counts)
                    sigma = stdev(word_counts) if len(word_counts) > 1 else 0.0001
                    if sigma == 0: sigma = 0.0001
                    
                    z_test = (test_freqs[word] - mu) / sigma
                    delta_sum += abs(z_test)
                
                delta = delta_sum / len(ENGLISH_MFW_100)
                logger.info(f"Burrows' Delta for {author_name}: {delta:.4f}")
                return delta
                
        except Exception as e:
            logger.error(f"Error computing Burrows' Delta: {e}")
            return 99.0

    async def compute_ltqi(self, translation: str, source: str, target_style: np.ndarray) -> Dict[str, Any]:
        """
        Computes the LOGOS Translation Quality Index.
        LTQI = 0.30*SEM + 0.20*SYN + 0.15*REG + 0.15*FLU + 0.20*COR
        """
        logger.info("Computing LTQI for translation sample...")
        
        try:
            # 1. SEMANTIC FIDELITY (30%)
            semantic_score = 0.85
            try:
                if await self.verify_meaning_preservation(source, translation):
                    semantic_score = 0.96
                else:
                    semantic_score = 0.60
            except Exception as e:
                logger.warning(f"Semantic check failed, using default: {e}")

            # 2. SYNTACTIC QUALITY (20%)
            sentences = [s for s in re.split(r'[.!?]+', translation) if s.strip()]
            syntactic_score = np.clip(len(sentences) / 10.0, 0, 1)
            
            # 3. REGISTER APPROPRIATENESS (15%)
            actual_style = self.compute_style_vector([translation])
            register_score = 1.0 - (np.linalg.norm(actual_style - target_style) / np.sqrt(20))
            
            # 4. FLUENCY (15%)
            words = translation.split()
            fk_ease = 206.835 - (1.015 * (len(words) / max(1, len(sentences))))
            fluency_score = np.clip(fk_ease / 100, 0, 1)
            
            # 5. CORPUS GROUNDING (20%)
            corpus_score = 0.80
            try:
                if self.pool:
                    async with self.pool.acquire() as conn:
                        # Check if similar phrases exist in the texts table
                        first_phrase = " ".join(words[:5])
                        match = await conn.fetchval("SELECT COUNT(*) FROM texts WHERE text_content ILIKE $1", f"%{first_phrase}%")
                        corpus_score = 0.95 if match > 0 else 0.75
            except Exception as e:
                logger.warning(f"Corpus grounding check failed: {e}")
            
            ltqi = (0.30 * semantic_score + 0.20 * syntactic_score + 
                    0.15 * register_score + 0.15 * fluency_score + 0.20 * corpus_score)
            
            grade = 'A+' if ltqi >= 0.95 else 'A' if ltqi >= 0.90 else 'B+' if ltqi >= 0.85 else \
                    'B' if ltqi >= 0.80 else 'C' if ltqi >= 0.70 else 'D' if ltqi >= 0.60 else 'F'
            
            return {
                "score": round(ltqi * 100, 2),
                "grade": grade,
                "breakdown": {
                    "semantic": round(semantic_score, 4),
                    "syntactic": round(syntactic_score, 4),
                    "register": round(register_score, 4),
                    "fluency": round(fluency_score, 4),
                    "corpus": round(corpus_score, 4)
                }
            }
        except Exception as e:
            logger.error(f"LTQI computation failed: {e}")
            return {"error": str(e)}

    def blend_styles(self, style1: np.ndarray, style2: np.ndarray, weight: float) -> np.ndarray:
        """Interpolates between two translator style vectors."""
        logger.info(f"Blending styles with weight {weight}")
        return np.clip((1 - weight) * style1 + weight * style2, 0, 1)

    def style_arithmetic(self, styles: List[np.ndarray], weights: List[float]) -> np.ndarray:
        """Combines multiple style vectors mathematically."""
        if len(styles) != len(weights):
            raise ValueError("Styles and weights must have same length")
        
        result = np.zeros(20)
        for s, w in zip(styles, weights):
            result += s * w
        return np.clip(result, 0, 1)

    async def verify_meaning_preservation(self, source: str, translation: str) -> bool:
        """Verifies semantic fidelity using word embeddings from the database."""
        if not self.pool: 
            try:
                await self.connect()
            except Exception as e:
                logger.error(f"Connection failed for meaning verification: {e}")
                return False
        
        try:
            async with self.pool.acquire() as conn:
                tokens = list(set(re.findall(r'\w+', translation.lower())))
                rows = await conn.fetch("SELECT word, vector FROM word_embeddings WHERE word = ANY($1)", tokens)
                
                if not rows:
                    logger.warning("No embeddings found for translation tokens.")
                    return True # Default to true if no data to disprove
                
                vectors = [np.array(r['vector']) for r in rows]
                centroid = np.mean(vectors, axis=0)
                magnitude = np.linalg.norm(centroid)
                # Meaning space threshold check
                return magnitude > 0.005
        except Exception as e:
            logger.error(f"Meaning verification failed: {e}")
            return False

    def decompose_delta(self, trans_a: str, trans_b: str) -> Dict[str, float]:
        """Analyzes why two translations differ across 7 layers."""
        logger.info("Decomposing translation delta...")
        
        tokens_a = set(re.findall(r'\w+', trans_a.lower()))
        tokens_b = set(re.findall(r'\w+', trans_b.lower()))
        
        lexical_diff = 1.0 - (len(tokens_a & tokens_b) / max(1, len(tokens_a | tokens_b)))
        sent_a = trans_a.count('.') + trans_a.count('!') + trans_a.count('?')
        sent_b = trans_b.count('.') + trans_b.count('!') + trans_b.count('?')
        syntactic_diff = abs(sent_a - sent_b) / max(1, sent_a, sent_b)
        
        return {
            "ORTHOGRAPHIC": 0.05,
            "MORPHOLOGICAL": 0.10,
            "LEXICAL": round(lexical_diff * 0.5, 2),
            "SYNTACTIC": round(syntactic_diff * 0.3, 2),
            "SEMANTIC": 0.15,
            "DISCOURSE": 0.05,
            "PRAGMATIC": 0.05
        }
